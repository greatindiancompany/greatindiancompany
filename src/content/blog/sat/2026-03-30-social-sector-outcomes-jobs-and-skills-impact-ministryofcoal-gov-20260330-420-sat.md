# ᱥᱤᱧᱚᱛ ᱨᱮᱱᱟᱜ ᱥᱟᱶᱛᱟᱨᱤ ᱚᱨᱡᱚ ᱯᱟᱹᱪᱷᱞᱟ ᱨᱮᱱᱟᱜ ᱯᱩᱭᱞᱩ ᱞᱟᱭᱤᱱ ᱠᱟᱹᱢᱤᱼᱦᱚᱲ

ᱥᱤᱧᱚᱛ ᱨᱮᱱᱟᱜ ᱥᱟᱶᱛᱟᱨᱤ ᱚᱨᱡᱚ ᱫᱚ ᱧᱩᱛᱩᱢ ᱠᱟᱹᱰᱨ ᱛᱮ ᱥᱮᱴᱮᱨᱟ, ᱢᱤᱫ ᱦᱮᱰᱞᱟᱭᱤᱱ ᱠᱟᱹᱢᱤ ᱫᱚᱨ ᱛᱮ ᱵᱟᱝ᱾ ᱯᱮᱭᱟ ᱥᱚᱨᱠᱟᱨᱤ ᱮᱞ ᱢᱤᱫ ᱥᱟᱫᱷᱟᱨᱚᱱ «ᱥᱟᱶᱛᱟᱨᱤ ᱠᱷᱚᱸᱰ ᱨᱮ ᱠᱟᱹᱢᱤ» ᱞᱟᱭᱤᱱ ᱠᱷᱚᱱ ᱵᱟᱹᱲᱛᱤ ᱢᱮᱱᱟᱜ-ᱟ: ᱯᱚᱥᱚᱱ ᱴᱨᱮᱠᱚᱨ (Poshan Tracker) ᱨᱮ ᱟᱝᱜᱟᱱᱣᱟᱲᱤ ᱠᱟᱹᱢᱤᱭᱟᱹ, UDISE+ ᱨᱮ ᱢᱟᱪᱮᱛ, ᱟᱨ MGNREGA ᱨᱮᱱᱟᱜ ᱡᱟᱭᱜᱟ ᱦᱟᱛᱟᱣ ᱟᱹᱭᱤᱱ ᱞᱟᱛᱟᱨ ᱟᱹᱛᱩ ᱵᱷᱟᱹᱰᱤ ᱤᱫᱤ ᱜᱷᱟᱨᱚᱸᱡᱽ᱾ ᱚᱠᱛᱚᱼᱵᱚᱫᱚᱞ ᱠᱟᱹᱢᱤᱼᱦᱚᱲ ᱥᱟᱨᱣᱮ (Periodic Labour Force Survey, PLFS) ᱛᱟᱭᱚᱢ ᱞᱟᱹᱭᱟ ᱡᱮ ᱡᱟᱹᱯᱩᱫ ᱠᱟᱹᱢᱤᱼᱵᱟᱡᱟᱨ ᱪᱮᱫ ᱠᱟᱹᱢᱤ ᱮᱫᱟ᱾ ᱱᱚᱣᱟ ᱵᱟᱝ ᱞᱟᱹᱭᱟ ᱡᱮ ᱚᱱᱟ ᱠᱟᱹᱢᱤᱭᱟᱹ ᱠᱚ ᱛᱟᱞᱟ ᱨᱮ ᱛᱤᱱᱟᱹᱜ ᱱᱚᱣᱟ ᱠᱟᱹᱢᱤᱦᱚᱨᱟ ᱵᱷᱤᱛᱨᱤ ᱨᱮ ᱢᱮᱱᱟᱜ ᱠᱚᱣᱟ᱾

## ᱯᱚᱥᱴᱤᱠ ᱠᱟᱹᱰᱨ ᱮᱞ ᱟᱠᱟᱱᱟ, ᱟᱨ ᱩᱱᱠᱩ ᱫᱩᱦᱲᱟᱹ ᱥᱮᱪᱮᱫ ᱦᱩᱭᱩᱜ ᱠᱟᱱᱟ

31 ᱡᱩᱞᱟᱭ 2026 ᱫᱷᱟᱹᱵᱤᱡ ᱯᱨᱮᱥ ᱤᱱᱯᱷᱚᱨᱢᱮᱥᱚᱱ ᱵᱭᱩᱨᱳ (Press Information Bureau) ᱫᱚ ᱥᱩᱨ ᱥᱩᱨ 13.99 ᱞᱟᱠᱷ ᱟᱝᱜᱟᱱᱣᱟᱲᱤ ᱛᱟᱞᱢᱟ ᱟᱨ 13.30 ᱞᱟᱠᱷ ᱟᱝᱜᱟᱱᱣᱟᱲᱤ ᱠᱟᱹᱢᱤᱭᱟᱹ ᱵᱟᱵᱚᱛ ᱞᱟᱹᱭ ᱠᱮᱫᱟ, ᱡᱟᱦᱟᱸᱭ ᱠᱚ ᱥᱩᱨ ᱥᱩᱨ 8.94 ᱠᱚᱨᱚᱲ ᱞᱟᱵᱷᱟᱱᱤᱭᱟᱹ ᱴᱷᱮᱱ ᱥᱮᱴᱮᱨᱚᱜ ᱠᱟᱱᱟ᱾ ᱚᱱᱟ ᱠᱟᱹᱢᱤᱭᱟᱹ ᱜᱮ ᱚᱱᱟ ᱦᱚᱲ ᱠᱟᱱᱟ ᱡᱟᱦᱟᱸᱭ ᱛᱟᱞᱢᱟ ᱮ ᱡᱷᱤᱡᱟ, ᱦᱟᱡᱤᱨᱟ ᱮ ᱪᱤᱱᱦᱟᱹᱭᱟ, ᱥᱮᱸᱜᱮᱞ ᱨᱟᱱᱫᱟ ᱡᱚᱢ ᱮ ᱮᱢᱟ ᱥᱮ ᱚᱲᱟᱜ ᱤᱫᱤ ᱡᱚᱢᱟᱜ (take-home ration) ᱮ ᱮᱢᱟ, ᱟᱨ ᱵᱟᱲᱦᱟᱣ ᱮ ᱚᱞᱟ᱾ ᱡᱟᱦᱟᱸ ᱯᱚᱥᱴᱤᱠ ᱚᱨᱡᱚ ᱵᱷᱮᱜᱟᱨ ᱵᱮᱵᱷᱟᱨᱤᱭᱟᱹ ᱮ ᱢᱟᱱᱟᱣᱟ — ᱞᱩᱯ ᱨᱮ ᱠᱟᱹᱢᱤᱭᱟᱹ ᱵᱟᱹᱱᱩᱜ ᱢᱤᱫ ᱮᱸᱜᱟᱛ ᱮᱯ — ᱚᱱᱟ ᱫᱚ ᱢᱚᱱᱛᱨᱟᱞᱚᱭ ᱥᱟᱹᱨᱤ ᱛᱮ ᱡᱟᱦᱟᱸ ᱥᱤᱥᱴᱚᱢ ᱮ ᱪᱟᱞᱟᱣ ᱮᱫᱟ ᱚᱱᱟ ᱵᱤᱨᱩᱫᱷ ᱨᱮ ᱰᱤᱡᱟᱭᱤᱱ ᱟᱠᱟᱱᱟ᱾

ᱚᱱᱟ ᱠᱟᱹᱰᱨ ᱪᱮᱛᱟᱱ ᱦᱩᱱᱟᱹᱨ ᱠᱷᱚᱡ ᱫᱚ ᱵᱷᱮᱜᱟᱨ ᱠᱟᱱᱟ᱾ ᱯᱚᱥᱚᱱ ᱴᱨᱮᱠᱚᱨ ᱪᱮᱛᱟᱱ ᱢᱟᱨᱪ 2026 ᱢᱚᱱᱛᱨᱟᱞᱚᱭ ᱛᱮᱞᱟ ᱞᱟᱹᱭ ᱞᱮᱫᱟ ᱡᱮ ᱢᱚᱦᱟᱲ ᱪᱤᱱᱦᱟᱹᱯ (facial recognition) ᱚᱲᱟᱜ ᱤᱫᱤ ᱡᱚᱢᱟᱜ ᱫᱚ ᱧᱩᱛᱩᱢ ᱚᱞ ᱞᱟᱵᱷᱟᱱᱤᱭᱟᱹ ᱴᱷᱮᱱ ᱥᱮᱴᱮᱨᱚᱜ ᱠᱟᱱᱟ ᱢᱮᱱᱛᱮ ᱯᱩᱥᱴᱟᱹᱣ ᱞᱟᱹᱜᱤᱫ ᱵᱮᱵᱷᱟᱨᱚᱜ ᱠᱟᱱᱟ, ᱟᱨ ᱡᱟᱹᱛᱤᱭᱟᱹᱨᱤ ᱤᱼᱥᱟᱥᱚᱱ ᱦᱟᱹᱴᱤᱧ (National e-Governance Division) ᱨᱮᱱ ᱠᱚᱼᱚᱨᱰᱤᱱᱮᱴᱚᱨ ᱠᱟᱹᱢᱤᱭᱟᱹ ᱠᱚ ᱚᱱᱟ ᱪᱮᱛᱟᱱ ᱥᱮᱪᱮᱫ ᱠᱚᱣᱟ᱾ ᱯᱚᱥᱚᱱ ᱵᱷᱤ ᱯᱚᱲᱦᱟᱣ ᱵᱷᱤ (Poshan Bhi Padhai Bhi) ᱞᱟᱛᱟᱨ, ᱥᱟᱵᱤᱛᱨᱤᱵᱟᱭ ᱯᱷᱩᱞᱮ ᱡᱟᱹᱛᱤᱭᱟᱹᱨᱤ ᱢᱟᱹᱭᱡᱤᱣ ᱟᱨ ᱜᱤᱫᱽᱨᱟᱹ ᱩᱛᱱᱟᱹᱣ ᱤᱱᱥᱴᱤᱪᱩᱴ (Savitribai Phule National Institute of Women and Child Development) 18 ᱢᱟᱨᱪ 2026 ᱫᱷᱟᱹᱵᱤᱡ 3,901 ᱯᱚᱱᱚᱛᱼᱛᱷᱚᱠ ᱢᱟᱥᱴᱚᱨ ᱴᱨᱮᱱᱚᱨ ᱠᱚ ᱥᱮᱪᱮᱫ ᱞᱮᱫ ᱠᱚᱣᱟ, ᱡᱟᱦᱟᱸᱭ ᱠᱚ ᱛᱟᱭᱚᱢ ᱠᱟᱹᱢᱤᱭᱟᱹ ᱠᱚ ᱥᱮᱪᱮᱫ ᱠᱚᱣᱟ᱾ ᱥᱮᱯᱴᱮᱢᱵᱚᱨ 2026 ᱯᱚᱥᱚᱱ ᱢᱟᱦ ᱱᱳᱴ ᱢᱤᱫ ᱵᱚᱫᱚᱞ ᱚᱲᱟᱜᱼᱫᱟᱣᱟᱨ ᱥᱮᱰᱩᱞᱚᱨ ᱮ ᱡᱚᱲᱟᱣᱟ, 0-3 ᱥᱮᱨᱢᱟ ᱵᱚᱭᱚᱥ ᱨᱮᱱ ᱜᱤᱫᱽᱨᱟᱹ ᱞᱟᱹᱜᱤᱫ 34 ᱩᱛᱱᱟᱹᱣ ᱢᱟᱭᱤᱞᱥᱴᱚᱱ ᱥᱟᱶ, ᱟᱨ ᱮᱛᱚᱦᱚᱵ ᱜᱤᱫᱽᱨᱟᱹ ᱧᱮᱞᱡᱚᱝ ᱞᱟᱹᱜᱤᱫ ᱱᱚᱵᱽᱪᱮᱛᱱᱟ (Navchetna) ᱟᱨ ᱟᱫᱷᱟᱨᱥᱤᱞᱟ (Aadharshila) ᱯᱟᱲᱦᱟᱣᱼᱯᱟᱛᱷ᱾ ᱠᱟᱹᱢᱤ ᱫᱚ ᱯᱚᱥᱴᱤᱠ ᱟᱨ ᱮᱛᱚᱦᱚᱵ ᱩᱥᱠᱟᱹᱣ ᱟᱨ ᱢᱤᱫ ᱵᱟᱭᱚᱢᱮᱴᱨᱤᱠ ᱵᱷᱟᱜᱮ ᱠᱷᱚᱸᱡ ᱠᱟᱱᱟ᱾ ᱢᱟᱥᱴᱚᱨᱼᱴᱨᱮᱱᱚᱨ ᱮᱞ ᱠᱷᱚᱱ ᱵᱟᱦᱨᱮ ᱨᱮᱱᱟᱜ ᱠᱷᱟᱹᱞᱤ ᱟᱨ ᱥᱮᱪᱮᱫ ᱠᱚᱵᱷᱚᱨ ᱱᱚᱣᱟ ᱱᱳᱴ ᱠᱚ ᱨᱮ ᱵᱟᱝ ᱞᱟᱹᱭ ᱟᱠᱟᱱᱟ᱾

ᱡᱟᱦᱟᱸᱨᱮ VB-G RAM G ᱠᱟᱹᱢᱤᱼᱡᱟᱭᱜᱟ ᱨᱮ ᱢᱚᱬᱮ ᱥᱮᱨᱢᱟ ᱞᱟᱛᱟᱨ ᱨᱮᱱ ᱢᱚᱬᱮ ᱥᱮ ᱚᱱᱟ ᱠᱷᱚᱱ ᱵᱟᱹᱲᱛᱤ ᱜᱤᱫᱽᱨᱟᱹ ᱢᱮᱱᱟᱜ ᱠᱚᱣᱟ, ᱟᱹᱛᱩ ᱠᱟᱹᱢᱤ ᱟᱹᱭᱤᱱ ᱢᱤᱫ ᱢᱟᱹᱭᱡᱤᱣ ᱠᱟᱹᱢᱤᱭᱟᱹ ᱮ ᱠᱷᱚᱡᱟ, ᱵᱷᱟᱹᱰᱤ ᱫᱚᱨ ᱨᱮ ᱯᱩᱨᱟᱹᱣ, ᱩᱱᱠᱩ ᱧᱮᱞ ᱞᱟᱹᱜᱤᱫ᱾ ᱚᱱᱟ ᱵᱷᱟᱹᱰᱤ ᱠᱟᱹᱢᱤᱦᱚᱨᱟ ᱟᱨ ᱜᱤᱫᱽᱨᱟᱹ ᱧᱮᱞᱡᱚᱝ ᱛᱟᱞᱟ ᱨᱮᱱᱟᱜ ᱢᱤᱫ ᱠᱷᱟᱴᱚ, ᱥᱟᱹᱨᱤ ᱡᱚᱲ ᱠᱟᱱᱟ᱾

## ᱢᱟᱪᱮᱛ ᱵᱟᱹᱲᱛᱤ ᱢᱮᱱᱟᱜ ᱠᱚᱣᱟ, ᱟᱨ ᱩᱱᱠᱩ ᱛᱟᱞᱟ ᱨᱮ ᱢᱟᱹᱭᱡᱤᱣ ᱵᱟᱹᱲᱛᱤ ᱢᱮᱱᱟᱜ ᱠᱚᱣᱟ

UDISE+ 2025-26 ᱫᱚ 1,02,73,020 ᱢᱟᱪᱮᱛ ᱮ ᱮᱞᱟ, 2022-23 ᱠᱷᱚᱱ 8.3% ᱵᱟᱲᱦᱟᱣ᱾ ᱢᱟᱹᱭᱡᱤᱣ ᱫᱚ ᱢᱟᱪᱮᱛ ᱨᱮᱱᱟᱜ 54.9% ᱠᱟᱱᱟ᱾ ᱠᱩᱲᱤ ᱫᱚ ᱵᱚᱞᱚ ᱨᱮᱱᱟᱜ 48.4% ᱠᱟᱱᱟ᱾ ᱪᱮᱫᱚᱜᱼᱜᱩᱨᱩ ᱟᱨ ᱢᱟᱪᱮᱛ ᱚᱱᱩᱯᱟᱛ ᱢᱩᱲᱩᱫ ᱨᱮ 10, ᱛᱟᱭᱟᱨᱤ ᱨᱮ 12, ᱛᱟᱞᱟ ᱨᱮ 17, ᱟᱨ ᱥᱮᱠᱮᱱᱰᱟᱨᱤ ᱨᱮ 21 ᱠᱟᱱᱟ, ᱡᱟᱦᱟᱸ ᱢᱚᱱᱛᱨᱟᱞᱚᱭ ᱡᱟᱹᱛᱤᱭᱟᱹᱨᱤ ᱥᱮᱪᱮᱫ ᱱᱤᱛᱤ (National Education Policy) ᱨᱮᱱᱟᱜ 30:1 ᱨᱮᱯᱷᱟᱨᱮᱱᱥ ᱥᱟᱶ ᱫᱚᱦᱚᱭᱟ᱾ ᱢᱤᱫᱼᱢᱟᱪᱮᱛ ᱵᱤᱨᱫᱟᱹᱜᱟᱲ ᱱᱤᱛᱦᱚᱸ 1,00,843 ᱢᱮᱱᱟᱜ-ᱟ᱾ ᱡᱟᱦᱟᱸ ᱠᱟᱹᱢᱤᱭᱟᱹ ᱚᱨᱡᱚ ᱛᱟᱦᱮᱸ ᱜᱮᱭᱟ, ᱚᱱᱟ ᱫᱚ ᱜᱷᱩᱨᱟᱹᱣ ᱠᱟᱱᱟ: ᱜᱷᱟᱹᱞ ᱚᱱᱩᱯᱟᱛ ᱟᱨᱟᱢ ᱞᱮᱠᱟ ᱧᱮᱞ ᱫᱟᱲᱮᱭᱟᱜ-ᱟ ᱡᱚᱠᱷᱚᱱ ᱢᱤᱫ ᱞᱟᱠᱷ ᱵᱤᱨᱫᱟᱹᱜᱟᱲ ᱨᱮ ᱱᱤᱛᱦᱚᱸ ᱢᱤᱫ ᱢᱟᱪᱮᱛ ᱢᱮᱱᱟᱜ ᱟᱭ᱾

UDISE+ ᱢᱤᱫ ᱯᱟᱲᱦᱟᱣ ᱮᱞ ᱵᱟᱝ ᱞᱟᱹᱭᱟ᱾ ᱪᱮᱫᱚᱜᱼᱜᱩᱨᱩ ᱵᱟᱵᱚᱛ ᱢᱤᱫ ᱦᱩᱱᱟᱹᱨ ᱞᱟᱹᱭ ᱫᱚ ᱚᱱᱟ ᱡᱚᱠᱷᱟ ᱞᱟᱹᱜᱤᱫ ᱛᱟᱹᱝᱜᱤ ᱞᱟᱹᱠᱛᱤ ᱡᱟᱦᱟᱸ ᱥᱮᱪᱮᱫ ᱢᱚᱱᱛᱨᱟᱞᱚᱭ (Ministry of Education) ᱚᱱᱟ ᱞᱮᱠᱟᱛᱮ ᱩᱪᱷᱟᱹᱱᱟ᱾ ᱱᱚᱣᱟ ᱩᱪᱷᱟᱹᱱ ᱡᱟᱦᱟᱸ ᱥᱟᱯᱚᱴ ᱮᱫᱟ, ᱚᱱᱟ ᱫᱚ ᱵᱤᱨᱫᱟᱹᱜᱟᱲ ᱨᱮ ᱢᱟᱨᱟᱝ ᱠᱟᱹᱢᱤ ᱵᱟᱵᱚᱛ ᱢᱤᱫ ᱞᱟᱹᱭ ᱠᱟᱱᱟ ᱟᱨ ᱜᱤᱫᱽᱨᱟᱹ ᱛᱟᱦᱮᱸᱱ ᱵᱟᱵᱚᱛ: ᱥᱮᱠᱮᱱᱰᱟᱨᱤ ᱛᱟᱦᱮᱸᱱ 51.9% ᱠᱟᱱᱟ᱾

## ᱟᱹᱛᱩ ᱵᱷᱟᱹᱰᱤ ᱠᱟᱹᱢᱤ ᱫᱚ ᱥᱮᱨᱢᱟ ᱛᱟᱞᱟ ᱨᱮ ᱟᱹᱭᱤᱱ ᱮ ᱵᱚᱫᱚᱞ ᱠᱮᱫᱟ

1 ᱡᱩᱞᱟᱭ 2026 ᱠᱷᱚᱱ ᱜᱷᱟᱨᱚᱸᱡᱽ ᱦᱚᱠ ᱫᱚ VB-G RAM G ᱞᱟᱛᱟᱨ 125 ᱫᱤᱱ ᱠᱟᱱᱟ, ᱢᱤᱫ ᱩᱪᱷᱟᱹᱱ ᱵᱷᱟᱹᱰᱤ ᱨᱮ ᱡᱟᱦᱟᱸ ᱚᱠᱟᱨᱮ ᱦᱚᱸ ₹300 ᱞᱟᱛᱟᱨ ᱨᱮ ᱵᱟᱹᱱᱩᱜ-ᱟ ᱟᱨ ᱜᱷᱟᱹᱞ ₹327.4 ᱠᱟᱱᱟ᱾ ᱯᱩᱨᱟᱹᱣ e-KYC ᱠᱟᱹᱢᱤ ᱠᱟᱨᱰ ᱫᱚ ᱱᱟᱣᱟ ᱜᱨᱟᱢᱤᱱ ᱨᱚᱡᱽᱜᱟᱨ ᱜᱮᱭᱟᱨᱮᱱᱴᱤ ᱠᱟᱨᱰ (Gramin Rozgar Guarantee Card) ᱮᱢ ᱦᱩᱭ ᱫᱷᱟᱹᱵᱤᱡ ᱵᱮᱰ ᱛᱟᱦᱮᱸᱱᱟ᱾ ᱢᱚᱱᱛᱨᱟᱞᱚᱭ 11 ᱚᱜᱚᱥᱴ 2026 ᱨᱮ ᱞᱳᱠ ᱥᱚᱵᱷᱟ (Lok Sabha) ᱮ ᱞᱟᱹᱭ ᱠᱮᱫᱟ ᱡᱮ ᱢᱚᱴᱟᱢᱩᱴᱤ ᱯᱩᱭᱞᱩ ᱪᱟᱸᱫᱚ ᱨᱮ 9.69 ᱠᱚᱨᱚᱲ ᱠᱷᱚᱱ ᱵᱟᱹᱲᱛᱤ ᱦᱚᱲᱼᱫᱤᱱ ᱵᱮᱱᱟᱣ ᱟᱠᱟᱱᱟ᱾ ᱚᱱᱟ ᱱᱟᱣᱟ ᱟᱹᱭᱤᱱ ᱞᱟᱛᱟᱨ ᱥᱟᱢᱵᱲᱟᱣ ᱠᱷᱚᱡ ᱠᱟᱱᱟ᱾ ᱱᱚᱣᱟ ᱢᱤᱫ ᱯᱩᱨᱟᱹᱼᱥᱮᱨᱢᱟ ᱜᱷᱟᱨᱚᱸᱡᱽ ᱜᱷᱟᱹᱞ ᱵᱟᱝ ᱠᱟᱱᱟ, ᱟᱨ ᱱᱚᱣᱟ FY 2025-26 MGNREGS ᱜᱩᱴ ᱥᱟᱶ ᱵᱟᱝ ᱡᱚᱲᱟᱣ ᱞᱟᱹᱠᱛᱤ ᱡᱮᱞᱠᱟ ᱢᱤᱫ ᱠᱟᱹᱢᱤᱦᱚᱨᱟ ᱢᱤᱫ ᱰᱮᱯᱷᱤᱱᱤᱥᱚᱱ ᱨᱮ ᱵᱟᱝ ᱛᱤᱸᱜᱩ ᱠᱟᱛᱮ ᱪᱟᱞᱟᱣ ᱞᱮᱱᱟ᱾

ᱢᱮ 2026 ᱱᱳᱴ ᱡᱟᱦᱟᱸ ᱯᱷᱤᱞᱰ ᱨᱚᱞ ᱮ ᱧᱩᱛᱩᱢᱟ — ᱯᱨᱚᱜᱨᱟᱢ ᱚᱯᱷᱤᱥᱚᱨ, ᱢᱮᱴ, ᱟᱨ ᱜᱨᱟᱢ ᱨᱚᱡᱽᱜᱟᱨ ᱥᱮᱵᱚᱠ (Gram Rozgar Sevaks) — ᱚᱱᱟ ᱠᱷᱟᱹᱛᱤᱨ ᱥᱟᱥᱚᱱᱤᱭᱟᱹ ᱠᱷᱚᱨᱚᱪ 6 ᱥᱟᱭᱠᱚᱲᱟ ᱠᱷᱚᱱ 9 ᱥᱟᱭᱠᱚᱲᱟ ᱨᱮ ᱵᱟᱲᱦᱟᱣ ᱮᱱᱟ᱾ ᱚᱱᱟ ᱯᱚᱨᱚᱛ ᱨᱮᱱᱟᱜ ᱦᱩᱱᱟᱹᱨ ᱫᱚ ᱦᱟᱡᱤᱨᱟ ᱛᱟᱹᱞᱠᱟᱹ (muster roll), ᱚᱯᱷᱞᱟᱭᱤᱱ ᱵᱷᱮᱜᱟᱨ ᱥᱟᱶ ᱢᱚᱦᱟᱲ ᱯᱚᱨᱢᱟᱱ (face authentication), ᱟᱨ ᱯᱩᱱ ᱟᱥᱮᱴ ᱛᱷᱤᱢ ᱞᱟᱛᱟᱨ ᱢᱤᱫ ᱜᱨᱟᱢ ᱥᱚᱵᱷᱟ (Gram Sabha) ᱨᱮ ᱵᱟᱪᱷᱟᱣ ᱠᱟᱹᱢᱤ ᱠᱟᱱᱟ᱾

## ᱡᱟᱹᱯᱩᱫ ᱠᱟᱹᱢᱤᱼᱵᱟᱡᱟᱨ ᱫᱚ ᱢᱤᱫ ᱵᱷᱮᱜᱟᱨ ᱥᱤᱨᱤᱡᱽ ᱠᱟᱱᱟ

ᱚᱜᱚᱥᱴ 2026 PLFS ᱪᱟᱸᱫᱚᱠᱤᱭᱟᱹ ᱵᱩᱞᱮᱴᱤᱱ, ᱱᱤᱛᱚᱜᱟᱜ ᱦᱟᱹᱯᱛᱟ ᱚᱵᱚᱥᱛᱟ (current weekly status), 15 ᱥᱮᱨᱢᱟ ᱟᱨ ᱚᱱᱟ ᱠᱷᱚᱱ ᱢᱟᱨᱟᱝ ᱦᱚᱲ, 15 ᱥᱮᱯᱴᱮᱢᱵᱚᱨ 2026 ᱨᱮ 3,70,160 ᱦᱚᱲ ᱨᱮᱱᱟᱜ ᱱᱟᱢᱩᱱᱟ ᱠᱷᱚᱱ ᱩᱪᱷᱟᱹᱱ ᱞᱮᱱᱟ᱾ ᱠᱟᱹᱢᱤᱼᱦᱚᱲ ᱥᱮᱞᱮᱫ 55.6% ᱛᱟᱦᱮᱸ ᱠᱟᱱᱟ᱾ ᱠᱟᱹᱢᱤᱭᱟᱹ ᱦᱚᱲ ᱚᱱᱩᱯᱟᱛ 52.8% ᱛᱟᱦᱮᱸ ᱠᱟᱱᱟ᱾ ᱵᱮᱼᱠᱟᱹᱢᱤ ᱜᱩᱴ ᱨᱮ 5.0%, ᱟᱹᱛᱩ ᱨᱮ 4.1%, ᱟᱨ ᱥᱚᱦᱚᱨ ᱨᱮ 6.8% ᱛᱟᱦᱮᱸ ᱠᱟᱱᱟ᱾ ᱢᱟᱹᱭᱡᱤᱣ ᱥᱮᱞᱮᱫ ᱜᱩᱴ ᱨᱮ 34.8%, ᱟᱹᱛᱩ ᱨᱮ 39.4%, ᱟᱨ ᱥᱚᱦᱚᱨ ᱨᱮ 25.4% ᱛᱟᱦᱮᱸ ᱠᱟᱱᱟ᱾ ᱥᱚᱦᱚᱨ ᱢᱟᱹᱭᱡᱤᱣ ᱫᱚᱨ ᱚᱜᱚᱥᱴ 2025 ᱠᱷᱚᱱ ᱞᱟᱛᱟᱨ ᱨᱮ ᱢᱮᱱᱟᱜ-ᱟ (26.1%)᱾ ᱟᱹᱛᱩ ᱢᱟᱹᱭᱡᱤᱣ ᱥᱮᱞᱮᱫ ᱢᱤᱫ ᱥᱮᱨᱢᱟ ᱞᱟᱦᱟᱨᱮ ᱠᱷᱚᱱ ᱪᱮᱛᱟᱱ ᱨᱮ ᱢᱮᱱᱟᱜ-ᱟ (37.4%)᱾

ᱠᱟᱹᱢᱤᱦᱚᱨᱟ ᱠᱟᱹᱢᱤ ᱵᱟᱲᱦᱟᱣ ᱫᱟᱲᱮᱭᱟᱜ-ᱟ ᱡᱚᱠᱷᱚᱱ ᱡᱟᱹᱛᱤᱭᱟᱹᱨᱤ ᱫᱚᱨ ᱠᱚᱢ ᱜᱮ ᱪᱟᱞᱟᱣᱟ, ᱪᱮᱫᱟᱜ ᱥᱮ PLFS ᱯᱩᱨᱟᱹ ᱠᱟᱹᱣᱰᱤᱼᱵᱮᱵᱚᱥᱛᱟ ᱮ ᱠᱚᱵᱷᱚᱨᱟ ᱟᱨ ᱠᱟᱹᱢᱤᱦᱚᱨᱟ ᱫᱚ ᱧᱩᱛᱩᱢ ᱚᱞ ᱠᱟᱹᱰᱨ ᱟᱨ ᱠᱟᱹᱢᱤ ᱠᱷᱚᱡ ᱜᱷᱟᱨᱚᱸᱡᱽ ᱠᱚ ᱠᱚᱵᱷᱚᱨᱟ᱾ ᱵᱟᱡᱟᱨ ᱞᱟᱹᱜᱤᱫ PLFS ᱵᱮᱵᱷᱟᱨ ᱢᱮ᱾ ᱠᱟᱹᱰᱨ ᱞᱟᱹᱜᱤᱫ ᱴᱨᱮᱠᱚᱨ, UDISE+, ᱟᱨ ᱟᱹᱛᱩ ᱩᱛᱱᱟᱹᱣ ᱛᱮᱞᱟ ᱵᱮᱵᱷᱟᱨ ᱢᱮ᱾

ᱜᱨᱮᱴ ᱤᱱᱰᱤᱭᱟᱱ ᱠᱚᱢᱯᱟᱹᱱᱤ (Great Indian Company) ᱫᱚ ᱠᱟᱹᱰᱨ ᱫᱚ ᱵᱷᱮᱜᱟᱨ ᱫᱚᱦᱚᱭᱟ ᱡᱮ ᱢᱤᱫ ᱠᱟᱹᱢᱤ ᱥᱞᱟᱭᱤᱰ ᱡᱟᱦᱟᱸ ᱥᱟᱨᱣᱮ ᱵᱟᱝ ᱩᱫᱩᱜ ᱟᱠᱟᱫᱟ ᱚᱱᱟ ᱵᱟᱝ ᱤᱫᱤᱭᱟ᱾ ᱚᱱᱟ ᱠᱟᱹᱢᱤᱭᱟᱹ ᱠᱚ ᱡᱟᱦᱟᱸ ᱱᱤᱭᱚᱢ ᱞᱟᱛᱟᱨ ᱨᱮ ᱠᱟᱹᱢᱤ ᱠᱚᱣᱟ ᱚᱱᱟ ᱫᱚ [ᱱᱤᱭᱚᱢ ᱱᱳᱴ](/blog/social-sector-outcomes-regulatory-outlook-ministryofcoal-gov-20260330-460-sat) ᱨᱮ ᱢᱮᱱᱟᱜ-ᱟ᱾ ᱩᱫᱩᱜ ᱫᱚ [greatindiancompany.com/blog](https://greatindiancompany.com/blog) ᱨᱮ ᱢᱮᱱᱟᱜ-ᱟ᱾
