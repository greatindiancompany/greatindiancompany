ᱱᱚᱶᱟ ᱥᱤᱥᱴᱚᱢ ᱞᱟᱹᱜᱤᱫ ᱛᱟᱞᱟ ᱚᱠᱛᱚ ᱨᱮᱱᱟᱜ ᱢᱮᱱ ᱫᱚ ᱥᱮᱨᱢᱟ ᱟᱠᱟᱱ ᱡᱚᱥ ᱧᱮᱞ ᱠᱟᱱᱟ ᱟᱨ ᱠᱩᱠᱞᱤ ᱠᱟᱱᱟ ᱡᱮ ᱱᱟᱣᱟ ᱥᱚᱨᱠᱟᱨᱤ ᱞᱮᱠᱷᱟ ᱪᱮᱫ ᱞᱟᱹᱭᱟ᱾ ᱚᱱᱟ ᱥᱮᱨᱢᱟ ᱢᱩᱫᱽ ᱨᱮ ᱛᱤᱱᱟᱹᱜ ᱱᱤᱛᱦᱚᱸ ᱞᱟᱦᱟ ᱠᱟᱱᱟ᱾ ᱢᱤᱫ ᱜᱚᱨᱚᱡᱽ ᱠᱟᱹᱨᱤᱜᱚᱞ ᱥᱮᱨᱢᱟ ᱫᱚ ᱞᱟᱦᱟ ᱠᱷᱚᱱ ᱯᱟᱨᱚᱢ ᱟᱠᱟᱱᱟ᱾ ᱢᱩᱞ ᱯᱟᱲᱦᱟᱣ ᱥᱮᱨᱢᱟ ᱜᱮ ᱱᱤᱛ ᱪᱟᱞᱟᱣ ᱮᱠᱟᱰᱮᱢᱤᱠ ᱥᱮᱨᱢᱟ ᱠᱟᱱᱟ᱾

## ᱡᱟᱦᱟᱸ ᱡᱚᱥ ᱨᱮᱱᱟᱜ ᱢᱟᱦᱟᱸ ᱥᱮᱴᱮᱨ ᱟᱠᱟᱱᱟ ᱟᱨᱵᱟᱝ ᱱᱚᱶᱟ ᱥᱮᱨᱢᱟ ᱠᱟᱱᱟ

[ᱡᱟᱹᱛᱤᱭᱟᱹᱨᱤ ᱥᱮᱪᱮᱫ ᱱᱤᱛᱤ (National Education Policy)](https://www.education.gov.in/nep/nep2020_final_eng.pdf) ᱞᱟᱹᱭᱟ ᱡᱮ 2025 ᱫᱷᱟᱹᱵᱤᱡ ᱟᱥᱲᱟ ᱟᱨ ᱪᱮᱛᱟᱱ ᱥᱮᱪᱮᱫ ᱥᱤᱥᱴᱚᱢ ᱨᱮᱱ ᱯᱟᱹᱴᱷᱩᱣᱟᱹ ᱢᱩᱫᱽ ᱨᱮ ᱠᱚᱢ ᱠᱷᱚᱱ ᱠᱚᱢ 50% ᱫᱚ ᱠᱟᱹᱨᱤᱜᱚᱞ ᱥᱮᱪᱮᱫ ᱨᱮᱱᱟᱜ ᱧᱮᱞ ᱧᱟᱢᱟ᱾ ᱥᱮᱯᱴᱮᱢᱵᱚᱨ 2026 ᱫᱚ ᱚᱱᱟ ᱢᱟᱦᱟᱸ ᱯᱟᱨᱚᱢ ᱟᱠᱟᱱᱟ᱾ ᱱᱤᱛᱤ ᱫᱚ ᱡᱚᱥ ᱟᱨ ᱴᱟᱭᱤᱢᱞᱟᱭᱤᱱ ᱥᱟᱶ ᱢᱤᱫ ᱮᱠᱥᱚᱱ ᱯᱞᱟᱱ ᱞᱟᱹᱠᱛᱤ ᱞᱮᱫᱟ᱾ ᱱᱚᱶᱟ ᱵᱨᱤᱯᱷ ᱫᱚ, ᱱᱚᱸᱰᱮ ᱵᱮᱵᱷᱟᱨ ᱟᱠᱟᱱ ᱩᱪᱷᱟᱹᱱ ᱨᱮ, ᱚᱱᱟ ᱥᱚᱨᱠᱟᱨᱤ ᱞᱮᱠᱷᱟ ᱵᱟᱝ ᱧᱟᱢ ᱟᱠᱟᱫᱟ ᱡᱟᱦᱟᱸ ᱞᱟᱹᱭᱟ ᱡᱮ 50% ᱧᱮᱞ ᱪᱤᱱᱦᱟᱹ ᱯᱩᱨᱟᱹᱣ ᱮᱱᱟ ᱟᱨᱵᱟᱝ ᱵᱟᱹᱲᱛᱤ ᱮᱱᱟ᱾ ᱡᱟᱦᱟᱸ ᱢᱮᱱᱟᱜᱼᱟ ᱚᱱᱟ ᱫᱚ ᱮᱴᱟᱜ ᱩᱫᱩᱜ ᱠᱟᱱᱟ: ᱦᱩᱱᱟᱹᱨ ᱢᱚᱱᱛᱨᱟᱞᱚᱭ ᱨᱮᱱᱟᱜ ᱟᱹᱭᱟᱹᱛ ᱡᱮ 15–29 ᱥᱮᱨᱢᱟ ᱩᱢᱮᱨᱤᱭᱟᱹ ᱛᱟᱨᱩᱱ ᱢᱩᱫᱽ ᱨᱮ ᱠᱟᱹᱨᱤᱜᱚᱞ ᱥᱮᱪᱮᱫ ᱧᱟᱢ ᱦᱟᱹᱴᱤᱧ 2017-18 ᱨᱮ 7.1% ᱠᱷᱚᱱ 2023-24 ᱨᱮ 26.1% ᱛᱮ ᱵᱟᱲᱦᱟᱣ ᱟᱠᱟᱱᱟ᱾ ᱱᱚᱶᱟ ᱫᱚ ᱛᱟᱨᱩᱱ ᱦᱚᱲ ᱢᱩᱫᱽ ᱨᱮ ᱠᱟᱹᱨᱤᱜᱚᱞ ᱥᱮᱴᱮᱨ ᱨᱮᱱᱟᱜ ᱩᱪᱟᱹᱲ ᱞᱮᱠᱟ ᱯᱟᱲᱦᱟᱣ ᱜᱟᱱᱟ᱾ ᱱᱚᱶᱟ ᱫᱚ ᱧᱮᱞ ᱡᱚᱥ ᱥᱟᱢᱟᱝ ᱨᱮ ᱱᱚᱢᱵᱚᱨ ᱞᱮᱠᱟ ᱵᱟᱝ ᱯᱟᱲᱦᱟᱣ ᱜᱟᱱᱟ, ᱪᱮᱫᱟᱜ ᱥᱮ ᱦᱚᱲᱮᱞ ᱵᱷᱮᱜᱟᱨᱟ᱾ ᱥᱟᱹᱨᱤ ᱛᱟᱞᱟ ᱚᱠᱛᱚ ᱟᱹᱭᱟᱹᱛ ᱫᱚ ᱱᱚᱶᱟ ᱠᱟᱱᱟ ᱡᱮ ᱡᱚᱥ ᱢᱟᱦᱟᱸ ᱯᱟᱨᱚᱢ ᱟᱠᱟᱱᱟ ᱟᱨ ᱢᱮᱥᱟ ᱩᱫᱩᱜ ᱫᱚ ᱱᱚᱶᱟ ᱥᱨᱚᱛ ᱨᱮ ᱵᱟᱝ ᱪᱷᱟᱯᱟ ᱟᱠᱟᱱᱟ᱾

ᱢᱩᱞ ᱥᱟᱶᱦᱮᱫ ᱫᱚ ᱮᱴᱟᱜ ᱜᱷᱚᱲᱤ ᱨᱮ ᱠᱟᱱᱟ᱾ ᱱᱤᱛᱤ ᱚᱞ ᱞᱟᱹᱭᱟ ᱡᱮ ᱡᱚᱛᱚ ᱯᱟᱹᱴᱷᱩᱣᱟᱹ ᱫᱚ ᱠᱞᱟᱥ 3 ᱫᱷᱟᱹᱵᱤᱡ ᱱᱚᱶᱟ ᱧᱟᱢᱟ, ᱟᱨ 2025 ᱨᱮᱱᱟᱜ ᱯᱨᱟᱭᱢᱟᱨᱤ ᱟᱥᱲᱟ ᱥᱤᱢᱟᱹ ᱫᱚᱦᱚᱭᱟ᱾ ᱱᱤᱯᱩᱱ ᱵᱷᱟᱨᱚᱛ (NIPUN Bharat), 5 ᱡᱩᱞᱟᱭ 2021 ᱨᱮ ᱮᱛᱚᱦᱚᱵ, ᱫᱚ 2026-27 ᱨᱮ ᱠᱞᱟᱥ 3 ᱢᱩᱪᱟᱹᱫ ᱫᱷᱟᱹᱵᱤᱡ ᱡᱚᱛᱚ ᱢᱩᱞ ᱥᱟᱶᱦᱮᱫ ᱟᱨ ᱮᱞᱠᱷᱟ ᱩᱫᱽᱫᱮᱥᱭᱚᱭᱟ᱾ ᱮᱠᱟᱰᱮᱢᱤᱠ ᱥᱮᱨᱢᱟ 2026-27 ᱜᱮ ᱚᱱᱟ ᱢᱤᱥᱚᱱ ᱥᱮᱨᱢᱟ ᱠᱟᱱᱟ᱾ UDISE+ ᱫᱚ ᱞᱟᱹᱭ ᱫᱟᱲᱮᱭᱟᱢ ᱡᱮ ᱢᱟᱪᱮᱛ ᱫᱚ ᱱᱤᱛ ᱢᱤᱫ ᱠᱚᱨᱚᱲ ᱯᱟᱨᱚᱢ ᱟᱠᱟᱫᱟ ᱟᱨ ᱢᱩᱞ ᱯᱟᱹᱴᱷᱩᱣᱟᱹ–ᱢᱟᱪᱮᱛ ᱟᱹᱨᱤ 10 ᱠᱟᱱᱟ᱾ ᱱᱚᱶᱟ ᱫᱚ, 28 ᱚᱜᱚᱥᱴ 2025 ᱱᱚᱴ ᱨᱮ, ᱠᱞᱟᱥ 3 ᱜᱤᱫᱽᱨᱟᱹ ᱨᱮᱱᱟᱜ ᱚᱱᱟ ᱦᱟᱹᱴᱤᱧ ᱵᱟᱝ ᱨᱤᱯᱚᱨᱴᱟ ᱡᱟᱦᱟᱸᱭ ᱢᱤᱥᱚᱱ ᱨᱮᱱᱟᱜ ᱞᱚᱠᱷᱭᱚ (Lakshyas) ᱞᱟᱹᱭᱟᱜ ᱮᱞᱠᱷᱟ ᱠᱟᱹᱢᱤ ᱟᱨ ᱯᱟᱲᱦᱟᱣ ᱫᱟᱲᱮᱭᱟᱜ ᱠᱚᱣᱟ᱾ ᱵᱷᱩᱨᱛᱤ ᱤᱱᱯᱷᱨᱟᱥᱴᱨᱚᱠᱪᱚᱨ ᱟᱨ ᱯᱟᱲᱦᱟᱣ ᱫᱚ ᱵᱷᱮᱜᱟᱨ ᱦᱚᱨ ᱨᱮ ᱢᱮᱱᱟᱜᱼᱟ᱾ ᱢᱤᱫ ᱫᱟᱵᱤ ᱡᱮ ᱢᱤᱥᱚᱱ ᱥᱟᱹᱨᱤ ᱮᱱᱟ ᱪᱮᱫᱟᱜ ᱥᱮ ᱛᱮᱭᱟᱨᱤ ᱛᱷᱚᱠ ᱵᱟᱹᱜᱤ 2.3% ᱛᱮ ᱠᱚᱢ ᱮᱱᱟ, ᱫᱚ ᱢᱤᱫ ᱫᱚᱦᱚ ᱞᱮᱠᱷᱟ ᱫᱚ ᱯᱟᱲᱦᱟᱣ ᱞᱮᱠᱷᱟ ᱞᱮᱠᱟ ᱵᱮᱵᱷᱟᱨ ᱦᱩᱭᱩᱜᱼᱟ᱾

PM-USHA ᱨᱮᱱᱟᱜ ᱞᱟᱹᱭ ᱟᱠᱟᱱ ᱟᱭᱢᱟ ᱥᱮᱨᱢᱟ ᱠᱷᱚᱨᱪᱟ, ᱞᱟᱦᱟ RUSA ᱫᱟᱭᱤᱛᱣᱚ ᱥᱟᱶ ₹12,926.10 crore, ᱫᱚ 2023-24 ᱠᱷᱚᱱ 2025-26 ᱫᱷᱟᱹᱵᱤᱡ ᱮᱥᱮᱫ ᱞᱮᱫᱟ᱾ ᱚᱱᱟ ᱚᱠᱛᱚ ᱵᱚᱸᱫᱽ ᱟᱠᱟᱱᱟ᱾ ᱪᱮᱛᱟᱱ ᱥᱮᱪᱮᱫ ᱵᱤᱵᱷᱟᱜᱽ (Department of Higher Education) ᱨᱮᱱᱟᱜ 2026-27 ᱛᱟᱞᱢᱟ ᱜᱚᱲᱚ ᱵᱮᱵᱚᱥᱛᱟ ₹1,850 crore ᱠᱟᱱᱟ, ₹800 crore ᱨᱮᱱᱟᱜ 2025-26 ᱵᱚᱫᱚᱞ ᱟᱱᱫᱟᱡᱽ ᱛᱟᱭᱚᱢ᱾ ᱢᱤᱫ ᱩᱛᱛᱚᱨᱟᱫᱷᱤᱠᱟᱨᱤ ᱡᱚᱡᱚᱱᱟ ᱫᱚ ᱚᱱᱟ ᱰᱤᱡᱟᱭᱤᱱ ᱤᱫᱤᱭᱟ ᱥᱮ ᱵᱟᱝ, ᱚᱱᱟ ᱫᱚ ᱛᱟᱭᱚᱢ ᱱᱚᱴᱤᱥ ᱞᱟᱹᱜᱤᱫ ᱠᱩᱠᱞᱤ ᱠᱟᱱᱟ, ᱵᱚᱸᱫᱽ ᱚᱠᱛᱚ ᱡᱚᱵᱟᱵᱽ ᱵᱟᱝ ᱮᱢᱟ᱾

## ᱡᱟᱦᱟᱸ ᱡᱚᱥ ᱱᱤᱛᱦᱚᱸ ᱢᱤᱫ ᱫᱟᱥᱚᱠ ᱫᱩᱨ ᱠᱟᱱᱟ

ᱱᱤᱛᱤ ᱫᱚ 2030 ᱫᱷᱟᱹᱵᱤᱡ ᱯᱨᱤᱥᱠᱩᱞ ᱠᱷᱚᱱ ᱢᱟᱫᱷᱭᱚᱢ ᱫᱷᱟᱹᱵᱤᱡ 100% ᱜᱩᱴ ᱵᱷᱩᱨᱛᱤ ᱠᱷᱚᱡᱟ᱾ UDISE+ 2024-25 ᱫᱚ ᱛᱟᱞᱟ ᱛᱷᱚᱠ ᱜᱩᱴ ᱵᱷᱩᱨᱛᱤ 90.3% ᱟᱨ ᱢᱟᱫᱷᱭᱚᱢ ᱛᱷᱚᱠ ᱜᱩᱴ ᱵᱷᱩᱨᱛᱤ 68.5% ᱫᱚᱦᱚᱭᱟ᱾ ᱢᱟᱫᱷᱭᱚᱢ ᱛᱷᱚᱠ ᱫᱚᱦᱚ 47.2% ᱠᱟᱱᱟ᱾ ᱢᱚᱬᱮ ᱥᱮᱨᱢᱟ ᱫᱚ ᱚᱱᱟ ᱟᱹᱨᱤ ᱡᱤᱞᱤᱧ ᱩᱪᱟᱹᱲ ᱞᱟᱹᱜᱤᱫ ᱠᱟᱹᱢᱤᱭᱟ, ᱟᱨ 68.5% ᱵᱷᱩᱨᱛᱤ ᱟᱨᱵᱟᱝ 47.2% ᱫᱚᱦᱚ ᱫᱚ ᱡᱚᱛᱚ ᱯᱩᱨᱟᱹᱣ ᱥᱩᱨ ᱢᱮᱱᱛᱮ ᱞᱟᱹᱭ ᱞᱟᱹᱜᱤᱫ ᱠᱟᱹᱢᱤ ᱵᱟᱝ᱾ ᱛᱟᱞᱟ ᱠᱷᱚᱱ ᱢᱟᱫᱷᱭᱚᱢ ᱛᱮ ᱩᱪᱟᱹᱲ ᱦᱟᱨ, 86.6%, ᱜᱮ ᱠᱟᱸᱪᱤ ᱠᱟᱱᱟ᱾ ᱡᱩᱫᱤ ᱚᱱᱟ ᱦᱟᱨ ᱛᱷᱩᱠᱩᱜᱼᱟ, 2030 ᱡᱚᱥ ᱫᱚ ᱢᱩᱪᱟᱹᱫ ᱟᱥᱲᱟ ᱥᱮᱨᱢᱟ ᱨᱮ ᱵᱷᱩᱞᱟ, ᱮᱛᱚᱦᱚᱵ ᱠᱞᱟᱥ ᱡᱚᱛᱚ ᱥᱩᱨ ᱛᱟᱦᱮᱸ ᱨᱮᱦᱚᱸ᱾ ᱛᱮᱭᱟᱨᱤ ᱵᱟᱹᱜᱤ ᱫᱚ ᱞᱟᱦᱟ ᱠᱷᱚᱱ 2.3% ᱛᱮ ᱞᱟᱛᱟᱨ ᱠᱟᱱᱟ᱾ ᱦᱚᱲᱮᱞ ᱞᱮᱠᱷᱟ ᱡᱟᱦᱟᱸ ᱛᱟᱦᱮᱸ ᱟᱠᱟᱱ ᱡᱟᱹᱛᱤᱭᱟᱹᱨᱤ ᱵᱷᱩᱞ ᱩᱫᱩᱜᱟ ᱚᱱᱟ ᱫᱚ ᱛᱟᱭᱚᱢ ᱠᱟᱱᱟ᱾

ᱪᱮᱛᱟᱱ ᱥᱮᱪᱮᱫ ᱜᱮ ᱡᱤᱞᱤᱧ ᱡᱚᱥ ᱠᱟᱱᱟ: 2035 ᱫᱷᱟᱹᱵᱤᱡ 50% ᱜᱩᱴ ᱵᱷᱩᱨᱛᱤ, ᱠᱟᱹᱨᱤᱜᱚᱞ ᱥᱮᱪᱮᱫ ᱥᱟᱶᱛᱮ, 2018 ᱨᱮ 26.3% ᱠᱷᱚᱱ᱾ ᱥᱤᱧᱚᱛ ᱪᱮᱛᱟᱱ ᱥᱮᱪᱮᱫ ᱥᱟᱨᱵᱷᱮ (All India Survey on Higher Education, AISHE) 2023-24 ᱫᱚ 18–23 ᱥᱮᱨᱢᱟ ᱞᱟᱹᱜᱤᱫ 30 ᱨᱤᱯᱚᱨᱴᱟ᱾ ᱮᱞᱟᱝ ᱢᱤᱫ ᱫᱟᱥᱚᱠ ᱨᱮ 30 ᱠᱷᱚᱱ 50 ᱫᱷᱟᱹᱵᱤᱡ ᱫᱚ ᱢᱤᱫ ᱢᱟᱨᱟᱝ ᱫᱟᱲᱮ ᱠᱟᱹᱢᱤ ᱠᱟᱱᱟ᱾ ᱱᱤᱛᱤ ᱞᱟᱹᱭᱟ ᱡᱮ ᱡᱟᱹᱥᱛᱤ ᱫᱚ ᱛᱟᱦᱮᱸ ᱟᱠᱟᱱ ᱥᱚᱝᱥᱛᱷᱟ ᱯᱟᱥᱱᱟᱣ ᱠᱷᱚᱱ ᱦᱤᱡᱩᱜ ᱞᱟᱹᱠᱛᱤᱭᱟ, ᱟᱨ 2030 ᱫᱷᱟᱹᱵᱤᱡ ᱡᱚᱛᱚ ᱦᱚᱱᱚᱛ ᱨᱮ ᱟᱨᱵᱟᱝ ᱥᱩᱨ ᱨᱮ ᱠᱚᱢ ᱠᱷᱚᱱ ᱠᱚᱢ ᱢᱤᱫ ᱢᱟᱨᱟᱝ ᱟᱭᱢᱟ ᱵᱤᱥᱚᱭ ᱥᱚᱝᱥᱛᱷᱟ ᱛᱟᱦᱮᱸ ᱞᱟᱹᱠᱛᱤᱭᱟ᱾ ᱡᱩᱞᱟᱭ 2026 ᱩᱪᱷᱟᱹᱱ ᱫᱚ ᱦᱚᱱᱚᱛ ᱡᱚᱥ ᱵᱟᱝ ᱱᱚᱢᱵᱚᱨᱟ᱾ ᱱᱚᱶᱟ ᱫᱚ ᱩᱫᱩᱜᱟ ᱡᱮ ᱵᱷᱩᱨᱛᱤ 4.50 crore ᱥᱮᱴᱮᱨ ᱮᱱᱟ, 2014-15 ᱨᱮ 3.42 crore ᱠᱷᱚᱱ, ᱢᱟᱭᱡᱤᱣ ᱵᱷᱩᱨᱛᱤ 2.24 crore ᱥᱟᱶᱛᱮ᱾ ᱦᱚᱨ ᱫᱚ ᱪᱮᱛᱟᱱ ᱥᱮᱫ ᱠᱟᱱᱟ᱾ ᱛᱷᱚᱠ ᱫᱚ ᱱᱤᱛᱦᱚᱸ ᱩᱫᱽᱫᱮᱥᱭᱚ ᱠᱷᱚᱱ ᱵᱟᱨᱜᱮᱞ ᱟᱹᱨᱤ ᱯᱚᱭᱮᱱᱴ ᱠᱚᱢ ᱠᱟᱱᱟ, ᱢᱮᱱ ᱨᱮᱱᱟᱜ ᱵᱷᱮᱜᱟᱨ ᱛᱚᱨᱠᱟ ᱞᱟᱦᱟ ᱜᱮ᱾

ᱡᱚᱛᱚ ᱟᱹᱨᱤ 30 ᱥᱟᱢᱟᱝ ᱨᱮ ᱚᱱᱩᱥᱩᱪᱤᱛ ᱡᱟᱹᱛ (Scheduled Tribe) ᱜᱩᱴ ᱵᱷᱩᱨᱛᱤ 22.8 ᱜᱮ ᱚᱱᱟ ᱡᱚᱥ ᱵᱷᱤᱛᱨᱤ ᱥᱚᱢᱟᱱ ᱦᱚᱨ ᱠᱟᱱᱟ᱾ 30 ᱠᱷᱚᱱ 50 ᱛᱮ ᱢᱤᱫ ᱡᱟᱹᱛᱤᱭᱟᱹᱨᱤ ᱩᱪᱟᱹᱲ ᱡᱟᱦᱟᱸ ᱡᱟᱹᱛ ᱟᱹᱨᱤ ᱯᱟᱹᱪᱷᱞᱟᱹ ᱫᱚᱦᱚᱭᱟ, ᱫᱚ ᱢᱤᱫ ᱞᱮᱠᱷᱟ ᱯᱩᱨᱟᱹᱣᱟ ᱟᱨ ᱱᱤᱛᱤ ᱨᱮᱱᱟᱜ ᱥᱮᱞᱮᱫ ᱯᱟᱹᱨᱥᱤ ᱵᱟᱹᱲᱛᱤᱭᱟ᱾ ᱚᱱᱩᱥᱩᱪᱤᱛ ᱡᱟᱹᱛᱤ (Scheduled Caste) ᱜᱩᱴ ᱵᱷᱩᱨᱛᱤ 27.8 ᱫᱚ ᱛᱟᱞᱟ ᱥᱩᱨ ᱠᱟᱱᱟ ᱟᱨ ᱱᱤᱛᱦᱚᱸ ᱚᱱᱟ ᱞᱟᱛᱟᱨ ᱠᱟᱱᱟ᱾

## ᱠᱷᱚᱨᱪᱟ ᱩᱫᱽᱫᱮᱥᱭᱚ ᱨᱮ ᱱᱚᱸᱰᱮ ᱱᱟᱣᱟ ᱥᱚᱨᱠᱟᱨᱤ ᱟᱹᱨᱤ ᱵᱟᱹᱱᱩᱜᱼᱟ

ᱱᱤᱛᱤ ᱫᱚ GDP ᱨᱮᱱᱟᱜ 6% ᱥᱚᱨᱠᱟᱨᱤ ᱠᱷᱚᱨᱪᱟ ᱟᱨᱦᱚᱸ ᱢᱟᱱᱟᱣᱟ ᱟᱨ 2017-18 ᱨᱮ ᱮᱞᱟᱝ 4.43% ᱚᱞᱟ᱾ ᱱᱚᱶᱟ ᱥᱟᱦᱴᱟ ᱨᱮ ᱛᱟᱭᱚᱢ ᱥᱚᱨᱠᱟᱨᱤ ᱟᱹᱨᱤ ᱵᱟᱝ ᱚᱞ ᱟᱠᱟᱱᱟ᱾ ᱛᱟᱞᱟ ᱚᱠᱛᱚ ᱨᱮᱱᱟᱜ ᱯᱷᱤᱥᱠᱟᱞ ᱥᱟᱹᱨᱤ ᱡᱟᱦᱟᱸ ᱞᱟᱹᱭ ᱜᱟᱱᱟ ᱚᱱᱟ ᱫᱚ ᱡᱚᱡᱚᱱᱟ ᱵᱚᱫᱚᱞ ᱵᱟᱵᱚᱛ ᱠᱟᱱᱟ, GDP ᱦᱟᱹᱴᱤᱧ ᱵᱟᱵᱚᱛ ᱵᱟᱝ᱾ 2025-26 ᱨᱮ, ᱡᱚᱛᱚᱠᱷᱚᱸᱰ ᱥᱮᱪᱮᱫ (Samagra Shiksha), PM SHRI, ᱟᱨ ᱦᱩᱱᱟᱹᱨ ᱢᱚᱱᱛᱨᱟᱞᱚᱭ ᱨᱮᱱᱟᱜ ᱜᱚᱲᱚ ᱡᱚᱡᱚᱱᱟ ᱫᱚ ᱡᱚᱛᱚ ᱟᱡᱼᱟᱡ ᱵᱟᱡᱮᱴ ᱟᱱᱫᱟᱡᱽ ᱞᱟᱛᱟᱨ ᱵᱚᱫᱚᱞ ᱮᱱᱟ, ᱟᱨ 2026-27 ᱟᱱᱫᱟᱡᱽ ᱫᱚ ᱢᱟᱨᱟᱝ ᱞᱮᱠᱷᱟ ᱨᱩᱣᱟᱹᱲ ᱫᱚᱦᱚᱭᱟ, PM-SETU ᱞᱟᱹᱜᱤᱫ ₹6,140.50 crore ᱥᱟᱶᱛᱮ᱾ ᱠᱷᱟᱹᱞᱤ ᱵᱟᱡᱮᱴ ᱯᱟᱹᱨᱥᱤ ᱠᱷᱚᱱ ᱟᱹᱭᱤᱡ ᱢᱤᱫ ᱦᱚᱨ ᱫᱚ ᱵᱚᱫᱚᱞ ᱟᱱᱫᱟᱡᱽ ᱨᱮᱠᱚᱨᱰ ᱠᱷᱚᱱ ᱰᱷᱮᱨ ᱥᱚᱢᱟᱱ ᱧᱮᱞᱚᱜᱼᱟ᱾

ᱛᱟᱭᱚᱢ ᱟᱭᱢᱟ ᱥᱮᱨᱢᱟ ᱞᱟᱹᱜᱤᱫ ᱠᱟᱹᱢᱤᱨᱮ ᱞᱟᱜᱟᱣ ᱪᱤᱛᱟᱹᱨ ᱫᱚ ᱚᱱᱟᱛᱮ ᱩᱫᱽᱫᱮᱥᱭᱚ ᱛᱮ ᱵᱷᱮᱜᱟᱨ ᱠᱟᱱᱟ᱾ ᱠᱟᱹᱨᱤᱜᱚᱞ ᱧᱮᱞ ᱨᱮᱱᱟᱜ ᱢᱟᱦᱟᱸ ᱯᱟᱨᱚᱢ ᱟᱠᱟᱱᱟ ᱟᱨ ᱱᱚᱶᱟ ᱥᱨᱚᱛ ᱨᱮ ᱪᱷᱟᱯᱟ ᱱᱚᱢᱵᱚᱨ ᱵᱟᱹᱱᱩᱜᱼᱟ᱾ ᱢᱩᱞ ᱯᱟᱲᱦᱟᱣ ᱨᱮᱱᱟᱜ ᱢᱤᱥᱚᱱ ᱥᱮᱨᱢᱟ ᱫᱚ ᱱᱤᱛᱚᱜᱟᱜ ᱠᱟᱱᱟ, ᱟᱨ ᱢᱤᱫ ᱦᱚᱲᱮᱞ ᱞᱮᱠᱷᱟ ᱫᱚ ᱱᱤᱛᱦᱚᱸ ᱚᱱᱟ ᱨᱮᱱᱟᱜ ᱡᱚᱵᱟᱵᱽ ᱵᱟᱝ ᱮᱢᱟ᱾ ᱟᱥᱲᱟ ᱵᱷᱩᱨᱛᱤ ᱫᱚ ᱮᱛᱚᱦᱚᱵ ᱟᱨ ᱛᱟᱞᱟ ᱛᱷᱚᱠ ᱨᱮ 2030 ᱡᱚᱥ ᱥᱩᱨ ᱥᱮᱴᱮᱨ ᱫᱟᱲᱮᱭᱟᱜᱼᱟ ᱟᱨ ᱱᱤᱛᱦᱚᱸ ᱢᱟᱫᱷᱭᱚᱢ ᱫᱚᱦᱚ ᱨᱮ ᱵᱷᱩᱞ ᱫᱟᱲᱮᱭᱟᱜᱼᱟ᱾ ᱪᱮᱛᱟᱱ ᱥᱮᱪᱮᱫ ᱫᱚ ᱵᱷᱩᱨᱛᱤ ᱡᱚᱲᱟᱣ ᱫᱟᱲᱮᱭᱟᱜᱼᱟ, 4.50 crore ᱫᱷᱟᱹᱵᱤᱡ ᱞᱮᱠᱟ, ᱟᱨ 2035 ᱨᱮ 50% ᱠᱷᱚᱱ ᱫᱩᱨ ᱛᱟᱦᱮᱸ ᱫᱟᱲᱮᱭᱟᱜᱼᱟ ᱡᱩᱫᱤ ᱥᱮᱨᱢᱟᱠᱤᱭᱟᱹ ᱵᱟᱲᱦᱟᱣ ᱫᱚ 29.5 ᱠᱷᱚᱱ 30 ᱨᱮᱱᱟᱜ ᱱᱟᱣᱟ ᱜᱚᱛᱤ ᱥᱩᱨ ᱛᱟᱦᱮᱸᱱᱟ᱾ ᱚᱱᱟ ᱫᱚ ᱵᱷᱮᱜᱟᱨ ᱫᱩᱨ ᱠᱟᱱᱟ᱾ ᱩᱱᱠᱩ ᱫᱚ ᱢᱤᱫ “ᱥᱤᱧᱚᱛ ᱦᱚᱨ ᱨᱮ ᱢᱮᱱᱟᱜᱼᱟ” ᱟᱹᱭᱟᱹᱛ ᱨᱮ ᱛᱟᱞᱟ ᱵᱟᱝ ᱦᱟᱛᱟᱣ ᱦᱩᱭ ᱞᱟᱹᱠᱛᱤᱭᱟ᱾
