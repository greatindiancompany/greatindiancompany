# ᱥᱤᱧᱚᱛ ᱨᱮᱱᱟᱜ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱵᱮᱱᱟᱣ ᱟᱨ ᱵᱮᱯᱟᱨᱤ ᱚᱠᱟᱥ ᱱᱤᱛᱤ ᱨᱮᱱᱟᱜ ᱢᱤᱫ ᱱᱤᱛᱤ-ᱱᱚᱠᱥᱟ

## ᱦᱟᱹᱛᱤᱭᱟᱹᱨ ᱵᱨᱤᱯᱷ ᱢᱮ, ᱱᱟᱨᱟ ᱵᱟᱝ

«Atmanirbhar defence» ᱢᱤᱫ ᱠᱟᱹᱢᱤᱦᱚᱨᱟ-ᱧᱩᱛᱩᱢ ᱠᱟᱱᱟ᱾ ᱠᱟᱹᱢᱤᱨᱮᱱ ᱚᱞ ᱟᱹᱰᱤ ᱠᱷᱟᱴᱚ ᱠᱟᱱᱟ, ᱟᱨ ᱩᱱᱠᱩ ᱵᱷᱮᱜᱟᱨ ᱵᱤᱵᱷᱟᱜᱽ ᱨᱮ ᱛᱟᱦᱮᱸᱱᱟ᱾ ᱡᱟᱦᱟᱸ ᱱᱤᱛᱤ-ᱪᱤᱠᱤ ᱩᱱᱠᱩ ᱢᱮᱥᱟᱭᱟ, ᱚᱱᱟ ᱵᱷᱩᱞ ᱫᱚᱯᱷᱛᱚᱨ ᱴᱷᱮᱱ ᱵᱷᱩᱞ ᱰᱤᱜᱟᱹᱣ ᱮᱢᱟ᱾ ᱞᱟᱛᱟᱨ ᱨᱮᱱᱟᱜ ᱱᱚᱠᱥᱟ ᱩᱪᱷᱟᱹᱱ ᱡᱟᱦᱟᱸ ᱞᱮᱠᱟ ᱵᱟᱰᱟᱭᱟ, ᱠᱟᱹᱢᱤ ᱨᱮᱱᱟᱜ ᱚᱱᱟ ᱫᱮᱠᱷᱟᱣ ᱦᱟᱹᱴᱤᱧ ᱠᱟᱱᱟ᱾

## ᱠᱤᱨᱤᱧᱤᱭᱟᱹ ᱫᱚ ᱟᱹᱜᱩ ᱛᱤᱝᱜᱩ ᱞᱟᱹᱜᱤᱫ ᱚᱠᱚᱭ ᱢᱮᱱ ᱟᱭᱟ

ᱯᱚᱡᱤᱴᱤᱵᱷ ᱤᱱᱰᱤᱡᱮᱱᱟᱭᱡᱮᱥᱚᱱ ᱞᱤᱥᱴ (positive indigenisation lists) ᱵᱟᱨᱭᱟ ᱡᱟᱭᱜᱟ ᱠᱷᱚᱱ ᱦᱤᱡᱩᱜ-ᱟ᱾ ᱢᱮ 2026 ᱦᱟᱹᱵᱤᱡ ᱨᱮᱱᱟᱜ ᱢᱤᱫ ᱯᱨᱮᱥ ᱤᱱᱯᱷᱚᱨᱢᱮᱥᱚᱱ ᱵᱭᱩᱨᱚ (Press Information Bureau, PIB) ᱯᱷᱮᱠᱴᱥᱤᱴ ᱯᱷᱟᱹᱫᱽ ᱠᱟᱹᱢᱤ ᱵᱤᱵᱷᱟᱜᱽ (Department of Military Affairs) ᱠᱷᱚᱱ ᱢᱚᱬᱮ ᱞᱤᱥᱴ ᱟᱨ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱩᱛᱱᱟᱹᱣ ᱵᱤᱵᱷᱟᱜᱽ (Department of Defence Production) ᱠᱷᱚᱱ ᱢᱚᱬᱮ ᱞᱤᱥᱴ ᱞᱮᱠᱷᱟᱭᱟ, ᱡᱟᱦᱟᱸ ᱢᱤᱫ ᱥᱟᱶ 5,521 ᱡᱤᱱᱤᱥ ᱥᱟᱸᱵᱽᱲᱟᱣᱟ᱾ 18 ᱚᱜᱚᱥᱴ 2026 ᱨᱮ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱩᱛᱱᱟᱹᱣ ᱵᱤᱵᱷᱟᱜᱽ 405 ᱡᱤᱱᱤᱥ ᱨᱮᱱᱟᱜ ᱛᱩᱨᱩᱭᱟᱜ ᱞᱤᱥᱴ ᱩᱪᱷᱟᱹᱱ ᱞᱮᱫ-ᱟ᱾ ᱯᱷᱟᱹᱫᱽ-ᱪᱟᱞᱟᱣ ᱞᱤᱥᱴ ᱟᱨ ᱩᱛᱱᱟᱹᱣ-ᱵᱤᱵᱷᱟᱜᱽ ᱞᱤᱥᱴ ᱧᱩᱛᱩᱢ ᱥᱟᱶ ᱛᱟᱦᱮᱸᱱ ᱢᱟ᱾ ᱚᱜᱚᱥᱴ ᱞᱤᱥᱴ ᱨᱮᱱᱟᱜ ᱚᱨᱡᱚ ᱫᱚ ᱥᱤᱫᱷ ᱩᱛᱱᱟᱹᱣ ᱛᱟᱭᱚᱢ ᱥᱤᱧᱚᱛᱤᱭᱟᱹ ᱩᱫᱽᱭᱚᱜ ᱠᱷᱚᱱ ᱠᱤᱨᱤᱧ ᱠᱟᱱᱟ, Make ᱦᱚᱨ (Make procedures) ᱟᱨ ᱚᱲᱟᱜ ᱵᱷᱤᱛᱨᱤ ᱠᱟᱹᱢᱤ ᱛᱮ, MSME ᱥᱮᱞᱮᱫ ᱛᱮ᱾

SRIJAN ᱮᱢ-ᱯᱞᱮᱴᱯᱷᱚᱨᱢ ᱠᱟᱱᱟ, ᱱᱤᱛᱤ-ᱱᱤᱭᱚᱢ ᱵᱟᱝ᱾ ᱱᱚᱶᱟ ᱚᱜᱚᱥᱴ 2020 ᱠᱷᱚᱱ ᱪᱟᱹᱞᱩ ᱢᱮᱱᱟᱜ-ᱟ᱾ «ᱮᱢ ᱟᱠᱟᱱ ᱡᱤᱱᱤᱥ», «ᱩᱪᱷᱟᱹᱱ ᱟᱠᱟᱱ ᱡᱤᱱᱤᱥ» ᱟᱨ «ᱫᱤᱥᱚᱢ ᱵᱷᱤᱛᱨᱤ ᱵᱮᱱᱟᱣ ᱟᱠᱟᱱ ᱡᱤᱱᱤᱥ» ᱢᱮᱥᱟ ᱞᱮᱠᱷᱟᱱ ᱞᱟᱦᱟᱸᱜ ᱰᱷᱮᱨ ᱢᱮᱱᱚᱜ-ᱟ᱾ ᱚᱜᱚᱥᱴ ᱩᱪᱷᱟᱹᱱ ᱟᱡᱛᱮ 33,000 ᱠᱷᱚᱱ ᱰᱷᱮᱨ ᱮᱢ ᱟᱠᱟᱱ ᱡᱤᱱᱤᱥ ᱫᱚ 15,700 ᱠᱷᱚᱱ ᱰᱷᱮᱨ ᱫᱤᱥᱚᱢ ᱵᱷᱤᱛᱨᱤ ᱵᱮᱱᱟᱣ ᱟᱠᱟᱱ ᱡᱤᱱᱤᱥ ᱠᱷᱚᱱ ᱵᱷᱮᱜᱟᱨ ᱫᱚᱦᱚᱭᱟ᱾

## ᱢᱟᱹᱞᱤᱠᱟᱱᱟ ᱚᱠᱚᱭ ᱫᱚᱦᱚᱭᱟ ᱟᱨ ᱠᱤᱨᱤᱧ-ᱱᱤᱛᱤ ᱚᱠᱚᱭ ᱫᱚᱦᱚᱭᱟ

DPIIT ᱨᱮᱱᱟᱜ Press Note 4 of 2020 ᱵᱟᱦᱨᱮ ᱮᱠᱣᱤᱴᱤ ᱫᱚᱦᱚᱭᱟ: 100 ᱥᱟᱭᱠᱚᱲᱟ ᱥᱤᱢᱟᱹ, ᱱᱟᱣᱟ ᱩᱫᱽᱭᱚᱜ ᱞᱟᱭᱥᱮᱱᱥ ᱞᱟᱹᱜᱤᱫ 74 ᱥᱟᱭᱠᱚᱲᱟ ᱦᱟᱹᱵᱤᱡ ᱟᱡ-ᱛᱮ, ᱚᱱᱟ ᱠᱷᱚᱱ ᱪᱮᱛᱟᱱ ᱱᱟᱣᱟ-ᱴᱮᱠᱱᱚᱞᱚᱡᱤ ᱠᱩᱠᱞᱤ ᱟᱨᱵᱟᱝ ᱚᱞ ᱟᱠᱟᱱ ᱠᱟᱨᱚᱱ ᱨᱮ ᱥᱚᱨᱠᱟᱨᱤ ᱦᱚᱨ, ᱟᱨ ᱱᱟᱣᱟ ᱞᱟᱭᱥᱮᱱᱥᱤᱭᱟᱹ ᱵᱟᱝ ᱠᱟᱱ ᱠᱚᱢᱯᱟᱹᱱᱤ ᱞᱟᱹᱜᱤᱫ ᱟᱹᱰᱤ ᱴᱟᱴᱠᱟ 49 ᱥᱟᱭᱠᱚᱲᱟ ᱦᱚᱨ᱾ ᱥᱩᱨᱚᱠᱠᱷᱟ ᱪᱷᱟᱹᱲ ᱟᱨ ᱫᱤᱥᱚᱢ-ᱨᱩᱠᱷᱤᱭᱟᱹ ᱧᱮᱞᱡᱚᱝ ᱚᱱᱟ ᱪᱤᱠᱤ ᱨᱮ ᱜᱮ ᱛᱟᱦᱮᱸᱱᱟ᱾ ᱩᱫᱽᱭᱚᱜ ᱞᱟᱭᱥᱮᱱᱥ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱟᱨ ᱵᱟᱦᱨᱮ ᱠᱟᱹᱢᱤ ᱥᱟᱶ ᱧᱮᱞᱚᱜ-ᱟ᱾

ᱠᱤᱨᱤᱧ-ᱱᱤᱛᱤ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱢᱚᱱᱛᱨᱟᱞᱚᱭ (Ministry of Defence) ᱨᱮᱱᱟᱜ ᱚᱞ ᱠᱟᱱᱟ᱾ 20 ᱱᱚᱵᱷᱮᱢᱵᱚᱨ 2025 ᱨᱮᱱᱟᱜ ᱢᱤᱫ PIB ᱪᱤᱠᱤ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱟᱨᱡᱟᱣ ᱦᱚᱨ 2020 (Defence Acquisition Procedure 2020) ᱵᱟᱰᱟᱭᱟ, Buy (Indian-IDDM) ᱫᱚ ᱯᱩᱭᱞᱩ ᱴᱷᱟᱶ ᱥᱟᱶ, ᱟᱨ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱠᱤᱨᱤᱧ ᱯᱚᱛᱚᱵ 2025 (Defence Procurement Manual 2025), ᱡᱟᱦᱟᱸ 1 ᱱᱚᱵᱷᱮᱢᱵᱚᱨ 2025 ᱠᱷᱚᱱ ᱪᱟᱹᱞᱩ ᱠᱟᱱᱟ, ᱟᱢᱫᱟᱡᱽ Rs 1 ᱞᱟᱠᱷ ᱠᱚᱨᱚᱲ ᱨᱮᱱᱟᱜ ᱨᱟᱡᱽᱥᱚ ᱠᱤᱨᱤᱧ ᱞᱟᱹᱜᱤᱫ᱾ FDI ᱱᱤᱛᱤ ᱪᱩᱠᱛᱤ ᱵᱟᱝ ᱮᱢᱟ᱾ ᱟᱨᱡᱟᱣ-ᱦᱚᱨ ᱵᱟᱦᱨᱮ ᱦᱤᱥᱟᱹᱭᱟᱹ ᱵᱟᱝ ᱪᱷᱟᱹᱲᱟ᱾

## ᱥᱤᱢᱟᱹ ᱚᱠᱚᱭ ᱠᱟᱵᱽᱡᱟᱭᱟ

ᱵᱟᱦᱨᱮ ᱵᱮᱯᱟᱨ ᱢᱟᱦᱟ ᱱᱤᱨᱫᱮᱥᱟᱞᱚᱭ (Directorate General of Foreign Trade, DGFT) SCOMET ᱠᱟᱵᱽᱡᱟ-ᱞᱤᱥᱴ ᱪᱷᱟᱯᱟᱣᱟ ᱟᱨ ᱱᱟᱣᱟ ᱯᱷᱟᱭᱤᱞ ᱟᱡᱟᱜ ᱥᱟᱭᱤᱴ ᱨᱮ ᱫᱚᱦᱚᱭᱟ᱾ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱩᱛᱱᱟᱹᱣ ᱵᱤᱵᱷᱟᱜᱽ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱵᱟᱦᱨᱮ-ᱨᱟᱠᱟᱵ ᱪᱷᱟᱹᱲ ᱞᱮᱠᱷᱟᱭᱟ: ᱠᱟᱹᱣᱰᱤ ᱥᱮᱨᱢᱟ (FY) 2024-25 ᱨᱮ 1,762᱾ DGFT ᱨᱮᱱᱟᱜ ᱨᱟᱹᱱᱤᱛᱤ-ᱵᱮᱯᱟᱨ ᱯᱚᱛᱚᱵ ᱜᱳᱞᱟᱹ-ᱦᱟᱹᱛᱤᱭᱟᱹᱨ ᱞᱤᱥᱴ ᱞᱟᱭᱥᱮᱱᱥ ᱫᱚ ᱵᱤᱵᱷᱟᱜᱽ ᱨᱮᱱᱟᱜ ᱦᱚᱨ ᱥᱮᱫ ᱩᱫᱩᱜᱟ᱾ ᱡᱟᱦᱟᱸ «ᱵᱟᱦᱨᱮ-ᱨᱟᱠᱟᱵ ᱩᱛᱱᱟᱹᱣ» ᱨᱮᱱᱟᱜ ᱱᱤᱛᱤ-ᱧᱮᱞᱡᱚᱝ ᱠᱷᱟᱹᱞᱤ Rs 23,622 ᱠᱚᱨᱚᱲ ᱵᱟᱦᱨᱮ-ᱨᱟᱠᱟᱵ ᱡᱩᱢᱤᱫ ᱧᱮᱞᱟ, ᱪᱷᱟᱹᱲ-ᱫᱟᱲᱮ ᱟᱨ ᱠᱟᱵᱽᱡᱟ-ᱞᱤᱥᱴ ᱵᱟᱝ, ᱚᱱᱟ ᱛᱟᱭᱚᱢᱤᱡ ᱢᱟᱞ-ᱫᱟᱹᱠᱟ ᱨᱮᱱᱟᱜ ᱵᱟᱹᱧᱪᱷᱚᱱ ᱟᱲᱟᱜᱟ᱾

ᱵᱟᱦᱨᱮ-ᱨᱟᱠᱟᱵ ᱡᱩᱢᱤᱫ ᱴᱷᱮᱱ ᱟᱡᱟᱜ ᱦᱟᱹᱴᱤᱧ ᱦᱚᱸ ᱞᱟᱹᱠᱛᱤᱜ-ᱟ᱾ FY 2024-25 ᱨᱮ ᱱᱤᱡᱮᱨᱟᱜ ᱯᱷᱟᱨᱢ Rs 15,233 ᱠᱚᱨᱚᱲ ᱮᱢ ᱞᱮᱫ-ᱟ ᱟᱨ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱥᱚᱨᱠᱟᱨᱤ ᱠᱷᱚᱸᱰ ᱩᱫᱽᱭᱚᱜ (defence public sector undertakings) Rs 8,389 ᱠᱚᱨᱚᱲ᱾ ᱚᱱᱟ ᱥᱮᱨᱢᱟ ᱨᱮᱱᱟᱜ ᱰᱷᱮᱨ ᱥᱚᱨᱠᱟᱨᱤ-ᱠᱷᱚᱸᱰ ᱵᱟᱦᱨᱮ-ᱨᱟᱠᱟᱵ ᱞᱟᱭᱤᱱ ᱨᱮ ᱛᱟᱦᱮᱸ ᱠᱟᱱᱟ᱾ ᱵᱮᱱᱟᱣ, ᱱᱚᱵᱷᱮᱢᱵᱚᱨ ᱪᱤᱠᱤ ᱞᱮᱠᱟᱛᱮ, ᱱᱤᱛ ᱦᱚᱸ ᱟᱢᱫᱟᱡᱽ 77 ᱥᱟᱭᱠᱚᱲᱟ ᱥᱚᱨᱠᱟᱨᱤ-ᱠᱷᱚᱸᱰ ᱠᱟᱱᱟ᱾ ᱩᱛᱱᱟᱹᱣ-ᱱᱤᱛᱤ ᱟᱨ ᱵᱮᱱᱟᱣ-ᱱᱤᱛᱤ ᱚᱱᱟ ᱦᱟᱹᱴᱤᱧ ᱵᱟᱝ ᱞᱟᱲᱟᱣ ᱮᱫ-ᱟ᱾

## ᱵᱮᱯᱟᱨᱤ ᱚᱠᱟᱥ ᱚᱠᱚᱭ ᱥᱟᱥᱚᱱᱟ

ᱚᱠᱟᱥ ᱵᱤᱵᱷᱟᱜᱽ (Department of Space) ᱨᱮᱱᱟᱜ ᱥᱤᱧᱚᱛᱤᱭᱟᱹ ᱚᱠᱟᱥ ᱱᱤᱛᱤ 2023 (Indian Space Policy 2023) IN-SPACe ᱫᱚ ᱵᱟᱝ-ᱥᱚᱨᱠᱟᱨᱤ ᱚᱠᱟᱥ-ᱠᱟᱹᱢᱤ ᱪᱷᱟᱹᱲ ᱟᱨ ᱩᱛᱱᱟᱹᱣ ᱞᱟᱹᱜᱤᱫ, ᱥᱤᱧᱚᱛᱤᱭᱟᱹ ᱚᱠᱟᱥ ᱯᱟᱱᱛᱮ ᱜᱟᱶᱛᱟ (Indian Space Research Organisation, ISRO) ᱫᱚ ᱯᱟᱱᱛᱮ ᱟᱨ ᱩᱛᱱᱟᱹᱣ ᱞᱟᱹᱜᱤᱫ, ᱟᱨ NewSpace India Limited ᱫᱚ ᱥᱚᱨᱠᱟᱨᱤ ᱯᱷᱚᱸᱰ ᱛᱮ ᱯᱟᱹᱭᱤᱥ ᱟᱠᱟᱱ ᱯᱞᱮᱴᱯᱷᱚᱨᱢ ᱵᱮᱯᱟᱨᱤ ᱵᱮᱱᱟᱣ ᱞᱟᱹᱜᱤᱫ ᱮᱢᱟ᱾ ᱨᱩᱠᱷᱤᱭᱟᱹ FDI ᱟᱨ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱵᱟᱦᱨᱮ-ᱨᱟᱠᱟᱵ ᱱᱤᱛᱤ IN-SPACe ᱵᱟᱝ ᱵᱟᱪᱷᱟᱣᱟ, ᱟᱨ ᱚᱠᱟᱥ ᱱᱤᱛᱤ ᱯᱚᱡᱤᱴᱤᱵᱷ ᱤᱱᱰᱤᱡᱮᱱᱟᱭᱡᱮᱥᱚᱱ ᱞᱤᱥᱴ ᱵᱟᱝ ᱩᱪᱷᱟᱹᱱᱟ᱾ ᱡᱚᱲᱟᱣ ᱵᱨᱤᱯᱷ ᱢᱮᱱ ᱢᱟ ᱡᱮ ᱦᱟᱛᱷ-ᱩᱪᱟᱹᱲ ᱚᱠᱟᱨᱮ ᱢᱮᱱᱟᱜ-ᱟ, ᱵᱤᱥᱮᱥ ᱠᱟᱛᱮ ᱚᱱᱟ ᱯᱷᱟᱨᱢ ᱞᱟᱹᱜᱤᱫ ᱡᱟᱦᱟᱸ ᱥᱮᱴᱮᱞᱟᱭᱤᱴ ᱟᱨ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱦᱟᱹᱴᱤᱧ ᱵᱟᱱᱟᱨ ᱵᱮᱱᱟᱣᱟ᱾

## ᱫᱟᱨᱦᱟᱠᱤᱭᱟᱹ ᱱᱤᱛᱤ-ᱧᱮᱞ ᱥᱟᱹᱨᱤ ᱛᱮ ᱪᱮᱫ ᱢᱮᱱ ᱫᱟᱲᱮᱭᱟᱜ-ᱟ

ᱱᱚᱶᱟ ᱢᱮᱱ ᱫᱟᱲᱮᱭᱟᱜ-ᱟ ᱡᱮ ᱱᱟᱣᱟ ᱞᱤᱥᱴ ᱩᱪᱷᱟᱹᱱ ᱦᱩᱭ ᱞᱮᱱᱟ ᱥᱮ ᱵᱟᱝ, ᱯᱩᱸᱡᱤ ᱠᱷᱚᱨᱚᱪ ᱵᱮᱵᱷᱟᱨ ᱦᱩᱭ ᱞᱮᱱᱟ ᱥᱮ ᱵᱟᱝ (FY 2025-26 ᱞᱟᱹᱜᱤᱫ ᱥᱩᱫᱷᱟᱹᱨ ᱟᱢᱫᱟᱡᱽ ᱨᱮ Rs 1.86 ᱞᱟᱠᱷ ᱠᱚᱨᱚᱲ, ᱯᱩᱨᱟᱹ ᱵᱮᱵᱷᱟᱨ), ᱟᱨ ᱢᱩᱪᱟᱹᱫ ᱵᱮᱱᱟᱣ ᱟᱨ ᱵᱟᱦᱨᱮ-ᱨᱟᱠᱟᱵ ᱩᱪᱷᱟᱹᱱ Rs 1,50,590 ᱠᱚᱨᱚᱲ ᱟᱨ Rs 23,622 ᱠᱚᱨᱚᱲ ᱵᱚᱫᱚᱞ ᱟᱠᱟᱫ-ᱟ ᱥᱮ ᱵᱟᱝ᱾ ᱱᱚᱶᱟ ᱚᱱᱟ ᱚᱨᱡᱚ ᱫᱚ ᱚᱞ ᱟᱠᱟᱱ 2029 ᱡᱚᱥ, Rs 3 ᱞᱟᱠᱷ ᱠᱚᱨᱚᱲ ᱵᱮᱱᱟᱣ ᱟᱨ Rs 50,000 ᱠᱚᱨᱚᱲ ᱵᱟᱦᱨᱮ-ᱨᱟᱠᱟᱵ, ᱥᱟᱶ ᱛᱩᱞᱟᱹᱣ ᱫᱟᱲᱮᱭᱟᱜ-ᱟ, ᱡᱚᱥ ᱫᱚ ᱦᱚᱨ ᱨᱮ ᱵᱚᱫᱚᱞ ᱵᱟᱹᱱᱩᱜ ᱛᱮ᱾ ᱱᱚᱶᱟ ᱡᱮᱜᱮᱛ-ᱴᱷᱟᱶ, ᱜᱩᱯᱛᱤ ᱠᱟᱹᱢᱤᱦᱚᱨᱟ, ᱟᱨᱵᱟᱝ ᱱᱚᱶᱟ ᱥᱟᱦᱴᱟ ᱠᱚ ᱨᱮ ᱵᱟᱹᱱᱩᱜ ᱠᱟᱹᱢᱤ-ᱡᱩᱢᱤᱫ ᱟᱞᱚ ᱵᱨᱤᱯᱷ ᱢᱟ᱾

ᱜᱨᱮᱴ ᱤᱱᱰᱤᱭᱟᱱ ᱠᱚᱢᱯᱟᱹᱱᱤ (Great Indian Company) ᱥᱟᱶᱛᱮᱱ ᱠᱷᱟᱴᱚ ᱪᱤᱠᱤ ᱪᱷᱟᱯᱟᱣᱟ ᱡᱮ ᱡᱚᱛᱚ ᱦᱟᱹᱛᱤᱭᱟᱹᱨ ᱟᱡᱟᱜ ᱩᱪᱷᱟᱹᱱ ᱨᱮ ᱯᱟᱲᱦᱟᱣ ᱜᱟᱱᱚᱜ-ᱟ᱾ ᱱᱤᱛᱤ-ᱠᱟᱹᱢᱤ ᱫᱚ ᱦᱟᱹᱛᱤᱭᱟᱹᱨ ᱠᱚ ᱵᱷᱮᱜᱟᱨ ᱫᱚᱦᱚ ᱠᱟᱱᱟ᱾
