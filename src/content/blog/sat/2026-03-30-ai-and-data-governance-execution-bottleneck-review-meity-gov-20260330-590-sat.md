ᱠᱟᱹᱢᱤ ᱨᱮᱱᱟᱜ ᱟᱴᱠᱟᱣ ᱫᱚ ᱢᱤᱫ ᱦᱩᱲᱟᱹᱜ GPU ᱵᱟᱝ ᱠᱟᱱᱟ ᱾ ᱱᱚᱣᱟ ᱢᱤᱫ ᱟᱹᱭᱟᱹᱛ ᱠᱟᱱᱟ ᱾ 14 ᱱᱚᱵᱷᱮᱢᱵᱚᱨ 2025 ᱨᱮ ᱯᱨᱮᱥ ᱤᱱᱯᱷᱚᱨᱢᱮᱥᱚᱱ ᱵᱭᱩᱨᱚ (Press Information Bureau, PIB) ᱢᱮᱱ ᱞᱮᱫᱟ ᱡᱮ ᱥᱚᱨᱠᱟᱨ DPDP ᱱᱤᱭᱚᱢ ᱩᱪᱷᱟᱹᱱ ᱟᱠᱟᱫᱟ, "DPDP ᱟᱹᱭᱤᱱ, 2023 ᱨᱮᱱᱟᱜ ᱯᱩᱨᱟᱹ ᱠᱟᱹᱢᱤᱭᱟᱹᱨᱮ ᱟᱹᱜᱩ ᱪᱤᱱᱦᱟᱹ ᱠᱟᱛᱮ" ᱾ ᱚᱱᱟ ᱚᱞ ᱛᱟᱭᱚᱢ ᱛᱮ ᱢᱤᱫ ᱜᱮᱞᱼᱤᱨᱟᱹᱞ ᱪᱟᱸᱫᱚ ᱨᱮᱱᱟᱜ ᱛᱷᱚᱠ ᱛᱷᱚᱠ ᱚᱠᱛᱚ ᱚᱞᱟ ᱾ ᱵᱟᱱᱟᱨ ᱟᱹᱭᱟᱹᱛ ᱚᱞ ᱨᱮ ᱢᱮᱱᱟᱜ-ᱟ ᱾ ᱚᱠᱟ ᱫᱷᱟᱨᱟ ᱚᱠᱟ ᱛᱷᱚᱠ ᱨᱮ ᱢᱮᱱᱟᱜ-ᱟ, ᱚᱱᱟ ᱫᱚ ᱨᱟᱡᱽᱯᱚᱛᱨᱚ (Gazette) ᱜᱮ ᱢᱮᱱᱟ ᱾

## "ᱯᱩᱨᱟᱹ" ᱪᱮᱫ ᱵᱟᱝ ᱢᱮᱱ ᱫᱟᱲᱮᱭᱟᱜ-ᱟ

G.S.R. 843(E), ᱛᱟᱹᱨᱤᱠᱷ 13 ᱱᱚᱵᱷᱮᱢᱵᱚᱨ 2025, ᱮᱦᱚᱵ ᱩᱪᱷᱟᱹᱱ ᱠᱟᱱᱟ ᱾ ᱱᱚᱣᱟ ᱟᱹᱭᱤᱱ ᱫᱚ ᱢᱤᱫ ᱵᱚᱞᱚᱠ ᱞᱮᱠᱟ ᱵᱟᱝ ᱟᱹᱜᱩᱭᱟ ᱾

ᱪᱷᱟᱯᱟ ᱠᱷᱚᱱ ᱯᱨᱚᱵᱷᱟᱣᱨᱮ: ᱫᱷᱟᱨᱟ 1(2), ᱫᱷᱟᱨᱟ 2, ᱫᱷᱟᱨᱟ 18 ᱠᱷᱚᱱ 26, ᱫᱷᱟᱨᱟ 35 ᱟᱨ 38 ᱠᱷᱚᱱ 43, ᱟᱨ ᱫᱷᱟᱨᱟ 44 ᱨᱮᱱᱟᱜ ᱩᱯᱼᱫᱷᱟᱨᱟ (1) ᱟᱨ (3) ᱾ ᱱᱤᱭᱚᱢ ᱨᱮ ᱱᱚᱣᱟ ᱱᱤᱭᱚᱢ 1, 2 ᱟᱨ 17 ᱠᱷᱚᱱ 21 ᱥᱟᱶ ᱢᱮᱥᱟᱜ-ᱟ: ᱧᱩᱛᱩᱢ, ᱩᱯᱨᱩᱢ, ᱟᱨ ᱵᱚᱨᱰ ᱵᱟᱦᱟᱞ ᱟᱨ ᱪᱟᱞᱟᱣ ᱨᱮᱱᱟᱜ ᱡᱚᱱᱛᱨᱚ ᱾

ᱪᱷᱟᱯᱟ ᱠᱷᱚᱱ ᱢᱤᱫ ᱥᱮᱨᱢᱟ, 13 ᱱᱚᱵᱷᱮᱢᱵᱚᱨ 2026: ᱫᱷᱟᱨᱟ 6(9) ᱟᱨ ᱫᱷᱟᱨᱟ 27(1)(d), ᱱᱤᱭᱚᱢ 4 ᱥᱟᱶ ᱾ ᱱᱚᱣᱟ ᱫᱚ ᱥᱚᱢᱢᱚᱛᱤ ᱢᱮᱱᱮᱡᱚᱨ (Consent Manager) ᱧᱩᱛᱩᱢ ᱚᱞ ᱟᱨ ᱧᱩᱛᱩᱢ ᱚᱞ ᱥᱚᱨᱛ ᱨᱮᱱᱟᱜ ᱵᱷᱟᱸᱜᱟ ᱵᱤᱰᱟᱹᱣ ᱨᱮᱱᱟᱜ ᱵᱚᱨᱰ ᱨᱮᱱᱟᱜ ᱟᱹᱭᱫᱟᱹᱨᱤ ᱠᱟᱱᱟ ᱾

ᱪᱷᱟᱯᱟ ᱠᱷᱚᱱ ᱜᱮᱞᱼᱤᱨᱟᱹᱞ ᱪᱟᱸᱫᱚ, 13 ᱢᱮ 2027: ᱫᱷᱟᱨᱟ 3 ᱠᱷᱚᱱ 5, ᱩᱯᱼᱫᱷᱟᱨᱟ (9) ᱵᱟᱹᱜᱤ ᱠᱟᱛᱮ ᱫᱷᱟᱨᱟ 6, ᱫᱷᱟᱨᱟ 7 ᱠᱷᱚᱱ 17, ᱫᱷᱟᱨᱟ 27 ᱨᱮᱱᱟᱜ ᱵᱟᱹᱠᱤ, ᱫᱷᱟᱨᱟ 28 ᱠᱷᱚᱱ 34, 36 ᱟᱨ 37, ᱟᱨ ᱫᱷᱟᱨᱟ 44(2) ᱾ ᱱᱤᱭᱚᱢ ᱨᱮ ᱱᱚᱣᱟ ᱱᱤᱭᱚᱢ 3, 5 ᱠᱷᱚᱱ 16, 22 ᱟᱨ 23 ᱠᱟᱱᱟ: ᱱᱚᱴᱤᱥ, ᱨᱩᱠᱷᱤᱭᱟᱹ, ᱵᱷᱟᱸᱜᱟ ᱨᱮᱱᱟᱜ ᱠᱷᱚᱵᱚᱨ, ᱢᱮᱴᱟᱣ, ᱜᱤᱫᱽᱨᱟᱹ ᱨᱮᱱᱟᱜ ᱥᱚᱢᱢᱚᱛᱤ, ᱢᱚᱦᱚᱛᱣᱯᱩᱨᱱᱚ ᱰᱟᱴᱟ ᱫᱟᱭᱤᱛᱣᱟᱱ (Significant Data Fiduciary) ᱨᱮᱱᱟᱜ ᱠᱟᱹᱢᱤ, ᱦᱚᱲ ᱨᱮᱱᱟᱜ ᱟᱹᱭᱫᱟᱹᱨᱤ, ᱫᱤᱥᱚᱢ ᱵᱟᱦᱨᱮ ᱩᱪᱟᱹᱲ, ᱯᱟᱱᱛᱮ ᱢᱟᱱ, ᱟᱯᱤᱞ, ᱟᱨ ᱥᱩᱪᱱᱟ ᱞᱟᱹᱜᱤᱫ ᱥᱚᱨᱠᱟᱨ ᱨᱮᱱᱟᱜ ᱠᱩᱞ ᱾

ᱡᱩᱫᱤ "ᱯᱩᱨᱟᱹ ᱠᱟᱹᱢᱤᱭᱟᱹᱨᱮ ᱟᱹᱜᱩ" ᱫᱚ "ᱠᱚᱢᱯᱟᱹᱱᱤ ᱢᱟᱲᱟᱝ ᱠᱷᱚᱱ ᱜᱮ DPDP ᱱᱚᱴᱤᱥ ᱮᱢ ᱦᱩᱭᱩᱜ ᱛᱟᱦᱮᱸᱱ ᱛᱟᱦᱮᱸᱱᱟ, ᱱᱤᱭᱚᱢ 6 ᱞᱮᱠᱟ ᱱᱤᱡᱮᱨ ᱰᱟᱴᱟ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱦᱩᱭᱩᱜ ᱛᱟᱦᱮᱸᱱᱟ, ᱟᱨ ᱥᱤᱰᱩᱞ ᱥᱟᱢᱟᱝ ᱦᱩᱭᱩᱜ ᱛᱟᱦᱮᱸᱱᱟ" ᱞᱮᱠᱟ ᱯᱟᱲᱦᱟᱣᱜ-ᱟ, ᱮᱱᱠᱷᱟᱱ ᱨᱟᱡᱽᱯᱚᱛᱨᱚ ᱚᱱᱟ ᱯᱟᱲᱦᱟᱣ ᱵᱤᱨᱩᱫᱷᱟ ᱾ ᱚᱱᱟ PIB ᱚᱞ ᱵᱷᱤᱛᱨᱤ ᱨᱮᱱᱟᱜ ᱛᱷᱚᱠ ᱛᱷᱚᱠ ᱟᱹᱭᱟᱹᱛ ᱜᱮ ᱩᱪᱷᱟᱹᱱ ᱥᱟᱶ ᱢᱮᱥᱟᱜ-ᱟ ᱾

## ᱵᱚᱨᱰ ᱫᱚ ᱫᱚᱥᱟᱨ ᱟᱴᱠᱟᱣ ᱠᱟᱱᱟ, ᱯᱟᱭᱨᱟ ᱚᱞ ᱵᱟᱝ

ᱫᱷᱟᱨᱟ 18 ᱢᱮᱱᱟ ᱡᱮ ᱵᱚᱨᱰ ᱫᱚ ᱛᱟᱞᱢᱟ ᱥᱚᱨᱠᱟᱨ ᱡᱟᱦᱟᱸ ᱛᱟᱹᱨᱤᱠᱷ ᱩᱪᱷᱟᱹᱱᱟ, ᱚᱱᱟ ᱛᱟᱹᱨᱤᱠᱷ ᱠᱷᱚᱱ ᱛᱷᱟᱯᱚᱱᱚᱜ-ᱟ ᱾ ᱫᱷᱟᱨᱟ 18 ᱯᱨᱚᱵᱷᱟᱣᱨᱮ ᱟᱹᱜᱩ ᱞᱮᱠᱷᱟᱱ ᱥᱚᱨᱠᱟᱨ ᱚᱱᱟ ᱛᱟᱹᱨᱤᱠᱷ ᱮᱢ ᱫᱟᱲᱮᱭᱟᱜ-ᱟ ᱾ ᱱᱚᱣᱟ ᱟᱡ ᱜᱮ ᱚᱱᱟ ᱛᱟᱹᱨᱤᱠᱷ ᱵᱟᱝ ᱠᱟᱱᱟ ᱾ ᱱᱤᱭᱚᱢ 17 ᱥᱚᱫᱚᱨᱼᱥᱚᱢᱼᱪᱚᱭᱚᱱ ᱠᱚᱢᱤᱴᱤ ᱵᱮᱱᱟᱣᱟ, ᱪᱮᱭᱟᱨᱯᱚᱨᱥᱚᱱ ᱞᱟᱹᱜᱤᱫ ᱠᱮᱵᱤᱱᱮᱴ ᱥᱮᱠᱨᱮᱴᱟᱨᱤ ᱟᱹᱭᱩᱨ ᱨᱮ, ᱟᱹᱭᱤᱱ ᱥᱮᱠᱨᱮᱴᱟᱨᱤ ᱟᱨ ᱤᱞᱮᱠᱴᱨᱚᱱᱤᱠᱥ ᱟᱨ ᱤᱱᱯᱷᱚᱨᱢᱮᱥᱚᱱ ᱴᱮᱠᱱᱚᱞᱚᱡᱤ ᱢᱚᱱᱛᱨᱟᱞᱚᱭ (Ministry of Electronics and Information Technology, MeitY) ᱨᱮᱱ ᱥᱮᱠᱨᱮᱴᱟᱨᱤ ᱥᱟᱶ ᱾ ᱱᱚᱣᱟ ᱥᱤᱨᱤᱡᱽ ᱛᱟᱭᱚᱢ ᱨᱮᱱᱟᱜ ᱡᱟᱦᱟᱸ ᱨᱟᱡᱽᱯᱚᱛᱨᱚ ᱵᱟᱝ ᱩᱫᱩᱜ ᱟᱠᱟᱫᱟ ᱡᱟᱦᱟᱸ ᱪᱮᱭᱟᱨᱯᱚᱨᱥᱚᱱ ᱧᱩᱛᱩᱢᱟ, ᱦᱮᱰᱠᱣᱟᱴᱚᱨ ᱛᱷᱤᱨᱟ, ᱥᱮ ᱵᱚᱨᱰ ᱱᱟᱞᱤᱥ ᱧᱟᱢᱮᱫ ᱠᱟᱱᱟ ᱢᱮᱱᱟ ᱾

PIB ᱢᱮᱱᱟ ᱡᱮ ᱵᱚᱨᱰ "ᱯᱩᱨᱟᱹ ᱰᱤᱡᱤᱴᱟᱞ ᱥᱚᱸᱥᱛᱷᱟ ᱞᱮᱠᱟ ᱠᱟᱹᱢᱤᱭᱟ" ᱟᱨ ᱱᱟᱜᱽᱨᱤᱠ ᱠᱚ ᱢᱤᱫ ᱯᱞᱮᱴᱯᱷᱚᱨᱢ ᱟᱨ ᱢᱳᱵᱟᱭᱤᱞ ᱮᱯ ᱛᱟᱞᱟᱛᱮ ᱱᱟᱞᱤᱥ ᱠᱚ ᱫᱚᱦᱚᱭᱟ, TDSAT ᱨᱮ ᱟᱯᱤᱞ ᱥᱟᱶ ᱾ "ᱠᱟᱹᱢᱤᱭᱟ" ᱫᱚ ᱚᱞ ᱨᱮᱱᱟᱜ ᱠᱟᱞ ᱠᱟᱱᱟ ᱾ TDSAT ᱨᱮᱱᱟᱜ ᱨᱳᱞ ᱢᱟᱲᱟᱝ ᱠᱷᱚᱱ ᱟᱹᱭᱤᱱ ᱨᱮ ᱚᱞ ᱟᱠᱟᱱᱟ: ᱫᱷᱟᱨᱟ 2 ᱮᱯᱮᱞᱮᱴ ᱴᱨᱟᱭᱵᱩᱱᱟᱞ ᱫᱚ ᱴᱮᱞᱤᱠᱚᱢ ᱰᱤᱥᱯᱭᱩᱴᱥ ᱥᱮᱴᱮᱞᱢᱮᱱᱴ ᱮᱱᱰ ᱮᱯᱮᱞᱮᱴ ᱴᱨᱟᱭᱵᱩᱱᱟᱞ (Telecom Disputes Settlement and Appellate Tribunal, TDSAT) ᱞᱮᱠᱟ ᱩᱯᱨᱩᱢᱟ, ᱟᱨ ᱫᱷᱟᱨᱟ 44(1), ᱢᱟᱲᱟᱝ ᱠᱷᱚᱱ ᱯᱨᱚᱵᱷᱟᱣᱨᱮ, TRAI ᱟᱹᱭᱤᱱ ᱚᱱᱟ ᱞᱮᱠᱟ ᱵᱚᱫᱚᱞᱟ ᱾ ᱱᱤᱭᱚᱢ 22 ᱨᱮᱱᱟᱜ ᱟᱯᱤᱞ ᱠᱟᱹᱢᱤ ᱜᱮᱞᱼᱤᱨᱟᱹᱞ ᱪᱟᱸᱫᱚ ᱨᱮᱱᱟᱜ ᱥᱮᱴ ᱥᱟᱶ ᱮᱦᱚᱵᱚᱜ-ᱟ ᱾ ᱱᱤᱭᱚᱢ 20 ᱨᱮᱱᱟᱜ ᱰᱤᱡᱤᱴᱟᱞ ᱚᱯᱷᱤᱥ ᱫᱚ ᱵᱚᱨᱰ ᱪᱮᱫ ᱞᱮᱠᱟ ᱠᱟᱹᱢᱤᱭᱟ ᱚᱱᱟ ᱱᱤᱭᱚᱢ ᱞᱮᱠᱟ ᱯᱨᱚᱵᱷᱟᱣᱨᱮ ᱢᱮᱱᱟᱜ-ᱟ ᱾ ᱱᱚᱣᱟ ᱱᱤᱛᱦᱚᱸ ᱢᱤᱫ ᱵᱚᱨᱰ ᱞᱟᱹᱠᱛᱤᱭᱟ ᱾

## ᱡᱟᱦᱟᱸ ᱛᱷᱚᱠ ᱱᱤᱛ ᱠᱟᱹᱢᱤ ᱨᱮ ᱵᱟᱝ ᱟᱹᱜᱩ ᱫᱟᱲᱮᱭᱟᱜ-ᱟ

ᱢᱤᱫ ᱢᱚᱦᱚᱛᱣᱯᱩᱨᱱᱚ ᱰᱟᱴᱟ ᱫᱟᱭᱤᱛᱣᱟᱱ ᱫᱚ ᱩᱱ ᱡᱚᱦᱚᱜ ᱜᱮ ᱛᱟᱦᱮᱸᱱᱟ ᱡᱚᱠᱷᱚᱱ ᱛᱟᱞᱢᱟ ᱥᱚᱨᱠᱟᱨ ᱫᱷᱟᱨᱟ 10 ᱞᱮᱠᱟ ᱢᱤᱫᱴᱟᱹᱝ, ᱥᱮ ᱢᱤᱫ ᱛᱷᱚᱠ, ᱩᱪᱷᱟᱹᱱᱟ ᱾ ᱫᱷᱟᱨᱟ 10 ᱢᱮ 2027 ᱨᱮᱱᱟᱜ ᱛᱟᱹᱞᱠᱟᱹ ᱨᱮ ᱢᱮᱱᱟᱜ-ᱟ ᱾ ᱟᱹᱭᱤᱱ ᱨᱮᱱᱟᱜ ᱠᱟᱨᱚᱱ ᱨᱮ ᱢᱟᱛᱨᱟ, ᱥᱚᱝᱵᱮᱫᱚᱱᱥᱤᱞᱛᱟ, ᱦᱚᱲ ᱪᱮᱛᱟᱱ ᱡᱚᱠᱷᱤᱢ, ᱥᱟᱨᱵᱷᱚᱨᱮᱱᱴᱤ, ᱱᱤᱨᱵᱟᱪᱚᱱ ᱞᱚᱠᱛᱟᱱᱛᱨᱚ, ᱯᱚᱱᱚᱛ ᱨᱮᱱᱟᱜ ᱨᱩᱠᱷᱤᱭᱟᱹ, ᱟᱨ ᱡᱟᱹᱛᱤᱭᱟᱹᱨᱤ ᱵᱽᱭᱵᱚᱥᱛᱷᱟ ᱥᱮᱞᱮᱫ ᱢᱮᱱᱟᱜ-ᱟ ᱾ ᱟᱹᱭᱤᱱ ᱡᱟᱦᱟᱸ ᱵᱮᱵᱷᱟᱨᱤᱭᱟᱹ ᱞᱮᱠᱷᱟ ᱵᱟᱝ ᱛᱷᱤᱨᱟ, ᱚᱱᱟ ᱯᱟᱨᱚᱢ ᱠᱟᱛᱮ ᱚᱠᱟ ᱯᱷᱚᱨᱢ ᱦᱚᱸ ᱢᱤᱫᱴᱟᱹᱝ ᱵᱟᱝ ᱵᱮᱱᱟᱣᱚᱜ-ᱟ ᱾ ᱛᱮᱥᱟᱨ ᱥᱤᱰᱩᱞ ᱨᱮᱱᱟᱜ ᱵᱮᱵᱷᱟᱨᱤᱭᱟᱹ ᱥᱤᱢᱟᱹ, ᱢᱟᱨᱟᱝ ᱤᱼᱠᱚᱢᱚᱨᱥ ᱥᱮ ᱥᱚᱥᱤᱭᱟᱞ ᱢᱤᱰᱤᱭᱟ ᱢᱚᱫᱷᱭᱚᱥᱛᱷᱤ ᱞᱟᱹᱜᱤᱫ ᱵᱟᱨ ᱠᱳᱴᱤ ᱟᱨ ᱚᱱᱞᱟᱭᱤᱱ ᱜᱮᱢᱤᱝ ᱢᱚᱫᱷᱭᱚᱥᱛᱷᱤ ᱞᱟᱹᱜᱤᱫ ᱢᱚᱬᱮ ᱜᱮᱞ ᱞᱟᱠᱷ, ᱫᱚ ᱱᱤᱭᱚᱢ 8 ᱞᱮᱠᱟ ᱢᱮᱴᱟᱣ ᱨᱮᱱᱟᱜ ᱴᱨᱤᱜᱟᱨ ᱠᱟᱱᱟ ᱾ ᱩᱱᱠᱩ ᱢᱚᱦᱚᱛᱣᱯᱩᱨᱱᱚ ᱰᱟᱴᱟ ᱫᱟᱭᱤᱛᱣᱟᱱ ᱯᱟᱨᱤᱠᱷᱟ ᱵᱟᱝ ᱠᱟᱱᱟ, ᱟᱨ ᱱᱤᱭᱚᱢ 8 ᱟᱡ ᱜᱮ ᱢᱮ 2027 ᱨᱮᱱᱟᱜ ᱛᱟᱹᱞᱠᱟᱹ ᱨᱮ ᱢᱮᱱᱟᱜ-ᱟ ᱾

ᱱᱤᱭᱚᱢ 13(4) ᱨᱮᱱᱟᱜ, ᱩᱞᱞᱮᱠᱷ ᱟᱠᱟᱱ ᱱᱤᱡᱮᱨ ᱰᱟᱴᱟ ᱟᱨ ᱚᱱᱟ ᱨᱮᱱᱟᱜ ᱴᱨᱟᱯᱷᱤᱠ ᱰᱟᱴᱟ ᱥᱤᱧᱚᱛ ᱵᱟᱦᱨᱮ ᱩᱪᱟᱹᱲ ᱪᱮᱛᱟᱱ ᱟᱴᱠᱟᱣ, ᱦᱚᱸ ᱚᱱᱟ ᱛᱟᱹᱨᱤᱠᱷ ᱞᱟᱹᱜᱤᱫ ᱛᱟᱺᱜᱤᱭᱟ, ᱟᱨ ᱚᱱᱟ ᱛᱟᱭᱚᱢ ᱢᱤᱫ ᱠᱚᱢᱤᱴᱤ ᱨᱮᱱᱟᱜ ᱥᱞᱟᱦᱟ ᱪᱮᱛᱟᱱ ᱥᱚᱨᱠᱟᱨᱤ ᱩᱞᱞᱮᱠᱷ ᱞᱟᱹᱜᱤᱫ ᱾ ᱩᱞᱞᱮᱠᱷ ᱵᱟᱹᱱᱩᱜ ᱦᱟᱹᱵᱤᱡ ᱚᱱᱟ ᱩᱯᱼᱱᱤᱭᱚᱢ ᱨᱮ "ᱢᱟᱱᱟᱣ" ᱞᱟᱹᱜᱤᱫ ᱪᱮᱫ ᱦᱚᱸ ᱵᱟᱹᱱᱩᱜ-ᱟ ᱾

## ᱪᱮᱫ ᱟᱴᱠᱟᱣ ᱵᱟᱝ ᱠᱟᱱᱟ

ᱤᱱᱰᱤᱭᱟᱮᱟᱭ ᱠᱚᱢᱯᱭᱩᱴ ᱯᱩᱞ, AIKosh, ᱟᱨ ᱥᱮᱯᱷ ᱮᱱᱰ ᱴᱨᱚᱥᱴᱮᱰ ᱮᱟᱭ (Safe and Trusted AI) ᱯᱨᱚᱡᱮᱠᱴ ᱠᱚ ᱰᱟᱴᱟ ᱨᱩᱠᱷᱤᱭᱟᱹ ᱵᱚᱨᱰ (Data Protection Board) ᱪᱮᱛᱟᱱ ᱵᱟᱝ ᱛᱟᱺᱜᱤ ᱠᱟᱱᱟ ᱾ ᱩᱱᱠᱩᱣᱟᱜ ᱟᱴᱠᱟᱣ, ᱡᱩᱫᱤ ᱛᱟᱦᱮᱸᱱ ᱠᱷᱟᱱ, ᱛᱟᱹᱞᱠᱟᱹᱨᱮ ᱥᱮᱞᱮᱫ ᱟᱨ ᱯᱟᱹᱦᱤᱞ ᱠᱟᱱᱟ, ᱟᱨ ᱚᱜᱚᱥᱴ 2026 ᱦᱟᱹᱵᱤᱡ ᱚᱞ ᱠᱚ ᱯᱩᱞ ᱵᱮᱵᱷᱟᱨ ᱨᱮ ᱩᱫᱩᱜᱟ: 237 ᱯᱨᱚᱡᱮᱠᱴ ᱟᱨ 93.18 ᱞᱟᱠᱷ GPU ᱴᱟᱲᱟᱝ ᱾ ᱥᱤᱧᱚᱛᱤᱭᱟᱹ ᱠᱚᱢᱯᱭᱩᱴᱚᱨ ᱤᱢᱟᱨᱡᱮᱱᱥᱤ ᱨᱮᱥᱯᱚᱱᱥ ᱴᱤᱢ (Indian Computer Emergency Response Team, CERT-In) ᱨᱮᱱᱟᱜ ᱨᱤᱯᱚᱨᱴ ᱦᱚᱸ ᱵᱟᱝ ᱛᱟᱺᱜᱤ ᱠᱟᱱᱟ ᱾

ᱵᱮᱵᱷᱟᱨᱤᱠ ᱵᱷᱩᱞ ᱫᱚ ᱢᱤᱫ ᱫᱚᱞ ᱠᱟᱱᱟ ᱡᱟᱦᱟᱸ 2026 ᱫᱚ ᱚᱱᱟ ᱵᱚᱨᱰ ᱞᱟᱹᱜᱤᱫ ᱜᱮ ᱵᱮᱱᱟᱣᱟ ᱡᱟᱦᱟᱸ ᱱᱤᱛ ᱵᱟᱭᱴᱷᱟᱹᱣ ᱢᱮᱱᱟᱜ-ᱟ ᱢᱮᱱᱛᱮ ᱵᱟᱝ ᱩᱫᱩᱜ ᱟᱠᱟᱱᱟ, ᱟᱨ ᱡᱟᱦᱟᱸ ᱢᱟᱲᱟᱝ ᱠᱷᱚᱱ ᱪᱟᱞᱟᱜ ᱠᱟᱱ CERT-In ᱨᱮᱱᱟᱜ ᱛᱩᱨᱩᱭ ᱴᱟᱲᱟᱝ ᱨᱮᱱᱟᱜ ᱜᱷᱚᱲᱤ ᱫᱚ ᱵᱟᱝ ᱧᱮᱞᱟ ᱾ [ᱛᱷᱟᱯᱚᱱᱤᱭᱟᱹ ᱥᱚᱸᱠᱷᱮᱯ](/blog/ai-and-data-governance-for-founders-and-operators-meity-gov-20260330-710-sat) ᱚᱱᱟ ᱛᱟᱹᱨᱤᱵ ᱩᱞᱴᱟᱹᱭᱟ ᱾

ᱜᱨᱮᱴ ᱤᱱᱰᱤᱭᱟᱱ ᱠᱚᱢᱯᱟᱹᱱᱤ (Great Indian Company) PIB ᱨᱮᱱᱟᱜ ᱚᱞ ᱟᱨ ᱨᱟᱡᱽᱯᱚᱛᱨᱚ ᱢᱤᱫ ᱥᱟᱶᱛᱮ ᱯᱟᱲᱦᱟᱣᱟ ᱾ ᱯᱩᱛᱷᱤ ᱚᱲᱟᱜ [greatindiancompany.com/blog](https://greatindiancompany.com/blog) ᱨᱮ ᱢᱮᱱᱟᱜ-ᱟ ᱾
