# ہِندوستانَن ہِندِن خوراک پروسیسنگ سکیٖمَن ہُند درمیٲنی مدت راستہ

یَتھ شعبَس مَنٛز درمیٲنی مدت چھِ ترٛےٚ سکیٖمَن ہِنٛز زندگی، نَہ زَن کھپتَس مُتعَلِق پانٛژِ وَریُک اندازٕ۔ پردھان منتری کشان سمپدا یوجنا (Pradhan Mantri Kisan Sampada Yojana, PMKSY) چھِ گوڈائے اَکھ لَٹہِ دُبارٕ ترتیب دِنہٕ آمٕژ۔ پردھان منتری فارملَیزیشن آف مایکرو فوڈ پروسیسنگ انٹرپرائزِز (Pradhan Mantri Formalisation of Micro Food Processing Enterprises, PMFME) چھُ پنِنِس کوریج ہدفَس نزدیٖک تہٕ اصل پانٛژِ وَریُکِس ڈیزائنَس مَنٛز لیکھِتھ مالی وَریَس پَتہٕ۔ پروڈکشن لنکڈ اِنسینٹِو (Production Linked Incentive, PLI) چھُ پنِنِس آخری شایع وَریَس مَنٛز۔ 2026-27 بجٹ چھُ ترٛےٚوے تخصیصَس پؠٹھ تھاوان۔ جانشیٖن سکیٖم، نٔوِس ناوَس تہٕ نٔوِس اخراجاتَس سٕتۍ، ٲس تِمَن دستاویزَن مَنٛز نَہ بیان کرنہٕ آمُت۔

## 2016 نِش 2021 تام: اَکھ چھتری، پَتہٕ فروخت اِنسینٹِو

PMKSY آو 3 مئی 2017 جٲری، تِمَن انفراسٹرکچر سکیٖمَن ہِنٛز چھتری پٲٹھۍ یِم وَزارَت گوڈائے چَلاوان ٲس۔ 2024-25 ہِنٛز سالانہ رِپورٹ چھِ وَنان زِ یہِ Rs 6,000 کروڑ سٕتۍ 2016-20 خٲطرٕ منظوٗر اوس، ژۄدَہِم فِنانس کمیشن (14th Finance Commission) سٕتۍ ہم آہنگ، تہٕ پَتہٕ 2020-21 تام وِستار۔ میگا فوڈ پارک (Mega Food Parks) تہٕ کولڈ چین چھِ چھتری کھۄتہٕ پرٲنۍ۔ دۄشوے چھِ وَزارَتِ ہِندِس بیانَس مَنٛز 2008 نِش۔

مارچ 2021 مَنٛز کابینہ کٔر بیاکھ اوزار شٲمِل۔ خوراک پروسیسنگ صنعتَس خٲطرٕ پروڈکشن لنکڈ اِنسینٹِو سکیٖم (Production Linked Incentive Scheme for Food Processing Industry, PLISFPI)، Rs 10,900 کروڑ سٕتۍ، چھُ 2021-22 نِش 2026-27 تام چَلان۔ یہِ چھُ مینوفیکچرنگ پیمانَس تہٕ غیر ملکی ریٹیلَس مَنٛز ہِندوستٲنۍ برانڈَن خٲطرٕ ادائیگی کَران، نَہ زَن کھیت دروازَس پؠٹھ کولڈ سٹورَس خٲطرٕ۔ PMFME اوس جون 2020 مَنٛز شُروٗع گومُت، 2,00,000 مایکرو یونٹَن کُن نشان، Rs 10,000 کروڑ سٕتۍ۔

پَندَہِم فِنانس کمیشن (15th Finance Commission) چکر شُروٗع گژھنہٕ تام وَزارَتَن فیصلہ کوٚر زِ پرٛٮ۪تھ پرٲنۍ جز نَوِس منظوٗری لائق چھُ نَہ۔ میگا فوڈ پارک، پٔتمِت تہٕ برٛونٛہِم ربط (backward and forward linkages)، تہٕ ہُنر تہٕ ہیزرڈ اینالسِس اینڈ کرِٹیکل کنٹرول پوائنٹ (Hazard Analysis and Critical Control Point, HACCP) کام روٗز صرف وعدَن خٲطرٕ۔ کولڈ چین، پروسیسنگ صلاحیت، کلسٹر، ٹیسٹنگ لیبارٹری، تحقیق، تہٕ آپریشن گرینز (Operation Greens) روٗز کھُلٕ۔

## 2025: بٔڑ PMKSY لفافہ، وَنان یہِ چکر اندر

31 جُلَے 2025 کابینہ کٔر 2021-22 نِش 2025-26 تام PMKSY اخراجات Rs 6,520 کروڑ تھاو، اِضافی Rs 1,920 کروڑ شٲمِل۔ تَتھ اِضافَس ہُند حِصہٕ، Rs 1,000 کروڑ، چھُ 50 اِریڈیئیشن یونٹَن تہٕ 100 نیشنل ایکریڈیٹیشن بورڈ فار ٹیسٹنگ اینڈ کیلیبریشن لیبارٹریز (National Accreditation Board for Testing and Calibration Laboratories, NABL) تصدیق شُدہ فوڈ ٹیسٹنگ لیبارٹریو سٕتۍ جوڈِتھ۔ سالانہ رِپورٹُک برٛونٛہہ والٕہ عدد Rs 5,520 کروڑ (Rs 4,600 کروڑ علاوٕ Rs 920 کروڑ) چھُ سُہ حساب یُس تَتھ کابینہ فیصلَہ برٛونٛہہ شایع اوس۔ راستہ چھُ یہِ سکیٖمُک وِستار، نَہ زَن اَتھُک بدل۔

9 جنوری 2026 رِپورٹ کَرنہٕ آمُت جسمٲنی پیش رفت چھُ ذخیرہ ہاوان یُس زِیادٕ تر بناونہٕ آمُت چھُ۔ جزَو شُروٗع گژھنہٕ پَتہٕ منظوٗر 1,618 PMKSY منصوٗبَن مَنٛز 1,185 چالو ٲسۍ، پروسیسنگ تہٕ تحفظ صلاحیت 270.51 لَکھ میٹرک ٹن۔ کولڈ چین باب، 31 دسمبر 2024 تاریخ، چھُ پکِمٕژ کتابَس سٕتۍ ہم آہنگ: 394 منصوٗبَن مَنٛز 286 مُکمل، 108 وَنان نٲفاذَس مَنٛز۔ PMKSY خٲطرٕ درمیٲنی مدت سوال چھُ اثاثَن ہُند مُکمل گژھُن تہٕ استعمال، علاوٕ نٔوِس اِریڈیئیشن تہٕ لیبارٹری ٹُکرَس، پارکَن ہِنٛز تازٕ لہر کھۄتہٕ زِیادٕ۔

## مایکرو یونٹ چھِ عددی ہدفَس نزدیٖک

PMFME ہُک ہدف چھُ 2,00,000 ادارٕ۔ جنوری 2026 ہُک جائزہ چھُ شُروٗعات نِش 1,72,707 کریڈِٹ سٕتۍ جُڑِتھ منظوٗر قرض گٔنٛٹیو، یِمَن مَنٛز 56,543 جنوری 2025 پَتہٕ آیہِ۔ یہِ چھُ کوریج ہدفَس خلاف پیش رفت۔ یہِ چھُ نَہ بیان زِ پرٛٮ۪تھ قرض بچِتھ کاروبار چھُ۔ سکیٖم اوس 2024-25 تام لیکھنہٕ آمُت تہٕ پَتہٕ 2025-26 تام وِستار۔ شُروٗعات نِش جٲری بیج سَرمایہ، 3.76 لَکھ سیلف ہیلپ گروپ (self-help group) ممبرَن خٲطرٕ Rs 1,282.98 کروڑ، چھُ سارِوٕے کھۄتہٕ لۄکٹِن پیداوَرَن خٲطرٕ برابر ٹریک۔

ڈیمانڈ نمبر 45 (Demand No. 45) چھُ وَنان 2026-27 مَنٛز PMFME خٲطرٕ Rs 1,700 کروڑ فراہم کَران۔ یہِ تخصیص چھِ رقم اَکھ وَریہِ تَتھ وِستارَس نیبر جٲری تھاوان یُس جائزہ بیان کَران چھُ۔ یہِ چھُ نَہ، بجٹ نوٹَس مَنٛز، دویِم پانٛژِ وَریُک مشن اعلان کَران۔

## PLI چھُ پنِنِس ادائیگی وَریَن مَنٛز

PLI تَحت رِپورٹ کَرنہٕ آمِژ سَرمایہٕ کاری چھِ Rs 9,702 کروڑ، 170 منظوٗر کیسَن مَنٛز 161 مَنٛز Rs 2,162.553 کروڑ اِنسینٹِو جٲری، تہٕ رِپورٹ کَرنہٕ آمُت روزگار 3.4 لَکھ۔ سکیٖمَس خٲطرٕ 2026-27 بجٹ تخمینہ چھُ Rs 1,200 کروڑ، 2025-26 ہِندِ بجٹ تہٕ نظرثانی شُدہ تخمینَن ہیوٗ۔ سکیٖم صفحَس ہُک مقصد چھُ سُہ یُس 2021 مَنٛز تھاونہٕ آو: تیار کرنہٕ خٲطرٕ تہٕ تیار کھؠنہٕ خٲطرٕ خوراک، پروسیس شُدہ پھَل تہٕ سبزی، سَمندری مصنوعات، تہٕ موزاریلا مَنٛز مینوفیکچرنگ چیمپین، علاوٕ اختراعی تہٕ نامیٲتی مصنوعات تہٕ بیرون ملک برانڈ۔ مِلِٹ جز آو تَتھ اخراجاتَس اندر بچتَہ نِش شٲمِل کرنہٕ۔

2026-27 پَتہٕ شایع PLI کیلنڈر چھُ ختم گژھان۔ ڈیمانڈ نمبر 45 مَنٛز کِہہ تہِ چھُ نَہ خوراک پروسیسنگَس خٲطرٕ دویِم PLI سلسلہ بیان کَران۔

## اَگہِ کینژِن وَریَن مُتعَلِق کِتھ کَن کٔرِو گَژھ

سُہ راستہ یُس دستاویزی چھُ، چھُ یِتھ پٲٹھۍ پَکان۔ انفراسٹرکچر گرانٹ چھِ زِیادٕ تر زمیٖنَس مَنٛز، متعین نامُکمل سیٹ تہٕ نٔوِس لیبارٹری تہٕ اِریڈیئیشن قسطَس سٕتۍ۔ مایکرو ادارہ رسمی بناوُن چھُ 2,00,000 منظوٗر قرضَن ہِندِس نظرَس اندر تہٕ وَنان بینکَن پؠٹھ منحصر۔ بٔڑِن ادارَن ہَن اِنسینٹِوَن چھُ آخری بجٹ وَریہِ۔ قدرِ اِضافہ تہٕ زرعی خوراک برآمدَن ہُند پروسیس شُدہ حِصہٕ چھِ تَتھ دہائی دوران ہیوٚر گومُت یُس وَزارَت حوالہٕ چھِ دِوان، یُس اقتصادی پس منظر چھُ، نَہ زَن سکیٖم ہدف۔ یِمَن ترٛےٚو خٲطرٕ تِہِندِ شایع کِھرَدِن پَتہٕ کِہہ بدل چھُ یِوان، تَتھ مُتعَلِق دعوٕہ چھُ سُہ دعوٕہ یُس بجٹ تہٕ سال اختتامُک جائزَن وَنان کۆر نَہ۔
