# ہِندوستانَس ہِنٛدِ بینک کریڈِٹ اعدادَن پؠٹھ بیٛاکھ کیاہ پَرُن

بیٛاکھ کریڈِٹ سرخی آسہِ زیٛادٕ تر فیصلَن خٲطرٕ غلط اگر یہِ اکی پَرنہٕ یِیہِ۔ یہِ چھُ سُہ ترتیب یَتھ منٛز اَکھ بانی، اَکھ قرضہ دِن وول، اَکھ سرمایہ کار، یا اَکھ پالیسی پَرن وول یہِ ریزرو بینک آف اِنڈیا (Reserve Bank of India, RBI) ہِنٛدِ پنُنِن ریلیزَن خلاف چیک کٔرِتھ ہیکہِ۔ ہیٚٹھُک اعداد چھِ تازٕ لنگر، نَہ پیشن گوئی۔

## 1. ماہانہ شعبہٕ نوٹَس سۭتۍ شروع کٔرِو، تہٕ سلسلُک ناو لِکھِو

کٲم کرن وول ریلیز چھُ «بینک کریڈِٹ ہِنٛز شعبہٕ تعیناتی» (Sectoral Deployment of Bank Credit)۔ تازٕ ریلیز، 31 اَگَست 2026 تاریخ، چھُ **31 جُلایی 2026** ختم گومُت پَندرَہ دۄہ (fortnight) محیط کران۔ غٲر خوراک کریڈِٹ (non-food credit)، تمام شیڈولڈ تجارتی بینکَن (scheduled commercial banks) خٲطرٕ دفعہ 42 ہِنٛدِس ریٹرنَس (Section-42 return) نِش، وُدُیو **19.1%**، اَکھ وَریہِ برٛونٛہہ کِس **9.9%** مقابلہ۔ اَتھ سۭتۍ ژور شعبہ شرح — کٔشی **17.0%**، صَنعت **20.0%**، خٔدمَت **22.9%**، ذٲتی قرضہٕ **16.2%** — چھِ 41 بینکَن تہٕ غٲر خوراک کریڈِٹُک لَگ بَگ 95% نِش یِوان۔

ییلہِ بیٛاکھ مہینہ نظرِ یِیہِ، نظر بننہٕ برٛونٛہہ ژور موازنہ لِکھِو:

- غٲر خوراک کریڈِٹ 19.1% مقابلہ، تہٕ نَوِس نوٹَس منٛز چھاپِتھ پٔتمِس وَریُک پَندرَہ دۄہ مقابلہ۔
- صَنعت ذٲتی قرضَن مقابلہ۔ مارچ پیٹھہٕ جُلایی 2026 تام رُوٗدِ ذٲتی قرضہ 16.2% پؠٹھ ییلہِ زن صَنعت 15.0% پیٹھہٕ 20.0% گَیہِ۔ اگر سُہ فرق بند گَژھِ یا کھُلِ، چکرُک کردار بدلاو۔
- مایکرو تہٕ لۄکُٹ صَنعت بمقابلہ بٔڈِ تہٕ درمیٲنی پؠٹھ صفت۔ جُلایی منٛز ٲسۍ بٔڈِ تہٕ درمیٲنی تیز گومُت تہٕ مایکرو تہٕ لۄکُٹ مستقل۔
- ترٛے ناو دار خٔدمَت سطَر: غٲر بینک فنانس کمپنی، تجارت، تہٕ تجارتی جائیداد (commercial real estate)۔

فُٹ نوٹ تہِ پَرِو۔ 31 دَسَمبَر 2025 پیٹھہٕ چھُ ٲخری رپورٹِنگ پَندرَہ دۄہ مہینُک ٲخری دۄہ۔ وَریَس مقابلہ (year-on-year) شرح چھِ تَتھ تاریخَس اَکھ وَریہِ برٛونٛہہ کِس پرانَس پَندرَہ دۄہَس سۭتۍ موازنہ کران۔

## 2. یہِ بیٛاکھَن زَن سرکٲری شرحَن سۭتۍ اوسط کرنُک انکار کٔرِو

اگر اَکھ مبصر **14.1%** یا **14.5%** 19.1% سۭتۍ نقل کَرِ، فرق چھُ نَہ خودکار پٲٹھۍ ترمیم۔

- **14.1%** چھُ مارچ 2026 ہِنٛدِس اَندَس تام BSR-1 منٛز باقی کریڈِٹ، علاقٲیی دیہی بینک (regional rural banks) شٲمِل، 11.1% پیٹھہٕ زِیادٕ۔ یہِ استعمال کٔرِو قرضہ ہؠن والِس قسمَس، سود شرح بینڈَن، تہٕ آبادی گروپَس خٲطرٕ۔ 9% کھوتہٕ کم قیمتَن وٲل قرضَن ہُنٛد حِصہٕ اوس **64.2%**۔ میٹروپولیٹن استعمال جایَن اوس **58.2%**۔
- **14.5%** چھُ جوٗن مالیاتی استحکام رپورٹَس (Financial Stability Report) منٛز پوٗرٕ 2025-26 خٲطرٕ نگرٲنی کریڈِٹ وُدھُن، **11.5%** جمع وُدھُنَس سۭتۍ۔ یہِ استعمال کٔرِو ییلہِ سوال وَریہِ، مارجِن، یا سرمایہ شرح آسہِ۔

تازٕ رفتارَس خٲطرٕ جُلایی غٲر خوراک شرح نقل کٔرِو۔ کَمِہ قرضہ ہؠو، تَس خٲطرٕ BSR نقل کٔرِو۔ یہِ زِ نظام رفتار چَلاوِتھ ہیکہِ، تَس خٲطرٕ استحکام رپورٹ نقل کٔرِو۔

## 3. معیار استحکام رپورٹ کیلنڈرَس پؠٹھ چیک کٔرِو، نَہ ماہانہ ریلیزَس پؠٹھ

ماہانہ نوٹ چھُ نَہ خراب قرضہ اپ ڈیٹ کران۔ جوٗن 2026 رپورٹَن کُل غٲر کارکردہ اثاثہ شرح (gross non-performing asset ratio) **1.8%**، سلِپیج شرح (slippage ratio) **1.2%**، تہٕ پروویژن کوریج شرح (provisioning coverage ratio) **75.6%** تھاو۔ کٔشی ٲس **5.1%** تہٕ کُل خراب قرضَن ہِنٛز **37.2%**۔ بنیٲدی دباؤ ٹیسٹ (baseline stress test) چھُ 46 بینکَن ہِنٛز خراب قرض شرح مارچ 2028 تام **1.9%** تام نِوان۔ زٕ ناسازگار کیس چھِ یہِ **3.8%** تہٕ **4.1%** تام نِوان۔ بیٛاکھ مالیاتی استحکام رپورٹ چھِ سُہ دستاویز یُس تَتھ وَتھ تصدیٖق کَرِ یا توڑِ۔ اَکھ ماہانہ کریڈِٹ بہتری چھِ نَہ کران۔

انتظارَس دوران چھِ فنڈنگ لکیٖر زِندہ ماہانہ سٲتھی۔ 2025-26 خٲطرٕ 11.5% جمعَن مقابلہ 14.5% پؠٹھ کریڈِٹ چھُ وجہ زِ رپورٹ فنڈنگ اَکھ چیلنج وَنان چھِ۔ اگر بیٛاکھ شعبہٕ ریلیز مضبوط آسہِ تہٕ بُلیٹنَس (Bulletin) منٛز جمع تبصرہ نَہ آسہِ، سُہ فرق چھُ خطرٕ، نَہ وُدھُن شرح۔

## 4. MSME خٲطرٕ الگ جدول استعمال کٔرِو

ترجیحی شعبہ ریٹرن (priority-sector returns)، نَہ صَنعت پیراگراف، چھِ MSME ذخیرہ۔ دَسَمبَر 2025 ہِنٛدِس اَندَس تام اوس باقی MSME کریڈِٹ **₹36.8 لَکھ کَروڑ**، **23.5%** زِیادٕ، **234.7 لَکھ** کھاتَن سۭتۍ — لَگ بَگ اَکھ وَریہِ برٛونٛہہ ہی کھاتہ شمار۔ 1 اپریل 2026 پیٹھہٕ چھِ بغیر ضمانت مایکرو تہٕ لۄکُٹ ادارہ حد (collateral-free micro and small enterprise ceiling) **₹20 لَکھ**، ₹10 لَکھ پیٹھہٕ زِیادٕ۔ اَکھ بانی گَژھِ نَہ اَکھ 20% صَنعت کریڈِٹ سرخی ثبوت مَنُن زِ یہِ حد گوڈَے ذخیرَس منٛز نظرِ چھِ یِوان۔ دَسَمبَر جدول چھُ قٲعدَس برٛونٛہہ، تہٕ جُلایی صَنعت نوٹ چھُ نَہ مایکرو تہٕ لۄکُٹ فیصد شایع کران۔

## 5. قیمتَس خٲطرٕ پالیسی بیان پَرِو، نَہ رفتارَس خٲطرٕ

ریپو شرح (repo rate) چھِ **5.25%**، غیر جانبدار موقفَس سۭتۍ، 3–5 اَگَست 2026 اجلاسَس تام۔ جوٗن 2026 تام ٲسۍ تازٕ روپے قرض شرح **80 بیسِس پوائنٹ** (basis points) کم، فَرؤری 2025 پیٹھہٕ مجموعی کَٹَن ہِنٛدِ **125 بیسِس پوائنٹ** مقابلہ۔ باقی قرض شرح ٲسۍ **91 بیسِس پوائنٹ** کم۔ بیان چھُ وَنان زِ تازٕ شرح پاس تھرو (pass-through) چھُ معتدل گومُت تِکیازِ کریڈِٹ مانگ مضبوط چھِ، تہٕ بینک کریڈِٹ اوس 15 جُلایی 2026 پؠٹھ **17.7%** زِیادٕ۔ اگر بیٛاکھ بیان کریڈِٹ وُنۍ تہِ تیز تہٕ تازٕ شرح پاس تھرو وُنۍ تہِ رُکاوِتھ ہاوِ، قیمت طاقت چھِ قرض مانگَس سۭتۍ۔ اگر تازٕ شرح دُبارٕ کم گَژھُن شروع کَرَن تہٕ کریڈِٹ وارٕ گَژھِ، چکر چھُ آسان چکر، نَہ مانگ چکر۔
