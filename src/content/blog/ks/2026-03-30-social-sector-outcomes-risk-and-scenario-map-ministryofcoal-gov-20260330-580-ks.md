# ہِندوستانَن ہِنٛد سَمٲجی نٔتیٖج خَطرَن ہُنٛد اَکھ منظرنامہ نَقشہ

یِم چھِ نَہ امکانیات۔ یِم چھِ راہ یِم گوڈَے شایع ہِندوستٲنی سلسلَن مَنٛز وُچھنہٕ چھِ یِوان، تہٕ نتیجہ اگر بیٚیُم جٲری پٔتمِس ہِنٛدِس مُشابِہ بۆسِہ۔ اَکھ خَطرہ رجسٹر یُس اَکھ فیصد موقع گَژھان چھُ، سُہ چھُ سُہ رجسٹرَس کھوتہٕ کَم کارآمد یُس سُہ اِشارہ ناو کران چھُ یُس خَطرہ تصدیٖق یا ریٹائر کَرِہ۔

## اگر قَد-عُمر کَمی وَتان روزِہ تہٕ وَزن نَہ

NFHS-6 بدلاو پانٛژھ وَرِیَن بۄن شُرٮ۪ن مَنٛز قَد-عُمر کَمی (stunting) 35.5% پؠٹھ 29.3%، شَدیٖد وَزن-قَد کَمی (severe wasting) 7.7% پؠٹھ 5.2%، وَزن-قَد کَمی (wasting) 19.3% پؠٹھ 19.0%، تہٕ کَم وَزن (underweight) 32.1% پؠٹھ 31.8%۔ اگر بیٚیُم سَروے سُہ نمونہ دُہراوِہ، اَکھ جائزہ یُس صِرف قَد-عُمر کَمی پؠٹھ بَناونہٕ آمُت چھُ سُہ کامیٲبی رپورٹ کَرِہ یِلہِ تَقریٖبَن پانٛژھَن مَنٛز اَکھ شُر وَنان تہِ وَزن-قَد کَم آسِہ تہٕ تَقریٖبَن ترٛےٚ مَنٛز اَکھ وَنان تہِ کَم وَزن آسِہ۔ عملی خَطرہ چھُ اَکھ اضافی پۄشتہِ تہٕ بَڑُن نِگرٲنی نظام یُس سُہ اِشارَس پؠٹھ نمبر چھُ لَگان یُس گوڈَے ہِلِتھ۔

## اگر شیر خوار کھٮ۪ن چھَوان وَنٛژھان روزِہ

جلدی شروع کرُن 41.8% پؠٹھ 50.1% تام بَڑاو۔ 6-23 رٮ۪تَن پؠٹھ کٲفی غِذا 11.0% پؠٹھ 15.3% تام بَڑۍ۔ شےٚ رٮ۪تَن بۄن صِرف ماے دۄد (exclusive breastfeeding) 63.7% پؠٹھ 55.8% تام وَتھ۔ شےٚ رٮ۪تَن بۄن وَنان دۄد چھَوان شُر 95.6% پؠٹھ روز۔ خَطرہ چھُ اَکھ ابلاغی مُہِم، پوشَن ماہ (Poshan Maah) سٕتی، یُس «دۄد چھَوان» کوٹ کَرِہ تہٕ صِرف ماے دۄد سَطر کُنہِ نَہ وُچھِہ۔ تِم چھِ الگ رویہ، تہٕ تِمَن مَنٛز صِرف اَکھ دۄشوَے سَروےَن درمیان خراب گۆو۔

## اگر بوٚڑ رِیاسَت وَنٛژھان روزَن

بِہار (Bihar) ہُنٛز قَد-عُمر کَمی 42.9% پؠٹھ 35.6% تام وَتھ تہٕ امہِ ہِنٛز وَزن-قَد کَمی 22.9% پؠٹھ 19.0% تام۔ مَدھَیَہ پرَدیش (Madhya Pradesh) ہُنٛز قَد-عُمر کَمی 35.7% پؠٹھ 31.4% تام وَتھ، یِلہِ وَزن-قَد کَمی 18.9% پؠٹھ 23.8% تام بَڑۍ تہٕ کَم وَزن 33.0% پؠٹھ 39.7% تام بَڑاو۔ اُتَر پرَدیش (Uttar Pradesh) ہُنٛز قَد-عُمر کَمی 39.7% پؠٹھ 31.5% تام وَتھ، یِلہِ وَزن-قَد کَمی 17.3% پؠٹھ 19.2% تام بَڑۍ تہٕ کَم وَزن 32.1% پؠٹھ 34.5% تام بَڑاو۔ «پۄشتہِ چھِ بہتر گَژھان» ہُنٛز اَکھ قومی تشخیص چھِ سُہ جواب سُہ رِیاسَتَس سوزن یَتہِ وَزن خراب گۆو۔ تصدیٖق کرن وول دَستاوے ز چھُ بیٚیُم رِیاسَت فیکٹ شیٹ، نَہ اَکہِ لَٹہِ اَکھ نٔو قومی اَوسَط۔

## اگر خون کَمی یَتھ سلسلَس مَنٛز غیر شایع روزِہ

29 مَے 2026 پریس نوٹ چھُ نَہ اَکھ قومی خون کَمی (anaemia) پھیلاو دِوان۔ مَے 2026 فیکٹ شیٹَن مَنٛز پۄشتہِ سَطر یِم ہیٚوٚر مُقابَلَن خٲطرٕ اِستعمال گٔیہِ، تِم تہِ چھِ نَہ۔ خَطرہ چھُ اَکھ حکمتِ عملی یا اَکھ مصنوع دعوٕ یُس NFHS-5 خون کَمی یِتھ کَنِہ سلوک کَرِہ یِتھ کَنِہ زِ NFHS-6 اَمِہ تازہ کۆر۔ یَتام زِ وِزارَت اَکھ ناو والٮ۪س سَروےَس مَنٛز اَکھ نٔو نمبر شایع کَرِہ، ایماندار سَطر چھِ یہِ زِ یہِ فیکٹ شیٹ دور چھُ نَہ خون کَمی تازہ کران۔

## اگر دیہی مَزدوٗری منتقلی رَلِہ

1 جُلایی 2026 پؠٹھ گارَنٹی چھِ اَکھ نٔوِ ایکٹَس تَل 125 دۄہ، 2005 ایکٹ چھُ منسوخ، تہٕ نوٹیفائی شُدہ مَزدوٗری چھِ کانٛہہ جایہِ ₹300 بۄن نَہ (قومی اَوسَط ₹327.4، ₹298.8 پؠٹھ)۔ رِیاسَت ہٮ۪کن شَٹھ دۄہَن تام زرعی وقفہ نوٹیفائی کٔرِتھ۔ بے روزگاری الاؤنس تہٕ اَکھ 0.05 فیصد دۄہَوار دیر معاوضہ چھِ لیکھنہٕ آمٕتۍ۔ وِزارَتَن لوک سبھاہس (Lok Sabha) 11 اَگَست 2026 وۆن زِ گوڈنِکِس رٮ۪تَس تَقریٖبَن مَنٛز 9.69 کروڑ کھوتہٕ زِیادٕ نفر-دۄہ پٲدٕ گٔیہِ۔ سُہ اَبتِدٲیی پیداوار چھُ نَہ خَطرہ ریٹائر کران۔ خَطرہ چھُ ظٲہِر گَژھان اگر وقفہ یِتھ کَنِہ نوٹیفائی گَژھَن زِ باقی وَرِیہ گَھیرِن، اگر جاب کارڈ تبدیلی مانگ روکِہ، یا اگر مَزدوٗری پَندَہ دۄہَن نیبر سِلِہ معاوضہ بغٲر یُس ایکٹ بیان چھُ کران۔ دویِم خَطرہ چھُ تجزیٲتی: مالی وَرِیہ 2025-26 MGNREGS کُل نٔوِ سلسلَس پؠٹھ جوڑُن یِتھ کَنِہ زِ قانوٗن بدلاو نَہ۔

## اگر سکوٗل فٲیدٕ سَہوٗلَت تہہ پؠٹھ روزَن

UDISE+ 2025-26 چھُ تیاری تہٕ ثانَوی سکوٗل ترٛاوُن بۄن، ثانَوی برقرار رکاوَٹ 47.2% پؠٹھ 51.9% تام ہیوٚر، تہٕ شاگِرد-اُستاد تَناسُب قومی تَعلیٖمی پالِسی (National Education Policy) ہِنٛدِس 30:1 حَوالَس اندر ہاوان۔ سُہ ٹیبلَس مَنٛز درمیٲنی سکوٗل ترٛاوُن سلسلہ چھُ 3.5% پؠٹھ 3.6% کُن گَژھان۔ اَکہِ اُستاد والؠ سکوٗل چھِ وَنان تہِ 1,00,843۔ خَطرہ چھُ اَکھ سَہوٗلَت مردم شُماری پؠٹھ اَکھ لَرنِنٛگ یا پوٗرٕ گَژھنُک نٔتیٖجہ اعلان کرُن۔ ثانَوی برقرار رکاوَٹ اَدھ قریب چھِ سُہ سطح یُس سُہ خَطرہ ٹھوس چھُ کران۔

## اگر مرکزی پِنشَن شرح تَتہِ روزِہ یَتہِ یہِ 2012 مَنٛز مُقَرر گۆو

21 جُلایی 2026 دیہی تَرقی ہِنٛز وِزارَت (Ministry of Rural Development) وۆن زِ مرکزی NSAP (National Social Assistance Programme) پِنشَن مَدَد 2012 مَنٛز ٲخری لَٹہِ نظرِثانی گۆو، زِ کانٛہہ نظرِثانی غورَس تَل چھِ نَہ، تہٕ زِ رِیاسَتی ٹاپ اَپ ₹50 پؠٹھ ₹5,700 تام چھِ۔ کینٛہہ رِیاسَتَن مَنٛز حٲصِل رقم چھِ اَوسَط تَقریٖبَن ₹1,100۔ خانَدٲنی بَہبوٗدَس خٲطرٕ خَطرہ، تہٕ کانٛہہ تہِ مصنوعَس خٲطرٕ یُس پِنشَن آمدَنی فرض کران چھُ، چھُ اَکھ قومی نمبر اِستعمال کرُن۔ وِزارَتَن یہِ تہِ وۆن زِ جائزَن مزید کٲفی مَدَد سفارش کۆر۔ سفارش چھِ نَہ اَکھ نظرِثانی شُدہ شرح۔

گریٹ اِنڈین کَمپَنی (Great Indian Company) چھِ منظرنامہ اَکھ شایع سیلَس سٕتی جوڑِتھ تھاوان تاکہ اَکھ پَتِم جٲری تِم بند کٔرِتھ ہٮ۪کہِ۔ سُہ درمیٲنی مُدَتی رَخ یَتہِ تِم سیل چھِ بِہِتھ، سُہ چھُ [رَخ نوٹَس](/blog/social-sector-outcomes-mid-term-trajectory-ministryofcoal-gov-20260330-540-ks) مَنٛز۔ فِہرِست چھِ [greatindiancompany.com/blog](https://greatindiancompany.com/blog) پؠٹھ۔
