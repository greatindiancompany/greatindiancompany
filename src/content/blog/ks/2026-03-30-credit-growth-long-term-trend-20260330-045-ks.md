# ہِندوستانُک بینک کریڈِٹ چکر کِتھ کَنِ وارٕ گۆو، پتہٕ تیز

مفید لمُک نظر چھُ ترٛے تاریخ وٲل وقفہ، نَہ اَکھ ہموار لکیٖر۔

## 2023-24 اوس تیز وَریہِ، 2024-25 وارٕ وَریہِ

ہِندوستانَس منٛز بینکاری ہِنٛز رُجحان تہٕ پیش رفت رپورٹ 2024-25 (Report on Trend and Progress of Banking in India 2024-25) چھِ شیڈولڈ تجارتی بینک (scheduled commercial banks) ناپان، علاقٲیی دیہی بینک (regional rural banks) باقِ رکِھتھ۔ تِہنٛد مشترکہ بیلنس شیٹ وُدُیو 2024-25 منٛز **11.2%**، 2023-24 کِس **15.5%** پتہٕ۔ اثاثَن ہِنٛدِس طرفَس پؠٹھ وُدُیو بینک کریڈِٹ **11.5%** تہٕ سرمایہ کاری **9.2%**۔ جمع وُدِیہِ **11.1%**۔

اَکھ انضمام (merger) چھُ تَتھ موازنَس بگاڑان۔ تَس باقِ رکِھتھ وُدُیو بینک کریڈِٹ 2024-25 منٛز **12.5%**، 2023-24 کِس **16.0%** مقابلہ۔ سرمایہ کاری وُدِیہِ 9.9%، 11.6% مقابلہ۔ جمع وُدِیہِ **11.4%**، **13.4%** مقابلہ۔ رپورٹُک پنُن خلاصہ چھُ سادہ ورژن: جمع تہٕ کریڈِٹ وُدِیہِ 2024-25 منٛز وُنۍ تہِ دۄگنہِ ہندسَن منٛز، تہٕ دۄشوَے ٲسۍ پٔتمِس وَریہِ کھوتہٕ وارٕ۔ یہِ وارٕ گَژھُن نظرِ آو بینک گروپَن پؠٹھ۔

سرکٲری شعبُک بینک (public-sector banks) ٲسۍ مارچ 2025 ہِنٛدِس اَندَس تام مشترکہ بیلنس شیٹُک **54.9%**، اَکھ وَریہِ برٛونٛہہ کِس 55.2% پیٹھہٕ کم۔ نجی شعبُک بینک (private-sector banks) ٲسۍ **37.1%**، 37.5% پیٹھہٕ کم۔ غیر ملکی بینکَو، سمال فائنانس بینکَو (small finance banks)، تہٕ پیمنٹس بینکَو (payments banks) حِصہٕ زِیادٕ کۆر۔ یہِ چھُ نظام کَس نِش چھُ، تَتھ منٛز اَکھ وارٕ تبدیٖلی، نَہ نظام ہُنٛد بدلاو۔

## 2025-26 کۆر کریڈِٹ طرف دُبارٕ ہیٚٹھ

2025-26 ہُنٛد سالانہ رپورٹ (Annual Report) چھُ وَنان زِ تجارتی شعبَس خٲطرٕ بینک کریڈِٹ وُدُیو وَریَس مقابلہ (year on year) **15.9%**، اَکھ وَریہِ برٛونٛہہ کِس **10.9%** مقابلہ۔ غٲر بینک ذرایعَن نِش کریڈِٹ وُدُیو **13.3%**، پٔتمِس وَریُک **15.1%** کھوتہٕ وارٕ۔ بینک، نَہ غٲر بینک چینل، ٲسۍ تَتھ وَریہِ تیز ثالث۔ کریڈِٹ وُدھُن گۆو جمع وُدھُنَس برٛونٛہہ، تہٕ کریڈِٹ-جمع شرح (credit-deposit ratio) وُدِیہِ۔ رپورٹ چھُ سمت بیان کران۔ یہِ چھُ نَہ شرح چھاپان۔

جوٗن 2026 ہِنٛز مالیاتی استحکام رپورٹ (Financial Stability Report) چھِ اَمی وَریہِ خٲطرٕ نگرٲنی جوٗرِ دِوان: کریڈِٹ وُدھُن **14.5%**، جمع وُدھُن **11.5%**۔ مدتی جمع (term deposits) رُوٗدِ جاری تہٕ بچت کھاتَن (current and savings accounts) کھوتہٕ برٛونٛہہ۔ سرکٲری شعبُک بینکَو وُنۍ تہِ نجی شعبُک بینکَن کھوتہٕ تیز کریڈِٹ وُدۆو۔ ذٲتی قرضہٕ تہٕ خٔدمَت ٲسۍ وَریُک وُدھُنُک اَہَم حصہ دار، تہٕ ژورَن اَہَم شعبَن منٛز کریڈِٹ وُدھُن اوس مارچ 2026 منٛز زِیادٕ۔

پَندرَہ دۄہَن ہُنٛد غٲر خوراک سلسلہ (fortnightly non-food series) پتہٕ گۆو تَتھ پوٗرٕ وَریُک نگرٲنی شرحَس برٛونٛہہ۔ یہِ اوس 31 مارچ 2026 پؠٹھ **15.9%** تہٕ 31 جُلایی 2026 پؠٹھ **19.1%**۔ جُلایی عدد چھُ 2026-27 سُنٛد۔ یہِ گَژھِ نَہ 2025-26 ہِنٛدِس اوسطَس منٛز واپس لیکھنہٕ۔

## معیار بہتر گۆو وارٕ گَژھنَس تہٕ تیز گَژھنَس دوران

خراب قرضَو وُدھُن چکر اَکھ اَکھ سۭتۍ نَہ لگا۔ رُجحان تہٕ پیش رفت پریس ریلیزَن کُل غٲر کارکردہ اثاثہ شرح (gross non-performing asset ratio) مارچ 2025 ہِنٛدِس اَندَس تام کئی دہٲیی ہُنٛد کم ترین **2.2%** تہٕ سَتَمبَر 2025 ہِنٛدِس اَندَس تام **2.1%** تھاو۔ جوٗن 2026 ہِنٛز استحکام رپورٹَن یہِ مارچ 2026 منٛز **1.8%** تھاو، دُبارٕ کئی دہٲیی ہُنٛد کم ترین بیان، بینک گروپَن پؠٹھ۔ سالانہ سلِپیج شرح (slippage ratio) کم گَیہِ 2025-26 منٛز **1.2%**۔ پروویژن کوریج شرح (provisioning coverage ratio) ٲس **75.6%**۔

سرمایہ گۆو 2024-25 ہِنٛدِس کمی پیٹھہٕ بیٛاکھ طرفَس۔ خطرٕ وزنی اثاثَن ہُنٛد سرمایہ شرح (capital to risk-weighted assets ratio) اوس مارچ 2025 ہِنٛدِس اَندَس تام **17.4%** تہٕ سَتَمبَر 2025 ہِنٛدِس اَندَس تام **17.2%**۔ مارچ 2026 ہِنٛدِس اَندَس تام اوس یہِ **17.7%**، عام ایکویٹی ٹیر 1 (common equity tier 1) **15.3%** پؠٹھ، دۄشوَے کئی دہٲیی ہِنٛد اُچ نکتہ بیان۔ منافع واری رُوٗد وارٕ وَریہِ دوران اُچ: اثاثَن پؠٹھ واپسی (return on assets) **1.4%** تہٕ ایکویٹی پؠٹھ واپسی (return on equity) 2024-25 منٛز **13.5%**، پتہٕ 2025-26 ہِنٛدِ گۆڈنِکِس اَدھَس منٛز **1.3%** تہٕ **12.5%**۔ استحکام رپورٹ چھِ وَنان زِ تِم واپسی ختم گومُتِس پوٗرٕ وَریہِ خٲطرٕ وسیع پٲٹھۍ مستحکم ٲسۍ، سرکٲری شعبُک بینک دویم اَدھَس منٛز تھوڑا مضبوط تہٕ غیر ملکی بینک نرم۔

## یہِ قوس کِس خٲطرٕ چھُ

اَکھ ترٛے وَریُک پَرن وول گَژھِ ترٛے بیان اَکی ساتہ تھاوُن۔ کریڈِٹ وُدھُن ٹھنڈ گۆو 2024-25 منٛز اَکِس واریاہ تیز 2023-24 پیٹھہٕ، انضمام کڈنہٕ پتہٕ تہِ۔ یہِ زِیادٕ گۆو 2025-26 منٛز، تہٕ غٲر خوراک شرح اوس جُلایی 2026 تام مزید تیز۔ قرض معیار تہٕ سرمایہ بہتر گۆو دۄشوَے رفتارَن منٛز گَزران۔ جمع وُدھُنَن تازٕ کریڈِٹ رفتارَس برابر نَہ کۆر، یِم وجہ سۭتۍ کریڈِٹ-جمع شرح وُدِیہِ تہٕ فنڈنگ چھِ استحکام رپورٹَس منٛز اَکھ زِندہ سوال پٲٹھۍ نظرِ یِوان۔

گریٹ اِنڈین کمپنی (Great Indian Company) چھِ یہِ قوس یِم وجہ استعمال کران زِ اَکِس مہینُک 19% چھاپَس اَکھ برٛونٛہہ تہٕ اَکھ پتہٕ آسہِ۔ مہینہ پانَے چھُ [کیاہ بدلاو](/blog/credit-growth-what-changed-20260330-005-ks) منٛز کھولنہٕ آمُت۔ تیز رفتارُک سرمایہ تہٕ خراب قرض ٹیسٹ چھُ [خطرٕ نوٹَس](/blog/credit-growth-risk-watch-20260330-075-ks) منٛز۔
