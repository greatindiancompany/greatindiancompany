# ہِندوستانُک بینک کریڈِٹُک بٔڈِ حِصہٕ چھُ میٹروپولیٹن مرکزَن منٛز استعمال گَژھان

ریزرو بینک آف اِنڈیا (Reserve Bank of India, RBI) سُنٛد مارچ 2026 ہُنٛد کریڈِٹ ریلیز چھُ نَہ پننِس پریس نوٹَس منٛز ریاست درجہ بندی کران۔ یہِ چھُ وَنان زِ قرضہ کَتِ چھُ استعمال گَژھان، جایِ ہِنٛدِس آبادی لحاظِ، تہٕ سُہ کَٹ چھُ سرکٲری علاقٲیی تصویر یۄس پَرن وول اَکھ ریاست جدول ایجاد کَرَن بغیر نقل کٔرِتھ ہیکہِ۔

## ژور آبادی گروپ، اَکھ تاریخ

مارچ 2026 ہُنٛد سالانہ بنیٲدی شماریٲتی ریٹرن (Basic Statistical Return) چھُ باقی کریڈِٹ تَتھ جایِ ہِنٛدِس آبادی لحاظِ درجہ بندی کران یِتِہ کریڈِٹ استعمال چھُ گَژھان، مردم شماری 2011 (Census 2011) استعمال کٔرتھ۔ دیہی (rural) مطلب چھُ 10,000 کھوتہٕ کم۔ نیم شہری (semi-urban) مطلب چھُ 10,000 پیٹھہٕ 1 لَکھ کھوتہٕ کم۔ شہری (urban) مطلب چھُ 1 لَکھ پیٹھہٕ 10 لَکھ کھوتہٕ کم۔ میٹروپولیٹن (metropolitan) مطلب چھُ 10 لَکھ تہٕ اَتھ کھوتہٕ زِیادٕ۔

مارچ 2026 ہِنٛدِس اَندَس تام ٲسۍ حِصہٕ یِتھ پٲٹھۍ:

- دیہی: **9.4%**
- نیم شہری: **14.5%**
- شہری: **17.9%**
- میٹروپولیٹن: **58.2%**

میٹروپولیٹن استعمال جایَن چھُ ذخیرٕچ واضح اکثریت۔ دیہی تہٕ نیم شہری جائے رَلِتھ چھِ ژۄتام کھوتہٕ کم۔ ژورَن گروپَن منٛز پریتھ اَکھ وُدُیو وَریَس مقابلہ (year on year) دۄگنہِ ہندسَن منٛز۔ ریلیز چھُ نَہ پریتھ گروپَس خٲطرٕ الگ وُدھُن شرح چھاپان، تِم وجہ سۭتۍ چھُ نَہ پریس نوٹَس منٛز اَکھ قطعی دیہی بمقابلہ میٹرو ریس۔ پوٗرٕ ریاست، ضلع تہٕ مرکز جدول چھِ BSR اشاعتَس سۭتۍ ہِندوستٲنۍ معیشت ہِنٛدِس ڈیٹابیسَس (Database on Indian Economy) پؠٹھ۔ تام نَہ تِم خانہ پَرنہٕ یِن، اَکھ قومی بریف گَژھِ نَہ سارِوے کھوتہٕ تیز ریاست ناو دِوان۔

ریٹرن چھُ استعمال جایِ پیٹھہٕ پَکان، نَہ بینک کِس لیٹر ہیڈَس پؠٹھ شہرَس پیٹھہٕ۔ اَکھ میٹروَس منٛز درج حد یۄس لۄکُٹ مرکزَس منٛز استعمال گَژھِ، تَس گَژھِ تَتھ مرکزَس سۭتۍ بیٹھُن۔ یہِ چھُ «کریڈِٹ کَتِ چھُ لَگان» خٲطرٕ صحیح تصور۔ یہِ چھُ وُنۍ تہِ 2011 ہِنٛز آبادی نقشہ، تِم وجہ سۭتۍ ہیکَن تِم قصبہ یِم مردم شماری پتہٕ اَکِس حدَس پیٹھہٕ وَدِمُت چھِ، پننِ موجودہ سائزَس کھوتہٕ ہیٚٹھ درجہ بندی گَژھِتھ۔

## سُہ نقشہ کَس چھُ قرضہ دِوان

اَمی ریٹرنَس پؠٹھ وۆت نجی شعبُک بینک کریڈِٹ وُدھُن مارچ 2026 ہِنٛدِس اَندَس تام **12.3%**، اَکھ وَریہِ برٛونٛہہ کِس **9.5%** پیٹھہٕ۔ سرکٲری شعبُک بینک وُدِیہِ تمام شیڈولڈ تجارتی بینکَن، علاقٲیی دیہی بینک (regional rural banks) شٲمِل، ہِنٛدِس 14.1% سرخی کھوتہٕ تیز۔ ریلیز چھُ نَہ ہائی لائٹَس منٛز سرکٲری شعبہ فیصد دِوان۔ سمال فائنانس بینک (small finance banks) رُوٗدِ نظامَس کھوتہٕ تیز وَدان تہٕ تُلِیو پننُن کُل کریڈِٹُک حِصہٕ **1.6%** تام، مارچ 2021 کِس **1.0%** پیٹھہٕ۔ تِہنٛد حِصہٕ چھُ وُنۍ تہِ لۄکُٹ۔ تِہنٛز سمت چھِ صرف میٹرو نظامَس کھوتہٕ اُلٹ، تِکیازِ سمال فائنانس بینک چھِ لۄکُٹ قرضہ ہؠن والَن کُن لائسنس، مگر پریس نوٹ چھُ نَہ تِہنٛد کھاتہ آبادی گروپَس لحاظِ تقسیم کران۔

اَتھ دورَس منٛز اَکھ نۆو ضلع سطح سلسلہ چھُ جنسَس لحاظِ افرادَن خٲطرٕ کریڈِٹ ہاوان۔ قومی سطحَس پؠٹھ ٲسۍ زنانہ **24.7%** افرادَن خٲطرٕ کریڈِٹُک، اَکھ وَریہِ برٛونٛہہ کِس 23.8% مقابلہ۔ ضلع فائل چھُ سُہ جایِ یِتِہ وُچھُن چھُ زِ سُہ حِصہٕ میٹرو حقیقت چھُ یا کھُل حقیقت۔ ہائی لائٹ چھُ صرف قومی عدد۔

## دیہی کریڈِٹ رَسان چھُ دویم سلسلہ

آبادی گروپ حِصہٕ چھِ نَہ کٔشی کریڈِٹ سکیٖمَن ہی چیز۔ فعال کسان کریڈِٹ کارڈ (operative Kisan Credit Cards) کم گَیہِ 2025-26 منٛز **11.7%**، تہٕ سکیٖمَس پؠٹھ باقی رقم کم گَیہِ **4.9%**۔ سالانہ رپورٹ (Annual Report) ہُنٛد جدول چھُ فعال کارڈ **256.2 لَکھ** تہٕ کُل باقی **₹5,71,444 کَروڑ** تھاوان، 2024-25 کِس 290.2 لَکھ کارڈَن تہٕ ₹6,00,975 کَروڑ مقابلہ۔ بینک چھُ وَنان زِ کمی چھِ زیٛادٕ تر کینٛہہ نجی شعبُک بینکَن تِم نمائش دُبارٕ درجہ بندی کرنہٕ نِش یِوان یِم کارڈ ہدایتَن منٛز فِٹ ٲسۍ نَہ۔ فعال کارڈَن ہُنٛد شمار چھُ غٲر کارکردہ کھاتہ (non-performing accounts) تہِ باقِ تھاوان۔ کم KCC ذخیرہ چھُ نَہ، بینک کِس پنُنِس نوٹَس پؠٹھ، ثبوت زِ دیہی استعمال جایَو کریڈِٹ رٔژِیو۔ تِم جائے ٲسۍ وُنۍ تہِ دۄگنہِ ہندسَن والِس قومی وُدھُنَس اندر۔

لیڈ بینک ذمہ داری (lead-bank responsibility) چھِ **778 ضلع** محیط کران، 12 سرکٲری شعبُک بینکَن تہٕ زَن نجی شعبُک بینکَن ذریعہ، جموں تہٕ کشمیر بینک (Jammu & Kashmir Bank) تہٕ ICICI Bank۔ یہِ چھُ ترجیحی شعبہ کریڈِٹ منصوبہ بندی ہِنٛز انتظامی نقشہ۔ یہِ چھُ نَہ وُدھُن درجہ بندی۔

## یہِ غلط ریاست کَتھَس بغیر کِتھ کَنِ استعمال کَرُن

اَکھ سرمایہ کار یا اَکھ ریاستی اہلکار یُس «اَتھ ریاستَس منٛز کریڈِٹ وُدھُن» چھُ یَژھان، تَس گَژھِ BSR ریاست جدول کھولُن۔ سُہ جملہ یُس پریس نوٹ سہارا دِیہِ چھُ تنگ تہٕ وُنۍ تہِ اہم: باقی بینک کریڈِٹُک لَگ بَگ ترٛے پانژھَم حِصہٕ چھُ میٹروپولیٹن مرکزَن منٛز استعمال گَژھان، بیٛاکھ ترٛے گروپ چھِ نَہ سِکُڑان، تہٕ کسان کریڈِٹ کارڈ باقی رقمَن منٛز کمی ہِنٛز بیان کردہ درجہ بندی وجہ چھِ۔ گریٹ اِنڈین کمپنی (Great Indian Company) چھِ ارادَتھ سۭتۍ تَتھ جملَس پؠٹھ رُکان۔ اَکھ بناوِتھ ریاست لیگ جدول آسہِ آسان پٲٹھۍ ژانڈنہٕ تہٕ بھروسَس خٲطرٕ خراب۔

تِم قومی شرح یِمَن سۭتۍ مقٲمی جدول موازنہ گَژھِ کرنہٕ، چھِ [اہم اعدادَن](/blog/credit-growth-key-numbers-20260330-015-ks) منٛز۔ ترجیحی شعبہ تہٕ MSME ذخیرہ یِمَن خلاف لیڈ بینک اصل پٲٹھۍ منصوبہ چھِ بناوان، چھِ [MSME نوٹَس](/blog/credit-growth-impact-on-smes-20260330-025-ks) منٛز۔
