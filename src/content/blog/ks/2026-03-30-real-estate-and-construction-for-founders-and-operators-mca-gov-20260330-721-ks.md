# ہِندوستانَس منز کِنَس یا تعمیٖر برٛونٛہہ پروموٹرَس کیاہ کھولُن چھُ

ہِندوستٲنۍ رہائشَس منز اَکھ بانی چھُ عام طور ترٛےٚ لوٗکَن منز اَکھ: اَکھ پروموٹر (promoter) یُس پلاٹ یا اپارٹمنٹ چھُ کِنان، اَکھ آپریٹر یُس اَکھ پی ایم اے وائی (PMAY) فائل چھُ پورا کَران، یا اَکھ بروکر، یَتھ منز اَکھ ویب سائٹ تہِ چھِ، یُس تِمَن درمیان چھُ بیٹھان۔ جُلایی 2025 ہُند رِیل اسٹیٹ (ضابطہ تہٕ ترقی) ایکٹَس (Real Estate (Regulation and Development) Act) پٮ۪ٹھ [ایم او ایچ یو اے FAQ (MoHUA FAQ)](https://www.mohua.gov.in/static/uploads/2025/12/542144159b61c980253a694f478d3a5d.pdf) چھُ گۄڈنِکِس تہٕ ترٛیِمِس خٲطرٕ لؠکھنہٕ آمُت۔ یہِ چھُ وَنان زِ یہِ رہنمائی چھِ، نہٕ زِ تُہندِس پروجیکٹَس پٮ۪ٹھ حکم۔ ایکٹ چھُ [اِنڈیا کوڈَس (India Code)](https://www.indiacode.nic.in/handle/123456789/2158?locale=en) پٮ۪ٹھ۔ رِیاستی قٲعدٕ چھِ وُنہِ تہِ فارم، فیس، تہٕ سودِ شرح طے کَران۔

## گۄڈنِکِس اشتہارَس برٛونٛہہ

پروجیکٹ اَتھ رِیاستَس ہِندِس رِیل اسٹیٹ ریگولیٹری اتھارٹیَس (Real Estate Regulatory Authority) سۭتۍ رجسٹر کٔرِو برٛونٛہہ زِ تُہۍ اشتہار دِیو، مارکیٹ کٔرِو، بُک کٔرِو، یا کانٛہہ تَس خریدَس خٲطرٕ دعوت دِیو۔ اتھارٹیَس چھِ 30 دوہ۔ مکمل درخواستَس پٮ۪ٹھ خاموشی چھِ ماننہٕ آمُت رجسٹریشن۔ اَکھ نامکمل فائل ہیکہِ واپس سوزنہٕ یِتھ تہٕ، سماعت پتہٕ، مسترد گَژھِتھ۔

تُہۍ چھِو ایکٹَس اندر اگر تُہۍ اَکھ عمارت چھِو بناوان، اَکھ عمارت اپارٹمنٹَن منز بدلاوان، یا زمین پلاٹَن منز کاٹان، فروختَس خٲطرٕ، اَکِس منصوبہ بندی علاقَس اندر۔ دَفہٕ 3(2) منز حد، یُس FAQ «یا» سۭتۍ دُہراوان چھُ، چھُ زمین نیبر تھاوان یُس پانٛژھ ہَتھ مربع میٹر کھوتہٕ زِیادٕ چھُ نہٕ، یا اَکھ پروجیکٹ یُس پرٛٮ۪تھ مرحلہ شمار کٔرِتھ آٹھ اپارٹمنٹ کھوتہٕ زِیادٕ چھُ نہٕ، علاوٕ پروجیکٹ یِمَن ایکٹ شُروٗع گَژھنہٕ برٛونٛہہ گوڈَے تکمیل سرٹیفکیٹ اوس، علاوٕ مرمت یُس کِنٛہہ نٔو چھُ نہٕ الاٹ کَران۔ اَتھ جملَس مرحلہ وار لے آؤٹَس پٮ۪ٹھ وکیلَس ہِندِس راےَس طورَس پٮ۪ٹھ مۂ تھٲوِو۔ رِیاستی اتھارٹیَس پرٛژھِو، تہٕ FAQ ہُند دستبرداری فائلَس منز تھٲوِو۔

اگر سُہ نفر یُس چھُ بنان تہٕ سُہ نفر یُس چھُ کِنان الگ آسَن، زٕوے چھِ پروموٹر تہٕ زٕوے چھِ ذمہ دار۔ اَکھ ترقیاتی اتھارٹی یُس عوامَس چھُ کِنان چھِ تہِ اَکھ پروموٹر۔ اَکھ مشترکہ ترقی پارٹنر چھُ نہٕ تعریفَس نیبر نِوان صرف زمین تھاونہٕ سۭتۍ۔

ایجنٹ، یِمن منز ویب پورٹل تہِ چھِ یِم رجسٹرڈ پروجیکٹ چھِ کِنان، پرٛٮ۪تھ رِیاستَس منز رجسٹر چھِ گَژھان ییتَن تِم چھِ کامہِ کَران۔ اَکھ قومی ویب سائٹ چھُ نہٕ اَکھ رجسٹریشن۔

## رقم، منصوبہ، تہٕ پانٛژھ ؤری ہُند پَتھ

پرٛٮ۪تھ پروجیکٹَس خٲطرٕ اَکھ الگ بینک اکاؤنٹ کھولِو۔ الاٹیَن یُس چھِ ادا کَران تَس ہُند سَتَتھ فیصد جمع کٔرِو۔ یہِ اَتھ پروجیکٹَس ہِندِس زمینِ لاگتَس تہٕ تعمیٖری لاگتَس خٲطرٕ اِستعمال کٔرِو۔ تکمیلَس ہِندِس تناسبَس منز نِکالِو، اَکِس انجینئرَس، اَکِس آرکیٹیکٹَس، تہٕ اَکِس پریکٹسَس منز چارٹرڈ اکاؤنٹنٹَس ہِندِن سرٹیفکیٹَن خلاف۔ یہِ چھُ نہٕ اَکھ ایسکرو (escrow) یُس ریگولیٹرَس چیکَس پٮ۪ٹھ دستخطَس پراران چھُ۔ نظم چھُ سرٹیفکیٹ تہٕ آڈٹ۔ آڈٹ چھُ مالی ؤری پتہٕ شےٚ رؠتَن اندر واجِب، تہٕ اَتھ چھُ ہاونہٕ زِ پروجیکٹ وصولی پروجیکٹَس سۭتۍ روز۔

فروختَس ہِندِس معاہدَس برٛونٛہہ 10 فیصد کھوتہٕ زِیادٕ مۂ نِیو۔ رِیاست چھِ معاہدہ مقرر کَران۔ اَمِس ہِندِس اِضافی شقن ہیکَن نہٕ ایکٹَس خلاف گَژھِتھ۔ تُہۍ چھِو ذمہ دار سُہ خٲطرٕ یُس اشتہار چھُ وَنان۔ غلط تفصیلَس سۭتۍ پٲدٕ گَژھَن واجِن نقصان چھُ دَفہٕ 12 تحت پروموٹر ذمہ داری، یِتھ کَنِ FAQ چھُ یہِ بیان کَران۔

کھُلا پارکنگ مۂ کِنِو۔ یہِ چھُ مشترکہ علاقہ۔ اَکھ گیراج یَتھ چھت تہٕ ترٛےٚ دیوار چھِ ہیکہِ الگ کِننہٕ یِتھ۔ پروجیکٹَس منز کمیونٹی تہٕ تجارتی سہولت چھِ مشترکہ علاقہ تہٕ ایسوسی ایشنَس چھِ گَژھان۔

بٔڑِن منصوبہ تبدیلیَن چھِ مجاز اتھارٹی تہٕ الاٹیَن ہُند زٕ ترٛیہِ حِصہٕ ضرورت، تُہندِس پنننِن اپارٹمنٹَن نیبر، پرٛٮ۪تھ الاٹیَس اَکھ ووٹ۔ اَکھ ہی رضامندی، علاوٕ اتھارٹی، چھِ ضرورت برٛونٛہہ زِ تُہۍ پروجیکٹَس منز پنننِن حقَن ہُند اکثریت منتقل کٔرِو۔ رجسٹریشن چھُ اَتھ تکمیلِ مدتَس تام روزان یُس تُہۍ بیان کۆر۔ فورس میجر (force majeure) فہرستَس نیبر توسیع، یُس قدرتی آفت تہٕ جنگ چھُ، چھُ اَکھ ؤری پٮ۪ٹھ محدود تہٕ قصورِ عدم چھُ ضرورت۔

حوالگی پٮ۪ٹھ، ساختی تہٕ مخصوص عیب چھِ پانٛژھ ؤری تُہندِس سۭتۍ، تہٕ FAQ چھُ وَنان زِ تُہۍ 30 دوہَن اندر درست کٔرِو۔ اگر تُہۍ قبضہٕ تاریخ چھوڑِو، الاٹی ہیکہِ سودَس سۭتۍ واپسی خٲطرٕ نیرِتھ یا روزِتھ تہٕ ماہانہ سود نِتھ۔ شرح چھِ اَتھ شرحَس برابر یُس الاٹی تُہۍ ادا کَرِ اگر تِم دیرِ آسَن۔

دَفہٕ 16 تحت ملکیت تہٕ تعمیٖر بیمہ چھُ صرف تَتھ وِزِ لاگو گَژھان یَلِہ تُہندِ رِیاست یہِ نوٹیفائی کَرِ۔ یہِ فرض مۂ کٔرِو زِ پالیسی گوڈَے لازمی چھِ کیازِ کِہ دَفہٕ موجود چھُ۔

## اگر کام پی ایم اے وائی گَر آسہِ نہٕ زِ نجی فروخت

گاہک چھُ اَکھ اہل خاندان، نہٕ زِ آزاد بازارُک الاٹی۔ [پی ایم اے وائی-یو 2.0 (PMAY-U 2.0)](https://pmaymis.gov.in/PMAYMIS2_2024/faq.html) چھُ اَکھ عمودی ادا کَران۔ فائدہ رَسانَس ہِندِس قیادتَس منز مرکزی مدد چھِ جیو ٹیگ کَرنہٕ آمِتِس پیش رفتَس خلاف 40:40:20 یِوان۔ تُہۍ ہیکِو نہٕ خاندانَس یہِ گرانٹ تہٕ سودِ سبسڈی زٕوے کِنِتھ۔ گَر گَژھِ زنانَس ہِندِس ناوَس منز یا مشترکہ ناوَس منز آسُن، یِتام نہٕ FAQ ہِند استثنا لاگو آسَن۔ خاندانَس چھُ نہٕ پانٛژھ ؤری یہِ کِنُن یا کرایَس پٮ۪ٹھ دِیُن۔ معیار چھُ ترٛیِم فریق مانیٹر وُچھان، تہٕ شکایت چھِ رِیاستَس، سی پی گرامز (CPGRAMS)، تہٕ مشن پورٹلَس ذریہ گَژھان۔

اگر تُہۍ نجی لانچ «بازارَس» خلاف قِیمت چھِو تھاوان، رجسٹریشن قِیمتُک پرنٹ چھُ [ریزرو بینک انڈیکس (RBI index)](https://www.rbi.org.in/Scripts/BS_PressReleaseDisplay.aspx?prid=63436)، نہٕ زِ پورٹل لِسٹنگ۔ نَوَد دوہَن ہِنز گھڑی چھِ [چیک لِسٹَس](/blog/real-estate-and-construction-next-90-days-checklist-mca-gov-20260330-481-ks) منز۔ گریٹ اِنڈین کمپنی چھِ یہِ ڈیسک نوٹ شایَع کَران تاکِ گۄڈنیُک دستاویز یُس پروموٹر کھولِ سُہ رِیاستی رجسٹر آسہِ، نہٕ زِ اَکھ پچ۔
