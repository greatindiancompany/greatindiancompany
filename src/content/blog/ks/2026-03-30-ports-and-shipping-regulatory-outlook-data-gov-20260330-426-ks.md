# اَہَم بَندَرَن، ساحلی جہازَن، تہٕ جہاز گھرَن خٲطرٕ ریگولیٹری نظر

ہِندوستٲنۍ سمندری ریگولیشن چھُ دفترَن ہُنٛد اَکھ سیٹ یِم اَکھ دوٚیمِس ہُنٛد بدل چھِ نَہ بنان۔ اَہَم بَندَرَس پیٹھ ٹیرف تنازعہ، سمندری مزدورَس ہُنٛد سرٹیفکیٹ، ریاستی بَندَرَس پیٹھ جیٹی لائسنس، تہٕ جہاز گھرَس پیٹھ سبسڈی دعوٕ چھِ مُختَلِف ڈیسکَن پیٹھ تران۔ سَتَمبر 2026 پیٹھ نظر چھُ یہِ زِ اَہَم بَندَر ڈیسک چھُ گوڈَے بدلِتھ، تہٕ جہاز گھر ڈیسک چھُ وۄنۍ اَکھ سکیٖم مینوئل، تقریر چھُ نَہ۔

## اَہَم بَندَر: بورڈ چھُ قیمت لگاون، ممبئی منٛز بورڈ چھُ فیصلہٕ کران

اَہَم بَندَر اتھارٹی ایکٹ، 2021 (Major Port Authorities Act, 2021) چھُ 3 نَوَمبر 2021 پیٹھ نافذ۔ اَتھَن اَہَم بَندَر ٹرسٹ ایکٹ، 1963 (Major Port Trusts Act, 1963) منسوخ کوٚر۔ بَندَر اتھارٹی بورڈ چھُ اَہَم بَندَر ریگولیٹ کران، چلاوان، تہٕ منصوٗبہٕ کران۔ سُہ چھُ پننین خدمَتَن خٲطرٕ تہٕ بَندَر اثاثَن تام رسٲیی خٲطرٕ ٹیرف مُقرر کران، زمیٖن سمیت۔ قانوٗن یِم یہِ عملی چھِ بناوان چھِ گوڈَے بَندَر وِنگ (Ports Wing) درج کرنہٕ آمٔتۍ: شرحَن تہٕ فیسَن ہُنٛد پیمانہٕ، ماسٹر پلان تہٕ فنڈَن ہُنٛد اِستعمال، حساب تہٕ آڈِٹ، کارپوریٹ سماجی ذمہ داری (corporate social responsibility)، سنکِنگ فنڈ (sinking fund)، تہٕ چیئرپرسن ہِنٛز شرط۔

فیصلہٕ کرن وول چھُ وۄنۍ اَہَم بَندَرَن ہِنٛز ٹیرف اتھارٹی (Tariff Authority for Major Ports) چھُ نَہ۔ یہِ ادارہ 13 اَگست 2025 ختم گو، ییٚلہِ اَہَم بَندَرَن ہُنٛد فیصلہٕ کن بورڈ (Adjudicatory Board for Major Ports) تشکیل گو۔ بورڈُک صدر افسر، وَزارَت صَفحَس پیٹھ نامزد، چھُ جسٹس آشیش جِتیندر دیسائی (Justice Ashish Jitendra Desai)۔ اَتھُک باقی رُکن چھِ شری چِراووری وِشواناتھ (Shri Chirravuri Viswanath) تہٕ ڈاکٹر سَنجیو رَنجَن (Dr Sanjeev Ranjan)۔ یہِ چھُ ممبئی (Mumbai) منٛز بیٹھِتھ۔ اَتھُک کٲم چھُ اِمتِیاز دارَن (concessionaires) تہٕ کیپٹِو بَرتھ اِستعمال کرن والین سیتہِ تنازعہ، دباوَس تحت منصوٗبَن ہُنٛد جائزہ، اِستعمال کرن والَن ہِنٛز شکایت، تہٕ پرٛون ٹیرف آرڈرَن پیٹھ باقی معاملہ۔ سُہ چھُ نَہ مرکزی ٹیرف مُقرر کرُن واپس انان۔

اَتھ مرکزَس گرد رہنُمٲیی چھِ آسان قانونَن سیتہِ غلط سمجھنہٕ یِوان۔ بَندَر وِنگ چھُ ہَرِت ساگَر (Harit Sagar) سَبز بَندَر رہنُمٲیی (مٔی 2023)، ساگَر آنکَلَن (Sagar Aankalan) کارکردگی رہنُمٲیی، اَکھ بَرتھِنگ پالیسی، زمیٖن پالیسی، بَندَرَس پیٹھ منحصر صنعتَس خٲطرٕ آبہِ کنارٕ پالیسی، تہٕ سَبز ٹَگ منتقلی خٲطرٕ اَکھ معیاری آپریٹنگ نوٹ شایع کران۔ تِم چھِ شکل دِوان زِ بَندَر کِتھہٕ پٲٹھۍ چھُ خرچ کران تہٕ کِتھہٕ پٲٹھۍ درجہٕ چھُ سپدان۔ تِم چھِ نَہ ایکٹ۔

وہے صَفہٕ چھُ 2021 ایکٹَس نِش لگِتھ ہِندوستٲنۍ بَندَر ایکٹ، 2025 (Indian Ports Act, 2025) درج کران۔ غٲر اَہَم بَندَر، تَقریٖباً 200، چھِ تَمہِ صَفحَس پنٕنِس بیانَس پیٹھ ریاستی سمندری بورڈَن یا ریاستی سرکارَن تحت روزان۔ یہِ مختصر چھُ نَہ 2025 ایکٹَس ہِنٛز دفعات خلاصہ کران۔ غٲر اَہَم جیٹی لائسنس دِن وول چھُ وۄنۍ تہِ ریاستی بورڈَس پیٹھ شروع کران۔

## جہاز: ڈائریکٹوریٹ، بَندَر چھُ نَہ

ڈائریکٹوریٹ جنرل آف شِپِنگ (Directorate General of Shipping) چھُ اَکھ منسلک دفتر، سَتَمبر 1949 منٛز بناونہٕ آمُت تہٕ ممبئی منٛز مبنی۔ سُہ چھُ تجارتی جہازرٲنی ہُنٛد انتظامی معاملہ کران۔ ساحلی آپریٹرَس ہُنٛد سرٹیفکیٹ، عملہ، تہٕ جہاز حفاظت چھِ تَتہِ بیٹھِتھ یِمَن وَقتَس تہِ ییٚلہِ مال صرف اَکھ ہِندوستٲنۍ بَندَرَس پیٹھ بیٛاکھَس کُن تران آسہِ۔ جہازرٲنی ڈویژن چھُ ساحلی جہازرٲنی اَکھ موڈ بیان کران یُس سڑک تہٕ ریل پیٹھ بوجھ کمنٛاوِتھ ہیکہِ۔ تَمہِ صَفحَس پیٹھ فی ٹن لاگت جُملہٕ چھُ آبہِ وَتہِ، ریل، تہٕ سڑک اَتھ طریقَس منٛز رَلان یُس چھُ نَہ پَرسان، تہِ یہِ چھُ یَتہِ ریگولیٹری حقیقت پٲٹھۍ اِستعمال کرنہٕ یِوان نَہ۔

اندرونی آبہِ وَتَہِ حوصلہٕ چھِ اَکھ بیٛاکھ آلہ۔ جہازرٲنی ڈویژن چھُ وَنان زِ جَل واہَک (Jal Vahak) سکیٖم، 15 دَسَمبر 2024 اعلان، چھُ اِنلینڈ اینڈ کوسٹَل شِپِنگ لمیٹڈ (Inland and Coastal Shipping Limited) ذریعہ نافذ، شِپِنگ کارپوریشن ہُنٛد ذیلی ادارہ، تہٕ چھُ مال مالکَن 35 فیصد حوصلہٕ دِوان یِم اندرونی آبہِ وَتہِ چھِ اِستعمال کران۔ یہِ حوصلہٕ چھُ نَہ اَکھ اَہَم بَندَر ٹیرف۔

## یارڈ: سکیٖم قانوٗن، ملکی مواد، تہٕ ری سائیکلِنگ قانون

جہاز گھر رقمَس چھُ وۄنۍ اَکھ مینوئل۔ 27 دَسَمبر 2025 پریس اِنفارمیشن بیورو (Press Information Bureau) نوٹ چھُ وَنان زِ 15 پیٹھ 25 فیصد مدد چھِ سنگِ میلَس پیٹھ مبنی، زِ آزاد تشخیص لازمی چھُ، تہٕ زِ جہاز توڑن ہُنٛد کریڈِٹ نوٹ چھُ ہِندوستانَس منٛز ری سائیکلِنگَس خٲطرٕ سکریپ قیمتَس ہُندِس 40 فیصد برابر۔ سکیٖم چھِ 31 مارچ 2036 تام چلان۔ وَزارَتَن چھُ مدد رہنُمٲیی پنٕنِس سائٹَس پیٹھ شایع کوٚرمُت۔ اہلیت چھُ معاہدٕ تاریخَن، جہاز قسمَس، تہٕ تِمَن رہنُمٲیین منٛز ملکی مواد شرطَن پیٹھ منحصر۔ یارڈ گَژھِ بولی ہِنٛز قیمت لگاونہٕ برٛونٛہہ رہنُمٲیی شیڈول پرُن، خلاصہ چھُ نَہ۔

ری سائیکلِنگ چھُ الگ ریگولیٹ سپدان۔ جہازرٲنی ڈویژن چھُ جہاز ری سائیکلِنگ ایکٹ، 2019 (Ship Recycling Act, 2019) کُن تہٕ اَلَنگ-سوسیا (Alang-Sosiya) پیٹھ ہانگ کانگ کنونشن (Hong Kong Convention) تعمیل کُن اشارٕ کران، یِتہِ سُہ چھُ وَنان زِ 131 الاٹ پلاٹَن منٛز 115 یہِ حیثیت پُر چھِ کران۔ سکریپ کرنَس خٲطرٕ کریڈِٹ نوٹ تہٕ مطابِق پلاٹ چھِ پالیسیَس منٛز جوڑِتھ تہٕ چھِ وۄنۍ تہِ ری سائیکلِنگ قانون تہٕ سکیٖم پٲٹھۍ انتظام سپدان۔

انفراسٹرکچر حیثیت، 19 سَتَمبر 2025 بَڑ ہِندوستٲنۍ جَہنڈٕ جہازَن خٲطرٕ تہٕ ہِندوستانَس منٛز بننہٕ آمٔتین چھوٹین جہازَن خٲطرٕ نوٹیفائی، چھُ قرضہٕ دِن والین اِستعمال کرن واجِن مالی درجہٕ بندی۔ یہِ چھُ نَہ ڈائریکٹوریٹ ہُنٛد جہاز سرٹیفکیٹ یا بَندَرَس ہُنٛد بَرتھِنگ قانوٗن بدل کران۔

گریٹ اِنڈین کمپنی (Great Indian Company) چھُ دفتر الگ تھاوان تاکہ تعمیل منصوٗبہٕ صحیح ڈیسک ناو دِیہِ۔ سلسلہٕ چھُ [greatindiancompany.com/blog](https://greatindiancompany.com/blog) پیٹھ۔
