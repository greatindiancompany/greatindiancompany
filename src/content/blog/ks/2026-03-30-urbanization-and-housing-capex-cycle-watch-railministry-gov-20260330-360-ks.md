# شَہری ہِندوستانَس منٛز سرمایہٕ خرچِ ہِنٛزۍ ترٛےٚ گَڑۍ

## سِکیٖمہٕ اَکِہ چارٹَس پؠٹھ رَلاونہٕ سۭتۍ چھُ چکر غلط بیان گَژھان

شَہری ہِندوستانَس منٛز سرمایہٕ خرچ چھُ آسٲنی سۭتۍ بڑاوِتھ وَننہٕ یِوان تِکیازِ رہٲیِشی تہٕ شَہری مَعامِلاتَن ہِنٛز وَزارَت (Ministry of Housing and Urban Affairs, MoHUA) چھِ مِشَنَن خٲطرٕ بٔڑ رۄپَے عَدَد شایع کَران یِم الگ طٔریٖقَن خرچ کَران چھِ۔ اَکھ گَریلوٗ اِمداد، اَکھ آب صاف کَرنُک پلانٹ، تہٕ اَکھ مُکَمَل ہوشیار سَڑک چھِ نَہ اَکِہ ؤریہِ عروجَس پؠٹھ واتان۔ شایع گَڑۍ چھِ یہِ۔

## گَڑی اَکھ: ہوشیار شَہر، مرکزَس پؠٹھ بند

مِشَن آو 25 جوٗن 2015 100 شَہرَن خٲطرٕ شُروٗع کَرنہٕ، پرٛٮ۪تھ شَہرِ کٮ۪ن چُنِمٕتٮ۪ن حِصن منٛز ریٹروفِٹِنگ، دُبارٕ تعمیٖر، تہٕ گرین فیلڈ کٲمِ ہِنٛدِس رَقبہ مَبنایِ ماڈَلَس پؠٹھ، نَہ زَن پوٗرٕ شَہر دُبارٕ بَناونَس۔ MoHUA ہُنٛد صَفہٕ چھُ مرکزی خرچ ₹48,000 کَروڑ تھاوان۔ مالی اِختِتام اوس 31 مارٕچ 2025۔ شَہرَن اوس ₹47,458 کَروڑ دعوٕے کۆرمُت، یُس صَفہٕ مرکزی حِصَس ہُنٛد 99 فیصَد وَنان چھُ۔ صَفہٕ چھُ یہِ تہِ وَنان زِ مَزیٖد بَجَٹی پروویجَن چھُنہٕ، تہٕ زِ کانٛہہ نٔو شَہر گَژھِہ نَہ رَٹنہٕ یِنِہ۔

6 اَگَست 2026 ہُنٛد لوک سَبھا (Lok Sabha) جَواب چھُ کٲمُک ذخیرٕ دِوان یِتھ کَن 31 مارٕچ 2026 تام رپورٹ سپد: 8,064 مَنصوٗبَہ، قۭمَت ₹1,64,811 کَروڑ، رَٹنہٕ، 7,790، قۭمَت ₹1,56,257 کَروڑ، مُکَمَل، تہٕ ₹46,377 کَروڑ مرکزی مَدَد اِستِمال۔ اصل منظوٗری ٲس صِرف 31 مارٕچ 2020 تام۔ ژور تَوسیعَن وَوت آخری مُکَمَلی ہَدف 31 مارٕچ 2025 تام۔ سُہ تٲریٖخَس پَتہٕ خرچ، یَتہِ سُہ آسِہ، چھُ رِیاسَت، شَہر، یا خاس مَقصَدٕچ کمپنی (special purpose vehicle) ہُنٛد زَر گوڈَے رَٹنہٕ آمٕتٮ۪ن مَنصوٗبَن پؠٹھ۔ سُہ چھُنہٕ اَکھ نٔو مرکزی سرمایہٕ خرچُک چکر۔

سُہ کِتابَس اندر ٲسۍ عَوامی-نِجی شِراکَت (public-private partnership) مَنصوٗبَہ 197، قۭمَت ₹9,277 کَروڑ، یِمَن منٛز 194، قۭمَت ₹9,190 کَروڑ، مُکَمَل۔ شِراکَت سَرمایہٕ اوس اصلی تہٕ ₹1,64,811 کَروڑ مَنصوٗبہٕ کِتابَس نسبت لَکوٗٹ۔

## گَڑی زٕ: AMRUT 2.0، اَکھ پانٛژھ ؤری ہِدایَت یۄس پَنُن آخری ؤری ناو چھِ دِوان

گۄڈنیُک AMRUT مِشَن (Atal Mission for Rejuvenation and Urban Transformation)، جوٗن 2015 منٛز شُروٗع، ٲس 500 شَہر محیط۔ پریٚس اِطلاع بیورو (Press Information Bureau, PIB) ہُنٛد 27 اکتوٗبَر 2021 ہُنٛد ہِدایَت جٲری چھُ AMRUT 2.0 ہُنٛد اِشارٲتی خرچ ₹2,99,000 کَروڑ تھاوان، شٲمِل ₹22,000 کَروڑ گۄڈنیُک مِشَن جٲری تھاونَس، مالی ؤری 2021-22 پؠٹھ 2025-26 تام۔ مرکزی حِصہٕ چھُ ₹86,760 کَروڑ، شٲمِل ₹10,000 کَروڑ یِمَن جٲری مَنصوٗبَن خٲطرٕ۔ فنڈ چھِ 20:40:40 جٲری گَژھان۔ ترٛیِم ؤریہِ ہُنٛد جٲری چھُ جائیداد ٹیکسَس تہٕ صٲرِف چارج اِصلاحَن پؠٹھ منحصر۔

جسمٲنی چیزَہ چھِ 4,700 قانوٗنی قَصبَن (statutory towns) منٛز نَلکہٕ تہٕ 500 AMRUT شَہرَن منٛز سیوریج یا سیپٹیج۔ تجویٖز جوڑ چھِ 2.68 کَروڑ نَلکہٕ تہٕ 2.64 کَروڑ سیوَر یا سیپٹیج جوڑ۔ ملین-پُلس شَہرَن (million-plus cities) چھُ پَننٮ۪ن مختص فنڈَن ہُنٛد 10 فیصَد عَوامی-نِجی شِراکَتَس ذَریعہٕ تھاوُن۔ شَہر چھِ اَکھ سِٹی واٹَر بیلَنس پلان (City Water Balance Plan) تہٕ اَکھ سِٹی واٹَر ایکشَن پلان (City Water Action Plan) داخل کَران۔ رِیاسَتہِ چھِ تِم اَکِہ سٹیٹ واٹَر ایکشَن پلانَس (State Water Action Plan) منٛز جمع کَران۔

مالی ؤری 2025-26 گوٚو 31 مارٕچ 2026 خَتم۔ یہِ مُختصَر چھُنہٕ کانٛہہ اَمہِ پَتہٕ خرچ اعلان کَران۔ یوتام MoHUA یا PIB کانٛہہ شایع کَرِ نَہ، سرمایہٕ خرچُک قیاس چھُ ہِدایَتِ ہِنٛدِس آخری ؤریہِ پؠٹھ رُکان، جمع سُہ مُکَمَلی رُکاوَٹ یُس رِیاسَتی منصوٗبَہ وِنہِ تہِ تھاوان چھِ۔

## گَڑی ترٛےٚ: PMAY چھِ گۄڈَے مَدَد، پَتہٕ سامان، پَتہٕ کٲم

دَسَمبَر 2024 ہُنٛد PIB نوٹ چھُ گۄڈنیُک PMAY-شَہری (Pradhan Mantri Awas Yojana – Urban) خلاف ₹8,07,393 کَروڑ سَرمایہٕ کاری رپورٹ کَران، تہٕ 9 اَگَست 2024 منظوٗر 2.0 سِکیٖمَس خٲطرٕ حُکوٗمَتی مَدَد ₹2.30 لاکھ کَروڑ، سَرمایہٕ کاری ہُنٛد عَدَد ₹10 لاکھ کَروڑ سۭتۍ۔ شِراکَت تہٕ فٲیدٕ گَژھَن وٲلٮ۪ن ہِنٛدٮ۪ن تعمیٖرَس تحت فی یونِٹ مَدَد چھِ ₹2.50 لاکھ۔ یہِ زَر چھُ اَکِہ اِہل گَریلوٗ خٲطرٕ اَکِہ گَرَس سۭتۍ جوڑِتھ۔ سُہ چھُ سیٖمینٛٹ، سٹیل، تہٕ سایِٹ مَزدوٗری بَنان یَلِہ گَر بَنان چھُ۔ سُہے نوٹ چھُ گۄڈنیُک سِکیٖمَس تحت 570 لاکھ میٹرک ٹَن سیٖمینٛٹ، 130 لاکھ میٹرک ٹَن سٹیل، تہٕ 917 کَروڑ نفر-دۄہ اندازٕ کَران، نوٹَس وِزِ «وۄنۍ تام» عَدَد کِس پٲٹھۍ۔ یہِ چھِ اِستِمال اندازٕ اَکِہ تٲریٖخ وٲلِس گَرَن ہِنٛدِس ذخیرٕ سۭتۍ جوڑِتھ، نَہ زَن 2026-27 خٲطرٕ آرڈَر کِتاب۔

سودِ ہِنٛز اِمدادی سِکیٖم (Interest Subsidy Scheme) چھِ قَرضَس ہِنٛز قۭمَت کَم کَران۔ سۄ چھِ نَہ میوٗنِسِپَل سرمایہٕ بَجَٹَس منٛز سَطَر۔

## کِیاہ وُچھُن

اَگَست 2026 ضمیمَس منٛز باقی ہوشیار شَہر مَنصوٗبَہ وُچھِو، نَہ زَن اَکھ تازٕ مرکزی منظوٗری۔ وُچھِو زِ AMRUT اِصلاح شَرط چھِ سُہ کِیاہ یُس رِیاسَت پیش کَران چھِ یَلِہ اَکھ پَتِم قِسط پَکان چھِ۔ PMAY وُچھِو عمودی حِصہٕ وار منظوٗری تہٕ مُکَمَلی ذَریعہٕ، تٲریٖخ سۭتۍ، نَہ زَن اَکِہ رَلِتھ «شَہری سرمایہٕ خرچ» سِلسِلَس ذَریعہٕ۔
