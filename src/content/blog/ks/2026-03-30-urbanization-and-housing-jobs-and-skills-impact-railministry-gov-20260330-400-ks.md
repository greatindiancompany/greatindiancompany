# شَہری رہٲیِشَس منٛز نوکری: واحد سرکٲری کٲمُک عَدَد، تہٕ اَمِچ تٲریٖخ

## رہٲیِشی وَزارَتَن کِیاہ گٔنِتھ

رہٲیِشی تہٕ شَہری مَعامِلاتَن ہِنٛزِ وَزارَتَھ (Ministry of Housing and Urban Affairs, MoHUA) چھُنہٕ، یِمَن جٲری کَرنَن منٛز یِم یَتہِ اِستِمال چھِ، شَہری تعمیٖرِ ہُنٛد کانٛہہ روزگار سَروے شایع کۆرمُت۔ قومی شماریٲتی دَفتَرَس (National Statistical Office, NSO) ہُنٛد وَقتی مَزدوٗر قوٗوَتُک سَروے (Periodic Labour Force Survey, PLFS) چھُ کٲم تہٕ بے روزگٲری ہُنٛد سرکٲری گَریلوٗ سَروے، شَہری تہٕ دیہی وَنڈَس سۭتۍ۔ یہِ مُختصَر چھُنہٕ سُہ سَروَس پؠٹھ کانٛہہ دَرَج تُلان۔ اَکِہ دَرَجَس چھُ مَخصوٗص PLFS ترٛیِم یا ؤری وار رپورٹ اَتھَس منٛز چھُ، تہٕ اَکھ رہٲیِشی نوٹ یُس جدول ورٲے اَکھ دَرَج اُدھار رَٹان چھُ، سُہ چھُ طٔریٖقہٕ یُس خراب عَدَد پھیلان چھُ۔

عَدَد یُس رہٲیِشی ریکارڈَس منٛز چھُ، سُہ چھُ PMAY-شَہری (Pradhan Mantri Awas Yojana – Urban) پؠٹھ دَسَمبَر 2024 ہِنٛدِس پریٚس اِطلاع بیورو (Press Information Bureau, PIB) پسِ مَنظَر نوٹَس منٛز۔ 1.18 کَروڑ گَرَن ہِنٛزِ منظوٗری تہٕ 88.32 لاکھ مُکَمَل یا حَوالہٕ سۭتۍ چھُ نوٹ وَنان زِ تعمیٖرَس منٛز اندازٕ 917 کَروڑ نفر-دۄہ کٲم پٲدٕ گٔیہِ، اندازٕ 570 لاکھ میٹرک ٹَن سیٖمینٛٹ تہٕ 130 لاکھ میٹرک ٹَن سٹیل سۭتۍ۔ نوٹ چھُ یہِ سَبُے پٔتمٮ۪ن اَٹھ پؠٹھ نَو ؤریَن ہُنٛد تَجرُبہٕ وَنان۔ یہِ چھُ سُہ گَرَن ہِنٛدِس ذخیرٕ سۭتۍ جوڑِتھ اَکھ اندازٕ۔ یہِ چھُنہٕ مَہوار تنخواہ فِہرِست، تہٕ یہِ چھُنہٕ PMAY-شَہری 2.0 (PMAY-Urban 2.0) خٲطرٕ اندازٕ۔

سُہے نوٹَس ہُنٛد عمودی وَنڈ چھُ ہاوان زِ کٲم کَتِہ ٲس: لگ بگ 75 لاکھ گَر فٲیدٕ گَژھَن وٲلٮ۪ن ہِنٛدٮ۪ن نِگرٲنی منٛز تعمیٖر (Beneficiary Led Construction) تحت، لگ بگ 18 لاکھ شِراکَت تہٕ جایہِ پؠٹھ کَچھی بَستی ہِنٛدِس دُبارٕ اِنتِظامَس تحت، تہٕ لگ بگ 25 لاکھ قَرضَس سۭتۍ جوڑِتھ اِمدادی سِکیٖمَس (Credit Linked Subsidy Scheme) تحت۔ فٲیدٕ گَژھَن وٲلٮ۪ن ہِنٛدۍ گَر چھِ گَریلوٗ ذَریعہٕ یا گَریلوٗ خٲطرٕ بَنان۔ شِراکَت گَر چھِ مَنصوٗبَن کِس پٲٹھۍ بَنان۔ اِمدادی گَر چھِ قَرضَس سۭتۍ ہیٚننہٕ یِوان۔ مَزدوری ہُنٛد مواد چھُنہٕ اَکھ، تہٕ نوٹ چھُنہٕ 917 کَروڑ نفر-دۄہ ترٛےٚن پؠٹھ وَنڈان۔

## اَمہِ پَتہٕ سِکیٖم کَسن ناو چھِ دِوان، تِمَن نوکری دِنہٕ ورٲے

PMAY-شَہری 2.0، سُہ نوٹَس منٛز، چھُ صَفائی کَرمچاریَن (safai karamcharis)، پی ایم سَوانِدھی (PM SVANidhi) تحت شناخْت کَرنہٕ آمٕتٮ۪ن سَڑکَن ہِنٛدٮ۪ن وِکرٮ۪ن، پی ایم وِشوَکَرما (PM Vishwakarma) تحت کاریگَرَن، اَنگَنوٲڑی (anganwadi) کٲم کَران وٲلٮ۪ن، عِمارَت تہٕ باقی تعمیٖری مَزدورَن، تہٕ کَچھی بَستیَن تہٕ چاؤلَن (chawls) ہِنٛدٮ۪ن باشندَن خاص توجہٕ دِوان۔ اَکھ گروپ فٲیدٕ گَژھَن وولَس ہِنٛدِس مرکز کِس پٲٹھۍ ناو دِیُن چھُ اہلیتُک فیصلہٕ۔ سُہ چھُنہٕ کانٛہہ تربیتِ ہُنٛد نشست یا مُزدوری شایع کَران۔ گۄڈنیُک سِکیٖمَس تحت لگ بگ 89 لاکھ گَر ٲسۍ زنانَن ہِنٛدِس ناؤَس پؠٹھ یا مُشتَرِکہٕ مِلکیَتَس منٛز۔ یہِ چھُ مِلکیَتُک نَتیٖجہٕ، یُس اَتھ باتَس خٲطرٕ اہم چھُ زِ اثاثہٕ کَس اَتھَس منٛز چھُ، تہٕ سُہ چھُ وِنہِ تہِ نَہ نوکریَن ہُنٛد شُمار۔

جُلاے 2025 ہُنٛد PIB جَواب چھُ یِم اِدارٕ بیان کَران یِمَن بند اِمداد سنبھال: قومی رہٲیِشی بینٛک (National Housing Bank)، سٹیٹ بینٛک آف اِنڈیا (State Bank of India)، رہٲیِشی تہٕ شَہری ترقی کارپوریشَن (Housing and Urban Development Corporation)، تہٕ پَتہٕ بینٛک تہٕ رہٲیِشی مالیات کَمپَنی۔ سودِ ہِنٛز اِمدادی سِکیٖم (Interest Subsidy Scheme) چھِ قَرضہٕ دیٖن وٲلٮ۪ن ذَریعہٕ چَلان۔ یِمَن فَرمَن اندر روزگارُک اَثَر چھُنہٕ رپورٹ سپدان۔

## ہُنَر یِم باقی مِشَن اصلَن بیان چھِ کَران

AMRUT 2.0 (Atal Mission for Rejuvenation and Urban Transformation 2.0) ہِدایَتَن ہُنٛد 27 اکتوٗبَر 2021 ہُنٛد جٲری چھُ اَکھ صلاحیَت بَناونُک حِصہٕ شٲمِل، یَمہِ سُنٛد سامعین فِہرِست چھُ: چُنِمٕتۍ نمایِندٕ، میوٗنِسِپَل عملہٕ، ٹھیکیدار، مینیجَر، پلمبَر، پلانٹ آپریٹَر، مَزدور، مشیر، طٲلِبِ عِلم، زنانَہ، تہٕ شہری۔ مِشَن چھُ یہِ تہِ وَنان زِ سُہ زنانَہ تہٕ نوجوان ہم وَخت رائے خٲطرٕ گِگ اِقتِصاد (gig-economy) ماڈَلَس تحت شٲمِل کَرِ۔ زٕنہٕ سَطَر چھِ تربیت تہٕ رائے ہُنٛد ڈیزایِن بیان کَران۔ کانٛہہ سَطَر چھُنہٕ وَنان زِ کیتَن لوٗکَن تربیت سپٕز۔

ہوشیار شَہرَن ہُنٛز کٲم، MoHUA ہِنٛدِس مِشَن صَفَس پؠٹھ، چھِ تَمام 100 شَہرَن منٛز کمانڈ سینٹَر، 31 شَہرَن منٛز آبَن ہِنٛدٮ۪ن نیٹ وَرکَن ہُنٛد SCADA نِگرٲنی، تہٕ 48 شَہرَن منٛز RFID لَگِتھ کوڑٕ گاڑۍ شٲمِل۔ یہِ نِظام چَلاوُن چھُ میوٗنِسِپَل ہُنَر۔ صَفہٕ چھُ اثاثہٕ تہٕ کَوریج رپورٹ کَران۔ سُہ چھُنہٕ آپریٹَرَن ہُنٛد سَر شُمار رپورٹ کَران۔ 31 مارٕچ 2025 مالی اِختِتامَس پَتہٕ چھُ 10 جوٗن 2025 ہُنٛد مَشوَرٕ نَمبَر 27 رِیاسَتَن پرژھان زِ خاس مَقصَدٕچ کمپَنی (special purpose vehicles) وِنہِ تہِ کِیاہ کَران۔ عملہٕ چھُ سُہ رِیاسَتی فیصلَس پَتہٕ یِوان۔

مَردَم شُمٲری 2027 (Census 2027) چھِ اَکھ بٔڑ عارضی گِنَتی عملہٕ نوکری دِوان، تہٕ PIB چھُ کیلنڈَر تہٕ ₹11,718.24 کَروڑ خرچ شایع کۆرمُت۔ مَرحَلہ I ہُنٛد رہٲیِشی مواد چھُ گَرٕچ حالَت تہٕ اَمۍ سہوٗلَت۔ گِنن وٲلٮ۪ن ہُنٛد شُمار چھُ مَردَم شُمٲری ہُنٛز کارروائی، نَہ زَن MoHUA رہٲیِشی سِکیٖمِ ہِنٛز نوکری، تہٕ یہِ مُختصَر چھُنہٕ سُہ اندازٕ کَران۔

## اَکھ قابِلِ اِستِمال پَٹ

917 کَروڑ نفر-دۄہ اِستِمال کٔرِو گۄڈنیُک PMAY-شَہری پؠٹھ تعمیٖری کٲمِ ہُنٛد تٲریٖخ وول وَزارَتی اندازٕ کِس پٲٹھۍ، تہٕ تَتہِ رُکِو۔ AMRUT سامعین فِہرِست اِستِمال کٔرِو آب تہٕ سیوریج کارروائی خٲطرٕ نوکری نقشہٕ کِس پٲٹھۍ۔ کمانڈ سینٹَر اثاثہٕ فِہرِست اِستِمال کٔرِو یِمَن نِظامَن ہُنٛد نقشہٕ کِس پٲٹھۍ یِمَن آپریٹَر چھِ، مِشَن بَجَٹ بند گَژھنہٕ پَتہٕ۔ مُزدوری، خٲلی جاے، یا شَہری بے روزگٲری ہِنٛدِس دَرَجَس خٲطرٕ PLFS جدولَس یا رِیاسَتی بھرتی نوٹِسَس کَھسِو، تہٕ سُہ دَستاویٖز مُدتَس سۭتۍ حوالہٕ کٔرِو۔
