# ہِندوستانٕکِس تِجارَتی خَسارَس، زَخیرَن تہٕ سرمایہٕ ہُندِس بہاوَس مَنٛز کیاہ وُچھُن

وُچھنَس لٲیِق بیرونی خطرٕ چھُ مخصوص۔ مالُک خَسارَہ چھُ اَکھ ؤری برٛونٛہہ کھۄتہٕ کھُلہٕ۔ پورٹ فولیو سَرمایَہ کٲری نؠر جوٗن سہ ماہیَس مَنٛز۔ مُلٕہ بَندِی کٔر زَخیرٕ جٲری کھاتَہٕ کھۄتہٕ زیادٕ بدلاو۔ جُلایی ہُند غَیر مُقیم جمع اندرُن ٲس اَتھ قَدَر بوڈ زِ رؠتہٕ وار مالِیاتی تصویرَس پٮ۪ٹھ غالِب سپُد، تہٕ یہِ چھُ وُنہِ اِبتِدٲیی نِشان والہٕ۔ یِمَن مَنٛز کانٛہہ چھُ نہٕ پیشنگوئی۔ پرٮ۪تھ اَکہِس چھُ اَکھ شایع سطح تہٕ اَکھ دستاویز یُس یہِ اپڈیٹ کَرِ۔

## مالُک خَسارَہ چھُ دۄشوَے گھَڑیَن پٮ۪ٹھ کھُلہٕ

ریزَرو بَینک آف اِنڈیا (Reserve Bank of India, RBI) گھڑیَس پٮ۪ٹھ ٲس اپریل–جوٗن 2026 ہُند مالِ خَسارَہ US$86.1 billion، اَکھ ؤری برٛونٛہہ US$68.9 billion خلاف۔ جُلایی ہُند مالِ خَسارَہ ٲس US$31.7 billion، US$28.2 billion خلاف۔ اپریل–جُلایی ٲس US$117.8 billion، US$97.1 billion خلاف۔

کَسٹَمز گھڑیَس پٮ۪ٹھ چھُ وَزارَتِ تِجارَت تہٕ صَنعَت (Ministry of Commerce and Industry) اپریل–جُلایی مالِ خَسارَہ US$118.60 billion تھاوان، US$96.66 billion خلاف۔ ڈائریکٹوریٹ جَنرَل آف کَمرشَل اِنٹَلیجَنس اینڈ سَٹیٹِسٹِکس (Directorate General of Commercial Intelligence and Statistics, DGCIS) عارِضی فوری اندازٕ چھِ اپریل–اَگَست بَرآمَد US$215,912.45 million تہٕ دَرآمَد US$363,000.93 million تھاوان۔ تِمَن دۄن سَطرَن درمیان خَسارَہ چھُ US$147,088.48 million۔ اَکھ ؤری برٛونٛہہ ٲس تِمَن دۄن کالمن ہُند فرق US$123,879.93 million۔ اَگَست پانےِ ٲس مضبوط بَرآمَد رؠتہٕ، 26.12 per cent بَڑِتھ، دَرآمَد 14.07 per cent بَڑِتھ۔ اَکہِ مضبوط رؠتھَن کٔر نہٕ پٲنٛژھ رؠتَن ہُند کَسٹَمز خَسارَہ پَتمِس ؤریَس کھۄتہٕ بون۔

یہِ وُچھُن کیاہ نرم کَرِ: اَکھ RBI رؠتہٕ یَتھ مَنٛز مالُک خَسارَہ اَکھ ؤری برٛونٛہہ بُنیادَس پٮ۪ٹھ تنگ سپدِ، یا اَکھ وَزارَتِ تِجارَت شایع یَتھ مَنٛز ؤری تام مالِ خَسارَہ کھُلہٕ گَژھُن رُکِ۔ یہِ کیاہ تیز کَرِ: بیاکھ سہ ماہی یَتھ مَنٛز مالِ دَرآمَد مالِ بَرآمَد تہٕ خِدمَتو ہُند فاضِل دۄشوَے پَتھ کھۄتہٕ برٛونٛہہ گَژھِ۔

خِدمَت آفسِٹ چھُ وُنہِ بوڈ۔ خالص خِدمَتی رَسید ٲسۍ اپریل–جوٗنَس مَنٛز US$51.6 billion تہٕ اپریل–جُلاییَس مَنٛز US$69.3 billion۔ اپریل–جوٗن خٲطرٕ جٲری کھاتَہٕ خَسارَہ ٲس مجموعی گھریلو پیداوار (gross domestic product, GDP) ہُند 0.5 per cent۔ خطرٕ چھُ اَکھ مالُک خَسارَہ یُس تَتھ آفسِٹَس کھۄتہٕ تیز بَڑِ، نہٕ مالِ خَسارَہ ہُند وجود۔

پٹرولِیَم چھُ مال کھاتَہٕ ہُند سو ٹُکرٕ یَتھ سارِوٕے کھۄتہٕ صاف بیرونی قِمَت چھِ۔ اپریل–جوٗن پٹرولِیَم، تیل تہٕ لُبرِکینٹ (petroleum, oil and lubricants, POL) خَسارَہ ٲس US$37.6 billion، US$32.2 billion خلاف۔ اَگہِ سہ ماہی ادایِگی ہُنٛد تُلن (balance of payments) چھُ سو دستاویز یُس یہِ اپڈیٹ چھُ کَران۔

## پورٹ فولیو نیرُن تہٕ جمع اندرُن چھِ الگ خطرٕ

اپریل–جوٗن 2026 مَنٛز ٲس زَرمُبادَل پورٹ فولیو سَرمایَہ کٲری US$9.6 billion ہُند خالص نیرُن۔ اَکھ ؤری برٛونٛہہ ٲس یہِ US$1.6 billion ہُند خالص اندرُن۔ سِدھی زَرمَبادَل سَرمایَہ کٲری (foreign direct investment, FDI) ٲس US$6.1 billion ہُند خالص اندرُن، اَکھ ؤری برٛونٛہہ US$5.2 billion کھۄتہٕ زیادٕ۔ سہ ماہی جٲری کھاتہٕ ٲس لَکُٹ۔ مالِیاتی میلاو ٲس نازُک ٹُکرٕ: اَکھ پورٹ فولیو نیرُن یُس جٲری کھاتَہٕ خَسارَس کھۄتہٕ بوڈ ٲس۔

جُلایی کٔر رؠتہٕ وار میلاو بدل۔ پورٹ فولیو سَرمایَہ کٲری ٲس US$4.1 billion ہُند خالص اندرُن۔ اپریل–جُلایی خٲطرٕ ٲس پورٹ فولیو سَرمایَہ کٲری وُنہِ US$5.5 billion ہُند خالص نیرُن۔ غَیر مُقیم جمع ٲس جُلاییَس مَنٛز US$33.5 billion ہُند خالص اندرُن تہٕ اپریل–جُلایی خٲطرٕ US$36.2 billion، اپریل–جُلایی 2025 ہَس مَنٛز US$4.7 billion خلاف۔ جُلایی ہُند مُجمَل تُلن ٲس US$20.8 billion ہُند فاضِل، اَمہِ کِنۍ بَڑۍ زَخیرٕ ادایِگی ہٕنٛدِس بُنیادَس پٮ۪ٹھ۔

جمع پرِنٹ چھُ اِبتِدٲیی۔ اَتھ قَدَر اَکہِ رؠتھَس ہُند اندرُن ہیٚکہِ اَکھ خاص کِھڑکی، اَکھ درجہٕ بندی، یا اَکھ بہاو آسِتھ یُس واپس گَژھِ۔ تصدیٖقی دستاویز چھُ اَگہِ سہ ماہی ادایِگی ہُنٛد تُلن، یَتہِ جُلایی پوٗرٕ مالِیاتی کھاتَس اندر ترمیم چھُ سپدان۔ تَتھ تام چھُ احتیاطی پرُن یہِ زِ جُلایی ہُند زَخیرٕ بَناو بَینکِنگ سرمایَس پٮ۪ٹھ جُکُت ٲس، تہٕ زِ اپریل–جُلایی پورٹ فولیو سَرمایَہ کٲری ٲس وُنہِ مَنفی۔

بیرونی تِجارَتی قَرضَن (external commercial borrowings, ECB) چھُ لیبلَن سۭتۍ وۄہے احتیاط ضروٗری۔ بَیلَنس آف پےمَنٹس مَینوئل چھَم (Balance of Payments Manual 6, BPM6) سہ ماہیَس مَنٛز ٲس ہِندوستانَس ECB خالص US$3.3 billion۔ تَمہِ سہ ماہیَس خٲطرٕ پرٛونؠ شکلُک بدلاو نوٹَس مَنٛز ٲس بیرونی تِجارَتی قَرض US$1.0 billion۔ تِم چھِ الگ سَطر۔ «ECB» وُچھُنَس چھُ میز ناؤ دِیُن ضروٗری۔

## مُلٕہ بَندِی ہیٚکہِ سُرخی تِجارَتی واقعہٕ بغٲر بدلاوِتھ

اپریل–جوٗن 2026 مَنٛز کھَسۍ زَخیرٕ نامزد شَرطَن مَنٛز US$22.5 billion۔ تَتھ مَنٛز ٲس US$14.4 billion مُلٕہ بَندِی، یُس RBI بنیٲدی طور کم سون قِمَت تہٕ بٔڑِن زَرَن خلاف مضبوط ڈالَرَس سۭتۍ جوڑان چھُ۔ ادایِگی بُنیادُک گِراو ٲس US$8.1 billion۔

ہَفتہٕ وار اسٹاک چھُ وۄہے وَنٛڈاو لَکُٹ پیمانَس پٮ۪ٹھ ہاوان۔ 11 سَتَمبَر 2026 تام ٲس کُل زَخیرٕ US$780,782 million، ہَفتہٕ پٮ۪ٹھ US$4,924 million کھۄتہٕ بون۔ سون ٲس US$111,225 million، ہَفتہٕ پٮ۪ٹھ US$2,591 million کھۄتہٕ بون۔ زَرمُبادَلَہٕکۍ اثاثہٕ ٲسۍ US$645,796 million، US$2,372 million کھۄتہٕ بون۔ تَتھ ہَفتَس مَنٛز کھَس سون اَجزا US$2,591 million، کُلَس مَنٛز US$4,924 million ہَس گِراوَس اندر۔ 4 سَتَمبَر تام ٲس اسٹاک US$785,706 million، تَتھ ہَفتَس پٮ۪ٹھ US$44,903 million بَڑِتھ۔ بوڈ ہَفتہٕ وار چال چھِ وُنہِ سلسلَس ہُند حصہٕ۔ تِم چھِ اَجزا پرنُک وجہ، نہٕ پرٮ۪تھ جُمعہِ نَو نظام ماننُک وجہ۔

مارچ 2026 کِس اَندَس پٮ۪ٹھ چھُ اسٹاک US$89,674 million بَڑِتھ۔ بَفر چھُ مالِیاتی ؤریَس بند گَژھنَس کھۄتہٕ بوڈ۔ وُچھُن چھُ اَگہِ چالُک ترکیب: لین دَین، سون، یا ڈالَر۔

## ارتکاز چھُ میزَن مَنٛز گوڈَے موجود بازار خطرٕ

2025-26 مَنٛز سافٹ ویَر بَرآمَد ٲس 54.1 per cent مُتَّحِدہ رِیاسَتھَس (United States) تہٕ 72.6 per cent ڈالَرَن مَنٛز انوٲیِس، US$221.4 billion کُلَس پٮ۪ٹھ بیرونِ مُلکِ تِجارَتی موجودگی بغٲر۔ اپریل–اَگَست 2026 مالِ بَرآمَدَن سوز US$215,912.45 million مَنٛز نِش US$42,790.23 million مُتَّحِدہ رِیاسَتھَس۔ مُتَّحِدہ عَرب اِمارَت (United Arab Emirates, UAE)، 2025-26 مَنٛز سارِوٕے کھۄتہٕ بوڈ ترجیحی تِجارَتی مُعاہِدٕ (free trade agreement, FTA) مال بازار US$37,359.11 million پٮ۪ٹھ، چھُ اپریل–اَگَست فوری اندازَن مَنٛز 10.61 per cent کھۄتہٕ بون۔

اَکہِ منزلَس مَنٛز مانگ یا پالیسی جھٹکہٕ چھُ گۄڈٕ یِمَن حِصن مَنٛز نظرِ یِوان، نہٕ تمام ہِندوستان کُلَس مَنٛز۔ DGCIS مُلک میز تہٕ اَگہِ سافٹ ویَر سروے چھِ اپڈیٹ۔ جُلایی وَزارَتِ تِجارَت شایعَن کٔر گوڈَے منزل بَڑُن شرح نِشان، شٲمِل مُتَّحِدہ رِیاسَتھ تہٕ چین (China)، «قِمَتَس مَنٛز بدلاؤ» بُنیادَس پٮ۪ٹھ۔ درجہٕ ضروٗری آسنَس پٮ۪ٹھ سطح میز استعمال کٔرِو، تہٕ یُس تیز سپُد تٔمِس ضروٗری آسنَس پٮ۪ٹھ بَڑُن فِہرِست استعمال کٔرِو۔

## یہِ وُچھنُک فِہرِست کیاہ چھُ نہٕ کَران

یہِ چھُ نہٕ 0.5 per cent جٲری کھاتَہٕ نِسبَت بحرانُک پُکار بناوان۔ یہِ چھُ نہٕ جُلایی ہُند جمع اندرُن سالانہٕ بناوان۔ یہِ چھُ نہٕ US$4,924 million ہُند ہَفتہٕ وار زَخیرٕ گِراو مُداخلَت پران۔ RBI ضمیمہٕ چھُ نہٕ تَتھ خانَس مَنٛز مُداخلَت شناخت کَران۔ پرن والٮ۪ن یِمَن مُداخلَتُک سوال چھُ، تِمَن چھُ اَکھ RBI بیان ضروٗری یُس آپریشَنَن متعلق بولِ، نہٕ صرف اسٹاک میز۔
