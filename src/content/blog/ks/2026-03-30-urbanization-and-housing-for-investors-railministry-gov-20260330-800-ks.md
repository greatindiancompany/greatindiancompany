# سَرمایہٕ کار ہِندوستانَس منٛز شَہری رہٲیِش کِتھ کَن پرِ

## پوزیشَنَس چھُ پَنُن نَقد بہاو ناو دِیُن

ہِندوستانَس منٛز شَہری رہٲیِش ہیٚکہِ رہنِ ہِنٛز اِمداد، ڈیوَلَپَرُک اِسکرو (escrow)، میوٗنِسِپَل آب ٹھیکہٕ، یا اَکھ مُکَمَل شہری اثاثہٕ مَزیٖد مرکزی گرانٹ ورٲے مطلب آسِتھ۔ یِمَن چھُنہٕ اَکھ پَیداوَر۔ سرکٲری ریکارڈ چھُ کافی وَننَس زِ کُس نَقد بہاو موجود چھُ۔ سُہ چھُنہٕ قۭمَت اِنڈیکس، تہٕ یہِ نوٹ چھُنہٕ کانٛہہ دِوان۔

## رہن کِتاب یُس وِنہِ تہِ کھُلہ چھُ

قَرضَس سۭتۍ جوڑِتھ اِمدادی سِکیٖم (Credit Linked Subsidy Scheme) چھِ بند کِتاب۔ پریٚس اِطلاع بیورو (Press Information Bureau, PIB) ہُنٛد 24 جُلاے 2025 ہُنٛد جَواب: فی گَر ₹2.67 لاکھ تام اِمداد، درمیٲنی آمدَنی کور گوٚو 31 مارٕچ 2021 خَتم، اِقتِصادی طور کمزوٗر تہٕ کَم آمدَنی کور گوٚو 31 مارٕچ 2022 خَتم، ₹58,885 کَروڑ جٲری 25,04,220 فٲیدٕ گَژھَن وٲلٮ۪ن۔ یِمَن وِنڈوَن تحت چھُنہٕ کانٛہہ چیز شُروٗع کَرنَس خٲطرٕ باقی۔

کھُلہ اِمداد چھِ PMAY-شَہری 2.0 (Pradhan Mantri Awas Yojana – Urban 2.0) اندر سودِ ہِنٛز اِمدادی سِکیٖم (Interest Subsidy Scheme)، 1 سَتَمبَر 2024 پؠٹھ نافِذ۔ سَبھ کھۄتہٕ زِیادٕ ₹1.80 لاکھ۔ آمدَنی ₹9 لاکھ تام۔ قَرضہٕ ₹25 لاکھ تام۔ جائیداد ₹35 لاکھ تام۔ کارپِٹ رَقبہ (carpet area) 120 مُرَبَع میٖٹَر تام۔ گۄڈَنیِکِس ₹8 لاکھَس پؠٹھ ژور فیصَد، 12 ؤریَن تام، پانٛژھ ؤری وار قِستہٕ، صِرف تَتھ وِزِ یَلِہ قَرضہٕ چالو چھُ تہٕ اَصل رَقَمُک اَدھا کھۄتہٕ زِیادٕ باقی چھُ۔ رہٲیِشی مالیات کَمپَنیَس منٛز سَرمایہٕ کار گَژھِہ قِست شَرط خطرٕ قَبوٗل کَرِتھ، نَہ زَن صِرف منظوٗری خط۔ پیشگی ادائیگی ہیٚکہِ اِمداد بند کٔرِتھ۔ حَدَن نٮ۪بَر قَرضہٕ اوس کُنہِ تہِ اَتھ منٛز نَہ۔

دَسَمبَر 2024 PIB نوٹَس ہُنٛد ₹10 لاکھ کَروڑ سَرمایہٕ کاری عَدَد تہٕ ₹2.30 لاکھ کَروڑ حُکوٗمَتی مَدَد چھِ 1 کَروڑ مَزیٖد خاندانوَن خٲطرٕ سِکیٖمُک شایع پیمانہٕ بیان کَران۔ تِم چھِ نَہ فِہرِست شُدٕ ڈیوَلَپَرِ ہِنٛدٮ۪ن پیشگی فروختَن خٲطرٕ رہنُمٲیی حد۔ شِراکَت تہٕ فٲیدٕ گَژھَن وٲلٮ۪ن ہِنٛدٮ۪ن تعمیٖرَس تحت ₹2.50 لاکھ فی یونِٹ مَدَد چھِ اَکِہ اِہل گَرَس سۭتۍ جوڑِتھ منتقلی۔ عام زمرٕ رِیاسَتَس منٛز چھُ سُہ مَدَدُک مرکزی حِصہٕ ₹1.50 لاکھ۔ ماڈَل یِم یِمَن رِیاسَتَن منٛز ₹2.50 لاکھ مرکزی آمد کِس پٲٹھۍ کِتابَس منٛز چھِ تھاوان، تِم چھِ وَزارَتِ ہِنٛدِس پانَنِہ وَنڈَس پؠٹھ غلط۔

## ڈیوَلَپَر کِتاب چھُ قانوٗن، نَہ زَن مِشَن

ریٖل اِسٹیٹ (ریگولیشَن اینڈ ڈیوَلَپمَنٹ) ایکٹ (Real Estate (Regulation and Development) Act) چھُ دَفہ 4 منٛز لازم کَران زِ اَلاٹیَن نِش حٲصِل رَقَمَن ہُنٛد سَتَتھ فیصَد زَمیٖن تہٕ تعمیٖر خٲطرٕ الگ شیڈول بینٛک کھاتَس منٛز آسِہ۔ یہِ چھُ گاہَک پیشگی رَقَمَن ہُنٛد اِستِمال محدود کَران۔ یہِ چھُ پرٛٮ۪تھ رِیاسَتَس منٛز مَنصوٗبہٕ نَقد بہاوِ ہِنٛز خصوصیت یَتہِ رجِسٹریشَن لاگوٗ چھِ۔ یہِ چھُ یہِ تہِ کیازِ زِ مَنصوٗبہٕ یُس رِیاسَتی رجِسٹَرَس پؠٹھ چھُنہٕ، سُہ چھُ سُہ مَنصوٗبہٕ نِش الگ خطرٕ یُس چھُ۔ رہٲیِشی تہٕ شَہری مَعامِلاتَن ہِنٛز وَزارَت (Ministry of Housing and Urban Affairs, MoHUA) چھِ رجِسٹریشَنِ ہِنٛزِ فورس میجَر (force majeure) تَوسیعَن پؠٹھ 31 جُلاے 2026 ہُنٛد مَشوَرٕ فِہرِست کَران۔ تَوسیع چھِ وَقتُک واقعہٕ۔ سۄ چھِ نَہ مانٛگُک واقعہٕ۔ کارپِٹ رَقبہ چھُ قانوٗنی پیمانہٕ تہٕ اِمدادُک پیمانہٕ۔ سوٗپَر بِلٹ اَپ رَقبہ (super built-up area) چھُ سۄ بُنیاد یۄس ایکٹَن رجِسٹَر فروختَن منٛز ختم کَرنَس خٲطرٕ لیکھ۔

قٲبلِ بَرداش کرایَہ وٲل رہٲیِش (Affordable Rental Housing) چھُ سُہ عمودی حِصہٕ یُس مَزدورَن، مہاجِرَن، تہٕ طٲلِبِ عِلمَن کُن چھُ۔ سُہ چھُ تھون تہٕ کرایَس پؠٹھ دِیُن ہُنٛد ایکسپوژَر اگر نِجی اِدارٕ سُہ رَٹِ۔ سُہ گَژھِہ نَہ فروختَس خٲطرٕ شُروٗع کِس پٲٹھۍ قۭمَت سپدُن۔

## میوٗنِسِپَل کِتاب چھُ مرکزَس پؠٹھ زیٛادٕ تر خرچ سپدِتھ

ہوشیار شَہرَن (Smart Cities) ہِنٛز مرکزی مَدَد چھِ نَہ بَڑَن وول سرمایہٕ خرچ کہانی۔ خرچ ₹48,000 کَروڑ، اِختِتام 31 مارٕچ 2025، دعوٕے ₹47,458 کَروڑ، کانٛہہ نٔو شَہر نَہ۔ لوک سَبھا (Lok Sabha) 31 مارٕچ 2026 تام رپورٹ کَرنہٕ آمُت، 8,064 مَنصوٗبَن منٛز 7,790 ٲسۍ مُکَمَل، تہٕ شِراکَت ذیلی سیٹ اوس 197 مَنصوٗبَہ، قۭمَت ₹9,277 کَروڑ، یِمَن منٛز 194 مُکَمَل۔ شِراکَت ٲس ₹1,64,811 کَروڑ مَنصوٗبہٕ کِتابَس ہُنٛد پَتَل حِصہٕ، تہٕ سُہ حِصہٕ چھُ زیٛادٕ تر بَناونہٕ آمُت۔ باقی قۭمَت چھِ کمانڈ سینٹَرَن ہِنٛز کارروائی تہٕ سُہ کِردار یُس رِیاسَتِ ہُنٛد چیف سیکرٹری (chief secretary) 10 جوٗن 2025 مَشوَرٕ تحت خاس مَقصَدٕچ کمپنی (special purpose vehicle) سونپان چھُ۔ اَتھ میوٗنِسِپَل فَریقَس سۭتۍ خِدمَت ٺیکَس کِس پٲٹھۍ قۭمَت دِیو، ضمیمَس منٛز شَہری سَطَر پرنہٕ پَتہٕ۔ 100 فیصَد پؠٹھ شَہر چھُنہٕ ٹینڈَر پائپ لایِن۔

AMRUT 2.0 (Atal Mission for Rejuvenation and Urban Transformation 2.0) ہُنٛد اِشارٲتی ₹2,99,000 کَروڑ، شٲمِل ₹86,760 کَروڑ مرکزی حِصہٕ، آو 2021-22 پؠٹھ 2025-26 تام شایع کَرنہٕ۔ ملین-پُلس شَہر مختص رَقَمَن ہُنٛد دَہ فیصَد اوس شِراکَت طٔریٖقَس منٛز آسُن، تہٕ کریڈِٹ ریٹِنگ تہٕ میوٗنِسِپَل بانڈ چھِ ناو دِنہٕ آمٕتۍ اِصلاح۔ اَکھ بانڈ یا شِراکَت بولی گَژھِہ وِنہِ تہِ رِیاسَتی آب ایکشَن پلانَس تہٕ جائیداد ٹیکسَس ہِنٛدِس دَروازَس کُن اِشارٕ کَرِتھ یُس پَتِم قِستہٕ بچاوان چھُ۔ یہِ نوٹ چھُنہٕ خرچ نٔو پانٛژھ ؤری مُدتَس منٛز رول کَران۔

## مَعلوٗمات یۄس ضربَس منٛز گُم چھِ

آخری سرکٲری شَہری حِصہٕ یُس MoHUA شایع چھِ کَران، سُہ چھُ 2011 جدول: 31.16 فیصَد، 377.1 ملین لوٗکھ۔ مَردَم شُمٲری 2027 ہُنٛد حَوالہٕ تٲریٖخ چھُ 1 مارٕچ 2027۔ یوتام یہِ جدول چھِ نَہ، اَکھ آخری بَڑَن ہُنٛد دَرَج یُس گول شَہری کَرنِہ کٮ۪ن فیصَدَن پؠٹھ بَناونہٕ آمُت چھُ، سُہ چھُ قیاس، تہٕ سُہ گَژھِہ قیاس کِس پٲٹھۍ لیبل سپدُن۔ گریٹ اِنڈین کَمپَنی (Great Indian Company) بدلاوِ نَہ گُم شُدٕ جدول عالمی پیمانَس سۭتۍ۔
