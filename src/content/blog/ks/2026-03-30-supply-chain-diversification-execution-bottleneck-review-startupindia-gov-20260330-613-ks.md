# ہِندوستانٕچ صَنعتی تِجارَت ہِنٛزِ تَنوعَس مَنٛز نافَذ کرنِک رُکاوَٹ

اِعلان چھِ نہٕ رُکاوَٹ۔ رُکُن چھُ عام طور ژورَن حوالَن مَنٛز اَکھ: قومی سکیٖمَس پؠٹھ لائن وَزارَتَس تام، اَصل کارخانَس پؠٹھ فراہم کُنَندَس تام، تِجارَتی معاہدَس پؠٹھ سَرٹیفِکیٹَس تام، یا کوریڈور نقشَس پؠٹھ ٲخرِی 50 کِلومیٖٹَرَن تام۔

## وَزارَتٕ چھِ نہٕ اَکھ سکیٖم پٲٹھۍ پَکان

2 اَگَست 2023 ہُند پریس اِنفارمیشن بیورو (Press Information Bureau, PIB) نوٹ چھُ صاف۔ تمام 14 شُعبَن خٲطرٕ پیداوار سۭتۍ جُڑِتھ حَوصلہٕ سکیٖم (Production Linked Incentive, PLI) ٲس نوٹیفائی کرنہٕ آمژٕ، تہٕ تِم ٲس مُتعلِق وَزارَتَن تہٕ محکَمَن ہِنٛدِ «نافَذ کرنِک الگ الگ مرحلَن مَنٛز»۔ صَنعت تہٕ اَندروٗنی تِجارَت ہِنٛزِ تَرقِی ہِنٛز محکَمہٕ (Department for Promotion of Industry and Internal Trade, DPIIT) ہیٚکہِ ڈھانچہٕ بیان کٔرِتھ۔ یِہ چھِ نہٕ پرٛؠتھ شُعبہٕ ہُند دعوٕ چلاوان۔ یُس فرم اَکھ کُنۍ PLI کِھڑکۍ منتظر روزِ، سُہ سُہے شُعبَس کھۄتہٕ پَتہٕ روزِ یُس گوڈائے دَرخواست منٛگان چھُ، تہٕ سُہے شُعبَس جلدی کَرِ یَمِیُک رہنُمٲیی وُنہِ تہِ ٹھہراونہٕ چھِ یِوان۔ سُہے جوابَس مَنٛز اوس وَنہٕ آمُت زِ کابینہِ اوس تَتھ تٲریٖخَس تام اِضافی شُعبہٕ منظوٗر کۆرمُت نہٕ۔ غیر فِہرِستی سامانَس خٲطرٕ «PLI» چالو پروگرام مانُن چھُ نافَذ کرنُک غلط، پالیسی تبدیلی نہٕ۔

فراہم کُنَندہٕ ہُند حوالہٕ چھُ سُہ رُکاوَٹ یُس نوٹ پانےٕ پیش کَران چھُ۔ اَصل اِدارَن (anchor units) پؠٹھ چھُ اُمید زِ تِم فراہم کُنَندَن ہِنٛز بُنیاد بناون، زیادٕ تَر خُرد، لَگوٗ تہٕ مَدرَم اِدارٕ (micro, small and medium enterprises, MSME)۔ تَتھ تٲریٖخَس 733 چُنہٕ آمَژَن دَرخواستَن مَنٛز 176 ٲس شُعبَن ہِنٛدِس اَکھ حِصَس مَنٛز MSME۔ اگر تِم فراہم کُنَندٕ اَوزار تیار کٔرِتھ نہٕ ہیٚکَن، اَصل اِدارٕ چھُ ذیلی جوڑ دَرآمَد کَران تہٕ مَقامی تَیارِی ہُند دعوٕ مالِ فِہرِستَس (bill of materials) مَنٛز ناکام گَژھِ۔ یِہ چھُ کارخانہٕ فرشُک رُکاوَٹ۔ یِہ Rs 1.97 lakh crore تَخصیٖصَس مَنٛز چھُ نہٕ وُچھنہٕ یِوان۔

## سَرحَدَس پؠٹھ کاغذی رُکاوَٹ

زٕ کاغذی رَستٕ چھِ جسمانی تَنوع رُکاون۔ کَوالِٹی کَنٹرول آرڈَر (Quality Control Order) ہیٚکہِ گَریہِ بازارَس خٲطرٕ اَکھ بَدل اَنٛدَرونی مال رُکٲوِتھ۔ خٲرجی تِجارَت پالیسی 2023 (Foreign Trade Policy 2023, FTP 2023) ہِنٛز پیشگی اِجازَت (Advance Authorisation)، بَرآمَد مُتَوَجِہ اِدارَن (export-oriented units, EOUs)، تہٕ خاص اِقتِصادی عَلاقَن (special economic zones, SEZs) خٲطرٕ چھوٹ چھِ اصلی تہٕ تنگ: یِہ چھِ بَرآمَد پیداوارَس خٲطرٕ، اَتھ شَرطَس سۭتۍ زِ اَنٛدَرونی مال گَریہِ محصولِ عَلاقَس (domestic tariff area) مَنٛز رَسِ نہٕ۔ یُس کارخانہٕ پنُن بَرآمَد تہٕ گَریہِ ذخیرہٕ الگ کَرِ نہٕ، سُہ «کسٹمزَس مَنٛز پھَسِتھ» وَچھِ یَلہٕ زَن سُہ پنِنہِ عمل ڈیزائنَس مَنٛز پھَسِتھ چھُ۔

تَرجیح چھِ سُہے پٲٹھۍ رُکان۔ اَکھ معاہدٕ یُس نافِذ چھُ نہٕ، یا اَکھ کھیپ یُس اصلِیتُک قٲیدٕ راوان چھُ، یا خٲرجی تِجارَت ہِنٛدِ ڈائریکٹوریٹ جنرل (Directorate General of Foreign Trade, DGFT) ضمیمہٕ 2B (Appendix 2B) ہِنٛدِس ایجنسی نِش غائب سَرٹیفِکیٹ، چھُ عام دَرجَس پؠٹھ روانہٕ گَژھان۔ خریٖدار تیٚلہِ نتیجہٕ کَڑان چھُ زِ ہِندوستان «کام اوس نہٕ کَران»، یَلہٕ زَن رُکاوَٹ دَستاوَز ٲس۔ متحدہ عرب امارات (United Arab Emirates, UAE) خٲطرٕ تِجارَت ہِنٛز محکَمہٕ (Department of Commerce) ہُند پنُن CEPA نوٹ چھُ ہاوان زِ کھولِتھ ورجن کِتھ کَن چھُ وَنان: 1 مے 2022 پؠٹھ نافِذ معاہدٕ، محصولِ سَطرَس سطحَس پؠٹھ مُقرَر ڈیوٹی خَتم کرُن سۭتۍ۔

## جسمانی رُکاوَٹ چھِ اکثر عِمارَت نہٕ

PIB ہُند 27 نَوَمبَر 2025 ہُند نوٹ چھُ 2023-24 خٲطرٕ مالِ رَسانی لاگَت جی ڈی پی (gross domestic product, GDP) ہُک 7.97 فیصد، کُل Rs 24.01 lakh crore، تہٕ لَکُٹ فرمَن خٲطرٕ زیادٕ لاگَت رپورٹ کَران۔ یِہ چھُ لَگ بَگ 600 کِلومیٖٹَرُک سفرَس ہِنٛدِ گۄڈنِک تہٕ ٲخرِی 50 کِلومیٖٹَرَن نِشان کَران۔ اَکھ کوریڈور ہیٚکہِ اصلی طور تیز آسِتھ — سُہے نوٹ چھُ وَنان زِ مَشرِقی مَخصوٗص مالِ گٲڑِ راہ (Eastern Dedicated Freight Corridor) ہُند ویگَن وَلُن 15-16 دۄہَن پؠٹھ 2-3 دۄہَن تام ووت — تہٕ کارخانہٕ ہیٚکہِ وُنہِ تہِ جہازُک وَکھ راوِتھ اگر رَسٲیی سَڑَک، ٹرک ٹرمینل، یا کنٹینر نظرُک تہہ کمزور آسِہِ۔ ULIP، یُس یونیفائیڈ لاجسٹِکس اِنٹرفیس پلیٹ فارم (Unified Logistics Interface Platform, ULIP) چھُ، تہٕ مالِ رَسانی ڈیٹا بینک (Logistics Data Bank) چھِ سُہ تہہ نظرَس مَنٛز انُنِک سَرکٲرِی کوشش۔ تِم چھِ نہٕ غائب رَسٲیی سَڑَک ہٹاون۔

رِیاستی صَلاحیَت چھِ دویِم جسمانی مُتغیر۔ قومی مالِ رَسانی پالیسی (National Logistics Policy) متعلق اَکھ PIB نوٹ چھُ وَنان زِ 26 رِیاستَو تہٕ مرکزی عَلاقَو (Union Territories) اوس رِیاستی مالِ رَسانی پالیسی نوٹیفائی کٔرمٕژ۔ LEADS، یُس رِیاستَن درمیان مالِ رَسانی ہِنٛز سَہولَت (Logistics Ease Across Different States, LEADS) چھُ، 13 مے 2026 دۄہ 2025 خٲطرٕ جٲری آو، چھُ طَریٖقہٕ یَمہِ سۭتۍ DPIIT باقی ہُند مُقابلہٕ کَران چھُ۔ یُس فراہم کُنَندہٕ جھُنڈ اَکھ ایسی رِیاستَس مَنٛز آسِہِ یَمہِ پالیسی یا بُنیٲدی ڈھانچہٕ میل کۆرمُت نہٕ، سُہ قومی کوریڈورُک اَوسَط رفتار وراثَتَس مَنٛز نِیہِ نہٕ۔

اَہَم معدنی عمل آوری چھِ سُہے قِسمُک اَکھ سست رُکاوَٹ۔ قومی اَہَم معدنی مشن (National Critical Mineral Mission) چھُ پارکَن تہٕ دُبارٕ اِستعمالَس خٲطرٕ تَخصیٖص تھاوان۔ یَتام زَن تِم پارک چلَن، اَکھ مَقامی سیل یا الیکٹرانک کارخانہٕ چھُ وُنہِ تہِ دَرآمَد کرنہٕ آمُت دَرمیٲنی مالَس خٲطرٕ قَطارَس مَنٛز روزان۔

## کیاہ اَگہٕ نِنہٕ

یُس حوالہٕ تُہۍ یِم سہہ ماہَس اصلی طور کنٹرولَس مَنٛز چھِو تھاوان، سُہ اَگہٕ نِیو: لائن وَزارَتہِ ہِنٛز رہنُمٲیی، فراہم کُنَندہٕ ہُند اَوزار تیار کرُن، اصلِیت فایل، یا ٲخرِی میٖلُک سروے۔ قومی تَخصیٖص ضمیمَس مَنٛز تھٲیو۔ گریٹ اِنڈین کَمپَنی (Great Indian Company) چھِ یِہ رُکاوَٹ نوٹ یِم خٲطرٕ لیکھان زِ نافَذ کرنِک مسلہٕ «پالیسی غیر یقینی» تحت درج گَژھَن نہٕ۔ ہوم پیج چھُ [greatindiancompany.com](https://greatindiancompany.com)۔
