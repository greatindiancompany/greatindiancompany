# شَہری رہٲیِشی مِشَنَن ہُنٛد بُنیاد تھاوَن وولَس ہُنٛد نقشہٕ

## لین دٮ۪ن پؠٹھ شُروٗع کٔرِو، نَہ زَن موضوعَس پؠٹھ

اَکِہ کَمپَنیَس یۄس شَہری ہِندوستانَس منٛز فروخت کَران چھِ، تَس چھُ زانُن چھُ زِ لین دٮ۪ن کُس سرکٲری دَروازٕ اِستِمال چھِ کَران۔ رہٲیِشی تہٕ شَہری مَعامِلاتَن ہِنٛز وَزارَت (Ministry of Housing and Urban Affairs, MoHUA) چھِ کینٛہہ چَلاوان، تہٕ تِم چھِ نَہ اَکھ گاہَک، اَکھ ٹینڈَر، یا اَکھ ڈیٹا کَمرٕ شیئر کَران۔ مَردَم شُمٲری 2011 (Census 2011)، وِنہِ تہِ وَزارَتِ ہُنٛد شایع شَہری بُنیاد 31.16 فیصَد تہٕ 377.1 ملین لوٗکھ، چھُ سیاق۔ سُہ چھُنہٕ فروختِ ہِنٛز پائپ لایِن۔ مَردَم شُمٲری 2027 بدلاوِ سُہ بُنیاد صِرف تَتھ وِزِ یَلِہ رجِسٹراڑ جَنرَل (Registrar General) جدول شایع کَرِ۔ 2026 دوران مَرحَلہ I چھُ رہٲیِشی حالَت تہٕ سہوٗلَت ریکارڈ کَران۔ سُہ چھُنہٕ ٹھیکہٕ دِوان۔

## اگر مصنوعات گَر قَرضَس یا اِمدادی یونِٹَس لَگان چھِ

سودِ ہِنٛزِ اِمدادی سِکیٖمِ (Interest Subsidy Scheme) ہِنٛدۍ قٲعدٕ اِستِمال کٔرِو یِم پریٚس اِطلاع بیورو (Press Information Bureau, PIB) 24 جُلاے 2025 بیان کٔر۔ تَتہِ ناو دِنہٕ آمُت پورٹَل چھُ pmay-urban.gov.in۔ مصنوعات ₹9 لاکھ آمدَنی، ₹25 لاکھ قَرضہٕ، ₹35 لاکھ جائیداد، تہٕ 120 مُرَبَع میٖٹَر کارپِٹ رَقبہ (carpet area) اندر فِٹ کٔرِو، یا اِمداد کِتابَس منٛز مَہ تھاوِو۔ خِدمَت نِظام یِتھ کَن بَناوِو زِ ؤری وار قِسط صِرف تَتھ وِزِ دعوٕے سپدِ یَلِہ قَرضہٕ چالو چھُ تہٕ اَصل رَقَمُک اَدھا کھۄتہٕ زِیادٕ باقی چھُ۔ سُہ جَوابَس منٛز سَبھ کھۄتہٕ زِیادٕ اِمداد چھِ ₹1.80 لاکھ، گۄڈَنیِکِس ₹8 لاکھَس پؠٹھ 4 فیصَد، 12 ؤریَن تام۔

اگر مصنوعات پانَے گَر چھُ فٲیدٕ گَژھَن وٲلٮ۪ن ہِنٛدٮ۪ن نِگرٲنی منٛز تعمیٖر (Beneficiary Led Construction) یا شِراکَتَس منٛز قٲبلِ بَرداش رہٲیِش (Affordable Housing in Partnership) تحت، تَیلِہ دَسَمبَر 2024 PIB نوٹَس ہِنٛز مَدَد چھِ فی یونِٹ ₹2.50 لاکھ، مرکزی حِصہٕ رِیاسَت یا مرکزی زیرِ اِنتِظام عَلاقَس (Union Territory) پؠٹھ منحصر۔ سُہ کٲمِ ہُنٛد ہیٚچھَن وول چھُ رِیاسَتی مِشَن تہٕ شَہری مَقٲمی اِدارٕ (urban local body)، یادداشتَس پَتہٕ۔ دَسَمبَر 2024 نوٹ چھُنہٕ جِنٛدٕ ٹینڈَر فِہرِست۔ موجودٕ منظوٗری خٲطرٕ رِیاسَت پرژھِو۔

اگر مصنوعات مَزدورَن، مہاجِرَن، یا طٲلِبِ عِلمَن خٲطرٕ کرایَہ وٲل رہٲیِش چھِ، فروختَس خٲطرٕ اپارٹمَنٹ پیش کَرنہٕ برٛونٛہہ قٲبلِ بَرداش کرایَہ وٲل رہٲیِش (Affordable Rental Housing) عمودی حِصہٕ پرِو۔ خاص توجہٕ خٲطرٕ ناو دِنہٕ آمٕتۍ گروپ، شٲمِل پی ایم سَوانِدھی (PM SVANidhi) تحت سَڑکَن ہِنٛدۍ وِکَری، تعمیٖری مَزدور، تہٕ صَفائی کَرمچاری (safai karamcharis)، چھِ فٲیدٕ گَژھَن وٲلٮ۪ن ہِنٛز زمرٕ۔ تِم چھِ نَہ وِتَرن شِراکَت یوتام رِیاسَت یہِ وَنِ نَہ۔

## اگر مصنوعات میوٗنِسِپَل نِظامَس لَگان چھِ

AMRUT 2.0 (Atal Mission for Rejuvenation and Urban Transformation 2.0)، 27 اکتوٗبَر 2021 ہِدایَتَن منٛز، چھُ نَلکہٕ، سیوریج، سیپٹیج، تہٕ تِمَن اندر پلان تہٕ اِصلاح ہیٚنان۔ ہیٚچھَن وول چھُ رِیاسَت تہٕ شَہر، سٹیٹ واٹَر ایکشَن پلانَس (State Water Action Plan) ذَریعہٕ، ترٛےٚ قِستَن منٛز فنڈَن سۭتۍ تہٕ ترٛیِم ؤریہِ پؠٹھ اِصلاحَس ہِنٛدِس دَروازَس سۭتۍ۔ ٹیکنالوجی ذیلی مِشَن چھُ آب سٹارٹ اَپ حوصلہٕ دِوان۔ یہِ چھُ سُہ جملہٕ یُس آب کَمپَنی حوالہٕ کَرِتھ ہیٚکہِ۔ جملہٕ یُس سۄ حوالہٕ کَرِتھ ہیٚکہِ نَہ، سُہ چھُ مالی ؤری 2025-26 پَتہٕ مرکزی منظوٗری ہُنٛد ضمانَت، یُس سُہ خرچَس پؠٹھ چھَپنہٕ آمُت آخری ؤری چھُ۔

ہوشیار شَہرَن (Smart Cities) ہِنٛدۍ کمانڈ سینٹَر چھِ تَمام 100 شَہرَن منٛز۔ مرکزی مِشَن بَجَٹ چھُنہٕ۔ MoHUA ہُنٛد صَفہٕ چھُ وَنان زِ اِختِتام اوس 31 مارٕچ 2025 تہٕ زِ کانٛہہ نٔو شَہر گَژھِہ نَہ رَٹنہٕ یِنِہ۔ اَکھ بُنیاد تھاوَن وول یُس سافٹ ویئر، کیمرٕ، یا کنٹرول روم کارروائی فروخت کَران چھُ، سُہ چھُ خاس مَقصَدٕچ کمپنی (special purpose vehicle) یا میوٗنِسِپَل اِدارَس فروخت کَران، رِیاسَتی ہِدایَت تحت، شٲمِل 10 جوٗن 2025 ہُنٛد مَشوَرٕ نَمبَر 27۔ 31 مارٕچ 2026 ضمیمہ جانٛچ کٔرِو اَمہِ برٛونٛہہ زِ قیاس کٔرِو زِ شَہرَس وِنہِ تہِ کھُلۍ مِشَن مَنصوٗبَہ چھِ۔ کینژن چھِ نَہ۔

## اگر مصنوعات فروختَس خٲطرٕ نِجی مَنصوٗبَس لَگان چھِ

ریٖل اِسٹیٹ (ریگولیشَن اینڈ ڈیوَلَپمَنٹ) ایکٹ، 2016 (Real Estate (Regulation and Development) Act, 2016)، چھُ چَلان وول نِظام۔ تَتہِ رجِسٹَر کٔرِو یَتہِ دَفہ 3 پؠٹھ 19 یہِ لازم چھِ کَران۔ اَلاٹی جمعَن ہُنٛد سَتَتھ فیصَد سُہ الگ شیڈول بینٛک کھاتَس منٛز تھاوِو یُس دَفہ 4 لازم چھُ کَران، تہٕ سُہ زَر زَمیٖن تہٕ تعمیٖر خٲطرٕ اِستِمال کٔرِو۔ کارپِٹ رَقبَس منٛز ماپِو، یُس اِمدادُک پیمانہٕ تہِ چھُ۔ رِیاسَتی اتھارٹی ہِنٛدۍ قٲعدٕ پرِو تہٕ، اگر آخِری تٲریٖخ پَکان چھُ، فورس میجَر (force majeure) تَوسیعَن پؠٹھ MoHUA ہُنٛد 31 جُلاے 2026 ہُنٛد مَشوَرٕ۔ اَکھ قومی «RERA پابَندی» بیج یُس رِیاسَتی رجِسٹَر نظرانداز چھُ کَران، سُہ چھُنہٕ پابَندی۔

## کَمپَنی کِس پؠٹھ مَہ بَناوِو

101 ہِم ہوشیار شَہر۔ قَرضَس سۭتۍ جوڑِتھ اِمدادی سِکیٖمِ (Credit Linked Subsidy Scheme) ہِنٛز بحالی، یۄس 2021 تہٕ 2022 منٛز بند گٔیہِ ₹58,885 کَروڑ 25,04,220 فٲیدٕ گَژھَن وٲلٮ۪ن واتنہٕ پَتہٕ۔ 2026 ہُنٛد سرکٲری شَہری کَرنُک دَرَج۔ اَکھ نوکری پٲدٕ کرنُک داواہ یُس گۄڈنیُک PMAY-شَہری (Pradhan Mantri Awas Yojana – Urban) پؠٹھ دَسَمبَر 2024 ہِنٛدِس 917 کَروڑ نفر-دۄہ اندازَس کھۄتہٕ برٛونٛہہ چھُ۔ پرٛٮ۪تھ اَکِہ یِمَن منٛز چھُ اَکھ دَستاویٖز یُس یا تَہ پروگرام خَتم کوٚر یا عَدَد وِنہِ تہِ شایع کوٚر نَہ۔
