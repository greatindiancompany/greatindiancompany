# سائبر کِردار تہٕ تَربِیَت گۄنٹۍ یِم سَرکٲری مَتَن ناو کران چھِ

سَرکٲری دستاویز چھِ دَفتَر تہٕ تَربِیَت پروگرام ناو کران۔ تِم چھِ نَہ سائبر سِکیورٹی نوکِریَن ہُند قومی گۄنٹ، نجی شعبَہِ خٲطرٕ خٲلی جَگہِ شرح، یا تَنخا ہَہ بینڈ شایع کران۔ کارآمد پَرھُن چھُ یہِ زِ قَواعِد کَہ کِردار لازِمی بناوان چھِ، تہٕ کَہ عوٲمی تَربِیَت اعداد اصل پٲٹھۍ رِپورٹ کَرنہٕ آمُت چھُ۔

## دَفتَر یِم ریگولیٹ فَرمَس بَرٕنۍ چھِ

**CERT-In خٲطرٕ اَکھ رابطہٕ نُقطہٕ۔** 1 سَتَمبَر 2025 ہٕنٛز خُرد، لَکُٹ تہٕ درمیٲنی اَدارَن (MSME) بیس لائن چھِ فَرمَن وَنان زِ تِم اَکھ سلامتی اِنچارج اَکی رابطہٕ نُقطہٕ پٲٹھۍ ہِندوستٲنؠ کَمپیوٹَر ایٚمَرجَنسی رِسپانس ٹیم (Indian Computer Emergency Response Team, CERT-In) تہٕ ریگولیٹَرَن خٲطرٕ مقرر کَرَن۔ اپریل 2022 ہٕنٛز ہِدایَت، یِم یہِ بیس لائن فَرمَن پٮ۪ٹھ چَلُن وَنان چھِ، چھِ تَمہِ رابطَس پَتہٕ رِپورٹِنگ نظام۔

**اَکھ بینک CISO ساختٲتی علحدگی سۭتۍ۔** 1 اپریل 2024 پٮ۪ٹھ نافِذ ریزرو بینک (Reserve Bank) ہٕنٛزِ آئی ٹی ہِدایَتَن تحت، اَکھ سینئر ایگزیکٹِو، ترجیٖحَن جنرَل منیجَر درجہٕ یا برابر، چھُ چیف اِنفارمیشن سِکیورٹی آفیسَر (Chief Information Security Officer, CISO) نامزَد کَرنہٕ یِوان۔ نَفرَس چھُ تَکنیکی پِسنظرِ ضروٗرَت۔ دَفتَر چھُ کاروبٲری حجم تہٕ ٹیکنالوجی پیچیدگی مطابق بَرنہٕ یِوان۔ CISO چھُ نَہ آئی ٹی سربراہَس سِدھہ رِپورٹ کران، تَس چھِ نَہ کاروبٲری ہَدف دِنہٕ یِوان، تہٕ سُہ چھُ رِسک نِگَرانی کَرن والٮ۪س ایگزیکٹِو ڈائریکٹَرَس یا برابرَس رِپورٹ کران۔ کِردارَس منٛز چھِ سائبر سِکیورٹی حکمتِ عملی چَلاوُن، اِطلاعٲتی اثاثہٕ پالیسی نافِذ کَرُن، بَرونی ایجنسیَن سۭتۍ ہم آہنگی، سلامتی آپریشن مرکز چَلاوُن، تہٕ بورڈَس یا تَہٕنٛدِس رِسک تہٕ آئی ٹی کمیٹیَن برٛونٛہہ ہر رٮ۪وارَس جٲیزٕ تھاوُن۔

**بیمہٕ کُنَندَس ہٕنٛز کمیٹی، نَہ اَکھ اَکی بیج۔** ہِندوستانُک بیمہٕ ریگولیٹری تہٕ ترقٲتی اتھارٹی (Insurance Regulatory and Development Authority of India, IRDAI) ہٕنٛز 2023 ہٕنٛز ہِدایَت چھِ چیف رِسک آفیسَر، CISO، چیف آئی ٹی سِکیورٹی آفیسَر، چیف سِکیورٹی آفیسَر، چیف ہیومن ریسورس آفیسَر، چیف ٹیکنالوجی آفیسَر، تہٕ آپریشن، فائنانس، قانونی تہٕ تعمیل سربراہَن اِنفارمیشن سِکیورٹی رِسک مینجمنٹ کمیٹیَس پؠٹھ بِہان۔ بیٹھَکَس چھُ CISO تہٕ کَم از کَم زٕ رُکن ضروٗرَت؛ سارِوٕے رُکن چھِ کَم از کَم ؤریَس منٛز زٕ لَٹہِ مِلان۔ CISO چھُ اِنفارمیشن تہٕ سائبر سِکیورٹی پالیسی ہُند مالک، رِسک نِگَرانی کَرن والٮ۪س ایگزیکٹِوَس کُن یا، تَس نَفرَس غَیر حاضِریَس منٛز، سی ای او (CEO) کُن رِپورٹ کران، تہٕ متعلقہٕ ریگولیٹَرَن واقَعَن متعلق مُطلِع کَرنُک ذِمہٕ دار۔ اگر CISO غَیر حاضِر آسہِ، چیف رِسک آفیسَر چھُ CISO ہُند فٲصلہٕ کِردار نِوان۔ ہِدایَت چھِ تہِ وَنان زِ CISO سَن گَژھِ معقوٗل کَم از کَم مُدت آسٕنۍ۔ تِم چھِ نَہ یہِ مُدت ؤریَن ہٕنٛدِس نَمبَرَس منٛز واضح کران۔

**اَکھ ٹیلی کام سلامتی اَفسَر یُس رہٲیِشی آسُن چھُ۔** پرٛٮ۪تھ ٹیلی مواصلٲتی ادارٕ چھُ اَکھ چیف ٹیلی کمیونیکیشن سِکیورٹی آفیسَر (Chief Telecommunication Security Officer) مقرر کران۔ قَواعِد چھِ لازِم کران زِ یہِ اَفسَر ہِندوستانُک شَہری تہٕ رہٲیِشی آسہِ، بورڈَس یا اَکھے حُکوٗمَتی ادارَس ذِمہٕ دار آسہِ، تہٕ نافِذ کَرنَس تہٕ واقَع رِپورٹِنگَس خٲطرٕ مرکزی حَکوٗمَتَس سۭتۍ ہم آہنگ کار آسہِ۔ اَفسَر تبدیلی چھِ فَوری مُطلِع کَرنہٕ یِوان۔

**اَکھ نَفَر یَس تام عوٲمی اصل پٲٹھۍ وٲتِتھ ہیکَن۔** پریس اِطلاع بیورو (Press Information Bureau, PIB) ہُند DPDP قَواعِدَن ہُند بیان چھُ وَنان زِ پرٛٮ۪تھ ڈیٹا فِڈیوشَری (data fiduciary) گَژھِ نامزَد اَفسَر یا ڈیٹا پروٹیکشن آفیسَر ہٕنٛز رابطہٕ اِطلاع ہاوٕنۍ تاکہ اَفٲدۍ پروسیسنگ متعلق سوال تُلِتھ ہیٚکَن۔ PIB چھُ تہِ وَنان زِ رَسٲیی، اصلاح، تازہ کاری، مِٹاؤن یا نامزدگی ہٕنٛزِ دَرخواستَن ہُند جَواب چھُ کَم از کَم 90 دۄہَن اَندر دِنہٕ یِوان۔ قاعِدہٕ 14، 18 رٮ۪تَن والٮ۪س نافِذ گروپَس منٛز، چھُ اَکھ شایع شِکایَت نِظام لازِم کران یُس معقوٗل مُدتَس منٛز جَواب دِیہِ یُس 90 دۄہَن کھۄتہٕ زِیادٕ آسہِ نَہ۔ اہم ڈیٹا فِڈیوشَری چھِ اِضافی آڈِٹ، اَثَر جٲیزٕ تہٕ ٹیکنالوجی جانچ فَرٲیِض تُلان یَلۍ تِم قَواعِد نافِذ گَژھَن۔

## تَربِیَت یُس عوٲمی رِکارڈ گَنان چھُ

CERT-In ہٕنٛز سالانہ رِپورٹ 2023 چھِ وَنان زِ اَتھ ٲس تَمہِ ؤریَس 26 تَربِیَت پروگرام منعقد کٔرمِت، 10,074 شرکاءَن خٲطرٕ۔ سامعین یِم یہِ فِہرِست کران چھُ تِمَن منٛز چھِ سِسٹَم تہٕ نیٹ وَرک ایڈمنِسٹریٹَر، ڈیٹابیس ایڈمنِسٹریٹَر، ایپلیکیشن ڈیولپَر، آئی ٹی منیجَر، CISO، CIO تہٕ آئی ٹی سلامتی پیشہٕ وَر، سَرکٲری، عوٲمی تہٕ نازُک شعبہٕ ادارَن پٮ۪ٹھ۔ اَمی رِپورٹَس منٛز سرگرمی جدول چھُ 26 تَربِیَت تہٕ 22 سلامتی مشق فِہرِست کران۔ بیانیہٕ چھُ الگ 14 ملکی سائبر بُحران مشق بیان کران۔ یہِ چھِ رِپورٹَس پنٕنۍ کاوُنٹَر؛ یہِ بریف چھُ نَہ تِم اَکھ کُلَس منٛز زورَس سۭتۍ تُلان۔

MSME کنٹرول چھِ ملازِمَن تہٕ ٹھیکیدارَن خٲطرٕ کَم از کَم ؤریَس منٛز زٕ لَٹہِ بُنیٲدی سائبر سِکیورٹی آگٲہی پرژھان، فِشِنگ، پاسوَرڈ، سوشَل انجینئرِنگ، آلہٕ اِستِعمال، تہٕ حساس اِطلاع ہَینڈلِنگ کور کٔرِتھ، علاوٕ CERT-In وَرکشاپَن تہٕ مشقَن منٛز شِرکَت۔

قانوٗن نافِذ کَرن والٮ۪ن طرفَس، 24 مارٕچ 2026 ہُک لوک سبھا جَواب چھُ وَنان زِ سائی ٹرین (CyTrain) پورٹلَس اوس، 31 جَنؤری 2026 تام، 1,51,081 کھۄتہٕ زِیادٕ پولیس تہٕ عدالتی اَفسَر دَرج تہٕ 1,42,025 کھۄتہٕ زِیادٕ سَرٹِفِکیٹ جٲری۔ زنانَن تہٕ بَچَن خِلاف سائبر جُرم رُکاوَٹ سکیٖم (CCPWC) تحت، 24,600 کھۄتہٕ زِیادٕ پولیس اَہلکار، عدالتی اَفسَر تہٕ پراسیکیوٹر اَس تَربِیَت دِنہٕ آمٕژ۔ یہِ اعداد چھِ عوٲمی تَفتیشی صَلاحِیَت بیان کران۔ تِم چھِ نَہ نجی شعبُک وَرک فورس سروے۔

اَکھ بھرتی منصوٗبہٕ ہیکہِ ناو دار دَفتَرَن پٮ۪ٹھ چَلِتھ: اَکھ وٲتِتھ ہیٚکہِ رابطہٕ، اَکھ CISO یُس آئی ٹی فراہمی سربراہ چھُ نَہ، اَکھ رہٲیِشی ٹیلی کام سلامتی اَفسَر یِتہٕ ٹیلی کام قَواعِد لاگو چھِ، تہٕ اَکھ تَربِیَت تال یُس MSME بیس لائن گوڈَے ؤریَس منٛز زٕ لَٹہِ تھاوان چھِ۔ اَکھ بھرتی منصوٗبہٕ یُس قومی ٹیلنٹ کمی مخصوص نَمبَرَس منٛز نقل کران چھُ، سُہ چھُ کینٛہہ نقل کران یُس یہِ دستاویز چھِ نَہ دِوان۔
