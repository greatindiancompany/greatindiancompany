# ہِندوستانٕکۍ ڈیجیٹل عوٲمی ریلَن پؠٹھ کھُلِہ دَروازٕ

موقع، یَتھ سیٹِنٛگَس مَنٛز، چھُ سُہ دَروازٕ یُس سرکٲری ڈیزایِنَن گوڈَے کٔٹمُت چھُ۔ یہِ چھُ نَہ بازار سایِز اندازٕ۔ ہِندوستانُک ڈیجیٹل عوٲمی بُنیاد (digital public infrastructure) چھُ کینٛہہ دَروازٕ تِمَن فَرمَن تہٕ عوٲمی ایجنسیَن خٲطرٕ کھُلہِ تران یِم متعین کردار تُلُن تیار چھِ۔ کردار چھِ الگ، دروازٕ چھِ الگ، تہٕ تِمَن مَنٛز کٔہہِ تہِ چھُ نَہ "یوٗ پی آی دُبارٕ بَناوِو" یا "قومی شِناخت فراہم کَرن وول بَنِو۔"

## ادایگی: شِرکَت دار بَنِو، نَہ دویُم نؠٹ ورک

یوٗنِفایڈ پیمنٹس اِنٹرفیس (Unified Payments Interface, UPI) چھُ مُشتَرِکہِ ادایگی ریل۔ نیشنل پیمنٹس کارپوریشن آف اِنڈیا (National Payments Corporation of India, NPCI) ہُند اَگَست 2026 پروڈَکٹ شُمار چھُ 752 بینک چالوٗ تہٕ 24,508.96 مِلین لین دین ہاوان۔ 29 جُلایی 2026 حُکوٗمَتی کھاتہٕ وَنان زِ نِظام چھُ 55.49 کروڑ کھۄتہٕ زیٛادٕ فَردَن تہٕ 6.5 کروڑ سوداگَرَن خِدمَت کَران۔ تِمَن حقیقتَن سۭتۍ مِلان موقع چھُ کنارَس پؠٹھ: اَکھ بینک یُس وۄنۍ تام 752 مَنٛز چھُ نَہ، اَکھ سوداگَر یُس وۄنۍ تام 6.5 کروڑ مَنٛز چھُ نَہ، یا اَکھ ورک فلو یُس منتقلی اِستِعمال چھُ کَران یُس ریل گوڈَے کٔرِتھ ہؠکِہ۔ پروڈَکٹ شُمار چھُ نَہ خالی قومی سوئچ ہاوان یُس نٔو کَمپنی چَلاوِ۔

سُہے کنارہ منطق چھُ ملحق عوٲمی ادایگی یوٹیلیٹی پؠٹھ لاگو صِرِف ییتہِ بنیادی صَفہٕ تِم تعریٖف چھُ کَران۔ یہِ مُختصَر چھُ نَہ سُہ پروڈَکٹَس مَنٛز فَرَق ایجاد کَران یُس NPCI اتہِ وَنمُت چھُ نَہ۔

## مالی معلومات: ترٛے نشست، تہٕ زیٛادٕ تر بَناون وٲل چھِ تِمَن مَنٛز اَکھ یَژھان

ریزرو بینک ہُند 28 نَوَمبَر 2025 کھاتہٕ جَمع کَرن وول (Account Aggregator) ہِدایات چھِ ناو وٲل نشستَن سۭتۍ بازار ڈھانچہٕ بَناوان۔

کھاتہٕ جَمع کَرن وول نشست چھےٚ لائسنس یافتہ این بی ایف سی (NBFC)۔ اَتھ چھُ کَمپنی، رجِسٹریشن سرٹیفِکیٹ، کَم از کَم ₹2 کروڑ خالص مِلکیَتِ فَنڈ (net owned funds)، تہٕ بورڈ منظور قۭمَت پالیسی ضروٗرت۔ یہِ ہؠکِہ مخصوص مالی معلومات تُلِتھ تہٕ یہِ گاہَکَس یا مالی معلومات اِستِعمال کَرن وٲلِس پیش کٔرِتھ۔ یہِ ہؠکِہ نَہ لین دین کٔرِتھ، ہؠکِہ نَہ معلومات تھاوِتھ، تہٕ ہؠکِہ نَہ جَمع کَرُن آوٹ سورس کٔرِتھ۔ زیٛادٕ تر پروڈَکٹ کَمپنیَن خٲطرٕ چھےٚ یہِ نشست غلط موقع۔ یہِ چھےٚ تَنٛگ یوٹیلیٹی۔

مالی معلومات اِستِعمال کَرن وول نشست چھےٚ سُہ یُس قرض دِن وول، بیمہ کَرن وول، یا بیٚیُہ ریگولیٹ فَرم اِستِعمال کٔرِتھ ہؠکِہ۔ اِستِعمال کَرن وول گَژھِ مالی شعبہٕ ریگولیٹرَس سۭتۍ دَرج آسُن۔ معلومات یُس پَکِتھ ہؠکِہ اَتھ مَنٛز چھِ بینک جمع، میوچل فَنڈ یونِٹ، ایکویٹی، بیمہ پالیسی، پینشن بیلنس، تہٕ جی ایس ٹی آر-1 (GSTR-1) تہٕ جی ایس ٹی آر-3 بی (GSTR-3B) مَنٛز جی ایس ٹی ریٹَرن، ہِدایَتن مَنٛز فِہرَست کَرنہٕ آمٕتۍ آیٹمَن درمیان۔ موقع چھُ دَستاوِز لَگُن رِضامَندی وول تُلنَس سۭتۍ بدلاوُن، سُہ کاروبارَس اندر یُس ریگولیٹر گوڈَے چھُ پہچانان۔

مالی معلومات فراہم کَرن وول نشست چھےٚ بینک، این بی ایف سی، فَنڈ، ڈیپازٹری، بیمہ کَرن وول، یا جی ایس ٹی این (GSTN) طرف: تُلُن دقیٖق جَواب دِیو۔ ہِدایات وَنان زِ اگر ایگریگیٹرُک بیان تہٕ فراہم کَرن وٲلِس ہُند کھاتہٕ اِختِلاف کٔرَن، فراہم کَرن وٲلِس ہُند موقف چھُ کھڑٕ۔ رِضامَندی دَرخاستَن ہُند قابلِ اعتماد جَواب بَناوُن چھُ تِمَن ادارَن خٲطرٕ آپریشن موقع یِم گوڈَے کھاتہٕ چھِ تھاوان۔

سَہامَتی (Sahamati) ہُند جُلایی 2026 ڈیش بورڈ چھُ تِمَن نشستَن آس پاس سرگرمی تصویر: 326.30 مِلین کُل جوڑِتھ کھاتہٕ، 538.32 مِلین کُل پُورٕ گَژھِتھ رِضامَندی، تہٕ صِرِف سَتھ رؠتہٕ 343.78 مِلین ڈیٹا فراہمی۔ پُورٕ گَژھِتھ رِضامَندی مَطلَب چھُ زِ ڈیٹا دِنہٕ آمُت، سَہامَتی ہُند تعریٖفَس پؠٹھ۔ موقع چھُ سَتھ دُہراو اِستِعمالَس مَنٛز، نَہ اَکِس لَٹہِ جوڑنُک مُہمَس مَنٛز۔

## دَستاوِزات، دستخط، تہٕ حُکوٗمَتی اے پی آی

ڈیجی لاکَر (DigiLocker) چھُ دَستاوِز دَروازٕ۔ 31 جُلایی 2026 تام ٲس اَتھ 72.43 کروڑ کھۄتہٕ زیٛادٕ دَرج شُدہ اِستِعمال کَرن وٲل تہٕ 5,437 دَستاوِز قِسم یا خِدمَت، یِمَن مَنٛز 4,776 رِیاسَتَن پؠٹھ۔ 29 جُلایی لوک سَبھا (Lok Sabha) جَوابَن رِپورٹ کٔر 936.03 کروڑ کھۄتہٕ زیٛادٕ دَستِیاب دَستاوِزات تہٕ 71.66 کروڑ کھۄتہٕ زیٛادٕ اِستِعمال کَرن وٲل، شُروع پؠٹھ ظٲہِر تَخصیٖص ₹168.58 کروڑ مُقابِل۔ محکمہٕ خٲطرٕ موقع چھُ جاری کَرن وول بَنُن یُس ہُند دَستاوِز حل گَژھِ۔ انحصار کَرن وٲلِ فَریق خٲطرٕ موقع چھُ سُہ دَستاوِز فوٹو کاپی بدَل قبول کَرُن۔ اِستِعمال کَرن وٲل بنیاد چھُ گوڈَے بٔڑ۔ زیٖٹھ پُچھ چھُ سُہ رِیاسَتی دَستاوِز قِسم یُس فِہرَستَس پؠٹھ چھُ تہٕ عملَس مَنٛز ناکام چھُ گَژھان۔

ای سائن (eSign) چھُ اَتھ نَذدیٖک دستخط دَروازٕ۔ 29 جُلایی پریس اِنفارمیشن بیورو (Press Information Bureau) کھاتہٕ وَنان زِ سٲری خِدمَت فراہم کَرن وٲلَن 155.56 کروڑ ای سائن جٲری کٔر، یِمَن مَنٛز 44.23 کروڑ سی ڈیکَن (CDAC)۔ اَکھ خِدمَت یُس وۄنۍ تہِ نفرَس پرنٹ، دستخط، تہٕ سکین پرژھان چھُ سۄ چھُ سُہ ریل اِستِعمال نَہ کَرُن چُنان یُس حُکوٗمَت سَتھ وَصفَس مَنٛز قانوٗنی طور قبول وَنان چھےٚ۔

اے پی آی سیتو (API Setu) چھُ فِہرَست دَروازٕ۔ سُہے کھاتہٕ وَنان زِ مرکزی تہٕ رِیاسَتی محکمَن پؠٹھ 8,904 اے پی آی تَتہِ شایع ٲسِہ، کھُلِہ اے پی آی پالیسی ہُند سَہارَس مَنٛز۔ میری پہچان (Meri Pehchaan) چھےٚ سنگل سائن آن پٲٹھۍ وَنان تاکہِ نفر پرؠتھ محکمہٕ خٲطرٕ نٔو لاگ اِن جمع کَرِ نَہ۔ اُمنگ (UMANG) چھُ موبایِل اَگہِ کِن دَروازٕ: 31 جُلایی 2026 تام 2,575 خِدمَت، 11.66 کروڑ کھۄتہٕ زیٛادٕ اِستِعمال کَرن وٲل۔ اَکھ محکمہٕ یُس تِمَن خِدمَتَن مَنٛز چھُ نَہ اَتھ چھُ ٹھوس شامِل کَرنُک کٲم۔ نجی فَرم ہؠکِہ نَہ اُمنگ بدلِتھ۔ یہِ ہؠکِہ اِضمام کٔرِتھ ییتہِ اے پی آی تہٕ سائن آن اِجازت دِیَن۔

مای سکیٖم (myScheme)، 29 جُلایی لوک سَبھا ضمیمَس مَنٛز، چھُ 4,750 کھۄتہٕ زیٛادٕ حُکوٗمَتی سکیٖم ہوسٹ کَران۔ اہلیتُک دریافت چھُ فایِدہ فراہمی کھۄتہٕ الگ موقع۔ ضمیمہ وَنان تہِ زِ نیشنل سکالرشِپ پورٹلَن (National Scholarship Portal) تَقریٖبَن 8.16 کروڑ سکالرشِپ تَقریٖبَن ₹53,300 کروڑ ڈایِرؠکٹ بِنِفِٹ ٹرانسفر (direct benefit transfer) ذریعہٕ جاری کٔر، تہٕ زِ ڈایِرؠکٹ بِنِفِٹ ٹرانسفرَن قومی سطحَس پؠٹھ ₹52 لَچھ کروڑ کھۄتہٕ زیٛادٕ منتقل کٔر۔ تِم رَقَم چھِ عوٲمی پائپ وَنان یِم گوڈَے چھِ ادا کَران۔ بَناون وٲلِس موقع چھُ سُہ پائپ آس پاس ٲخری میل تجربہ یُس رِیاسَت چھےٚ چَلان، نَہ سکالرشِپ پلیٹ فارمُک نجی کلون۔

## صحت: سینڈ باکس، پَتہٕ حِفاظَتی جانچ

آیوشمان بھارت ڈیجیٹل مِشن (Ayushman Bharat Digital Mission) سایِٹ چھُ سَتَمبَر 2021 مَنٛز شُروع گَژھِتھ مِشن وَنان۔ وَزارَتُک 24 جُلایی 2026 جَواب وَنان زِ ایپلیکیشن چھِ سینڈ باکسَس مَنٛز تصدیٖق گَژھان تہٕ اِضمام برٛونٛہہ ویب ایپلیکیشن حِفاظَتی جانچَس مَنٛز گَژھان۔ مرکزی کلینِکَل ذخیرہٕ چھُ نَہ۔ تَبادُل چھُ مریض رِضامَندی پَیر کَران۔ مِشن سایِٹَس پؠٹھ ناو دِنہٕ آمٕتۍ عمارَتی بلاکَن مَنٛز چھِ اے بی ایچ اے (ABHA) نمبر، ہیلتھ فَسِلِٹی رِجِسٹری (Health Facility Registry)، ہیلتھ کیئر پروکِشنَلز رِجِسٹری (Healthcare Professionals Registry)، تہٕ یوٗنِفایڈ ہیلتھ اِنٹرفیس (Unified Health Interface)۔

20 جُلایی 2026 تام ٲس ذخیرہٕ 94.87 کروڑ اے بی ایچ اے شِناخت، 5.36 لَچھ سہوٗلیت، تہٕ 10.09 لَچھ پیشہٕ ور۔ موقع چھُ سافٹ ویئر تہٕ آپریشن یِم سہوٗلیت یا عوٲمی پروگرامَس سُہ تَبادُلَس سۭتۍ شٲمِل گَژھنَس مَنٛز مدد کَران چھِ۔ جَواب چھُ گوڈَے مربوط پروگرام فِہرَست کَران، پی ایم-جے (PM-JAY) تہٕ نِکشَے (Nikshay) سَمیت، تہٕ رِیاسَتی نِظام یِتھ کٔنۍ گُجَراتُک (Gujarat) TECHO، اُتَر پردیشُک (Uttar Pradesh) eKavach، راجَستھانُک (Rajasthan) حمل تہٕ بچہ ٹریکِنٛگ نِظام، تہٕ تَمِل ناڈو (Tamil Nadu) ہُند صحت بیمہ سکیٖم۔ اَکھ کَمپنی یُس سُہ اِضمام سُہ سہوٗلیتَس خٲطرٕ دُہراوِتھ ہؠکِہ یُس وۄنۍ تام 5.36 لَچھ نؠبر چھُ سۄ چھےٚ نقشَس ہُند کھُلہِ حِصَس پؠٹھ کامہِ کَران۔ اَکھ کَمپنی یُس صحت ریکارڈَن ہُند قومی گودام تجویز چھُ کَران سۄ چھےٚ بیان کَرنہٕ آمُت ڈیزایِنَس خلاف کامہِ کَران۔

## تجارت: نؠٹ ورک شِرکَت دار، نَہ حُکوٗمَتی دوکانَس پؠٹھ لِسٹِنٛگ

ONDC ہُند موقع چھُ سَتھ پؠٹھ پَکان یُس تجارت وَزارَت وَنان چھےٚ زِ یہِ چھُ نَہ۔ 4 اَگَست 2026 جَواب وَنان زِ یہِ چھےٚ نَہ پلیٹ فارم، نَہ ایپلیکیشن، نَہ بازار۔ یہِ چھےٚ سؠکشن 8 کَمپنی (section 8 company) تہٕ کھُلہِ نؠٹ ورک۔ خریٖدار تہٕ فروش چھِ ایپلیکیشنَن درمیان لین دین کَران۔ نشست چھِ خریٖدار طرف ایپ، فروش طرف ایپ، لوژِسٹِکس، تہٕ، تِمَن شعبَن مَنٛز یِم جَواب فِہرَست چھُ کَران، نقل و حرکت تہٕ مالی خِدمَت تہٕ پرچون تہٕ زراعت۔

جُلایی 2026 تصویر چھےٚ 500 مِلین کھۄتہٕ زیٛادٕ کُل آرڈَر تہٕ تَقریٖبَن 2 لَچھ یا زیٛادٕ چالوٗ پرچون سوداگَر، تَقریٖبَن 60 فیصد لۄکُٹ فروش۔ ڈیجی دُکان (DigiDukaan)، کِرانا شامِل کَرنہٕ خٲطرٕ، اوس حیدَرآباد تہٕ جَے پوٗرَس مَنٛز چالوٗ۔ ₹277.35 کروڑ عوٲمی سَہار چھُ 2026-27 تام مایکرو تہٕ لۄکُٹ ادارَن خٲطرٕ چَلان، 50 فیصد زنانَن ہُند ہَدف سۭتۍ۔ عملی موقع چھُ فروش ایپ، لوژِسٹِکس شراکت دار، یا فیلڈ شامِل کَرنُک شراکت دار بَنُن تَتہِ ییتہِ ڈیجی دُکان وَنمُت چھُ نَہ۔ یہِ چھُ نَہ "ONDC پؠٹھ لِسٹ گَژھِو" یِتھ پٲٹھۍ زِ حُکوٗمَتی دوکان موجود آسہِ۔

## تٲلیٖم: سُہ پلیٹ فارمَس مَنٛز حصہٕ تُلِو یُس رِیاسَت گوڈَے چھِ چَلان

دِکشا (DIKSHA) ہُند تعارُفی صَفہٕ چھُ وسیع تٲلیٖم ماحول، غیر حُکوٗمَتی تنظیمَن سَمیت، مُشتَرِکہِ پلیٹ فارمَس پؠٹھ شِرکَت خٲطرٕ دعوت دیوان۔ رِیاسَت چھِ یہِ کِتھ پٲٹھۍ اِستِعمال کَرُن چھُ اِنتِخاب تھاوان۔ پلیٹ فارم چھُ 36 زبان سَہار کَران۔ یہِ چھُ سَنبَرڈ (Sunbird) پؠٹھ بَناونہٕ آمُت، یُس صَفہٕ ایم آی ٹی لائسنس کھُلہِ ذٔریع (MIT-licensed open source) وَنان چھُ۔ اِستِعمال کیس یِم یہِ ناو چھُ دیوان یِمَن مَنٛز چھِ توانائی وٲل درسی کِتاب، کورس، کوئز، سُوال بینک، تہٕ اُستاد پروگرام یِتھ کٔنۍ نِشتھا (NISHTHA)۔ مواد لائسنس چھُ بیان: این سی ای آر ٹی (NCERT) درسی کِتاب CC BY-NC-ND تَحت، تہٕ وسائل CC BY-NC-SA تَحت۔ موقع چھُ تِمَن لائسنسَن اندر مواد تہٕ رِیاسَتی پروگرام، خصوصی ضروٗرت وٲلِ بچَن خٲطرٕ مواد سَمیت، یُس صَفہٕ وَنان چھُ زِ گوڈَے زمرہ چھُ۔ یہِ چھُ نَہ خصوصی قومی درسی کِتاب فرنچایِز۔

## یِمَن دَستاوِزَن پؠٹھ کوتاہ موقع چھُ نَہ

"DPI بازار آمدنی" ہُند شایع ذخیرہٕ چھُ نَہ یُس تقسیم کَرنہٕ یِیہِ۔ ڈیجی لاکَرُک شُروع پؠٹھ ظٲہِر تَخصیٖص ₹168.58 کروڑ چھُ عوٲمی بِناوُن خَرچ، نَہ فروخت ہَدف۔ کھاتہٕ جَمع کَرن وٲل ہؠکَن نَہ ڈیٹا پؠٹھ بیٖہِتھ تہٕ یہِ بیورو پٲٹھۍ کماوِتھ۔ ONDC ہُند آرڈَر شُمار چھُ نَہ مال قۭمَت۔ آدھار چھُ نَہ فروختُک ڈیٹا سیٹ؛ UIDAI ہُند سیچوریشن رِپورٹ چھُ کوریج شماریات۔

دَروازٕ یِم کھُلِہ چھِ چھِ کردار: ریگولیٹ کَرنہٕ آمُت معلومات اِستِعمال کَرن وول، دَستاوِز جاری کَرن وول، اے پی آی شایع کَرن وول، صحت ایپلیکیشن یُس جانچ پاس کَرِ، نؠٹ ورک شِرکَت دار، رِیاسَتی تٲلیٖم پروگرام، سوداگَر یا بینک یُس وۄنۍ تام شایع شُمارَن نؠبر چھُ۔ پرؠتھ کردارَس چھُ دَستاوِز یُس وَنان چھُ زِ کُس داخِل گَژھِتھ ہؠکِہ۔ کردارَس پؠٹھ شُروع کَرُن چھُ سُہ طٔریٖقہٕ یِتھ پٲٹھۍ یہِ منظر اصل چھُ روزان۔
