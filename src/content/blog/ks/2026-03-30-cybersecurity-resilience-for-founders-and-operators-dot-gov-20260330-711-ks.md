# ہِندوستانٕکٮ۪ن سائبر قَواعِدَن تحت بٲنی تہٕ آپریٹَر کیاہ واجب چھِ

شُروع کٔرِو یہِ پرژھِتھ زِ ادارَس کَہ لیبَل لاگو چھِ۔ زِیادٕ تَر بَڑان فَرمَن چھِ اَکھ کھۄتہٕ زِیادٕ۔ فَرٲیِض چھِ نَہ سارِوٕے کھۄتہٕ سخت اَکی پالیسیَس منٛز کَم گَژھان۔

## اگر تُہۍ باڈی کارپوریٹ، اِنٹَرمیڈیَری، ڈیٹا سینٹَر، یا سَرکٲری ادارٕ چھِو

ہِندوستٲنؠ کَمپیوٹَر ایٚمَرجَنسی رِسپانس ٹیم (Indian Computer Emergency Response Team, CERT-In) ہٕنٛز 28 اپریل 2022 ہٕنٛز ہِدایَت چھِ خِدمَت دِن والٮ۪ن، اِنٹَرمیڈیَریَن، ڈیٹا سینٹَرَن، باڈی کارپوریٹَن تہٕ سَرکٲری ادارَن پؠٹھ لاگو۔ مَتن منٛز چھِ زٕ آپریٹَر سطحٕک فَرٲیِض واضح۔

ہِدایَت ہٕنٛدِس ضمیمَس منٛز سائبر واقَع گَژھَن تِم لَبنہٕ یا تِمَن متعلق وَننہٕ یِنہٕ پَتہٕ 6 گَنٛٹَن اَندر رِپورٹ کَرنہٕ۔ ہِدایَتَس منٛز چھَپنہٕ آمٕتۍ چینل چھِ incident@cert-in.org.in، فون 1800-11-4949 تہٕ فیکس 1800-11-6969۔ 6 گَنٛٹہٕ چھِ نوٹِسَس پٮ۪ٹھ، نَہ تَفتیش مُکمل گَژھنہٕ پٮ۪ٹھ۔

آئی سی ٹی (ICT) نِظامَن ہٕنٛد لاگ چالو کٔرِو تہٕ تِم ہِندوستٲنؠ حَدَس اَندر رولِنگ 180 دۄہَن تام محفوٗظ تھٲوِو۔ تِم واقَع رِپورٹَس سۭتۍ پیش کٔرِو، یا یَلۍ CERT-In حُکم دِیہِ۔

پَرِو ضمیمہٕ زِ کَہ واقَع قٲبِل چھِ، تہٕ پَرِو 27 جوٗن 2022 ہٕنٛز توسیع اگر تُہۍ اَکھ خُرد، لَکُٹ یا درمیٲنی ادارٕ (MSME) چھِو یا اگر تُہۍ ڈیٹا سینٹَر، وی پی ایس (VPS)، کلاؤڈ یا وی پی این (VPN) خِدمَت چَلاوان چھِو۔ توسیعُک عنوان، CERT-In ہٕنٛدِس ہِدایَت صَفَس پؠٹھ، چھُ MSME تہٕ سبسکرایبَر تصدیٖقی اِنتِظامَن خٲطرٕ نافِذ گَژھنُک وَقت کور کران۔ فایِل پنُنۍ چھِ تَمہِ تٲریٖخَس خٲطرٕ ذَریعہٕ یُس اپریل ہِدایَتَہِ نِش الگ چھُ۔

اَکھ نَفَر ناو کٔرِو یُس CERT-In سۭتۍ کَتھ کٔرِتھ ہیکہِ۔ MSME بیس لائن چھُ تَس CERT-In تہٕ ریگولیٹَرَن خٲطرٕ اَکھ اَکی رابطہٕ نُقطہٕ وَنان۔

## اگر تُہۍ MSME درجہٕ بندیَس منٛز یِوان چھِو

1 سَتَمبَر 2025 ہٕنٛز بیس لائن چھِ تِمَن ادارَن پؠٹھ لاگو یِم خُرد، لَکُٹ تہٕ درمیٲنی اَدارَن ہٕنٛدِس وَزارَتَس (Ministry of MSME) ہٕنٛزِ 1 جوٗن 2020 ہٕنٛزِ اِطلاع S.O. 1702(E) تحت چھِ۔ یہِ چھُ اَکھ کَم از کَم: 15 کنٹرول، 45 سفارش، پینل شُدہ ادارَہٕ سۭتۍ تجویز شُدہ سالانہ آڈِٹ، تہٕ اَکھ اِخبار زِ تُہۍ پنِنٮ۪ن خطرَن خٲطرٕ وُنہِ تہِ برٛونٛہہ گَژھُن چھُ۔

کنٹرول یِم آپریٹَر یِمہِ رٮ۪تھَس اَکھ مالکَس سونپِتھ ہیکہِ چھِ ٹھوس۔ ہارڈویئر، سافٹ ویئر تہٕ اِطلاعٲتی اثاثَن ہُند فِہرِست۔ حَدَس پؠٹھ تہٕ ہوسٹَس پؠٹھ فایَروال تھٲوِو۔ WPA2 یا WPA3 اِستِعمال کٔرِو، تہٕ مہمان وائی فائی الگ تھٲوِو۔ SPF، DKIM تہٕ DMARC چالو کٔرِو۔ لائسنس یافتہٕ اینڈ پواینٹ حِفاظَت اِستِعمال کٔرِو تہٕ سائبر سَوَچھتا کیندر (Cyber Swachhta Kendra) سۭتۍ آن بورڈ کٔرِو۔ USB آٹو رَن محدود کٔرِو۔ ڈیفالٹ پاسوَرڈ ہٹٲوِو۔ وینڈَر تہٕ CERT-In صلاحَن پٮ۪ٹھ پیچ کٔرِو۔ اَکھ واقَع منصوٗبہٕ لِکھِو تہٕ تَس ٹیسٹ کٔرِو۔ 180 دۄہَن ہٕنٛد لاگ تھٲوِو۔ اَفٲدۍ ؤریَس منٛز زٕ لَٹہِ تَربِیَت دِیو۔ وینڈَرَن کَم از کَم یہِ بیس لائن لازِم تھٲوِو۔ بیک اپ کٔرِو، بحالی ٹیسٹ کٔرِو، تہٕ نازُک ایپلیکیشنَن خٲطرٕ اَکھ لَکُٹ تسلسُل منصوٗبہٕ تھٲوِو۔ رَسٲیی کَم از کَم ہر رٮ۪وارَس تہٕ پرٛٮ۪تھ نؠبر نیرنَس پؠٹھ جٲیزٕ کٔرِو۔

## اگر تُہۍ ڈِجِٹَل ذٲتی ڈیٹا پروسیس کران چھِو

تُہۍ چھِو 11 اَگَست 2023 نافِذ گٔمِتِس ایکٹَس تحت ڈیٹا فِڈیوشَری (Data Fiduciary) عَلاقَس منٛز۔ 24 سَتَمبَر 2026 تام، ٹیمَس یہِ وَننہٕ برٛونٛہہ زِ اَکھ فریضہٕ نافِذ چھُ، نافِذ گَژھنُک شق وُچھِو۔ قَواعِد 1، 2 تہٕ 17 پٮ۪ٹھ 21 چھِ نافِذ۔ قاعِدہٕ 4، رضامندی منیجَر (Consent Managers)، چھُ 13 نَوَمبَر 2025 پَتہٕ اَکھ ؤری یِوان۔ نوٹِس، حِفاظَت، خلاف ورزی اِطلاع تہٕ اہم فِڈیوشَری قَواعِد چھِ تَمہِ اشاعَتَس پَتہٕ 18 رٮ۪تَن یِوان۔

آپریٹَرَن گَژھِ وُنہِ تہِ دستاویز تَیار کَرٕنۍ، تہٕ تِمَن گَژھِ نافِذ گَژھنُک تٲریٖخ لیبل کَرُن۔ یَلۍ قاعِدہٕ 6 نافِذ آسہِ، کَم از کَم حِفاظَت فِہرِستَس منٛز چھِ انکرِپشن، اوبفیسکیشن، ماسکِنگ یا ورچوَل ٹوکن؛ رَسٲیی کنٹرول؛ لاگ تہٕ جٲیزٕ؛ بیک اپ؛ اَکھ ؤری لاگ رکاوَٹ یِلہِ نَہ بیٛاکھ قانوٗن الگ وَنِ؛ پرٛٮ۪تھ ڈیٹا پروسیسرَس خٲطرٕ اَکھ معاہدٕ شق؛ تہٕ تَکنیکی تہٕ تنظیمی اِقدام یِم حِفاظَت حقیقی بناوان چھِ۔ قاعِدہٕ 7 چھُ پَتہٕ متٲثِر اَفٲدۍ سادٕ زَبانَس منٛز بغیر دیر نوٹِس، بورڈَس فَوری بیان، تہٕ 72 گَنٛٹَن اَندر تفصیٖلی رِپورٹ لازِم کران، یِلہِ نَہ بورڈ زیوٗٹھ وَقت دِیہِ۔

PIB چھُ وَنان زِ اَفٲدۍ گَژھِ اَکھ رابطہٕ لَبِتھ ہیٚکِن، تہٕ زِ رَسٲیی، اصلاح، تازہ کاری، مِٹاؤن تہٕ نامزدگی ہٕنٛزِ دَرخواستَن ہُند جَواب چھُ کَم از کَم 90 دۄہَن اَندر دِنہٕ یِوان۔ قاعِدہٕ 14، یُس 18 رٮ۪تَن والٮ۪ن سیٹَس سۭتۍ نافِذ چھُ گَژھان، چھُ شِکایَت نِظام پنُنۍ شایع کَرُن تہٕ معقوٗل مُدتَس منٛز جَواب دِیُن لازِم کران یُس 90 دۄہَن کھۄتہٕ زِیادٕ آسہِ نَہ۔

## اگر اَکھ ریگولیٹَر گوڈَے تُہٕنٛز نِگرانی کران چھُ

بینک، مخصوص این بی ایف سی (NBFC)، کریڈِٹ اِنفارمیشن کمپنی تہٕ ناو دار آل اِنڈیا فائنیشَل اِنسٹِٹیوشَن چھِ 1 اپریل 2024 پٮ۪ٹھ نافِذ ہِندوستانُک ریزرو بینک (Reserve Bank of India) ہٕنٛزِ ہِدایَتَن پٮ۪ٹھ چَلان، شٲمِل اَکھ چیف اِنفارمیشن سِکیورٹی آفیسَر (Chief Information Security Officer, CISO) یُس آئی ٹی سربراہَس رِپورٹ چھُ نَہ کران، تہٕ CERT-In تہٕ ریزرو بینکَس مُطلِع کَرنُک فریضہٕ۔ ہاؤسِنگ فاینانس کمپنی چھِ سائبر واقَع قومی ہاؤسِنگ بینکَس (National Housing Bank) رِپورٹ کران۔

سیکیورٹیز اینڈ ایکسچینج بورڈ آف اِنڈیا (Securities and Exchange Board of India, SEBI) تحت ادارٕ چھِ سائبر سِکیورٹی اینڈ سائبر ریزِلیَنس فریم وَرک (Cybersecurity and Cyber Resilience Framework, CSCRF) پٮ۪ٹھ چَلان، شٲمِل یہِ زَمرٕ یَس منٛز تِم یِمہِ مالی ؤریَس خٲطرٕ تھاونہٕ آیہِ تہٕ سلامتی آپریشن مرکز (security operations centre, SOC) وَتھ یُس تَمہِ زَمرَس ضروٗرَت چھُ۔ بیمہٕ کُنَندہٕ تہٕ ہِندوستانُک بیمہٕ ریگولیٹری تہٕ ترقٲتی اتھارٹی (Insurance Regulatory and Development Authority of India, IRDAI) اِنٹَرمیڈیَری چھِ 24 اپریل 2023 ہٕنٛزِ ہِدایَتَن تہٕ اِنفارمیشن سِکیورٹی رِسک کمیٹی پٮ۪ٹھ چَلان۔

ٹیلی مواصلٲتی ادارٕ چھِ 6 گَنٛٹَن تہٕ 24 گَنٛٹَن ہٕنٛدِ رِپورٹَن پٮ۪ٹھ چَلان، اَکھ رہٲیِشی ہِندوستٲنؠ چیف ٹیلی کمیونیکیشن سِکیورٹی آفیسَر (Chief Telecommunication Security Officer) مقرر کران، تہٕ 2024 ہٕنٛزِ قَواعِدَن منٛز SOC تہٕ آڈِٹ فَرٲیِض چَلاوان۔ فَرم یِم لائسنس یافتہٕ چھِ نَہ مگر گاہَکَن پہچانُنَس خٲطرٕ ٹیلی کام شناسہٕ اِستِعمال کران چھِ، تِم گَژھَن اَکتوٗبَر 2025 ہٕنٛز ترمیم پَرُن برٛونٛہہ زِ تِم پٲن پانَس ٹیلی کام سائبر قَواعِدَن نؠبر مانِن۔

آپریٹَر عادت یُس تِمَن سارِوٕے لیبَلَن پَتہٕ بچان چھُ چھُ اَکھ صَفُک میٹرِکس: واقَع، کَس وَنُن چھُ، کوتاہ گَنٛٹَن منٛز، تہٕ کَہ لاگ یہِ ثابِت کران چھُ۔
