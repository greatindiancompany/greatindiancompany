# ہِندوستانَن ہِنٛد سَمٲجی نٔتیٖجَن ہُنٛد ناہَموار درمیٲنی مُدَتی رَخ

پٔتمِس سَروے وقفَس پؠٹھ ہِندوستانَن ہِنٛد سَمٲجی نٔتیٖج چھِ نَہ اَکھ وِکر۔ قَد-عُمر، بیٖمہ، تہٕ اِدارَتی زَچِہ ہِلِتھ۔ وَزن-قَد، کَم وَزن، تہٕ زَرخیزی ہِلِتھ نَہ، اتھہِ طریقَس پؠٹھ یُس اَکھ منصوٗبہ بند تبدیلی وَنِہ۔ سکوٗل شِرکَت ژور UDISE+ وَرِیَن پؠٹھ بہتر گٔیہِ تہٕ وَنان تہِ چھِ تَقریٖبَن اَدھ شاگِرد ثانَوی برقرار رکاوَٹ شرحَس نیبر ترٛاوان۔ درمیٲنی مُدَت چھِ سُہ وَنٛژھُن۔

## شُر پۄشتہِ: اَکھ صاف فٲیدٕ، زٕ ہَموار سَطر، اَکھ اُلٹاؤ

NFHS-5 چھُ 2019-21 کُن اشارہ کران۔ NFHS-6 مَیدٲنی کٲم چھِ 2023-24، 29 مَے 2026 جٲری۔ پانٛژھ وَرِیَن بۄن شُرٮ۪ن خٲطرٕ، قَد-عُمر کَمی (stunting) 35.5% پؠٹھ 29.3% تام وَتھ۔ وِزارَت ہُنٛد سُرخی چھُ یہِ 17% کَمی وَنان، یُس پرٛون بُنیادَس پؠٹھ نِسبَتی بدلاو چھُ (تَقریٖبَن شےٚ فیصد نُکتہ)، نَہ سَتَہ نُکتَن ہُنٛد گرُن۔ شَدیٖد وَزن-قَد کَمی (severe wasting) 7.7% پؠٹھ 5.2% تام وَتھ۔ وَزن-قَد کَمی (wasting) 19.3% پؠٹھ 19.0% تام گٔیہِ۔ کَم وَزن (underweight) 32.1% پؠٹھ 31.8% تام گۆو۔

سُہ وقفَس پؠٹھ شیر خوار کھٮ۪ن چھَوان چھُ تہِ وَنٛژھِتھ۔ زَیِشَس پؠٹھ اَکھ گھَنٹَس اندر شروع کرُن 41.8% پؠٹھ 50.1% تام بَڑاو۔ 6-8 رٮ۪تَن پؠٹھ ماے دۄدَس سٕتی ٹھوس یا نیم ٹھوس کھٮ۪ن 45.9% پؠٹھ 59.5% تام بَڑاو۔ 6-23 رٮ۪تَن پؠٹھ کٲفی غِذا 11.0% پؠٹھ 15.3% تام بَڑۍ۔ شےٚ رٮ۪تَن بۄن صِرف ماے دۄد (exclusive breastfeeding) 63.7% پؠٹھ 55.8% تام وَتھ۔ شےٚ رٮ۪تَن بۄن موجوٗد دۄد چھَوان ہیوٚر روز، 95.6% بمقابلہ 95.1%۔ اَکھ رَخ سلایِڈ یُس صِرف قَد-عُمر کَمی ہاوان چھُ، سُہ چھُ کھٮ۪ن چھَوان اُلٹاؤ تہٕ تِم وَزن اِشارٕ ترٛاوان یِم قَدَس پَتہٕ گٔیہِ نَہ۔

یُس پروگرام یَتھ وَقتَس آس پاس چھُ، سُہ چھُ مِشَن پوشَن 2.0 (Mission Poshan 2.0)، یُس سکیٖم ہِدایَت چھِ وَنان زِ 15 ہم فِنٲنس کَمِشَن (Finance Commission) وَرِیَن 2021-22 پؠٹھ 2025-26 تام منظوٗر آو، اَنٛگنواڑی (Anganwadi) خٔدمَت، پوشَن اَبھِیان (Poshan Abhiyaan)، تہٕ اَسپیریشَنَل ضِلعن تہٕ شُمال مَشرِق (North East) مَنٛز کِشوٗر کوٗرٮ۪ن ہِنٛز مَدَد جوڑِتھ۔ سَروے کِھڑکۍ تہٕ سکیٖم کِھڑکۍ چھِ اوورلیپ کران۔ تِم چھِ نَہ اَکھ چیز: اَکھ خانَدٲنی سَروے چھُ شُر ناپان، تہٕ اَکھ سکیٖم منظوٗری چھُ خرچ کرنُک اِجازَت ناپان۔ ہِدایَت چھِ نَہ دعوٕ کران زِ منظوٗری پانَے 6.2 نُکتہ قَد-عُمر کَمی گرُن پٲدٕ کۆر۔

## زَچِہ دیکھ بھال تہٕ مالی حِفاظَت زَرخیزی کھوتہٕ زیٖادٕ ہِلِتھ

اِدارَتی زَیِش 88.6% پؠٹھ 90.6% تام بَڑۍ۔ ماہِر حاضِری 89.4% پؠٹھ 91.3% تام بَڑۍ۔ زٕ دۄہَن اندر نوزائیدَس خٲطرٕ زَیِش پَتہٕ دیکھ بھال 79.1% پؠٹھ 85.3% تام بَڑۍ۔ ژور یا زِیادٕ زَیِش برٛونٛہہ دورٕ 58.5% پؠٹھ 65.2% تام بَڑۍ۔ 180 دۄہ یا زِیادٕ خٲطرٕ لۄہ تہٕ فولِک تیزاب 26.0% پؠٹھ 37.8% تام بَڑاو۔ 12-23 رٮ۪تَن پؠٹھ کارڈَن پؠٹھ پوٗرٕ ویکسین 83.8% پؠٹھ 87.1% تام بَڑۍ۔ اَکھ صِحَتی بیٖمہ یا مالی سکیٖم سٕتی خانَدٲنی کَوَریج 41.0% پؠٹھ 60.2% تام بَڑاو۔

کُل زَرخیزی شرح 2.0 پؠٹھ روز۔ مانع حمل پھیلاو 66.7% پؠٹھ 69.1% تام بَڑاو۔ رَخ چھُ پرَٛتھ زَیِش آس پاس مزید دیکھ بھال، تہٕ اَکھ زَرخیزی سطح یُس پٔتمِس سَروےَس مَنٛز گوڈَے تبدیلی سطحَس تام وٲتِتھ اوس تہٕ پَتہٕ روز۔ سیزیرین زَیِش 21.5% پؠٹھ 27.2% تام بَڑۍ۔ یہِ چھُ خٔدمَت مِکسَس مَنٛز اَکھ رَخ، تہٕ تَس چھُ اَکھ معیار پرُن درکار یُس فیکٹ شیٹ چھُ نَہ دِوان۔

خون کَمی (anaemia) چھِ یَتھ رَخَس پؠٹھ غائب۔ 29 مَے 2026 نوٹ چھُ نَہ اَکھ نٔو پھیلاو دِوان، تہٕ فیکٹ شیٹ پۄشتہِ سَطر یِم ہیٚوٚر مُقابَلَن خٲطرٕ اِستعمال گٔیہِ، تِم تہِ چھِ نَہ۔ NFHS-5 خون کَمی یِتھ کَنِہ برٛونٛہہ نِنُن یِتھ کَنِہ زِ یہِ اَکھ NFHS-6 نتیجہ آسِہ، چھُ سلسلہ توڑان۔

## سکوٗل: ژور وَرِیَن ہِنٛز رَساہی، اَکھ ثانَوی شرح یُس وَنان تہِ بۄن چھُ

7 جُلایی 2026 جٲری UDISE+ چھُ اَکھ ژور وَرِیُک سَہوٗلَت تہٕ شِرکَت رَخ دِوان، 2022-23 پؠٹھ 2025-26 تام۔ اُستاد 94,83,294 پؠٹھ 1,02,73,020 تام بَڑۍ۔ تیاری سکوٗل ترٛاوُن گۆو 8.7%، 3.7%، 2.3%، 1.8%۔ ثانَوی سکوٗل ترٛاوُن گۆو 13.8%، 10.9%، 8.2%، 7.0%۔ سُہ ٹیبلَس مَنٛز درمیٲنی سلسلہ گۆو 8.1%، 5.2%، 3.5%، 3.6%۔ ثانَوی برقرار رکاوَٹ 2022-23 مَنٛز 44.1% پؠٹھ 2024-25 مَنٛز 47.2% تہٕ 2025-26 مَنٛز 51.9% تام گٔیہِ۔ تازہ وَرِیَس مَنٛز ثانَوی کُل داخِلہ چھُ 71.7%، 68.5% پؠٹھ۔

سکوٗل ترٛاوُنَس پؠٹھ سمت چھُ اصلی۔ ثانَوی برقرار رکاوَٹَس پؠٹھ سطح چھُ مطلب زِ درمیٲنی مُدَتی مسئلہ «شُر چھِ جَماعَت 3 مَنٛز ترٛاوان» پؠٹھ «شُر چھِ نَہ ثانَوی مَرحَلہ پوٗرٕ کران» کُن بدلاو۔ سکوٗلَن مَنٛز کَمپیوٗٹَر 2022-23 مَنٛز 47.7% پؠٹھ 2025-26 مَنٛز 69.9% تام بَڑۍ۔ یہِ چھُ اَکھ اِنفراسٹرکچَر رَخ۔ یہِ چھُ نَہ اَکھ لَرنِنٛگ رَخ۔ یَتھ جٲری مَنٛز چھِ نَہ کامیٲبی سکور۔

## بیٚیُم سرکٲری نُکتہ کیاہ ژٔٹِہ

بیٚیُم پۄشتہِ نُکتہ گَژھِہ بیاکھ سَروے یا اَکھ صاف لیبل شُدہ اِنتِظامی سلسلہ آسُن، نَہ 29.3% ہُنٛد دُبارٕ بیان۔ بیٚیُم سکوٗل نُکتہ گَژھِہ وَنُن زِ ثانَوی برقرار رکاوَٹ 51.9% پؠٹھ چھا ہِلان تہٕ زِ درمیٲنی سکوٗل ترٛاوُن 3.5% پؠٹھ 3.6% تام ہُنٛد ٹِک چھا اُلٹان۔ دیہی مَزدوٗری رَخ چھُ 1 جُلایی 2026 پؠٹھ اَکھ نٔوِ قانوٗنَس پؠٹھ دُبارٕ شروع گومُت، تَتھ کِنۍ مالی وَرِیہ 2025-26 MGNREGS کُل چھِ اَکھ سلسلہ ہُنٛد اَنٛد، نَہ بیٚیُمِک بُنیاد، یَتام زِ اَکھ وِزارَتی اشاعت تِم صاف جوڑِہ۔
