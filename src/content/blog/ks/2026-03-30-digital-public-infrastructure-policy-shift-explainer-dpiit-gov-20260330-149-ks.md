# ہِندوستان کِتھ پٲٹھۍ ڈیجیٹل عوٲمی بُنیاد پنٛنہِ دَستاوِزَن مَنٛز وَنان چھُ

ہِندوستانُک ڈیجیٹل عوٲمی بُنیاد (digital public infrastructure) چھُ نعرٕ پٲٹھۍ غلط اِستِعمال کَرُن آسان، پالیسی پٲٹھۍ پرُن مشکل۔ سرکٲری دَستاوِز چھِ اَکھ مخصوص تبدیٖلی وَنان: پرؠتھ محکمہٕ ہُند سافٹ ویئر بَند نِظام مانُن بَند کٔرِو، تہٕ کینٛژ مُشتَرِکہِ کٲم شایع کٔرِو یِم واریاہ ایجنسی تہٕ فَرم اِستِعمال کٔرِتھ ہؠکَن۔

سَتھ تبدیٖلی چھےٚ پروگرامَس مَنٛز اَکھ شُروع تٲریٖخ، تہٕ پَتہٕ اَکھ تٲریٖخ ییلہِ ہِندوستانَن بیٚیہِ حُکوٗمَتَن سُہے زبان اِستِعمال کَرنُک وَن۔

## پروگرام، پَتہٕ فقرہ

پریس اِنفارمیشن بیورو (Press Information Bureau)، 29 جُلایی 2026 حُکوٗمَتی کھاتہٕ رِپورٹ کَران، وَنان زِ ڈیجیٹل اِنڈیا (Digital India) پروگرام جُلایی 2015 مَنٛز شُروع گٔیہِ۔ سُہے اِطلاع وَنان زِ حُکوٗمَتَن اَکھ ڈیجیٹل عوٲمی بُنیاد طٔریٖقہٕ اختیار کٔر تاکہِ ڈیجیٹل خِدمَت آبادی پیٖمانَس پؠٹھ کھُلِہ، باہمی چَلن وٲل، تہٕ بَڑاونہٕ یِن وٲل پلیٹ فارمَن ذریعہٕ دِنہٕ یِیہِ۔

اِطلاع چھُ پَتہٕ کٲم سادٕ لَفزن مَنٛز ناو دیوان۔ بُنیٲدی ڈیجیٹل عوٲمی بُنیاد، یہِ وَنان چھُ، چھےٚ آبادی پیٖمانَس پؠٹھ شِناخت (آدھار، Aadhaar) خٲطرٕ، ادایگی (یوٗنِفایڈ پیمنٹس اِنٹرفیس، Unified Payments Interface, UPI) خٲطرٕ، تہٕ ڈیٹا تَبادُل (ڈیجی لاکَر، DigiLocker، تہٕ اے پی آی سیتو، API Setu) خٲطرٕ نافذ کَرنہٕ آمٕژ۔ سُہ جُملہ چھُ گَریلوٗ پالیسی بیان۔ یہِ چھُ "ڈیجِٹائزیشن" ہُند عام دعوٲ کھۄتہٕ زیٛادٕ ٹھوس۔

یِمَن ترٛؠن کٲمَن آس پاس وَنان سُہے اِطلاع تِم نِظام یِم اَکھ سکیٖم کھۄتہٕ مُشتَرِکہِ پَلَمبنٛگ چھِ: میری پہچان (Meri Pehchaan) اَکھ سنگل سائن آن پٲٹھۍ، اے پی آی سیتو حُکوٗمَتی اے پی آی ہُند فِہرَست پٲٹھۍ (سَتھ کھاتَس مَنٛز 8,904 اے پی آی شایع)، ای سائن (eSign) آن لایِن دستخط خِدمَت پٲٹھۍ، تہٕ کامن سروس سینٹر (Common Service Centre) سَہایَتی رَسایِی پٲٹھۍ ییتہِ اَکھ نفر لین دین پَنہٕ پانَے مُکَمَل چھُ نَہ کَران۔

## جی ٹوَنٹیَن اَتھ کوتاہ وَنُن منظور کٔر

ہِندوستانٕکۍ 2023 صدارتَس دوران لؠکھ جی ٹوَنٹیَن (G20) یہِ خیال حُکوٗمَتَن ہُند وسیع گروپَس خٲطرٕ۔ 9 سَتَمبَر 2023 ہُند نئی دِلّی لیڈَرز ڈِکلیریشن (New Delhi Leaders' Declaration) چھُ ڈیجیٹل عوٲمی بُنیاد اَکھ بَدلان تصور وَنان: مُشتَرِکہِ ڈیجیٹل نِظامَن ہُند سیٹ، یُس عوٲمی تہٕ نجی دۄشوٕے شعبَن چھُ بَناوان تہٕ اِستِعمال کَران، محفوظ تہٕ لَچکدار بُنیادَس پؠٹھ، یُس کھُلِہ معیارات تہٕ مُشَخَصاتَن پؠٹھ تہٕ تہِ کھُلِہ ذٔریع سافٹ ویئرَس پؠٹھ بَنِتھ ہؠکِہ، تہٕ یُس سماجی پیٖمانَس پؠٹھ خِدمَت دِتھ ہؠکِہ۔

اعلان چھُ ڈیجیٹل عوٲمی بُنیاد نِظامَن خٲطرٕ اَکھ رضاکارانہ، تجویز کَرنہٕ آمُت جی ٹوَنٹی فریم ورک (G20 Framework for Systems of Digital Public Infrastructure) خوش آمدید وَنان۔ ڈیجیٹل معیشت وَزیرَن ہُند فریم ورک، 19 اَگَست 2023 تٲریٖخ، چھُ صاف زِ رُکن چھِ ڈیجیٹل تبدیٖلی الگ طٔریٖقَن سۭتۍ کَران، تہٕ تصور ہؠکہِ ڈھالِتھ تہٕ تہِ الگ ناو دِتھ۔ "عوٲمی،" سَتھ مَتن مَنٛز، ہؠکِہ عوٲمی فایِدہ تہٕ رَسایِی مَطلَب آسِتھ، عوٲمی اتھارٹی ہُند حکمرانی تہٕ نگرانی تَحت۔ اَمہِ مَطلَب چھُ نَہ زِ پرؠتھ جُز گَژھِ اَکھ حُکوٗمَتی محکمہٕ آسُن۔

وَزیرَن ہُند ضمیمہ چھُ ترٛے بُنیٲدی کٲم ناو دیوان:

- شِناخت: لوٗکھ تہٕ کاروبار ہؠکَن شِناخت محفوظ پٲٹھۍ تصدیٖق کٔرِتھ، تکمیٖلی اعتماد خِدمَتَن سۭتۍ یِتھ کٔنۍ الیکٹرانک دستخط تہٕ تصدیٖق یِن وٲل اسناد۔
- ادایگی: لوٗکھَن، کاروبارَن، تہٕ حُکوٗمَتَن درمیان پونٛسہٕ ہُند آسان تہٕ فوری منتقلی۔
- رِضامَندی سۭتۍ ڈیٹا شریک کَرُن ییتہِ یہِ لاگو چھُ: ذٲتی ڈیٹا ہؠکِہ عوٲمی تہٕ نجی شعبَن درمیان پَکِتھ، لاگو ڈیٹا حکمرانی فریم ورکَس تَحت حفاظَتَن سۭتۍ۔

ہِندوستانُک سِٹیک چھُ سَتھ فِہرَستَس پؠٹھ زیٛادٕ تَرجَمہٕ بغیر بیٹھان۔ آدھار چھُ شِناخت نِظام۔ UPI چھُ ادایگی نِظام۔ ڈیجی لاکَر، اے پی آی سیتو، تہٕ کھاتہٕ جَمع کَرن وول (Account Aggregator) فریم ورک چھِ ڈیٹا شریک کَرنُک الگ جَواب۔ تِم چھِ نَہ اَکھ جَواب۔ ڈیجی لاکَر چھُ اتھارٹی جاری کَرنہٕ آمُت دَستاوِز تھاوان یا تُلان۔ اَکھ کھاتہٕ جَمع کَرن وول، ریزرو بینک (Reserve Bank) ہُند ہِدایتَن تَحت، چھُ مالی معلومات تُلان تہٕ یہِ گَژھِ نَہ تھاوُن۔

## ہِندوستٲنۍ نِظامَن مَنٛز "کھُلہِ" کوتاہ مَطلَب چھُ

کھُلہِ مَطلَب چھُ نَہ بے قاعِدہ، تہٕ یہِ مَطلَب چھُ نَہ اَکھ حُکوٗمَتی بازار۔

کھاتہٕ جَمع کَرن وول ڈیزایِن چھُ سَبہٕ کھۄتہٕ صاف قانوٗنی بیان۔ ریزرو بینک آف اِنڈیا (نان بینکِنٛگ فِنانشَل کَمپنی – اکاوٗنٹ ایگریگیٹر) ہِدایات، 2025 (Reserve Bank of India (Non-Banking Financial Companies – Account Aggregator) Directions, 2025)، 28 نَوَمبَر 2025 جاری، چھِ کاروبار یِتھ پٲٹھۍ تعریٖف کَران زِ مخصوص مالی معلومات تُلِن یا جَمع کَرِن تہٕ یہِ گاہَکَس یا مالی معلومات اِستِعمال کَرن وٲلِس پیش کَرِن۔ گاہَکَس ہُند مالی معلومات چھُ نَہ کھاتہٕ جَمع کَرن وٲلِس ہُند مِلکیَت تہٕ یہِ گَژھِ نَہ کانٛہہ بیٚیِس طٔریٖقَس مَنٛز اِستِعمال کَرنہٕ۔ ہِدایات وَنان زِ کھاتہٕ جَمع کَرن وول دِیہِ خِدمَت صِرِف صریٖح رِضامَندی پؠٹھ، لین دینَس سَہار دِیہِ نَہ، تہٕ مالی معلومات رُزِ نَہ اَتھ سۭتۍ۔ رِضامَندی گَژھِ اَکِس معیاری شے مَنٛز پَکُن یُس مقصد، وصول کَرن وٲل، تہٕ مُکَمَل گَژھنُک وَقت وَنان آسہِ۔

تجارت وَرژن سُہے خیالُک چھُ ادارَتی، نَہ ادایگی قاعِدہ۔ 4 اَگَست 2026 وون تجارت تہٕ صنعت ہُندِ وَزارَتَن (Ministry of Commerce and Industry) لوک سَبھا (Lok Sabha) ہَس زِ اوپن نیٹ ورک فار ڈیجیٹل کامرس (Open Network for Digital Commerce) اَکھ کھُلہِ، باہمی چَلن وول نؠٹ ورک چھُ یُس خریٖدار تہٕ فروشَن ایپلیکیشنَن درمیان لین دین کَرُن چھُ دیوان بغیر اَکِس ایپلیکیشنَس سۭتۍ بَنٛد روزنہٕ۔ یہِ چھےٚ کَمپنی ایکٹ، 2013 (Companies Act, 2013) تَحت اَکھ سؠکشن 8 کَمپنی (section 8 company)۔ جَواب وَنان زِ یہِ چھےٚ نَہ پلیٹ فارم، نَہ ایپلیکیشن، نَہ بازار۔ فروش طرف تہٕ خریٖدار طرف ایپلیکیشن چھِ اَکھ مُشتَرِکہِ مُشَخَصات نافذ کَران۔ نؠٹ ورک چَلان وول چھُ نَہ دوکان۔

صحت چھُ وۄنۍ تہِ بیٚیُہ نمونہ اِستِعمال کَران۔ آیوشمان بھارت ڈیجیٹل مِشن (Ayushman Bharat Digital Mission) سایِٹ چھُ سَتَمبَر 2021 مَنٛز شُروع گَژھِتھ مِشن وَنان تاکہِ اَکھ مربوط، شَہری مرکز ڈیجیٹل صحت ماحول بَنِہِ۔ صحت وَزارَتُک 24 جُلایی 2026 جَواب وَنان زِ ڈیزایِن ذریعہٕ رازدٲری اَکھ رہنما اُصول چھُ تہٕ صحت ڈیٹا ہُند مرکزی ذخیرہٕ چھُ نَہ۔ تَبادُل چھُ مریضَس ہُند رِضامَندی پَتہٕ گَژھان۔ ایپلیکیشن چھِ سینڈ باکسَس (sandbox) مَنٛز تصدیٖق گَژھان تہٕ شٲمِل گَژھنہٕ برٛونٛہہ ویب ایپلیکیشن حِفاظَتی جانچَس مَنٛز گَژھان۔

تٲلیٖم چھےٚ اِرادَتن وَفاق۔ دِکشا (DIKSHA)، نیشنل کَونسل آف ایجوکیشنل ریسرچ اینڈ ٹرینِنٛگ (National Council of Educational Research and Training) ہُند نالِج شیئرِنٛگ خٲطرٕ ڈیجیٹل بُنیاد (Digital Infrastructure for Knowledge Sharing)، وَنان زِ پرؠتھ رِیاسَت تہٕ مرکزی زیرِ اِنتِظام عَلاقہ (Union Territory) چھُ پلیٹ فارم پَنہٕ پنُن طٔریٖقہٕ اِستِعمال کَران۔ قومی پرت چھُ عمارَتی بلاک دیوان، یِمَن مَنٛز تِم تہِ یِم 29 جُلایی 2021 شُروع گَژھِتھ نیشنل ڈیجیٹل ایجوکیشن آرکِٹؠکچر (National Digital Education Architecture) سۭتۍ وابستہٕ چھِ۔ رِیاسَت چھےٚ پروگرام فٲصلہٕ کَران۔

## تبدیٖلیَن کوتاہ بدلِیہِ

پرٛون نمونہ اوس محکمہٕ پورٹل: اَکھ سکیٖم، اَکھ لاگ اِن، اَکھ ڈیٹابیس، اَکھ وؠنڈر۔ اُوپر اِطلاع مَنٛز چھِ وۄنۍ تہِ پورٹل۔ اُمنگ (UMANG) چھُ واریاہ خِدمَتَن ہُند موبایِل اَگہِ کِن دَروازٕ۔ مای سکیٖم (myScheme) چھُ سکیٖمَن ہُند لَبنُک سایِٹ۔ ای آفِس (eOffice) چھُ فایلن خٲطرٕ کامہِ ہُند نِظام۔ یِم چھِ کارآمَد، تہٕ لوک سَبھاہَن چھُ یِمن خٲطرٕ چَلانُک شُمار شایع کَرمُت۔ تِم چھِ نَہ پَنہٕ سٲنۍ پالیسی تبدیٖلی۔

تبدیٖلی چھےٚ سُہ حِصہٕ یُس بیٚیُہ ادارہٕ کال کٔرِتھ ہؠکِہ۔ اَکھ بینک ہؠکِہ UPI ادایگی سوزِتھ بغیر شِناخت دُبارٕ بَناونہٕ۔ اَکھ رِیاسَتی بورڈ ہؠکِہ دَستاوِز ڈیجی لاکَرَس مَنٛز جٲری کٔرِتھ بغیر قومی بَٹوہٕ بَناونہٕ۔ اَکھ ریگولیٹ کَرنہٕ آمُت قرض دِن وول ہؠکِہ کھاتہٕ جَمع کَرن وٲلِس سُہ بیان پرٛژھِتھ یَتھ گاہَکَن رِضامَندی دِژمٕژ چھےٚ، پی ڈی ایف جَمع کَرنہٕ بدَل۔ اَکھ فروش ایپ ہؠکِہ سُہ خریٖدار ایپ لَبِتھ یُس اَمہِ سُند چھُ نَہ، اگر دۄشوٕے ONDC ہُند مُشَخَصات بولَن۔

## پالیسی کوتاہ چھےٚ نَہ طے کَران

جی ٹوَنٹی مَت چھُ رضاکارانہ۔ یہِ چھُ نَہ ہِندوستانَس، یا کانٛہہ بیٚیِس رُکنَس، پرؠتھ ڈیٹا سیٹ کھُلُن لازِم کَران۔ ہِندوستانُک پنُن ڈیٹا تحفُظ قانوٗن چھُ مرحلہٕ وار گھٲڑی پؠٹھ، یُس یَمہِ کھۄتہٕ الگ مُختصَر چھُ۔ ڈیجیٹل پَرسَنَل ڈیٹا پروٹؠکشن قاعِدے، 2025 (Digital Personal Data Protection Rules, 2025) آیہِ نَوَمبَر 2025 مَنٛز نوٹِفای کَرنہٕ۔ زیٛادٕ تر عملی فرایِضَن آو پَتہٕ کِن شُروع تٲریٖخ دِنہٕ۔ اَکھ مُشتَرِکہِ ریل تہٕ اَکھ مُکَمَل رازدٲری ضابطہ چھِ نَہ اَکھ پروجیکٹ، تہِ اگر سُہے وَزارَت دۄشوٕے خٲطرٕ بولِہ۔

پالیسی چھےٚ تہِ نَہ وعدٕ کَران زِ پرؠتھ رہائشی پرؠتھ ریلَس پؠٹھ چھُ۔ UIDAI ہُند سیچوریشن رِپورٹ 30 جوٗن 2026 تام چھُ چالوٗ آدھار رَسایِی قومی سطحَس پؠٹھ 94.59 فیصد تھاوان، تہٕ کینٛژن شُمالی مشرقی رِیاسَتَن مَنٛز کٲفی کَم۔ اَکھ ڈیزایِن یُس عالمگیر شِناخت فرض کَرِ سُہ ناکام گَژھِ تَتہِ ییتہِ شِناخت رِپورٹ پَنہٕ نشان چھُ لگاون۔

پالیسی پرِو ترٛے عوٲمی کٲم جمع اَکھ قاعِدہ زِ ڈیٹا کُس تھاوِتھ ہؠکِہ۔ شِناخت چھےٚ اَکھ نمبر تہٕ اَکھ تصدیٖق خِدمَت۔ ادایگی چھےٚ بینکَن درمیان اَکھ پَیغام۔ ڈیٹا چھُ پَکان ییلہِ نفر اِتِفاق کَرِ، تہٕ سُہ درمیانی یُس اِتِفاق چھُ تُلان اَکثر پے لوڈ تھاوُن منع چھُ۔ یہِ چھےٚ تبدیٖلی۔ اَتھ پؠٹھ ایپ چھِ پَتہٕ کِن، نجی اِنتِخاب۔
