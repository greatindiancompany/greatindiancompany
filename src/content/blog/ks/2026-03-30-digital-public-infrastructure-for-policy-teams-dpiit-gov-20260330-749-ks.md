# پالیسی ٹیٖمَن ہِندوستانٕکۍ ڈیجیٹل ریلَن پؠٹھ کوتاہ ماپُن گَژھِہ

پالیسی ٹیٖمَن چھُ نَہ ڈیجیٹل عوٲمی بُنیاد (digital public infrastructure) خٲطرٕ بیٚیُہ نعرٕ ضروٗرت۔ تِمَن چھُ سیریزَن ہُند لۄکُٹ فِہرَست ضروٗرت یُس تِم فایِل نوٹَس مَنٛز دفاع کٔرِتھ ہؠکَن، تہٕ انکار فِہرَست تِمَن دعوٲن ہُند یِم موجودٕ اشاعت تُلِتھ ہؠکَن نَہ۔ خوشخبری چھےٚ یہِ زِ کینٛہہ شایع کَرن وٲل چھِ گوڈَے اُچَھ معیار پورٕ کَران۔ کٲم چھےٚ سُہ معیار نقل کَرُن، تہٕ پٔتھمِس ریلَس پؠٹھ رَقَم اُدھار تُلُن بَند کَرُن۔

## آدھار نوٹ نقل کٔرِو، احتیاط سَمیت

یونیک آئیڈؠنٹِفِکیشن اتھارٹی آف اِنڈیا (Unique Identification Authority of India, UIDAI) ہُند سیچوریشن رِپورٹ 30 جوٗن 2026 تام چھُ نمونہ۔ یہِ چھُ پیشین کَرنہٕ آمُت آبادی، خام تفویض شُمار، چالوٗ اندازٕ، تہٕ رَسایِی شرح شایع کَران، پرؠتھ رِیاسَت تہٕ مرکزی زیرِ اِنتِظام عَلاقہ (Union Territory) خٲطرٕ۔ پَتہٕ چھُ یہِ زٕ چیز وَنان یِم تشریح چھِ بدلاوان۔ ہولڈر چھِ آدھارَس (Aadhaar) پؠٹھ پَتَس حساب گَنان، تِکیازِ تِم ہؠکَن نَہ رِیاسَتِ مَنٛز مقیم آسِتھ۔ چالوٗ اندازٕ چھُ رجِسٹرار جَنرَل (Registrar General) موت شرح اِستِعمال کَرِتھ موتَن خٲطرٕ ایڈجسٹ کَران۔ قومی چالوٗ رَسایِی چھُ 94.59 فیصد۔ ناگالینڈ (Nagaland) چھُ 62.41 فیصد۔ کیرَلا (Kerala) چھُ 105.02 فیصد۔ ترٛےو رَقَم ہؠکَن سچ آسِتھ۔

رِیاسَتی حُکوٗمَت یُس سکیٖم نوٹ چھُ لؠکھان گَژھِ پَنِنِس رِیاسَتُک سَطر تہٕ زٕ احتیاط چِپکاوُن، نَہ صِرِف قومی 94.59 فیصد۔ بِہار (Bihar) ہُند سَطر چھُ 88.29 فیصد تہٕ 11,67,82,847 اندازٕ چالوٗ تفویض پیشین کَرنہٕ آمُت آبادی 13,22,65,000 مُقابِل۔ گُجَرات (Gujarat) چھُ 89.74 فیصد۔ اَکھ نوٹ یُس وَنان چھُ زِ "آدھار عالمگیر چھُ" سُہ آسہِ تِمَن رِیاسَتَن مَنٛز UIDAI ہُندِ پنٛنِس صَفَس پؠٹھ غلط۔ اَکھ نوٹ یُس 100 فیصد کھۄتہٕ اُوپر رَقَم فاضِل آبادی چھُ مانان سُہ تہِ آسہِ غلط۔ یہِ چھُ پَتہٕ بنیاد شماریات۔

اگر سکیٖم تصدیٖق چھےٚ کَران، پالیسی میٹرک چھُ چالوٗ کالم، نَہ خام کالم۔ خام تفویض قومی سطحَس پؠٹھ چھِ 1,44,80,05,767۔ چالوٗ اندازٕ چھِ 1,34,76,52,046۔ خام ذخیرہٕ اِستِعمال کَرُن چھُ زِندٕ ہولڈرَن ہُند شُمار زیادتی وَنان۔

## او این ڈی سی پؠٹھ سُہ جَدوَل پرٛژھِو نَہ یُس یہِ وَنان چھُ زِ اَتھ چھُ نَہ

4 اَگَست 2026 جَواب دِژ تجارت وَزارَتَن لوک سَبھا (Lok Sabha) سُوالَس یَتھ فروش، تاجر، تہٕ ایم ایس ایم ای اوپن نیٹ ورک فار ڈیجیٹل کامرس (Open Network for Digital Commerce, ONDC) پؠٹھ پرٛژھ، رِیاسَت پتھ۔ جَواب وَنان زِ نؠٹ ورک چھُ نَہ پرؠتھ فروشُک الگ الگ منسوب شُمار تھاوان، تِکیازِ فروش چھِ نؠٹ ورک شِرکَت دارَن ذریعہٕ شٲمِل گَژھان تہٕ ہؠکَن پَکِتھ۔ تِم رَقَم یِمَن پَتہٕ یہِ کھڑٕ رُزِ، جُلایی 2026 تام، چھِ قومی: 500 مِلین کھۄتہٕ زیٛادٕ کُل آرڈَر، منٹَس پؠٹھ 500 کھۄتہٕ زیٛادٕ آرڈَر، تَقریٖبَن 2 لَچھ یا زیٛادٕ چالوٗ پرچون سوداگَر، یِمَن مَنٛز تَقریٖبَن 60 فیصد لۄکُٹ فروش، تہٕ تَقریٖبَن 10 لَچھ یا زیٛادٕ نقل و حرکت تہٕ لوژِسٹِکس خِدمَت فراہم کَرن وٲل۔

اَکھ رِیاسَتی دعوٲ یِتھ کٔنۍ "او این ڈی سی چھُ سٲنِس رِیاسَتِ مَنٛز اتھہِ ایم ایس ایم ای شامِل کَرمٕتۍ" چھُ نَہ سَتھ جَوابَس سَہار۔ پالیسی ٹیٖم یِمَن رِیاسَتی رَقَم ضروٗرت چھِ تِمَن گَژھِ یہِ فروش ایپلیکیشنَن سۭتۍ بَناوُن، یا رِیاسَتِ پنٛنِس شامِل کَرنُک سکیٖمَس مَنٛز رِپورٹِنٛگ فقرہ ضروٗرت کَرُن۔ TEAM اِقدام، 2026-27 تام ₹277.35 کروڑ منظور تہٕ سُہ ہَدف زِ فایِدہ تُلن وٲلِ ادارَن ہُند اَدھ زنانَن ہُند آسہِ، چھُ سُہ جَگہِ ییتہِ رِیاسَت رِپورٹِنٛگ جوڑِتھ ہؠکِہ، تِکیازِ یہِ چھُ حُکوٗمَتی سَہار پروگرام نَہ سؠکشن 8 کَمپنی (section 8 company) ہُند مِلکیَت۔ ڈیجی دُکان (DigiDukaan)، یِتھ پٲٹھۍ وَننہٕ آمُت، اوس حیدَرآباد (Hyderabad) تہٕ جَے پوٗر (Jaipur) مَنٛز چالوٗ۔ بیٚیہِ رِیاسَت گَژھِہ نَہ یہِ کوٹ کَرُن یِتھ پٲٹھۍ زِ یہِ گوڈَے مقٲمی آسہِ۔

## صحتَس چھُ ڈیش بورڈ۔ اَتھ تٲریٖخ لگٲوِو۔

صحت وَزارَتُک 24 جُلایی 2026 جَواب چھُ 20 جُلایی 2026 تام منجمد قومی تصویر دیوان: 94.87 کروڑ اے بی ایچ اے (ABHA) شِناخت، 5.36 لَچھ سہوٗلیت، 10.09 لَچھ پیشہٕ ور۔ پَتہٕ چھُ یہِ رِیاسَتی تہٕ ضِلعی تقسیمَس خٲطرٕ dashboard.abdm.gov.in کُن اِشارٕ کَران، تہٕ یہِ چھُ تَمِل ناڈو (Tamil Nadu) سَتھ کَٹَس مَنٛز 2.57 کروڑ اے بی ایچ اے دیوان۔ صحیح پالیسی عمل چھُ پارلیمانی قومی کُل کوٹ کَرُن ییلہِ تُہۍ مستحکم اینکر ضروٗرت آسہِ، تہٕ ڈیش بورڈ سُہ تٲریٖخ سۭتۍ کوٹ کَرُن یَتھ تُہۍ یہِ ایکسپورٹ کٔر ییلہِ تُہۍ ضِلع ضروٗرت آسہِ۔ چالوٗ سکرین ترٛے رؠتہٕ پَتہٕ "20 جُلایی تام" وَنِو نَہ۔

سُہے جَواب چھُ ڈیزایِن قَید وَنان یِم صحت محکمہٕ سرکلرَس مَنٛز اُلٹاوِ نَہ: صحت ریکارڈَن ہُند مرکزی ذخیرہٕ نَہ، رِضامَندی پَتہٕ تَبادُل، سینڈ باکس (sandbox) تصدیٖق، تہٕ اِضمام برٛونٛہہ ویب ایپلیکیشن حِفاظَتی جانچ۔ اَکھ رِیاسَتی پروجیکٹ یُس "تجزیہ خٲطرٕ" کلینِکَل ریکارڈَن ہُند یکجا نقل چھُ پرژھان سُہ چھُ کِہہ پرژھان یُس قومی جَواب وَنان چھُ زِ مِشن چھُ نَہ کَران۔ ناو وٲل اِضمام، پی ایم-جے (PM-JAY)، نِکشَے (Nikshay)، تہٕ مخصوص رِیاسَتی پلیٹ فارم سَمیت، چھِ اِجازت نمونہ ہاوان۔

جوٗن پؠٹھ اَگَست 2025 تریِمہِ فَریق جانزہ چھُ انتظار وَقت کَم، لاگَت بچاو، تہٕ بہتر رَسایِی پٲٹھۍ خُلاصہٕ، خاص کَرِتھ زنانَن تہٕ کَم خِدمَت یافتہ گروپَن خٲطرٕ۔ جَواب چھُ نَہ مطالعہٕ ہُند رَقَم چھاپان۔ پالیسی نوٹ گَژھِ نَہ تِم جُملہ فیصدَن مَنٛز بَڑاوُن۔ اگر مطالعَس جَدوَل چھُ، مطالعہ کوٹ کٔرِو۔ اگر چھُ نَہ، نتیجہٕ معیاری تھٲوِو۔

## خرچ نوٹ گَژھِ سُہ کھاتہٕ اِستِعمال کَرُن یُس دعوٲ سۭتۍ مِلان چھُ

MeitY ہُند 29 جُلایی 2026 لوک سَبھا جَواب چھُ صاف زِ ڈیجیٹل اِنڈیا (Digital India) اَکھ چھَترِ چھُ تہٕ زِ MeitY ہُند پنُن آٹھ مرکزی شعبہٕ ذیلی سکیٖم چھِ نَہ رِیاسَتی بَجَٹ۔ تِمَن سکیٖمَن خٲطرٕ 2025-26 مَنٛز شایع لِفاف اوس ₹5,144.45 کروڑ تَخصیٖص تہٕ ₹4,638.28 کروڑ خرچ۔ ڈیجی لاکَر (DigiLocker) ہُند پنُن شُروع پؠٹھ ظٲہِر تَخصیٖص اوس ₹168.58 کروڑ، ₹105.14 کروڑ خرچ سۭتۍ، 936.03 کروڑ کھۄتہٕ زیٛادٕ دَستِیاب دَستاوِزَن نَذدیٖک۔ اُمنگ (UMANG) ہُند اوس ₹149.91 کروڑ تَخصیٖص تہٕ ₹129.15 کروڑ اِستِعمال۔

اَکھ رِیاسَتی فایِل یُس "ڈیجیٹل عوٲمی بُنیاد سَرمایَہٕ خَرچ" وَنان چھُ تہٕ پَتہٕ صِرِف MeitY چھَترِ کوٹ چھُ کَران سُہ چھُ وَزارَتی لِفاف فایبَر، صحت، تہٕ ادایگی سۭتۍ رَلاومُت۔ بھارت نیٹ (BharatNet) چھُ سُہ رَقَم یُس جوڑُن چھُ ییلہِ موضوع دیہی رَسایِی آسہِ: جوٗن 2026 تام، 2.21 لَچھ گام پنچایت (gram panchayat) خِدمَت خٲطرٕ تَیار، تہٕ سَتھ رؠتہٕ دوران 80,472 گام پنچایتَن پؠٹھ چالوٗ حاضِری ہُند نُکتہٕ (points of presence)۔ دۄشوٕے رَقَم گَژھِ نوٹَس مَنٛز آسُن۔ صِرِف خِدمَت خٲطرٕ تَیار شُمار کوٹ کَرُن زیادتی وَنِہ زِ پنچایت کوتاہ اِستِعمال کٔرِتھ ہؠکِہ۔

یوٗ پی آی (UPI) گَژھِ شایع ماہانہ جَدوَل وول ادایگی نِظام پٲٹھۍ بحث کَرنہٕ، نَہ MeitY خَرچ سَطر پٲٹھۍ۔ اَگَست 2026 نیشنل پیمنٹس کارپوریشن آف اِنڈیا (National Payments Corporation of India, NPCI) ہُند پروڈَکٹ شُمارَس پؠٹھ اوس 752 بینک، 24,508.96 مِلین لین دین، تہٕ ₹29,82,355.95 کروڑ۔ 29 جُلایی پریس اِنفارمیشن بیورو (Press Information Bureau) کھاتہٕ، یُس 731 بینک تہٕ 55.49 کروڑ کھۄتہٕ زیٛادٕ فَرد وَنان چھُ، چھُ گۄڈنیُک عکس۔ رؠتہٕ لیبل لگٲوِو۔

## خِدمَت فراہمی چھُ چینل میٹرک ضروٗرت کَران، صِرِف ایپ میٹرک نَہ

راجیہ سَبھا (Rajya Sabha) بیان 31 جُلایی 2026 تام وَنان زِ اُمنگ تہٕ ڈیجی لاکَر چھِ آدھار، ای سائن (eSign)، تہٕ اے پی آی سیتو (API Setu) سۭتۍ مربوط، تہٕ زِ 4,07,122 کامن سروس سینٹر (Common Service Centre) 30 جوٗن 2026 گام پنچایت سطحَس پؠٹھ چالوٗ ٲسِہ، یِمَن مَنٛز 70,069 زنانَن چَلاونہٕ آمٕتۍ۔ مئی 2026 PIB کَٹَن وَن زِ 5.01 لَچھ CSC کُل چالوٗ ٲسِہ، یِمَن مَنٛز 3.91 لَچھ گام پنچایت سطحَس پؠٹھ، 800 کھۄتہٕ زیٛادٕ خِدمَتَن سۭتۍ۔

دیہی سکیٖمَس خٲطرٕ چھُ پالیسی میٹرک یہِ زِ رہائشی مرکزَس پؠٹھ مُکَمَل کٔرِتھ ہؠکِہ کیاہ، نَہ صِرِف یہِ زِ ایپ موجود چھا۔ پرادھان مَنتری گرامین ڈیجیٹل ساکشارتا اَبھِیان (Pradhan Mantri Gramin Digital Saksharta Abhiyan, PMGDISHA)، یَتھ 6.39 کروڑ لوٗکھ تربیت دِژ، 31 مارچ 2024 مُکَمَل گٔیہِ۔ یہِ چالوٗ ساتھ اقدام پٲٹھۍ فِہرَست کٔرِو نَہ۔ اگر ڈیجیٹل خواندگی وۄنۍ تہِ سکیٖمُک حِصہٕ چھُ، سُہ آلہ ناو دِیو یَتھ یہِ بدلِیہِ۔ جُلایی 2026 کھاتَس مَنٛز کوٹ کَرنہٕ آمُت 2022-23 سَروے، یَتھ مَنٛز 15 پؠٹھ 24 ؤری وٲنٛسہِ لوٗکھَن ہُند 78.4 فیصد منسلکہ سوزِتھ ہؠکِہ، چھُ سَتھ سَروے ؤریہِ ہُند بنیاد۔ یہِ چھُ نَہ 2026 انتظامی شماریات۔

زبان چھےٚ بیان کَرنہٕ آمُت شمولیت اوزار۔ 29 جُلایی کھاتہٕ چھُ بھاشِنی (BHASHINI) ہِندوستٲنۍ زبان ٹؠکنالوجی خٲطرٕ عوٲمی پلیٹ فارم وَنان۔ دِکشا (DIKSHA) چھُ 36 زبان سَہار کَران تہٕ پرؠتھ رِیاسَتَس پَنہٕ پنُن پروگرام چَلاون چھُ دیوان۔ اَکھ سکیٖم نوٹ یُس صِرِف ہِندی یا صِرِف اَنٛگریٖزی بہاو چھُ سوزان گَژھِ یہِ وَنُن، تِمَن زٕ حقیقتَن مُقابِل۔

## قانوٗنی کیلنڈر چھُ وۄنۍ سکیٖم ڈیزایِنُک حِصہٕ

13 نَوَمبَر 2025 ہُند G.S.R. 843(E) کٔر ڈیجیٹل پَرسَنَل ڈیٹا پروٹؠکشن ایکٹ، 2023 (Digital Personal Data Protection Act, 2023) ہُند کینٛہہ پروویژن فوری نافذ، سُہے دۄہ قایم گَژھِتھ ڈیٹا پروٹؠکشن بورڈ (Data Protection Board) ہُند بنیاد سَمیت۔ اَتھ مُقَرَر کٔر دفعہ 6(9) تہٕ دفعہ 27(1)(d) اَکھ ؤری پَتہٕ، تہٕ اہم عملی دفعہ اَٹھارَہ رؠتہٕ پَتہٕ۔ سکیٖم آرکِٹؠکٹ یِم 2026 ٲخرِس مَنٛز چھِ لؠکھان تِمَن گَژھِ مئی 2027 تٲریٖخ فایِلَس مَنٛز تھاوُن تہٕ گَژھِ ایکٹَس مَنٛز زٕ نَوَمبَر 2026 فقرے پرُن۔ شعبہٕ قاعِدے چھِ نَہ سَتھ تٲریٖخَس انتظار کَران۔ کھاتہٕ جَمع کَرن وول (Account Aggregator) رِضامَندی فرایِض چھِ گوڈَے ریزرو بینک ہُند نَوَمبَر 2025 ہِدایَتن مَنٛز۔ اَکھ سکیٖم یُس مالی معلومات چھُ شیئر کَران گَژھِ وۄنۍ تِم ہِدایات پَیر کَرُن، تہٕ گَژھِ نَہ شِرکَت دارَن وَنُن زِ "ڈی پی ڈی پی شُروع گَژھِتھ چھُ نَہ" چادر جُملہ پٲٹھۍ۔

## جانزہ: جَدوَل پرٛژھِو

MeitY وون لوک سَبھاہَس زِ آی آی پی اے (IIPA) جانزہ کٔر الیکٹرانک گورننس سکیٖم نَوَمبَر 2025 مَنٛز تہٕ لٔب بہتر رَسایِی، کَم وَقت، تہٕ بغیر کاغذ فراہمی۔ چھاپنہٕ آمٕتۍ نتیجہٕ چھِ معیاری۔ سُہے چھُ اے بی ڈی ایم جانزہ خُلاصَس متعلق سچ۔ پالیسی ٹیٖمَن گَژھِ تَل جَدوَل کمیشن یا بازیافت کَرُن برٛونٛہہ وَزیٖرُک مُختصَر "کَم وَقت" رَقَمَس مَنٛز بدلِہ۔ یوت تام، ایماندار سَطر چھےٚ یہِ زِ آزاد جانزہ گٔیہِ تہٕ زِ پارلیمانی خُلاصہ چھُ نَہ اَتھ مقدار کَران۔

## اَکھ فایِل نوٹ یُس سُوالَس پَتہٕ بچِہ

ریل، میٹرک، تٲریٖخ، تہٕ شایع کَرن وول ؤنِو۔ UIDAI طرز احتیاط ؤنِو اگر آسہِ۔ اگر شایع کَرن وٲلَن وَنمُت چھُ زِ رِیاسَتی کَٹ موجود چھُ نَہ، ؤنِو یہِ، تہٕ یہِ اندازٕ کٔرِو نَہ۔ اگر زٕ جَواب زٕ ڈیجی لاکَر اِستِعمال کَرن وٲل شُمار دِیَن، دۄشوٕے تٲریٖخ ہٲوِو: 29 جُلایی لوک سَبھا جَوابَس مَنٛز 71.66 کروڑ کھۄتہٕ زیٛادٕ، تہٕ راجیہ سَبھا بیانَس مَنٛز 31 جُلایی تام 72.43 کروڑ کھۄتہٕ زیٛادٕ۔ اِنتِظام چھُ سُست۔ یہِ چھُ تہِ سُہ یُس ڈیجیٹل بُنیاد دعوٲ قابلِ یقین چھُ بَناوان ییلہِ بیٚیُہ سُوال پرژھنہٕ یِیہِ۔
