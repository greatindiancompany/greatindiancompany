# پلانٹَس خٲطرٕ خطر قبول کَرنہٕ برونہہ فارما PLI ایوارڈ کِتھ کَن پرِو

اَکھ پیداوار سٟتؠ جوڑِتھ اِمدادی (production-linked incentive) چھُ پلانٹ پیداوار کَرنہٕ پتہٕ اہل فروختَس پؠٹھ اَکھ دعوٕ، سکیٖم رہنما اصولَن ہِنٛدِس پیماش طریقَس پؠٹھ۔ سُہ چھُ نَہ سُہ چیک یُس بورڈ سرمایی اخراجات (capital expenditure, capex) منظوٗر کَران وِزِ وَسان چھُ۔ ہِندوستٲنۍ دوا سازی خٲطرٕ خطر قبول فائلَس وَنُن چھُ زِ خطَس پؠٹھ دوا سازِ محکمہٕ (Department of Pharmaceuticals) ہِنٛز زٕ سکیٖمَن مَنٛز کَم چھِ، کیازِ کہ نقد بہاؤ ہُند شکل چھُ الگ۔

## بَلک ڈرَگ ایوارڈ چھِ گرین فیلڈ تہٕ فہرستَس سٟتؠ بَند

21 مارچ 2020 ہُند کابینہٕ فٲصلہٕ چھُ شناخت شُدہ بَلک ڈرَگَن ہِنٛدِن اہل تیاری کُنندَن 2019–20 ہِنٛدِس بُنیٲدی وَرِیَس پؠٹھ اضافٲتی فروختَس خٲطرٕ شےٚ ؤری ادا کَران۔ تَتھ فٲصلَس مَنٛز شرح چھِ خمیرَس پؠٹھ (fermentation-based) مصنوعاتَن خٲطرٕ 20 فیصد تہٕ کیمیاوی ترکیب (chemical synthesis) مصنوعاتَن خٲطرٕ 10 فیصد۔ لِفافہٕ چھُ اَٹھَن ؤرِیَن مَنٛز ₹6,940 کَروڑ۔ اصل شناخت شُدہ سیٹ ٲس 53 مصنوعات۔ 6 فَرؤری 2026 ہُند لوک سَبھا (Lok Sabha) جواب چھُ وَنان زِ سکیٖم چھِ گرین فیلڈ منصوبَن (greenfield projects) ذٔریعہٕ تَیار کَرنہٕ آمٕتِن 41 شناخت شُدہ مصنوعاتن ہِنٛزِ فروختَس پؠٹھ اِمدادی دِوان، 33 بَلک ڈرَگَن خٲطرٕ 48 منصوٗبہ منظوٗر۔

اَکِس کریڈٹ کمیٹی خٲطرٕ عملی سوال چھِ منظوٗری خطَس تہٕ موجودٕ رہنما اصولَن پؠٹھ، نَہ اَکِس شعبہ مضمونَس مَنٛز۔ کیاہ مالیکیول یِمَن مصنوعاتَن مَنٛز اَکھ چھُ یِم فرمَس واقعی بَناونہٕ خٲطرٕ منظوٗر آیہ؟ کیاہ منصوٗبہ گرین فیلڈ چھُ، یِتھ پٲٹھۍ 2026 ہُند جواب یَتھ سکیٖمَس خٲطرٕ چھُ منگان؟ اِمدادی حساب کَم بُنیٲدی وَرِیَس ہِنٛز فروخت استعمال کَرِ؟ کیاہ محکمہٕ ہُند پتہٕ درستگی نامہٕ (corrigendum)، بشمول سکیٖم انڈیکسَس پؠٹھ 4 مے 2026 ہُند، سُہ دعوٕ قٲعدہ بدلاو یُس ماڈل وُنہِ تہِ 2020 ہِنٛدِس کابینہٕ متنَس پؠٹھ فرض کَران چھُ؟ یِہ مختصر چھُ نَہ سُہ مے درستگی نامہٕ دُبارٕ بیان کَران۔ پی ڈی ایف چھُ کَران۔

1 سَتَمبر 2022 ہُند اَکھ سرکٲری نوٹ اوس وَنان زِ تَتھ وِزِ منظوٗر 51 منصوبَن مَنٛز 14 چالو کرنہٕ آمٕتۍ۔ سُہ تَتھ تٲریٖخَس تام اَکھ چالو کَرنُک حقیقت چھُ۔ سُہ چھُ یاد دہٲنی زِ منظوٗری تہٕ گۄڈنیُک تجارتی فروخت چھِ الگ واقعہ۔ اِمدادی چھُ نَہ یَتھ خٲطرٕ شُروٗع گَژھان زِ اَکھ منصوٗبہ فہرستَس پؠٹھ نظرِ یِوان چھُ۔

## دوا سازی ایوارڈ ہٮ۪کَن براؤن فیلڈ آسِتھ، تہٕ ذخیرہ چھُ گروپ

دوا سازی PLI، 3 مارچ 2021 دۄہ ₹15,000 کَروڑ تخصیصَس سٟتؠ نوٹیفائی، چھُ درخواست دہندگان مالی سال 2019–20 ہِنٛزِ دوا ساز سامانَن ہِنٛزِ عالمی تیاری آمدنی لحاظَہ گروپ کَران۔ گزٹ چھُ تَقریٖبَن ₹11,000 کَروڑ گروپ اے (₹5,000 کَروڑ یا زیادٕ آمدنی)، ₹2,250 کَروڑ گروپ بی (₹500 کَروڑ پؠٹھ ₹5,000 کَروڑ)، تہٕ ₹1,750 کَروڑ گروپ سی (₹500 کَروڑ کھۄتہٕ کم) مختص کَران، گروپ سی اندر اَکھ MSME ذیلی گروپ سٟتؠ۔ ذخیرَن مَنٛز چھِ انتظامی اخراجات۔ سِڈبی (Small Industries Development Bank of India, SIDBI) اوس منصوٗبہ انتظام ایجنسی نامزد ییلہِ درخواست 2021 مَنٛز پورٹل pli-pharma.udyamimitra.in ذٔریعہٕ کھُلِیہ۔

فَرؤری 2026 ہُند جواب چھُ وَنان زِ 55 درخواست دہندگان منظوٗر چھِ، دوا سازی تہٕ اِن وِٹرو تشخیص (in-vitro diagnostics) خٲطرٕ، گرین فیلڈ تہٕ براؤن فیلڈ (brownfield) منصوبَن ذٔریعہٕ۔ براؤن فیلڈ چھُ خطر قبولَس خٲطرٕ اہم: موجودٕ جاے پؠٹھ اضافٲتی اہل فروختَس پؠٹھ اِمدادی چھُ نَہ ہِی خطر یُس اَکھ نَوِ خمیر بلاک چھُ۔ جواب چھُ تہِ وَنان زِ یِہ سکیٖم چھِ یِم اہم شُروٗعاتی مادٕ، اِنٹرمیڈیٹ تہٕ API پوٗرٕ کَران یِم بَلک ڈرَگ سکیٖم تحت نوٹیفائی چھِ نَہ، علاوٕ اعلیٰ قۭمَت زمرٕ یِتھ پٲٹھۍ حیاتی دوا (biopharmaceuticals)، پیچیدٕ جنرک، پیٹنٹ یا ختم گَژھن وٲلِس نزدیٖک دَوا، خود مدافعتی دَوا تہٕ کینسر خلاف دَوا۔ اَکھ ماڈل یُس ہِی نوٹیفائی مصنوعَس پؠٹھ بَلک ڈرَگ PLI شرح دوا سازی PLI پؠٹھ رَلاوان چھُ، متن تراوِتھ چھُ۔

یَتھ دوٚیِمِس سکیٖمَس خٲطرٕ ؤری بہ ؤری شرح چھِ 1 جوٗن 2021 ہِنٛدِن عملی رہنما اصولَن تہٕ پتٕمِن درستگی نامَن مَنٛز۔ سُہ جدول استعمال کٔرِو۔ 20 تہٕ 10 ہِنٛد کابینہٕ فیصد چھِ بَلک ڈرَگ سکیٖم ہِنٛدِس 2020 فٲصلَس سٟتؠ۔

## پارک چھِ لاگت ڈھیر بدلاوان، اَکِس سرکٲری وقت جدولَس پؠٹھ

اگر پلانٹ اَکِس بَلک ڈرَگ پارکَس اندر تجویز چھُ، سرکٲری سرمایی اخراجات چھِ مشترَکہ بنیادی ڈھانچہ، نَہ کرایہ دارَس ہُند ری ایکٹر لاگت۔ فَرؤری 2026 ہُند جواب چھُ جَمبوسر، ہَرولی تہٕ نَکّاپَلّی خٲطرٕ کُل منصوٗبہ لاگت تہٕ مشترَکہ بنیادی ڈھانچہٕ ہِنٛز لاگت دِوان، تہٕ پرٮ۪تھ پارکَس پؠٹھ ₹1,000 کَروڑ حد وٲل مرکزی گرانٹ۔ سُہ چھُ تہِ وَنان زِ سکیٖم مالی سال 2026–27 تام چھِ چَلان، تہٕ سُہ چھُ وُنہِ تہِ جٲری کٲم بیان کَران۔ اَکھ کرایہ دار ماڈل یُس فرض کَران چھُ زِ بَہار، گَندِ آب صَفٲیی تہٕ سالوینٹ واپسی (solvent recovery) پیٹھہٕ دستیاب چھِ، تَتھ جوابَس نِش اَگہِ چھُ۔ پلاٹ قۭمَت، الاٹمنٹ حال تہٕ رِیاسَتی نافِذ ایجنسی ہُند جدول چھِ رِیاسَتی دستاویز۔ مرکزی جواب چھُ تُہۍ گرانٹ حد تہٕ مشترَکہ کٲم ہُند پَڑاو وَنان۔

## ایوارڈ کِیاہ قۭمَت چھُ نَہ تھاوان

اَٹھَن ؤرِیَن مَنٛز ₹46,400 کَروڑ اضافٲتی فروخت ہِنٛز کابینہٕ اُمید چھِ سکیٖم ہِنٛز بیان کَرنہٕ آمٕژ اُمید، نَہ اَکِس کمپنی ہِنٛز آمدنی لائن۔ دَوا قٲعدَن تحت معیاری نظام، بشمول 28 دَسَمبر 2023 دۄہ نوٹیفائی کَرنہٕ آمُت نظرثانی شُدہ شیڈول ایم (Schedule M)، ہٮ۪کَن تجارتی فروخت رُکٲوِتھ تَتھ وِزِ تہِ ییلہِ PLI خط موجود آسہِ۔ مرکزی دَوا معیاری کنٹرول تنظیم (Central Drugs Standard Control Organisation, CDSCO) تہٕ رِیاسَتی لائسنس اختیار چھِ دوا سازِ محکمہٕ نِش اَکِس الگ فائلَس پؠٹھ۔ اَکھ سَرمایہٕ کار یَس PLI خط چھُ تہٕ تیاری لائسنس چھُ نَہ، تَس چھُ اجازت سیٹُک اَدھ۔

سَرمایہ مختص کُنندَن خٲطرٕ لِکھنہٕ آمُت جانچ فہرست چھُ [سَرمایہٕ کار فائلَس](/blog/pharma-and-biotech-for-investors-mnre-gov-20260330-794-ks) مَنٛز۔ گریٹ اِنڈین کمپنی (Great Indian Company) چھِ سکیٖم حد اَکِس جاے تھاوان تاکِہ اَکھ شرط نامہ سُہ گزٹ ناو کٔرِتھ ہٮ۪کہِ یَس پؠٹھ سُہ منحصر چھُ: [greatindiancompany.com](https://greatindiancompany.com)۔
