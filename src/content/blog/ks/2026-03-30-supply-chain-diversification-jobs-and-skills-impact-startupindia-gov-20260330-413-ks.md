# ہِندوستانٕچ صَنعتی تِجارَت ہِنٛزِ تَنوعَس اندر روزگار تہٕ ہُنَر

رَسانہٕ زنٛجیرُک تَنوع چھُ لوکھ تیٚلہِ ملازمتَس مَنٛز انان یَتِ اَکھ کام چھُ جاے بدلاوان۔ سَرکٲرِی نوٹ چھِ سُہ طریقہٕ بیان کَران۔ تِم چھِ نہٕ، یِمَن دَستاوَزَن مَنٛز یِہ مُختصَر اِستعمال کَران چھُ، «رَسانہٕ زنٛجیر تَنوع» خٲطرٕ اَکھ کُنۍ قومی روزگار کُل شایع کَران۔ سلائیڈَس پؠٹھ اَکھ گول نَمبَر دُہراوُن آسِہِ اَندازٕ۔ کارآمد پَرُن چھُ یِہ زِ پالیسی کِم کامَن پؠٹھ بناوُن اُمید چھِ کَران، تہٕ کِم ہُنَرَن خٲطرٕ کارخانَس پانےٕ بَجَٹ تھاوُن چھُ۔

## PLI نوٹ کامَس مُتعلِق اصلی طور کیاہ چھُ وَنان

2 اَگَست 2023 ہُند پریس اِنفارمیشن بیورو (Press Information Bureau, PIB) نوٹ، تِجارَت تہٕ صَنعت ہِنٛدِ ریاستی وَزیر (Minister of State for Commerce and Industry) پؠٹھ، چھُ وَنان زِ سکیٖمَن چھِ لَگ بَگ اَگہٕ کٮ۪ن پانٛژھ ؤرِیَن پیداوار، روزگار، تہٕ تَرقِی بَڑاوُنِک صلاحیت۔ یِہ چھُ یِہ تہِ وَنان زِ 14 شُعبٕ چھِ جزوی طور یِم خٲطرٕ چُنہٕ آمٕتۍ زِ تِم ہیٚکن روزگار، بَرآمَد، تہٕ وسیع فٲیدٕ ضرب کٔرِتھ۔ «صلاحیت» تہٕ «ہیٚکن» چھِ سُہے جوابَس مَنٛز فعل۔ تِمَن نَکھ چھُ نہٕ کانٛہہ تعداد۔

سُہے نوٹَس مَنٛز ٹھوس مَزدوٗری ڈھانچہٕ چھُ فراہم کُنَندہٕ سلسِلہٕ۔ اَصل اِدارَن (anchor units) پؠٹھ چھُ قدرِ زنٛجیرَس پؠٹھ فراہم کُنَندَن ہِنٛز بُنیاد بناوُن اُمید، زیادٕ تَر معاون اِدارَن خُرد، لَگوٗ تہٕ مَدرَم اِدارَن (micro, small and medium enterprises, MSME) ہِنٛدِس شُعبَس مَنٛز سۭتۍ۔ تَتھ تٲریٖخَس چُنہٕ آمَژَن 733 دَرخواستَن مَنٛز 176 MSME ٲس بَلک دَواہِ (bulk drugs)، طِبی آلات (medical devices)، دواسازی (pharma)، ٹیلی کام (telecom)، سفید سامان (white goods)، خورٲکی عمل آوری (food processing)، کَپُڑ (textiles)، تہٕ ڈرونَن (drones) مَنٛز فٲیدٕ وَٹھن وٲلۍ۔ یِہ چھُ نقشہٕ زِ کارخانہٕ فرشُک کام تہٕ اَوزار کمرُک کام کَتِ بیٹھُن چھُ: صرف اَصل اِدارَس اندر نہٕ، بلکہ تِمَن فرمَن مَنٛز تہِ یِم اَتھ کھاون چھِ۔ یُس اَصل اِدارٕ سُہے کِٹَس دَرآمَد کَران چھُ یُس سُہ گوڈَے دَرآمَد کَران اوس، تَمہِ سُہ دویِم تہہ روزگار چھُ نہٕ بناومُت۔

شُعبٕ چھِ الگ۔ تکنیکی کَپُڑ (technical textiles)، خورٲکی عمل آوری، تہٕ سفید سامانُک جوڑُن چھِ خاص فولاد (specialty steel) یا ایڈوانسڈ کیمسٹری سیلَن (advanced chemistry cells) نِش الگ ہُنَر کَڈان۔ اَکھ روزگار دعوٕ یُس تِمَن اَکھ «صَنعتی روزگار» سَطرَس مَنٛز جمع کَران چھُ، سُہ تربیت مسلہٕ چھُ چُھپاوان۔

## مالِ رَسانی تہٕ معدنی نوٹ یِم ہُنَر نامزد چھِ کَران

قومی مالِ رَسانی پالیسی (National Logistics Policy)، 17 سَتَمبَر 2022 دۄہ شروع، چھِ لوکھ مالِ رَسانی نِظامُک حِصہٕ مانان، سَڑکَن ہِنٛدِس حاشیَس مَنٛز نوٹ نہٕ۔ سُہے پالیسی متعلق اَکھ PIB پیش رفت نوٹ چھُ وَنان زِ نافَذ کرُن جامع مالِ رَسانی عَمَل مَنصوٗبہٕ (Comprehensive Logistics Action Plan) ذریعہٕ چھُ چلان، یَمہِ مَنٛز انسانی وسیلہٕ تَرقِی چھِ ڈیجیٹل مالِ رَسانی نِظامَن، اثاثہٕ معیاری کرُن، رِیاستی شُمولیت، تہٕ بَرآمَد-دَرآمَد مالِ رَسانی سۭتۍ۔ سُہے نوٹ چھُ وَنان زِ پالیسی چھِ صَلاحیَت بَڑاوَن وٲلۍ کمیشن (Capacity Building Commission) سۭتۍ تربیت تہٕ صَلاحیَت بَڑاوُن زور دِوان۔ یِہ چھُ نہٕ ڈرائیوَرَن، گودام مُلٲزِمَن، یا کسٹمز کلرکَن ہُند حساب کَران یِم ملازم آیہِ۔

اَہَم معدنیات چھِ اَکھ پَتَل، زیادٕ مخصوص تہہ جمع کَران۔ قومی اَہَم معدنی مشن (National Critical Mineral Mission) متعلق 24 مارچ 2025 ہُند PIB نوٹ چھُ وَنان زِ مشنَس مَنٛز ہُنَر تَرقِی تہٕ تحقیق سَہارٕ ہِنٛز تَخصیٖص چھِ، عمل آوری پارکَن خٲطرٕ Rs 500 crore تہٕ دُبارٕ اِستعمالَس خٲطرٕ Rs 1,500 crore نَکھ۔ عمل آوری تہٕ دُبارٕ اِستعمال چھِ تِم روزگار یِم کَھننہٕ آمُت یا واپس اننہٕ آمُت مواد اَکھ ایسا اَنٛدَرونی مال بناوان چھِ یُس اَکھ سیل کارخانہٕ یا الیکٹرانک لائن اِستعمال کٔرِتھ ہیٚکہِ۔ تلاش روزگار تہٕ عمل آوری روزگار چھِ نہٕ سُہے پیشہٕ۔

تِجارَتی تَنوع چھُ اَکھ کلری تہٕ کَوالِٹی ہُنَر جمع کَران یُس کارخانہٕ دور چھِ راوان۔ نافِذ معاہدَس تحت تَرجیح چھِ اصلِیتُک سَرٹیفِکیٹَس (certificate of origin) سۭتۍ دعوٕ کرنہٕ یِوان۔ کارخانَس یا کسٹمز ڈیسکَس مَنٛز کَہَن چھُ اصلِیتُک قٲیدٕ زانُن۔ خٲرجی تِجارَت ہِنٛدِ ڈائریکٹوریٹ جنرل (Directorate General of Foreign Trade, DGFT) ہُند کام چھُ سُہ نِظام چلاوُن۔ یُس فرم صرف جوڑَن وٲلۍ ملازم کَران چھِ تہٕ کانٛہہ نہٕ یُس اصلِیت دَستاوَزَس مَنٛز انِتھ ہیٚکہِ، سُہ حکمتِ عملی ہُند تِجارَتی حِصہٕ راوِ۔

## بُنیاد تھاوَن وول کیاہ بَجَٹَس مَنٛز تھٲوِ

ترٛےٚ ٹیمَن خٲطرٕ بَجَٹ تھٲیو، تہٕ تِمَن قومی اعداد و شمارَس مَنٛز مَہ بدلِو۔ گۄڈنیُک، PLI سامانَس خٲطرٕ اَصل اِدارَس ہِنٛد پنِنہِ ٹیکنیشَن تہٕ کَوالِٹی مُلٲزِم۔ دویِم، MSME فراہم کُنَندَن ہُند اَوزار تہٕ عمل ہُنَر، تِکیازِ یِہ چھُ سُہ سلسِلہٕ یُس 2023 نوٹ چھُ بیان کَران۔ ترٛیِم، مالِ رَسانی تہٕ اصلِیت دَستاوَز، تِکیازِ اَکھ پرزٕ یُس پَکِتھ نہٕ ہیٚکہِ یا تَرجیح دعوٕ کٔرِتھ نہٕ ہیٚکہِ سُہ چھُ نہٕ تَنوع کرنہٕ آمُت ذَریعہٕ۔ اگر پَتہٕ کن اَکھ PIB نوٹ نامزد سکیٖمَس خٲطرٕ روزگار رَقَم شایع کَرِ، سُہ نوٹ تہٕ تَتھ ہِنٛز تٲریٖخ حوالہٕ دِیو۔ تَتھ تام خانہٕ خٲلی تھٲیو۔

گریٹ اِنڈین کَمپَنی (Great Indian Company) چھِ روزگار دعوٕ پنِنہِ مُختصَرَن مَنٛز سُہے پٲٹھۍ تھاوان: گۄڈَے طریقہٕ، نَمبَر تیٚلہِ یَلہٕ سَرکٲرِی نوٹ سُہ وَنِ۔ ہوم پیج چھُ [greatindiancompany.com](https://greatindiancompany.com)۔
