# ہِندوستانَس چِ ژور مزدور کوڈ نافِذ گَژھنہٕ پتہٕ کیاہ بدلاؤ آو

21 نومبر 2025 کۆر مزدوری تہٕ روزگارَچ وَزارَت (Ministry of Labour and Employment) وَنُن زِ ژور مزدور کوڈ (Labour Codes) چھِ ہُیِ دۄہ نافِذ کَرنہٕ یِوان۔ وزیٖرِ اعظمَن کۆر ہُیِ دۄہ یہِ قدم خوش آمدید تہٕ کوٚرُن اَکھ اصلاح بیان یُس مزدورَن اختیار چھُ دِوان، تعمیل چھِ آسان کَران، تہٕ کاروبار کَرنُک سہولت چھُ سہارا دِوان۔ اقتصادی سروے 2025–26 (Economic Survey 2025–26) کۆر پتہٕ پنٛنہِ شروٗعاتی بیانَس منز یہِ اِطلاع ریکارڈ تہٕ کوٚرُن وَنُن زِ قٲعدٕ ٲسۍ اَمِہ پتہٕ کینٛہَن رؠتَن منز متوقع۔

بدلاؤ چھُ اصلی۔ یِہ چھُ زِیادٕ وَنُن تِہ آسان۔ کوڈ چھِ قانوٗنی چوکھٹ بدلاوان۔ تِم چھِ نَہ، پنُن پانَے، آغازِ تٲریخَس پؠٹھ پرٮ۪تھ مزدورَس اکاوٗنٹَس منز اَکھ فائدٕ جمع کَران۔

## یُس چوکھٹ بدلاونہٕ آو

وَزارَتُک بیان چھُ یِہ زِ پرون قانوٗنُک واریاہ حِصہٕ اوس 1930 تہٕ 1950 ہِنٛدِس دہایَن نِش تہٕ 29 مرکزی ایکٹَن منز بیٹھِتھ اوس۔ کم از کم اُجرت ٲس شیڈول شُدٕ ملازمتَن پؠٹھ لاگو، اَمہِ کِنۍ مزدورَن ہِنٛدِ بٔڑِس گروپَن سو قانوٗنی حق نِش نیبر ٲسۍ۔ تَقرری ہِنٛدِ خط ٲسۍ نَہ لازمی۔ سماجی تحفُظُک دائرہٕ اوس محدود۔ زَنانَن ہُند راتِ ہُند تہٕ کینٛہَن پیشَن منز روزگار اوس محدود۔ ESIC ہُند دائرہٕ اوس اِطلاع شُدٕ علاقَن سۭتۍ جوڈِتھ تہٕ، عام طور، 10 یا زِیادٕ ملازِم والٮ۪ن ادارَن سۭتۍ۔ اَکھ آجرَس ٲس واریاہ دَرج، لائسنس، تہٕ ریٹَرنَن سۭتۍ واسطہٕ۔

یہِ بیان چھُ وَزارَتُک 21 نومبر 2025 ہُند گۄڈَے-تہٕ-پتہٕ نوٹ۔ یِہ چھُ کوڈ پَرنہٕ خٲطرٕ صحیح بُنیاد۔ یِہ چھُ نَہ یہِ دعوٕ زِ ہِندوستانَس منز پرٮ۪تھ ادارٕ پرٮ۪تھ پرون ایکٹَس نیبر اوس۔

## نوِ چوکھٹ قانوٗنَس منز کیاہ چھِ تھاوان

ژور کوڈ چھِ وۄنۍ یہِ میدان تُلان۔

اُجرتَن ہُند کوڈ، 2019 (Code on Wages, 2019) چھُ کم از کم اُجرتُک قانوٗنی حق پرون ملازمت شیڈولَس کھۄتہٕ نیبر تام واتناوان۔ وَزارَت چھِ اَکھ قومی فرش اُجرت تِہ بیان کَران، تاکہِ کانہہ مزدور تَمہِ کم از کم زندگی معیارَس کھۄتہٕ کَم اُجرت مِلِ یُس مرکز تھاوان چھُ، تہٕ چھِ اُجرتِ ہِنٛز وَقتَس پؠٹھ ادایگی لازمی بیان کَران۔ آئی ٹی تہٕ آئی ٹی-فعال خدماتَن (IT and IT-enabled services) خٲطرٕ چھُ ہُیِ نوٹ وَنان زِ تنخواہ گَژھِ پرٮ۪تھ رؠتھِ ہِنٛدِ 7 تام جٲری کَرنہٕ۔

صَنعتی تعلقاتَن ہُند کوڈ، 2020 (Industrial Relations Code, 2020) تہٕ سماجی تحفُظَس دفعات، یِتھ کٔنۍ وَزارَت چھِ تِمَن خلاصہٕ کَران، چھِ مُقَرَر مُدَت ملازِمَن (fixed-term employees) مستقل مزدورَن ہِوِ چُھٹی، طبی، تہٕ سماجی تحفُظ فائدٕ ہِنٛدِ اہل بناوان، برابر اُجرتَس پؠٹھ، تہٕ پانٛژِ ؤری بجائے اَکی ؤری مسلسل خدمتَس پتہٕ گراچوٹی سۭتۍ۔ بیان کٔرمُت مقصد چھُ پرتہِ پرتہِ ٹھیکہٕ پؠٹھ کم انحصارَس سۭتۍ سِدِی نوکری۔

پیشَہٕ وَرٲے حفاظَت، صحت تہٕ کٲمِ ہِنٛدِ حالاتَن ہُند کوڈ، 2020 (Occupational Safety, Health and Working Conditions Code, 2020)، دُبارٕ وَزارَتُک بیانَس منز، چھُ تمام مزدورَن خٲطرٕ تَقرری ہِنٛدِ خط لازمی کَران، زَنانَن راتِ تہٕ تمام قٕسمِ کٲمَس منز کٲمِ ہُند اِجازت دِوان — تِمَن منز زیرِ زمین کان کنی تہٕ بَری مشینری — رضامندی تہٕ حفاظتی اِقدامَن تحت، تہٕ 40 کھۄتہٕ زِیادٕ وَہَس والٮ۪ن مزدورَن خٲطرٕ اَکھ مُفت سالانہٕ صحت جانچ لازمی کَران۔ ESIC ہُند دائرہٕ چھُ تمام ہِندوستان بیان کَرنہٕ یِوان: 10 کھۄتہٕ کم ملازِم والٮ۪ن ادارَن خٲطرٕ رضاکارانہٕ، تہٕ لازمی یَتہِ تِہ اَکھ ملازم خطرناک عملَس منز لگاونہٕ آسہِ۔

تعمیل گَژھِ اَکی دَرج، اَکی تمام ہِندوستان لائسنسَس، تہٕ اَکی ریٹَرنَس پؠٹھ سَمٹُن، اَکی اِنسپکٹر-کم-سہولتکار کردارَس سۭتۍ یُس نفاذَس سۭتۍ سۭتہِ رہنمائی طرف تِہ مُڑِتھ آسہِ۔

## سو حِصہٕ یُس خاص طور کٲمِ ہِنٛدِس مستقبلِ ہِنٛدِس شکلن متعلق چھُ

سماجی تحفُظَس پؠٹھ کوڈ، 2020 (Code on Social Security, 2020) چھُ گۄڈنیُک مرکزی قانوٗن یُس یہِ کٲم متعارف کَران چھُ۔

اَکھ گِگ مزدوٗر چھُ سو نفر یُس کٲم کَران چھُ، یا کٲمِ ہِنٛدِس اَکھ انتظامَس منز شِرکت کَران چھُ، تہٕ روایتی آجر-ملازم تعلقَس نیبر تَمہِ سرگرمی نِش کَماوان چھُ۔ اَکھ ایگریگیٹر (aggregator) چھُ اَکھ ڈیجیٹل درمیٲنی یا اَکھ بازار یُس خریدار یا استعمال کرن والَس فروخت کرن والَس یا خدمت فراہم کرن والَس سۭتۍ جوڑان چھُ۔ پلیٹ فارم کٲم چھِ سو کٲم یُس اَتھۍ آن لائن پلیٹ فارمَس ذریعہٕ مُنَظَم چھُ۔

دفعہ 114 چھُ مرکزَس گِگ تہٕ پلیٹ فارم مزدورَن خٲطرٕ سکیٖم بناونُک اِجازت دِوان یِم زندگی تہٕ معذوری کور، حادثُک بیمہ، صحت تہٕ زچگی فائدٕ، بُڈَے پنُک تحفُظ، کریش، تہٕ بیاکھ فائدٕ چھِ چھانٹان یِم مرکز طے کَران چھُ۔ فنڈنگ ہٮ۪کہِ مرکزَس نِش، رِیاسَتَن نِش، ایگریگیٹرَن نِش، فائدہ تُلن والٮ۪ن نِش، کارپوریٹ سماجی ذمہٕ داری فنڈَن نِش، یا اَکی جوڑَس نِش یِتھ۔

سَتَم شیڈولَس (Seventh Schedule) منز درج زُمرَن ہِنٛدِ ایگریگیٹرَن خٲطرٕ چھُ دفعہ 114(4) سو حصہٕ تھاوان یُس اَمِہِک اَکھ حِصہٕ فنڈ کَرِ۔ شرح چھِ مرکز اِطلاع کَران۔ یِہ ہٮ۪کہِ نَہ 1 فیصد کھۄتہٕ کَم یا 2 فیصد کھۄتہٕ زِیادٕ آسِتھ سالانہٕ ٹرن اوورَس پؠٹھ۔ یَتھ مقصدَس خٲطرٕ ٹرن اوور چھُ مرکزَس دِنہٕ آمُت ٹیکس، لیوی، تہٕ سیس نیبر تھاوان۔ حصہٕ ہٮ۪کہِ نَہ گِگ تہٕ پلیٹ فارم مزدورَن دِنہٕ آمٕتِ یا دِنہٕ واجِب رقمَس 5 فیصد کھۄتہٕ زِیادٕ گَژھِتھ۔ سَتَم شیڈول چھُ سواری شیئرنگ، خوراک تہٕ گروسری ترسیل، لاجسٹکس، ای-بازار، پیشہٕ ورانہٕ خدمات، صحت، سفر تہٕ مہمان نوازی، مواد تہٕ میڈیا، تہٕ کانہہ تِہ بیاکھ سامان تہٕ خدمت پلیٹ فارم درج کَران۔

دفعہ 114(5) چھُ سو جملہٕ یُس یَتھ خودکار لیوی بننہٕ نِش رُکان چھُ۔ سو تٲریخ یَتھ پؠٹھ ایگریگیٹرَن ہُند حصہٕ شروٗع گژھان چھُ، سو چھُ پَنہٕ پانَے مرکز اِطلاع کَران۔ دفعہ 114(6) چھُ قومی سماجی تحفُظ بورڈ (National Social Security Board) گِگ تہٕ پلیٹ فارم مزدورَن ہُند بورڈ بناوان، ییلہِ سو یہِ کٲم کَران چھُ تیلہِ اَکی بدلاونہٕ آمٕتِ ترکیبَس سۭتۍ۔

وَزارَتُک 21 نومبر نوٹ چھُ یہِ تِہ جمع کَران زِ اَکھ آدھارَس سۭتۍ جوڈِتھ یونیورسل اکاوٗنٹ نمبر چھُ یہِ فائدٕ رِیاسَتَن درمیان پورٹیبل بناونہٕ خٲطرٕ۔

## آغازِ تٲریخَس سۭتۍ کیاہ نَہ پھِرِیو

ہُیِ پریس نوٹ چھُ وَنان زِ حُکوٗمَت کَرِ مشورٕ ییلہِ سو قٲعدٕ، ضابطہٕ، تہٕ سکیٖم بناوان چھُ، تہٕ زِ منتقلی دوران موجودٕ ایکٹَن تہٕ تِمَن ہِنٛدِ قٲعدَن، اِطلاعن، تہٕ سکیٖمَن ہِنٛدِ متعلقہٕ دفعات چھِ جٲری روزان۔ یُس پَرَن وول 21 نومبر 2025 سو دۄہ مانِ یَتھ پرٮ۪تھ پرون فارم، حد، تہٕ سکیٖم مَرِ، سو کَرِ عملی حیثیت غلط۔

زٕ مزید حد چھِ اہم۔ وَزارَتُک موازنہٕ جدول چھُ وَنان زِ تمام مزدورَن، تِمَن منز گِگ تہٕ پلیٹ فارم مزدور، مِلہِ سماجی تحفُظ، تہٕ اَکی سطرَس منز چھُ وَنان زِ تمام مزدورَن مِلہِ پروویڈنٹ فنڈ، ESIC، بیمہ، تہٕ بیاکھ فائدٕ۔ ہُیِ نوٹَس منز ESIC ہُند زیادٕ صحیٖح قٲعدٕ چھُ سو یُس پالیسی ڈیزائن کَرنَس وِزِ استعمال گَژھِ: تمام ہِندوستان دائرہٕ، 10 ملازِمَن کھۄتہٕ کَم رضاکارانہٕ، خطرناک عملَن خٲطرٕ لازمی۔ پروویڈنٹ فنڈ رکنیت چھُ وۄنۍ تِہ اَمہِ پؠٹھ منحصر زِ ادارٕ دائرَس منز آسہِ تہٕ زِ حصہٕ اصلَس منز جمع آسہِ کَرنہٕ یِوان۔ تہٕ سو «64 فیصد کھۄتہٕ زِیادٕ» سماجی تحفُظ دائرہٕ یُس نوٹ 2025 خٲطرٕ کوٹ کَران چھُ، 2015 منز تَقریٖبَن 19 فیصد پؠٹھ کھۄتہٕ، چھُ وَزارَتُک دائرہٕ بیان۔ یِہ چھُ نَہ وَقتی مزدور طاقت سروے (Periodic Labour Force Survey) ہُند نتیجہٕ، تہٕ یِہ چھُ نَہ ہُیِ چیز زَن اَکھ گِگ مزدور سکیٖم یُس 1–2 فیصد جمع کَرُن شروٗع کوٚر۔

پالیسی بدلاؤ، سخت وَنہٕ آمُت، چھُ یِہ۔ ہِندوستانَس چھُ وۄنۍ اَکھ اُجرت کوڈ، اَکھ صَنعتی تعلقات کوڈ، اَکھ سماجی تحفُظ کوڈ، تہٕ اَکھ حفاظَت کوڈ، 21 نومبر 2025 پؠٹھ نافِذ، گِگ تہٕ پلیٹ فارم کٲم قانوٗنَس منز نامزد۔ اُجرتُک حق، تَقرری ہُند خط، تہٕ مُقَرَر مُدَت گراچوٹی قٲعدٕ چھِ تِم بدلاؤ یِم وَزارَت فوری ڈیزائن کِس طورَس پؠٹھ پیش کَران چھِ۔ گِگ حصہٕ چھُ اَکھ بینڈ تہٕ اَکھ آغازِ تٲریخ یُس وۄنۍ تِہ اِطلاع پؠٹھ انتظارَس منز چھُ، تہٕ پرون قٲعدٕ چھِ متبادل روزان یوتہِ تام نوِ قٲعدٕ تِمَن بدل کَرِ۔
