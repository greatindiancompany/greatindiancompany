# پالِسی ٹیٖم کِتھ کَنِہ پرِن ہِندوستانَن ہِنٛد سَمٲجی نٔتیٖج جٲری

اَکھ پالِسی مختصر چھُ ناکام گَژھان یَلہِ یہِ اَکھ مُستَفیٖد شُمار تہٕ اَکھ جِسم ناپ اَکھ جملَس مَنٛز تھاوان چھُ تہٕ دۄشوَے نٔتیٖجہ وَنان چھُ۔ وِزارَت چھِ تِم گوڈَے الگ کران۔ پالِسی ٹیم ہُنٛد کٲم چھُ سُہ الگ کرُن تھاوُن، سُہ حُکوٗمَت سطح ناوُن یُس عمل کٔرِتھ ہٮ۪کہِ، تہٕ سُہ بیٚیُم دَستاوے ز منٛگنہٕ یُس نتیجہ بدلاوِہ۔

## پیداوار تہٕ نٔتیٖجہ الگ سَطرَن پؠٹھ تھاوِو

پیداوار، اِنتِظامی نظامَن پؠٹھ: 31 جُلایی 2026 تام تَقریٖبَن 8.94 کروڑ پوشَن ٹریکَر (Poshan Tracker) مُستَفیٖد؛ مَے 2026 تام 0-5 وَرِیَن ہِنٛد 6.3 کروڑ کھوتہٕ زِیادٕ شُرٮ۪ن خٲطرٕ بَڑُن نِگرٲنی؛ نٔوِ مَزدوٗری ایکٹَس گوڈنِکِس رٮ۪تَس تَقریٖبَن مَنٛز 9.69 کروڑ کھوتہٕ زِیادٕ دیہی نفر-دۄہ، یِتھ کَنِہ لوک سبھاہس (Lok Sabha) 11 اَگَست 2026 وَننہٕ آو؛ UDISE+ 2025-26 مَنٛز 1,02,73,020 سکوٗلی اُستاد۔

نٔتیٖجہ، سَروےَن تہٕ شِرکَت تَناسُبَن پؠٹھ: قَد-عُمر کَمی (stunting) 29.3%، وَزن-قَد کَمی (wasting) 19.0%، کَم وَزن (underweight) 31.8%، صِرف ماے دۄد (exclusive breastfeeding) 55.8%، 6-23 رٮ۪تَن پؠٹھ کٲفی غِذا 15.3%، اِدارَتی زَیِش (institutional births) 90.6%، کارڈَن پؠٹھ پوٗرٕ ویکسین 87.1%، ثانَوی برقرار رکاوَٹ 51.9%، اَگَست 2026 مَنٛز موجوٗد ہَفتہٕ وار حیثیت (current weekly status) پؠٹھ مَزدور طاقتَس مَنٛز شِرکَت 55.6%۔ اَکھ جملہ یُس وَنان چھُ زِ «8.94 کروڑ مُستَفیٖد چھِ ہاوان زِ پۄشتہِ چھِ بہتر گٔمٕژ» چھُ پیداوار نٔتیٖجَس ہِنٛدِس روپَس مَنٛز اِستعمال کران۔

## وَنِو زِ بدلاو نُکتَن مَنٛز چھُ یا نِسبَتی

29 مَے 2026 سُرخی چھِ وَنان زِ قَد-عُمر کَمی 17% تہٕ شَدیٖد وَزن-قَد کَمی (severe wasting) 32% وَتھ۔ سطح چھِ 35.5% پؠٹھ 29.3%، تہٕ 7.7% پؠٹھ 5.2%۔ سطح مختصر کٔرِو۔ اگر تُہۍ سُرخی دُہراوِو، وَنِو زِ یہِ NFHS-5 بُنیادَس سٕتی نِسبَتی چھُ۔ سُہ نظم «کانٛہہ تہِ زَیِش برٛونٛہہ دیکھ بھال چھِ 95.9%» سٕتی «180 دۄہ لۄہ تہٕ فولِک تیزاب چھُ 37.8%» سٕتی تہِ لاگو چھُ۔ دۄشوَے ہٮ۪کن سَچ آسِتھ۔ صِرف اَکھ چھُ پوٗرٕ کورس۔

## ناوِو کُس نَفاذ چھُ کران

مِشَن پوشَن 2.0 (Mission Poshan 2.0) چھُ پالِسی تہٕ مَنصوٗبہ بندی مرکزَس سٕتی ترٛاوان تہٕ دۄہَرار نَفاذ رِیاسَتَن سٕتی۔ اَنٛگنواڑی (Anganwadi) پیکجَس اندر ویکسین، صِحَت جانچ، تہٕ حَوالہ چھِ قومی صِحَت مِشَن (National Health Mission) ذریعہٕ دِنہٕ یِوان۔ اَکھ جائزہ یُس صِرف زَنانن تہٕ شُرٮ۪ن ہِنٛز تَرقی ہُنٛد مَحکَمہ (Department of Women and Child Development) سَمن کرِہ، سُہ چھُ نَہ پیکج ہُنٛد صِحَتی ترٛیِم حِصہٕ کور کران۔ قومی سَمٲجی مَدَد پروگرام (National Social Assistance Programme) چھُ پِنشَنَن پؠٹھ اَکھ آئینہ کیس: مرکزَن پَنُن مَدَد 2012 مَنٛز ٲخری لَٹہِ نظرِثانی کۆر، 21 جُلایی 2026 وۆن زِ کانٛہہ نظرِثانی غورَس تَل چھِ نَہ، تہٕ رِیاسَتی ٹاپ اَپ ₹50 پؠٹھ ₹5,700 تام ریکارڈ کٔر۔ اَکھ مختصر یُس اَکھ قومی پِنشَن کوٹ کران چھُ، سُہ چھُ اَکھ شرح بیان کران یُس رِیاسَتَن گوڈَے بدلاومُت چھُ۔

دیہی مَزدوٗری گارَنٹی چھُ وۄنۍ 1 جُلایی 2026 پؠٹھ VB-G RAM G ایکٹ (VB-G RAM G Act)۔ رِیاسَت چھِ سکیٖم نوٹیفائی کران، ہٮ۪کن شَٹھ دۄہَن تام زرعی وقفہ نوٹیفائی کٔرِتھ، تہٕ بے روزگاری الاؤنس تہٕ دیر معاوضہ چھُ واجِب یَلہِ ایکٹ ہِنٛد شرط پوٗرٕ گَژھَن۔ اَکھ رِیاسَتی جائزَس خٲطرٕ سوال چھِ حقیقت: کیاہ وقفہ نوٹیفائی گۆو، تہٕ کِمَن تاریٖخَن خٲطرٕ؟ کیاہ پرٛون جاب کارڈ وَنان تہِ ماننہٕ چھِ یِوان یَتہِ ای-کے وائی سی (e-KYC) پوٗرٕ چھُ؟ کیاہ دیر-معاوضہ کیس ادا گٔیہِ؟ نفر-دۄہ اَکہِ لَٹہِ چھِ نَہ تِمَن جواب دِوان۔

## ضِلہٕ شیٹ اِستعمال کٔرِو، تہٕ امہِ ہِنٛز سرحد مانِو

ہِندوستان تہٕ رِیاسَت NFHS-6 فیکٹ شیٹ 29 مَے 2026 آیہِ۔ ضِلہٕ مجموعہ 20 تہٕ 21 اَگَست 2026 آیہِ۔ مَنی پوٗر (Manipur) چھُ نَہ مجموعَس مَنٛز۔ IIPS (International Institute for Population Sciences) چھُ پرن والٮ۪ن پَنٮ۪نِس اشاعت حِصَس تہٕ اَکھ مہمان لاگ اِن کُن اشارہ کران۔ اَکھ رِیاسَتی اَوسَط یُس اَکھ دیہی-شَہری خَلأ چھُپان چھُ — گُجَرات (Gujarat) 39.5% دیہی قَد-عُمر کَمی بمقابلہ 28.2% شَہری — چھُ ضِلہٕ مَنصوٗبَس خٲطرٕ اَکھ کمزور اِکٲیی۔ فیکٹ شیٹَس مَنٛز قَوسین چھِ اَکھ پَتَل نمونہ نشان لگاون۔ تِم سیل مستحکم مَہ مَنِو۔

اَکھ NFHS-6 خون کَمی (anaemia) نمبر مختصر مَہ کٔرِو۔ یَتھ دورُک پریس نوٹ تہٕ پۄشتہِ سَطر چھِ نَہ اَکھ شایع کران۔ UDISE+ پؠٹھ اَکھ لَرنِنٛگ نٔتیٖجہ مختصر مَہ کٔرِو۔ سُہ شِرکَت اِستِثنا مختصر کٔرِو یُس سکوٗل جٲری خود چھُ رَٹان: تیاری تہٕ ثانَوی سکوٗل ترٛاوُن وَتھ، تہٕ سُہ ٹیبلَس مَنٛز درمیٲنی سلسلہ 3.5% پؠٹھ 3.6% کُن گۆو۔

## اَکھ لَکُٹ جائزہ ایجنڈا

پرَٛتھ رِیاسَتَس پؠٹھ ژور چیز منٛگِو یِم یہِ گوڈَے واجِب چھِ: پوشَن ماہ (Poshan Maah) پؠٹھ اَنٛگنواڑی اِنتِظامی کَمیٹی (Anganwadi Management Committee) بیٹھَک، ہاوان کھٮ۪ن مینوٗ، پَردھان مَنتٔری ماترو وَندَنا یوجَنا (Pradhan Mantri Matru Vandana Yojana) ڈیش بورڈَس پؠٹھ زَچِہ فٲیدٕ زیرِ التوا، تہٕ نوٹیفائی شُدہ مَزدوٗری-وقفہ کیلنڈر اگر کانٛہہ آسِہ۔ تِم NFHS-6 رِیاسَت شیٹَس سٕتی جوڑِو، قَد-عُمر کَمی، وَزن-قَد کَمی، تہٕ کَم وَزن خٲطرٕ، ترٛےٚ اِشارَن ہِنٛدِس طورَس پؠٹھ پرنہٕ آمٕتۍ۔ یہِ چھُ اَکھ جائزہ۔ اَکھ قومی فیصد چھُ اَکھ دیباچہ۔

گریٹ اِنڈین کَمپَنی (Great Indian Company) چھِ دیباچہ یِتھ کَنِہ لیکھان تاکہ اَکھ سیکرٹریٹ نوٹ اَکھ جٲری تام وَاتِہ۔ سلسلہ تعریٖف چھِ [پَیمایِش نوٹَس](/blog/social-sector-outcomes-evidence-snapshot-ministryofcoal-gov-20260330-700-ks) مَنٛز۔ فِہرِست چھِ [greatindiancompany.com/blog](https://greatindiancompany.com/blog) پؠٹھ۔
