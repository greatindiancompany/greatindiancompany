# ہِندوستان سرکٲری طور کیاہ وَنان چھُ، تہٕ اَکھ براڈ بینڈ درجہ بندی کیاہ چھِپاونہٕ

اَکھ مُلکَن درمیان براڈ بینڈ درجہ بندی چھِ صرف اَتھ تعریفَس پٲٹھۍ اصل یُس اَتھ بۄن چھُ۔ ہِندوستٲنؠ سرکٲری متن چھُ عالمی دعوٕ کَران۔ ہِندوستانُک ٹیلی کام ریگولیٹری اتھارٹی (Telecom Regulatory Authority of India, TRAI) ہِند پنِنہِ فُٹ نوٹ چھِ ہاونہٕ زِ تِم دعوٕ تہٕ اَکھ «فی کس براڈ بینڈ سبسکرائبر» لیگ جدول الگ سوالَن جواب چھِ دِوان۔ یِہ بریف پریس اِنفارمیشن بیورو (Press Information Bureau, PIB) تہٕ TRAI لفظَن اندر چھُ روزان۔ یہِ چھُنہٕ اَکھ ITU درجہ، اَکھ OECD اوسط، یا اَکھ موازنہ مُلک شٲمِل کَران یُس تِم صفحہ حساب چھِنہٕ کَران۔

## دعوٕ یِم حُکوٗمَتھ شایع کٔر

PIB سُند 21 December 2024 بھارت نیٹ (BharatNet) نوٹ چھُ وَنان زِ ہِندوستانَن دُنیاہَس مَنٛز 5G سروسُک سارِوٕے کھۄتہٕ تیز رول آؤٹ وُچھ، October 2024 تام 779 ضِلَن مَنٛز 4,62,084 5G بیس ٹرانسیور سٹیشن لگِتھ۔ وُہی پیراگراف چھُ 783 ضِلَن مَنٛز 24,96,644 4G بیس سٹیشن، March 2014 مَنٛز Rs 269 فی GB پؠٹھ Rs 9.08 فی GB تام ڈیٹا لاگَتَس مَنٛز کمی، March 2014 مَنٛز 1.30 Mbps پؠٹھ 95.67 Mbps تام میڈین موبائل براڈ بینڈ رفتارَس مَنٛز اضافہ، تہٕ 6,44,131 گامَن مَنٛز 6,15,836 مَنٛز 4G موبائل کوریج دِوان۔ نوٹ چھُ بھارت نیٹ دُنیاہَس مَنٛز سارِوٕے کھۄتہٕ بۆڈ دیہی براڈ بینڈ کنیکٹیوٹی پروگرام وَنان۔ یہِ چھُنہٕ «سارِوٕے کھۄتہٕ تیز» جملَس سۭتۍ باقٕے مُلکَن ہِند بیس سٹیشن گنتی چھاپان۔ اَکھ پَرَن والہ دعوٕ کوٹ کٔرِتھ ہیکہِ تہٕ وُنہِ موازنہ کنندَس واجب آسہِ۔

PIB سُند 12 February 2026 پارلیمانی نوٹ چھُ وزیرَن ہِندِس لفظَن مَنٛز اَکھ طویل رسائی کہانی دُہراوان: موبائل سبسکرائبر 2014 مَنٛز 930 million پؠٹھ 1.2 billion، موبائل رسائی 75 percent پؠٹھ 92 percent، انٹرنیٹ سبسکرائبر 250 million پؠٹھ 1 billion کھۄتہٕ زیادٕ، انٹرنیٹ رسائی 20 percent پؠٹھ تقریباً 71.8 percent، تہٕ براڈ بینڈ سبسکرائبر 61 million پؠٹھ 1 billion کھۄتہٕ زیادٕ۔ اوسط فکسڈ براڈ بینڈ رفتار چھُ تقریباً 61.55 Mbps دِنہٕ آمُت۔ تَتھ ریلیزَس مَنٛز لِکھنہٕ آمُت بیان چھُ 61.55 Mbps قومی اوسط December 2025 تام Ookla سُند Speedtest انڈیکس حوالہ دِوان۔ قومی براڈ بینڈ مشن 2.0 (National Broadband Mission 2.0) تحت فکسڈ رفتار ہَدف چھُ 2030 تام 100 Mbps۔ موبائل میڈین رفتار تہٕ فکسڈ اوسط رفتار چھِ الگ سلسلہ۔ 95.67 Mbps تہٕ 61.55 Mbps اَکِس «ہِندوستان تیز چھُ» جملَس مَنٛز تھاوُن چھُ تِم رَلاوان۔

وُہی 2026 جواب چھُ ترمیم شُدہ بھارت نیٹ پروگرام (Amended BharatNet Programme) عالمی سطحَس پؠٹھ سارِوٕے کھۄتہٕ بۆڈ حُکوٗمت قیادت والہ کنیکٹیوٹی پروگرام وَنان تہٕ اَکھ $16.9 billion لیبل لَگان۔ December 2024 نوٹ چھُ پروگرام لاگَت Rs 1,39,579 crore تھاوان۔ خرچَس خٲطرٕ روپیہ عدد استعمال کٔرِو تہٕ عالمی خصوصیتَس خٲطرٕ PIB جملہ۔

## TRAI ہِنٛز تعریف اَکِس درجہ بندی سۭتۍ کیاہ چھِ کَران

end-July 2026 تام TRAI اَس 1,094.48 million براڈ بینڈ سبسکرائبر گٔنۍ۔ تِمَن مَنٛز 1,027.50 million موبائل وائرلیس ٲسۍ۔ فکسڈ وائرڈ رسٲیی ٲس 48.14 million۔ اَکھ مُلک یُس صرف فکسڈ لائن گَنِ چھُ، سُہ سبسکرائبر چارٹَس پؠٹھ «پَتہٕ» وُچھِ تہٕ فائبر ٹو دی ہوم چارٹَس پؠٹھ برونہہ ہیکہِ آسِتھ۔ TRAI چھُ ٹیلی کثافت (tele-density) تہِ 2020 تکنیکی گروپ رپورٹَس پؠٹھ آبادی اندازَن سۭتۍ حساب کَران، تہٕ یہِ چھُ سرخی ٹیلی کثافتَس مَنٛز مشین-پؠٹھ-مشین (machine-to-machine) کنیکشن شٲمِل کَران۔ کُل ٹیلی کثافت ٲس 94.68 percent۔ شَہری ٹیلی کثافت ٲس 154.12 percent۔ دیہی ٹیلی کثافت ٲس 61.05 percent۔ وائر لائن ٹیلی کثافت ٲس 3.36 percent، تہٕ دیہی وائر لائن ٹیلی کثافت ٲس 0.57 percent۔

دِلہِ لائسنس یافتہ سروس علاقہ ٹیلی کثافت ٲس end-July تام 368.33 percent، تہٕ بہارُک ٲس 64.19 percent۔ TRAI سُند نوٹ چھُ خبردار کَران زِ دِلہِ ہُند ٹیلی فون ڈیٹا چھُ غازی آباد، نوئیڈا، گُرگرام، تہٕ فریدآباد پؠٹھ خدمت کرنہٕ آمِتۍ وائرلیس سبسکرائبر شٲمِل کَران۔ اَکھ عالمی چارٹ یُس دِلہِ اَکِس شَہر رِیاسَتھ پٲٹھۍ برتاؤ چھُ کَران یُس اَکِس مُلکَس سۭتۍ موازنہ چھُ، سُہ ہِندوستٲنؠ سلسلہ پنِنہِ فُٹ نوٹَس خلاف استعمال چھُ کَران۔

TRAI ہِندِس March 2026 سہ ماہی اَس فی 100 آبادی انٹرنیٹ سبسکرائبر 76.59، شَہری 126.80، تہٕ دیہی 48.31 تھاو۔ February 2026 مشن بیان چھُ 47.16 دیہی انٹرنیٹ سبسکرائبر فی 100 اَکِس 60 ہِندِس 2030 ہَدفَس کُن بیس لائن پٲٹھۍ استعمال کَران۔ یِم چھِ سرکٲری گھریلو بینچ مارک۔ تِم صحیح موازنہ چھِ یَتام اَکھ بنیٲدی ہِندوستٲنؠ ماخذ اَکھ یکساں غیر ملکی سلسلہ شایع کَرِ۔

تِم لائسنس یافتہ علاقہ ہِندوستانَس اندر کِتھ پٲٹھۍ الگ چھِ، سُہ چھُ [رِیاسَتھ نمونہ نوٹ](/blog/telecom-broadband-state-wise-patterns-mohfw-gov-20260330-312-ks) ہُند موضوع۔
