# ہِندوستانٕچ صَنعتی تِجارَت ہِنٛزِ تَنوعَس مَنٛز سَرمایہٕ خرچُک چکر وُچھُن

ہِندوستانٕچ صَنعتی تَنوع ہِنٛزِ کَتھَس مَنٛز سارِوٕے کھۄتہٕ غلط پَرنہٕ یِوان رَقَم چھُ Rs 1.97 lakh crore۔ پریس اِنفارمیشن بیورو (Press Information Bureau, PIB) ہِنٛدِس 2 اَگَست 2023 ہِنٛدِس نوٹَس مَنٛز یِہ رَقَم چھُ 14 شُعبَن خٲطرٕ پیداوار سۭتۍ جُڑِتھ حَوصلہٕ سکیٖمَن (Production Linked Incentive, PLI) ہُند حَوصلہٕ تَخصیٖص۔ یِہ چھُ نہٕ آرڈَر کرنہٕ آمَژَن مشینَن ہِنٛزِ گِنتی، بناونہٕ آمَژَن عِمارَتَن ہِنٛزِ گِنتی، یا مول ہٮ۪ونہٕ آمٕتۍ فراہم کُنَندہٕ اَوزارَن ہُند حساب۔ یُس سَرمایہٕ خرچ (capital expenditure, capex) وُچھَن وول یِہ تَخصیٖص خرچَس مَنٛز لِکھِ، سُہ وُچھِ اَکھ تیزی یُس دَستاوَزَن مَنٛز درج چھِ نہٕ۔

## ترٛےٚ الگ رَقَمَن ہِنٛدِ ڈھیر

ترٛےٚ ڈھیر الگ تھٲیو۔

گۄڈنیُک چھُ حَوصلہٕ ہُند لِفافہٕ۔ 2 اَگَست 2023 ہُند لوک سبھا (Lok Sabha) جواب چھُ وَنان زِ سکیٖمَن ہُند اِعلان سَرمایہٕ تہٕ ٹیکنالوجی انُنہٕ خٲطرٕ، پیمانہٕ بَناونہٕ خٲطرٕ، تہٕ صَنعتی تَیارِی تہٕ بَرآمَدَس سَہارٕ دِنہٕ خٲطرٕ آو کَرنہٕ۔ اَتھ ڈیزائنُک حَوصلہٕ چھُ پرٛؠتھ نافِذ کَرن واجِن وَزارَتہِ ہِنٛدِ کارکردگی قٲیدَن ہِنٛدِس مُقابلَس مَنٛز دِنہٕ یِوان۔ یَتام زَن پَتہٕ کن اَکھ سَرکٲرِی نوٹ ادا کرنہٕ آمُت دعوٕ رپورٹ کَرِ، تَخصیٖص چھُ حَوصلہٕ ہِنٛز گُنجٲیش، خرچ کرنہٕ آمُت سَرمایہٕ خرچ نہٕ۔

دویِم چھُ دَرخواست دِن وول سُند پنُن سَرمایہٕ خرچ۔ سُہے جواب چھُ وَنان زِ سکیٖمَن ہُند اَکھ سلسِلہٕ وار اَثَر آسُن چھُ: اَصل اِدارٕ (anchor units) چھِ فراہم کُنَندَن ہِنٛز بُنیاد تھاوان، تہٕ زیادٕ تَر معاون اِدارٕ چھِ خُرد، لَگوٗ تہٕ مَدرَم اِدارَن (micro, small and medium enterprises, MSME) ہِنٛدِس شُعبَس مَنٛز اُمید کرنہٕ یِوان۔ تَتھ تٲریٖخَس ٲس 733 دَرخواستٕ چُنہٕ آمژٕ، تہٕ فٲیدٕ وَٹھن وٲلِیَو مَنٛز 176 ٲس MSME، بَلک دَواہِ (bulk drugs)، طِبی آلات (medical devices)، دواسازی (pharma)، ٹیلی کام (telecom)، سفید سامان (white goods)، خورٲکی عمل آوری (food processing)، کَپُڑ (textiles)، تہٕ ڈرونَن (drones) مَنٛز۔ چُناو چھُ اَکھ دروازٕ۔ یِہ چھُ ثُبوت زِ فرمَو سکیٖمَس اندر آسُن منٛگ۔ یِہ چھُ نہٕ چالو کرنُک سَرٹیفِکیٹ۔

ترٛیِم چھُ کارخانہٕ کِس آسَس پؠٹھ سَرکٲرِی سَرمایہٕ خرچ: صنعتی کوریڈور (corridors)، پارک، تہٕ معدنی زنٛجیر۔ PIB ہُند 27 نَوَمبَر 2025 ہُند نوٹ چھُ نیشنل اِنڈسٹریل کوریڈور ڈیولپمنٹ کارپوریشن (National Industrial Corridor Development Corporation, NICDC) ہِنٛدِ پٕلَگ اینڈ پلے پارکَن (plug-and-play parks) تہٕ مالِ کوریڈورَن ہِنٛدِس کامَس بیان کَران، بشمول عالَمی بَینکَس (World Bank) سۭتۍ جُڑِتھ مالی مدد یُس سُہ مَشرِقی مَخصوٗص مالِ گٲڑِ راہ (Eastern Dedicated Freight Corridor) تہٕ ریل مالِ رَسانی خٲطرٕ USD 1.96 billion وَنان چھُ، تہٕ گَنگا آبی رَستہٕ ہِنٛزِ تَرقِی خٲطرٕ USD 375 million۔ تِم چھِ صَنعتی تَیارِی کِس آسَس پؠٹھ بُنیٲدی ڈھانچہٕ ہُند خرچ، PLI ہِنٛد جٲری رَقَم نہٕ۔ قومی اَہَم معدنی مشن (National Critical Mineral Mission)، یُس مرکزی کابینہِ (Union Cabinet) 29 جَنؤری 2025 دۄہ منظوٗر کۆر، چھُ پنِنہِ تَخصیٖص تھاوان: اَہَم معدنی عمل آوری ہِنٛدِ پارکَن خٲطرٕ Rs 500 crore تہٕ دُبارٕ اِستعمالَس خٲطرٕ Rs 1,500 crore، یِتھ کَنہِ PIB ہِنٛدِس 24 مارچ 2025 ہِنٛدِس نوٹَس مَنٛز درج چھُ۔ یِہ چھُ اَنٛدَرونی مالَس پؠٹھ سَرمایہٕ خرچ، تہٕ یِہ چھُ اَکھ بَجَٹ تَخصیٖص، نہٕ اَکھ ایسا پارک یُس گوڈائے کیتھوڈ مواد (cathode material) بناوان چھُ۔

## کَتِ چھُ چکر بَوَہ

پرٛؠتھ PLI شُعبہٕ چھُ نہٕ یکساں سَرمایہٕ خرچ وول۔ ایڈوانسڈ کیمسٹری سیل بیٹری (advanced chemistry cell batteries)، ہائی ایفیشَنسی سولر پی وی ماڈیول (high-efficiency solar PV modules)، خاص فولاد (specialty steel)، تہٕ الیکٹرانک پرزٕ چھِ کَٹُن تہٕ سِیوُن وٲلۍ لائنَس کھۄتہٕ کئی ؤرِیَن ہِنٛدِس اثاثہٕ چکرَس زیٛادٕ نزدیٖک۔ سَرکٲرِی فِہرِست چھِ تِمَن خورٲکی سامانَس تہٕ کَپُڑَس سۭتۍ سُہے 14 شُعبَن ہِنٛدِس چوکَٹَس مَنٛز تھاوان۔ اَکھ سَرمایہٕ خرچ بورڈ (dashboard) یُس تِمَن ہُند اَوسَط کَڑِ، سُہ چکر گَژھِ راوان۔ سَرمایہٕ بَوَہ شُعبَن مَنٛز زَمیٖن، بِجلِی، تہٕ فراہم کُنَندہٕ اَوزارَن ہِنٛدِ آرڈَر وُچھِو، تہٕ مَزدوٗری بَوَہ شُعبَن مَنٛز اِستعمال تہٕ بَرآمَد آرڈَر وُچھِو۔

2023 ہُند جواب اوس یِہ تہِ وَنان زِ سکیٖمَن ٲس نافَذ کرنِک الگ الگ مرحلَن مَنٛز، تہٕ زِ کابینہِ اوس تَتھ دۄہَس تام نَو شُعبہٕ منظوٗر کۆرمُت نہٕ۔ اَکھ چکر ہیٚکہِ رُکِتھ تِکیازِ اَکھ شُعبہٕ ہِنٛز رہنُمٲیی چھِ وُنہِ تہِ نافَذ گَژھان، اگرچہ قومی تَخصیٖص بۆڈ چھُ وَنان۔ شُعبَن ہِنٛز فِہرِست جَمَد ماننہٕ برٛونٛہہ، تہٕ یِہ ماننہٕ برٛونٛہہ زِ سۄ بَڑِمُت چھِ، تازٕ PIB نوٹ پَرِو۔

## سہہ ماہَس پؠٹھ سَرمایہٕ خرچَس صَفحَس پؠٹھ کیاہ تھٲیو

ژور سَطر، پرٛؠتھ اَکھ تٲریٖخَس سۭتۍ: حَوصلہٕ تَخصیٖص (Rs 1.97 lakh crore، یِتھ کَنہِ اَگَست 2023 مَنٛز وَنہٕ آو، یَتام زَن اَکھ نٔو نوٹ اَتھ جاے نِیہِ)؛ چُنہٕ آمٕتۍ دَرخواست دِن وٲلۍ، تہٕ سَرمایہٕ ہِنٛدِ دعوٕ یا جٲری کرنہٕ آمُت حَوصلہٕ ہُند کُنہِ پَتہٕ کُن رَقَمَس سۭتۍ مُقابلہٕ؛ سَرکٲرِی کوریڈور تہٕ پارک مَنصوٗبہٕ یِم تُہۍ اَکھ PIB یا صَنعت تہٕ اَندروٗنی تِجارَت ہِنٛزِ تَرقِی ہِنٛزِ محکَمہٕ (Department for Promotion of Industry and Internal Trade, DPIIT) نوٹَس سۭتۍ جوڑِتھ ہیٚکِو؛ تہٕ معدنی عمل آوری ہِنٛز تَخصیٖص اگر تُہٕندِ مالِ فِہرِستَس (bill of materials) لِتھیَم، کوبالٹ، یا اَتھ پٲٹھۍ اَنٛدَرونی مالَس ضرورت چھِ۔ پیشَنگوئی سۭتۍ بَرُنَس بَدل خٲلی تھٲیو۔

گریٹ اِنڈین کَمپَنی (Great Indian Company) چھِ یِہ سَرمایہٕ خرچُک پَرُن روزگار نوٹَس تہٕ خطرٕ نقشَس نِش الگ تھاوان۔ کَمپَنی چھِ یِہ سیٹ [greatindiancompany.com](https://greatindiancompany.com) پؠٹھ شایع کَران۔
