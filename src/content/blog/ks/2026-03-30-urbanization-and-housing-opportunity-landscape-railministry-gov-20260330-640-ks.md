# شَہری رہٲیِشی کٲم یۄس سرکٲری قٲعدٕ وِنہِ تہِ کھُلہ چھِ تران

## موقع چھُ اَکھ قٲعدٕ یَس وِنہِ تہِ اَکھ چَلاوَن وول چھُ

اَکھ منظر یُس «ہوشیار شَہر، قٲبلِ بَرداش رہٲیِش، پراپ ٹیک (proptech)» قٲعدٕ ورٲے فِہرِست چھُ کَران، سُہ چھُ کلیدی لفظَن ہِنٛز فِہرِست۔ سرکٲری ریکارڈ چھُ پانٛژھ قٕسم کٲم کھُلہ تران تِکیازِ دَستاویٖز چھُ وَنان زِ کُس وِنہِ تہِ عمل کَرِ۔

## سودِ ہِنٛزِ اِمدادَس اندر شُروٗعات

PMAY-شَہری 2.0 (Pradhan Mantri Awas Yojana – Urban 2.0) ہِنٛز سودِ ہِنٛز اِمدادی سِکیٖم (Interest Subsidy Scheme, ISS) چھِ یِمَن شَرطَن پؠٹھ جِنٛدٕ یِم پریٚس اِطلاع بیورو (Press Information Bureau, PIB) 24 جُلاے 2025 رپورٹ کٔر۔ اَکھ قَرضہٕ دیٖن وول یا رہٲیِشی مالیات کَمپَنی ہیٚکہِ اَکھ قَرضہٕ شُروٗع کٔرِتھ یُس ₹9 لاکھ تام آمدَنی، ₹25 لاکھ تام قَرضہٕ، ₹35 لاکھ تام جائیداد، تہٕ 120 مُرَبَع میٖٹَر تام کارپِٹ رَقبہ (carpet area) منٛز فِٹ سپدِ، تہٕ گۄڈَنیِکِس ₹8 لاکھَس پؠٹھ 4 فیصَد پؠٹھ ₹1.80 لاکھ تام حٲصِل کٔرِتھ، 12 ؤریَن تام، پانٛژھ قِستَن منٛز۔ کھُلہ عملی مسلہٕ چھُ قَرضہٕ جٲری اِمتِحانَس اندر تھاوُن: چالو، تہٕ پرٛٮ۪تھ قِستَس پؠٹھ اَصل رَقَمُک اَدھا کھۄتہٕ زِیادٕ باقی۔ بند قَرضَس سۭتۍ جوڑِتھ اِمدادی سِکیٖم (Credit Linked Subsidy Scheme) چھِ نَہ موقع۔ سۄ گوٚو درمیٲنی آمدَنی وٲلٮ۪ن قَرضہٕ رَٹن وٲلٮ۪ن خٲطرٕ 2021 منٛز تہٕ باقی اِہل گروپَن خٲطرٕ 2022 منٛز خَتم، ₹58,885 کَروڑ جٲری گَژھنہٕ پَتہٕ۔

شِراکَت تہٕ فٲیدٕ گَژھَن وٲلٮ۪ن ہِنٛدۍ یونِٹ چھِ ₹2.50 لاکھ حُکوٗمَتی مَدَد تھاوان، مرکزی حِصَس سۭتۍ یُس دَسَمبَر 2024 نوٹ عَلاقَس مُطابِق چھُ تھا۔ کھُلہ کٲم چھِ رِیاسَتِ ہِنٛز یادداشت، شَہری مَقٲمی اِدارَس (urban local body) ہِنٛز فٲیدٕ گَژھَن وٲلٮ۪ن ہِنٛز فِہرِست، تہٕ شِراکَت مَنصوٗبَن خٲطرٕ اَکھ شَریک یُس سُہ مَدَدَس تام بَناوِ۔ نوٹَس ہُنٛد «6 لاکھ کھۄتہٕ زِیادٕ» اُصولی منظوٗری ٲس 29 رِیاسَتَن تہٕ مرکزی زیرِ اِنتِظام عَلاقَن پؠٹھ دَسَمبَر 2024 ہُنٛد مَوقِف۔ اَکھ فَرم گَژھِہ موجودٕ رِیاسَتی منظوٗری پرِتھ، نَہ زَن سُہ جملہٕ، سایِٹ ٹیٖم نوکری دِنہٕ برٛونٛہہ۔

## کرایَہ ذخیرٕ یُس سِکیٖم ناو چھِ دِوان تہٕ پانَے چھُنہٕ بَناوان

قٲبلِ بَرداش کرایَہ وٲل رہٲیِش (Affordable Rental Housing) چھُ 2.0 عمودی حِصہٕ کٲم کَران وٲلٮ۪ن زنانَن، صنعتی مَزدورَن، شَہری مہاجِرَن، یِمَن بے گَر یا نادار چھِ، طٲلِبِ عِلمَن، تہٕ اَتھ کٮ۪ن فٲیدٕ گَژھَن وٲلٮ۪ن خٲطرٕ۔ دَسَمبَر 2024 نوٹ چھُ وَنان زِ عَوامی تہٕ نِجی اِدارَن پؠٹھ حصہٕ رَٹُن چھُ توقع۔ یہِ چھُ اَکھ دعوت یۄس سِکیٖم دَستاویٖزَس منٛز لیکھنہٕ آمٕژ چھِ۔ سۄ چھِ مَنصوٗبہٕ صِرف تَتھ وِزِ بَنان یَلِہ رِیاسَت تجویٖز انان چھِ تہٕ اَکھ اِدارٕ سِکیٖم قٲعدَن تحت کرایَہ وٲل رہٲیِش مالِک یا چَلاوُن تَیار چھُ۔ سۄ چھِ الگ سودِ ہِنٛزِ اِمدادی سِکیٖمَس منٛز اپارٹمَنٹ فروخت کَرنہٕ نِش۔

## فنڈِ ہِنٛدِس آخری چھَپنہٕ آمٕتِس ؤریہِ پَتہٕ آب نِظام

AMRUT 2.0 (Atal Mission for Rejuvenation and Urban Transformation 2.0) ہِدایَت چھِ وِنہِ تہِ اثاثہٕ بیان کَران: 4,700 قانوٗنی قَصبَن (statutory towns) خٲطرٕ نَلکہٕ، 500 شَہرَن خٲطرٕ سیوریج یا سیپٹیج، شَہری آب بیلَنس پلان، تہٕ اَکھ ٹیکنالوجی ذیلی مِشَن یُس آب سیکٹَرَس منٛز سٹارٹ اَپ حوصلہٕ چھُ دِوان۔ تِم چھِ تجارتی شَرط تہِ بیان کَران: ملین-پُلس شَہر (million-plus city) فنڈَن ہُنٛد 10 فیصَد شِراکَت طٔریٖقَس منٛز، تہٕ میوٗنِسِپَل بانڈ تہٕ کریڈِٹ ریٹِنگ اِصلاح کِس پٲٹھۍ۔ اِشارٲتی خرچُک عرصہٕ چھُ مالی ؤری 2025-26 تام چَلان۔ موقع یُس سُہ تٲریٖخَس پَتہٕ زِنٛدٕ چھُ روزان، سُہ چھُ کارروائی، غٲر آمدَنی آب (non-revenue water)، صاف کَرنہٕ آمُت آبُک دُبارٕ اِستِمال، تہٕ کانٛہہ رِیاسَتی منصوٗبہٕ یُس وِنہِ تہِ مَنصوٗبَہ مُکَمَل کَران چھُ۔ سُہ چھُ کمزور موقع اَکِہ فَرمَس خٲطرٕ یَس 2026-27 منٛز تازٕ مرکزی منظوٗری چھِ تہٕ پَتِم حُکوٗمَتی فیصلہٕ چھُنہٕ۔

پے جَل سَرویکشَن (Pey Jal Survekshan)، سُہے ہِدایَتَن منٛز، چھُ آبِ ہِنٛزِ کوالیٹی تہٕ مقدارِ ہُنٛد شایع جائزٕ۔ اَکھ کَمپَنی یۄس رِساو ماپان چھِ یا رسد ٹیسٹ کَران چھِ، تَس چھُ اَکھ ناو دِنہٕ آمُت عَوامی اِستِمال۔ ہِدایَت چھِ نَہ خریدِ ہُنٛد کوٹہ وعدٕ کَران۔

## نِظام گوڈَے بَناونہٕ آمٕتۍ، بَجَٹ گوڈَے بند

تَمام 100 ہوشیار شَہرَن (Smart Cities) چھِ کمانڈ سینٹَر، MoHUA صَفَس پؠٹھ تہِ تہٕ اَگَست 2026 لوک سَبھا (Lok Sabha) جَوابَس پؠٹھ تہِ۔ صَفہٕ چھُ 31 شَہرَن منٛز آبَن ہِنٛدٮ۪ن نیٹ وَرکَن پؠٹھ SCADA، 48 شَہرَن منٛز RFID کوڑٕ گاڑۍ، تہٕ ہوشیار سَڑکَہ تہِ بیان کَران۔ 10 جوٗن 2025 ہُنٛد مَشوَرٕ نَمبَر 27 چھُ رِیاسَتَن پرژھان زِ خاس مَقصَدٕچ کمپَنی (special purpose vehicles) دُبارٕ مقصد دِیَن۔ کھُلہ کٲم چھِ کارروائی، دیکھ بھال، تہٕ اِنٹیگریشَن اَکِہ شَہری کَمیٹیَس خٲطرٕ یۄس فیصلہٕ کَران چھِ زِ کمپنیَس وِنہِ تہِ کِردار چھُ۔ بند کٲم چھِ مرکزی فنڈ وول نٔو شَہر۔ MoHUA چھُ وَنمُت زِ کانٛہہ نٔو شَہر گَژھِہ نَہ رَٹنہٕ یِنِہ۔

## اِظہار اَکھ مصنوعات کِس پٲٹھۍ

ریٖل اِسٹیٹ (ریگولیشَن اینڈ ڈیوَلَپمَنٹ) ایکٹَس (Real Estate (Regulation and Development) Act) ہُنٛد دَفہ 4 چھُ لازم کَران زِ اَلاٹی جمعَن ہُنٛد سَتَتھ فیصَد زَمیٖن تہٕ تعمیٖر خٲطرٕ الگ شیڈول بینٛک کھاتَس منٛز آسِہ۔ رِیاسَتی اتھارٹی چھِ رجِسٹَر شایع کَران۔ سافٹ ویئر، اِسکرو (escrow) کارروائی، کارپِٹ رَقبَس ہُنٛد پیمانہٕ، تہٕ ایجَنٹ رجِسٹریشَن چھِ کٲم یۄس قانوٗن چھُ بَناوان۔ تِم چھِ نَہ MoHUA مِشَن سرکیولَرَس سۭتۍ بَنان، تہٕ تِم چھِ رِیاسَتی قٲعدَس مُطابِق بدلِتھ۔

گۄڈنیُک PMAY-شَہری نوٹَس ہُنٛد 917 کَروڑ نفر-دۄہ ہُنٛد اندازٕ چھُ تٲریٖخ وول تعمیٖری اندازٕ، نَہ زَن خٲلی جاے فِہرِست۔ ہُنَر تہٕ سُہ اندازٕ چھِ نوکری بَریٖفَس منٛز۔
