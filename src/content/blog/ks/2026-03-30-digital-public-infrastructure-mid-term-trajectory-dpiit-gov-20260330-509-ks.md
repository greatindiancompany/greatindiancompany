# ہِندوستانٕکۍ ڈیجیٹل ریل گوڈَے کوتاہ کَرنُک شیڈول چھُ

راستہٕ چھُ سُہ وَتھ یَتھ کٔہہِ عزم کٔر، نَہ اختیارُک اندازٕ۔ ہِندوستانٕکۍ ڈیجیٹل عوٲمی بُنیادَس (digital public infrastructure) پؠٹھ چھُ بیٚیِس ؤریہِ تہٕ اَدھَس خٲطرٕ عزم وول وَتھ لۄکُٹ تہٕ مخصوص۔ ڈیجیٹل پَرسَنَل ڈیٹا پروٹؠکشن ایکٹ (Digital Personal Data Protection Act) ہُند زیٛادٕ تر عملی حِصہٕ چھُ 13 نَوَمبَر 2025 پَتہٕ اَٹھارَہ رؠتہٕ خٲطرٕ مُقَرَر۔ اوپن نیٹ ورک فار ڈیجیٹل کامرس (Open Network for Digital Commerce, ONDC) خٲطرٕ لۄکُٹ ادارٕ شامِل کَرنُک سکیٖم چھُ مالی ؤری 2026-27 تام چَلان۔ عوٲمی ریکارڈَس مَنٛز باقی سٲری چھُ گوڈَے پروڈکشنَس مَنٛز نِظام، یُس ہُند بیٚیُہ قدم مسلسل اشاعت چھُ نَہ تٲریٖخ وول نٔو قانوٗن۔

## قانوٗنی وَتھَس چھِ زٕ بَقیہ تٲریٖخ

13 نَوَمبَر 2025 ہُند نوٹیفِکیشن G.S.R. 843(E) وَنٛڑ ڈیجیٹل پَرسَنَل ڈیٹا پروٹؠکشن ایکٹ، 2023 ترٛؠن شُروعَن مَنٛز۔ گۄڈنیُک، اشاعت پؠٹھ، کٔر ادارَتی پروویژن کَور، یِمَن تَحت ڈیٹا پروٹؠکشن بورڈ (Data Protection Board) سُہے دۄہ G.S.R. 844(E) ذریعہٕ قایم گٔیہِ۔ بورڈُک ہیڈ آفِس چھُ قومی رازدٲنی عَلاقَس (National Capital Region) مَنٛز۔ سُہ حِصہٕ چھُ نَہ مستقبلَس مَنٛز۔

دویُم شُروع چھُ اشاعت پؠٹھ اَکھ ؤری: دفعہ 6(9) تہٕ دفعہ 27(1)(d)۔ سُہ تٲریٖخ چھُ نَوَمبَر 2026 مَنٛز یِوان۔ ترٛیُم چھُ اشاعت پؠٹھ اَٹھارَہ رؠتہٕ: دفعہ 3 پؠٹھ 5، دفعہ 6 ہُند ذیلی دفعہ (1) پؠٹھ (8) تہٕ (10)، دفعہ 7 پؠٹھ 17، دفعہ 27 ہُند باقی فقرہ گوڈَے کَور کَرنہٕ آمُت علاوٕ، دفعہ 28 پؠٹھ 34، دفعہ 36 تہٕ 37، تہٕ دفعہ 44(2)۔ 13 نَوَمبَر 2025 پؠٹھ اَٹھارَہ رؠتہٕ چھِ مئی 2027 مَنٛز یِوان۔

قاعِدے، 2025 (Rules, 2025) آیہِ 14 نَوَمبَر 2025 شایع کَرنہٕ، 16 دَسَمبَر 2025 تصحیح سۭتۍ۔ سَتَمبَر 2026 تہٕ مئی 2027 درمیان چھُ شیڈول قانوٗنی کٲم تِکیازِ نَوَمبَر 2026 مَنٛز زٕ فقرے تہٕ مئی 2027 مَنٛز اہم عملی بلاک جذب کَرُن۔ یہِ صَفہٕ چھُ نَہ فقرے دُبارٕ وَنان۔ راستہٕ چھُ کیلنڈر پَنہٕ۔ بَناون وٲل یِم ایکٹ گوڈَے پوٗری پٲٹھۍ نافذ چھِ مانان، تہٕ بَناون وٲل یِم یہِ وۄنۍ تہِ مسودٕ چھِ مانان، چھِ دۄشوٕے گزٹ غلط پران۔

کھاتہٕ جَمع کَرن وول (Account Aggregator) رول بُک چھُ نَہ یَتھ کیلنڈرَس پؠٹھ۔ ریزرو بینک ہُند 28 نَوَمبَر 2025 ہِدایات چھِ گوڈَے نافذ۔ بیٚیِس ؤریہِ تہٕ اَدھَس پؠٹھ چھِ تِم "آن" گَژھان نَہ۔ تِم چھِ جاری ایگریگیٹرَن لین دین چَلاونہٕ، مالی معلومات تھاونہٕ، تہٕ جَمع کَرنُک کاروبار آوٹ سورس کَرنہٕ منع کَران۔ کانٛہہ درمیانی کَتھ یَتھ مَنٛز کھاتہٕ جَمع کَرن وٲل قرض دِن وٲل یا ادایگی کَمپنی بَنان چھِ تِمَن ہِدایَتن مَنٛز تبدیٖلی ضروٗرت۔ اتہِ اِستِعمال کَرنہٕ آمٕتۍ دَستاوِزَن مَنٛز چھُ نَہ ایساہ تبدیٖلی۔

## تجارت سکیٖمَس چھُ مُکَمَل گَژھنُک ؤری

4 اَگَست 2026 لوک سَبھا (Lok Sabha) جَواب چھُ ایم ایس ایم ای ٹریڈ اِنیبلمؠنٹ اینڈ مارکِٹِنٛگ (MSME Trade Enablement and Marketing) اِقدامَس تَحت مالی ؤری 2024-25 پؠٹھ 2026-27 تام ₹277.35 کروڑ منظور کَران۔ پونٛسہٕ چھُ ONDC سۭتۍ شٲمِل گَژھن وٲل مایکرو تہٕ لۄکُٹ ادارَن سَہار کَران: آگاہی، تربیت، فِہرَست بَناوُن، شامِل کَرُن، کھاتہٕ اِنتِظام، لوژِسٹِکس، تہٕ بَند کَرُن۔ فایِدہ تُلن وٲلِ ادارَن ہُند اَدھ چھُ زنانَن ہُند ہَدف۔ مالی ؤری 2026-27 چھُ 31 مارچ 2027 مُکَمَل گَژھان، یُس مئی 2027 ڈیٹا تحفُظ شُروعَس سۭتۍ سُہے کِھرِکۍ اندر چھُ بیٹھان۔

یہِ چھُ نؠٹ ورکَس کنارَس پؠٹھ شیڈول عوٲمی کوشِش، نَہ سؠکشن 8 کَمپنی (section 8 company) پَنہٕ ہُند شیڈول توسیع۔ جَواب چھُ ONDC گوڈَے پرچون، نقل و حرکت، لوژِسٹِکس، مالی خِدمَت، تہٕ زراعتَس پؠٹھ چالوٗ وَنان۔ جُلایی 2026 تام رِپورٹ کٔر اَتھ 500 مِلین کھۄتہٕ زیٛادٕ کُل آرڈَر، منٹَس پؠٹھ 500 کھۄتہٕ زیٛادٕ آرڈَر، تَقریٖبَن 2 لَچھ یا زیٛادٕ چالوٗ پرچون سوداگَر (تَقریٖبَن 60 فیصد لۄکُٹ فروش)، تہٕ تَقریٖبَن 10 لَچھ یا زیٛادٕ نقل و حرکت تہٕ لوژِسٹِکس خِدمَت فراہم کَرن وٲل۔ ڈیجی دُکان (DigiDukaan)، جَوابَس مَنٛز ناو دِنہٕ آمُت کِرانا شامِل کَرنُک کوشِش، اوس حیدَرآباد (Hyderabad) تہٕ جَے پوٗر (Jaipur) مَنٛز چالوٗ۔ درمیانی پرُن چھُ یہِ زِ فروش سَہارَس چھُ مارچ 2027 تام بَجَٹ تہٕ زٕ شَہَرَن مَنٛز ہاونہٕ آمُت فیلڈ ماڈل۔ یہِ چھُ نَہ بیٚیہِ پَنجاہ شَہَرَن ہُند شایع فِہرَست۔

ONDC ہُند تعارُفی صَفہٕ، یُس نؠٹ ورکُک پنُن کھاتہٕ چھُ نَہ پارلیمانُک، چھُ سُہے شعبَن ہُند پھیلاو وَنان تہٕ پرن وٲلَن سالانہ رِپورٹ تہٕ سہ ماہی اَپ ڈیٹ کُن اِشارٕ کَران۔ تِم دَستاوِز چھِ صحیح بیٚیُہ پرُن یُس انتظامُک نظریہ چھُ یَژھان۔ تِم چھِ نَہ اُوپر پارلیمانی کُلَن ہُند بدل۔

## نِظام یِم پروڈکشنَس مَنٛز چھِ، تہٕ تِکیازِ تِمَن چھُ نَہ "لانچ" بَقیہ

کینٛہہ ریل چھِ سُہ نقطہ پار ییتہِ راستہٕ لانچ تٲریٖخ مَطلَب آسِتھ ہؠکِہ۔ تِہُند عوٲمی وَتھ چھُ گوڈَے بیان کَرنہٕ آمُت ڈیزایِنُک گہرا اِستِعمال۔

شِناخت۔ یونیک آئیڈؠنٹِفِکیشن اتھارٹی آف اِنڈیا (Unique Identification Authority of India, UIDAI) ہُند سیچوریشن رِپورٹ 30 جوٗن 2026 تام چھُ چالوٗ آدھار (Aadhaar) رَسایِی قومی سطحَس پؠٹھ 94.59 فیصد، ناگالینڈَس (Nagaland) مَنٛز 62.41 فیصد، اروناچل پردیشَس (Arunachal Pradesh) مَنٛز 77.77 فیصد، تہٕ بِہارَس (Bihar) مَنٛز 88.29 فیصد تھاوان۔ رِپورٹَس مَطلَب راستہٕ، اگرچہ ہَدف پٲٹھۍ وعدٕ نَہ، چھُ تِمَن فَرَقَن ہُند بند۔ رِپورٹ چھُ نَہ تٲریٖخ مُقَرَر کَران زِ ناگالینڈ قومی شرحَس کَنِس تام وَتِہ۔ اَکھ درمیانی منصوٗبہٕ یُس پرؠتھ ضِلعَس مَنٛز عالمگیر تصدیٖق فرض کَرِ سُہ چھُ شایع کَرن وٲلِس برٛونٛہہ۔

ادایگی۔ نیشنل پیمنٹس کارپوریشن آف اِنڈیا (National Payments Corporation of India, NPCI) ہُند ماہانہ جَدوَل چھُ گوڈَے اَگَست 2026 مَنٛز 752 بینک تہٕ 24,508.96 مِلین لین دین ہاوان۔ حُکوٗمَتَن الگ وَن، 29 جُلایی 2026 پریس اِنفارمیشن بیورو (Press Information Bureau, PIB) کھاتَس مَنٛز، زِ یوٗنِفایڈ پیمنٹس اِنٹرفیس (Unified Payments Interface, UPI) چھُ 55.49 کروڑ کھۄتہٕ زیٛادٕ فَردَن تہٕ 6.5 کروڑ سوداگَرَن خِدمَت کَران۔ شیڈول رویہ چھُ بیٚیِس رؠتہٕ ہُند سَطر۔ یِمَن کاغذاتَن مَنٛز چھُ نَہ دویُم "UPI لانچ۔"

دَستاوِزات تہٕ اے پی آی۔ ڈیجی لاکَرَس (DigiLocker)، 31 جُلایی 2026 تام، ٲس 72.43 کروڑ کھۄتہٕ زیٛادٕ اِستِعمال کَرن وٲل تہٕ 5,437 دَستاوِز قِسم، یِمَن مَنٛز زیٛادٕ تر رِیاسَتی حُکوٗمَتَن پؠٹھ (4,776، مرکزی 661 مُقابِل)۔ اے پی آی سیتو (API Setu) ٲس 29 جُلایی کھاتَس مَنٛز 8,904 شایع اے پی آی۔ برٛونٛہہ کٲم، تِمَن رَقَمَن ہُند چہرَس پؠٹھ، چھےٚ زیٛادٕ جاری کَرن وٲل تہٕ زیٛادٕ اے پی آی، یُس بتدریج شامِل کَرُن چھُ۔ MeitY ہُند ظٲہِر ڈیجی لاکَر تَخصیٖص شُروع پؠٹھ، 29 جُلایی لوک سَبھا جَوابَس مَنٛز ₹168.58 کروڑ، چھُ نَہ نٔو سَرمایہٕ پروگرام وَنان۔

صحت۔ آیوشمان بھارت ڈیجیٹل مِشن (Ayushman Bharat Digital Mission) آو، اَمہِ پنٛنِس سایِٹَس حساب، سَتَمبَر 2021 مَنٛز شُروع کَرنہٕ۔ 20 جُلایی 2026 تام رِپورٹ کٔر وَزارَتَن 94.87 کروڑ اے بی ایچ اے (ABHA) شِناخت، 5.36 لَچھ سہوٗلیت، تہٕ 10.09 لَچھ پیشہٕ ور، صحت ریکارڈَن ہُند مرکزی ذخیرہٕ بغیر تہٕ اِضمام برٛونٛہہ سینڈ باکس (sandbox) جمع حِفاظَتی جانچ سۭتۍ۔ ناو وٲل عوٲمی نِظام، پی ایم-جے (PM-JAY) تہٕ نِکشَے (Nikshay) سَمیت، چھِ گوڈَے جوڑِتھ۔ جَوابَس مَنٛز درمیانی وَتھ چھُ سُہے ڈیزایِنُک مزید اِضمام، یوٗنِفایڈ ہیلتھ اِنٹرفیس (Unified Health Interface) دریافت پرت پٲٹھۍ سَمیت۔ جَواب چھُ نَہ سہوٗلیت رجِسٹریشنَن خٲطرٕ ہَدف مُقَرَر کَران۔

تٲلیٖم۔ دِکشا (DIKSHA) ہُند تعارُفی صَفہٕ چھُ پلیٹ فارم وَنان یُس رِیاسَت گوڈَے چھِ اِستِعمال کَران، 36 زبانَن مَنٛز، نِشتھا (NISHTHA) اُستاد تٲلیٖم پروگرام پٲٹھۍ تہٕ نیشنل ڈیجیٹل ایجوکیشن آرکِٹؠکچر (National Digital Education Architecture)، 29 جُلایی 2021 شُروع، عمارَتی بلاک فریم پٲٹھۍ۔ راستہٕ چھُ وَفاقی: پرؠتھ رِیاسَت چھےٚ مُشتَرِکہِ پلیٹ فارمَس پؠٹھ پَنہٕ پنُن پروگرام چَلاونُک آزادی تھاوان۔ اَکھ قومی کورس شُمار چھُ نَہ ڈیزایِن۔

زبان۔ 29 جُلایی PIB کھاتہٕ چھُ بھاشِنی (BHASHINI)، نیشنل لینگویج ٹرانسلیشن مِشن (National Language Translation Mission)، ہِندوستٲنۍ زبان ٹؠکنالوجی خٲطرٕ عوٲمی ڈیجیٹل پلیٹ فارم وَنان۔ یہِ چھُ نَہ اتہِ اِستِعمال کَرنہٕ آمُت حِصَس مَنٛز اِستِعمال کَرن وٲل کُل شایع کَران۔ عزم چھُ پلیٹ فارمُک وجود بیٚیہِ خِدمَتَن خٲطرٕ شمولیت پرت پٲٹھۍ، ای گورننس انٹرفیسَن خٲطرٕ تِم خصوصیت سَمیت یِم MeitY ذکر کٔر۔

## پرؠتھ ڈیجیٹل وَتھَس پؠٹھ جِسمٲنی قَید

جوٗن 2026 تام وَنان سُہے PIB کھاتہٕ زِ 2.21 لَچھ گام پنچایت (gram panchayat) بھارت نیٹَس (BharatNet) تَحت خِدمَت خٲطرٕ تَیار ٲسِہ، ییلہِ حاضِری ہُند نُکتہٕ (points of presence) سَتھ رؠتہٕ دوران 80,472 گام پنچایتَن پؠٹھ چالوٗ ٲسِہ۔ بیٚیِس ؤریہِ تہٕ اَدھَس پؠٹھ چھُ UPI، ڈیجی لاکَر، دِکشا، یا ٹیلی ہیلتھ بہاوُک دیہی اِستِعمال وۄنۍ تہِ اَتھ پؠٹھ منحصر زِ تِمَن زٕ رَقَمَن مَنٛز کُس چھُ پَکان۔ کھاتہٕ چھُ نَہ چالوٗ حاضِری ہُند نُکتَن خٲطرٕ مُکَمَل گَژھنُک تٲریٖخ شایع کَران۔ یہِ چھُ فَرَق سرکٲری بَناوان۔ ڈیجیٹل ریلَن ہُند راستہٕ یُس یہِ نظر اَنداز کَرِ سُہ چھُ شَہَرَن تہٕ تِمَن لوٗکھَن ہُند راستہٕ یِمَن گوڈَے بیٚیُہ رَبط چھُ۔

کامن سروس سینٹر (Common Service Centre) چھِ اِنسٲنی حل ییتہِ رَبط یا مہارَت پَتَل آسہِ۔ جوٗن 2026 راجیہ سَبھا (Rajya Sabha) کَٹَن رِپورٹ کٔر 4,07,122 مرکز گام پنچایت سطحَس پؠٹھ چالوٗ۔ تِم چھِ درمیانی فراہمی نِظامُک حِصہٕ، نَہ وِراثَت یُس فرض کَرِتھ ہَٹاونہٕ یِیہِ۔

## کوتاہ جوڑُن چھُ نَہ

یِمَن دَستاوِزَن پؠٹھ یِم یہِ مُختصَر انحصار چھُ کَران چھُ نَہ سرکٲری 2030 ہَدف۔ ONDC آرڈَر قۭمَتُک شایع وَتھ چھُ نَہ، تِکیازِ لوک سَبھا جَوابَن آرڈَر شُمار شایع کٔر تہٕ رۄپے نَہ۔ UPI الگ پروڈَکٹَس مَنٛز شیڈول دُبارٕ لؠکھُن چھُ نَہ۔ اَگَست تہٕ سَتَمبَر 2023 ہُند جی ٹوَنٹی (G20) فریم ورک چھُ رضاکارانہ بیرونی وَصف روزان، نَہ ہِندوستٲنۍ نافذ کیلنڈر۔

شیڈول آیٹم چھِ نَوَمبَر 2026 تہٕ مئی 2027 ڈیٹا تحفُظ تٲریٖخ، تہٕ مارچ 2027 تام TEAM خرچ۔ درمیانی باقی چھُ آپریشن: بیٚیہِ سرکٲری شُمار شایع کٔرِو، تہٕ تِم فَرَق بَند کٔرِو یِم موجودٕ شُمار گوڈَے مانان چھِ۔
