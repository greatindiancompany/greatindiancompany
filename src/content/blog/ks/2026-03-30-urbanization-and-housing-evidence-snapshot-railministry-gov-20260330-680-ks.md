# شَہری رہٲیِشی عَدَد، پرٛٮ۪تھ اَکِہ تٲریٖخَس تہٕ دَستاویٖزَس سۭتۍ

## فِہرِست کِتھ کَن پرِن

بون پرٛٮ۪تھ عَدَد چھُ اَکِہ دَستاویٖزَس تہٕ تٲریٖخَس سۭتۍ جوڑِتھ۔ یَتہِ زٕ سرکٲری صَفہٕ اِختِلاف چھِ کَران، تِم زٕنہٕ چھِ ہاونہٕ یِوان تہٕ تٲریٖخ چھِ تھاونہٕ یِوان۔ یہِ صَفہٕ چھُنہٕ دَسَمبَر 2024 ہُنٛد ذخیرٕ قیاسَس سۭتۍ سَتَمبَر 2026 تام تازٕ کَران۔

## آبادی تہٕ قَصبہٕ

رہٲیِشی تہٕ شَہری مَعامِلاتَن ہِنٛزِ وَزارَتَھ (Ministry of Housing and Urban Affairs, MoHUA) ہُنٛد شَہری بَڑَن ہُنٛد صَفہٕ، مَردَم شُمٲری 2011 (Census 2011) 1 مارٕچ 2011 تام دُہراوان: کُل آبادی 1,210.2 ملین؛ شَہری آبادی لگ بگ 377.1 ملین؛ دَہَس منٛز خالص شَہری اِضافہٕ 91.0 ملین۔ جدول چھُ شَہری حِصہٕ 2011 منٛز 31.16 فیصَد تہٕ 2001 منٛز 27.81 فیصَد چھَپان۔ سُہے صَفَس پؠٹھ نَثر چھِ 31.6 فیصَد تہٕ 3.35 فیصَد نُقطَن ہُنٛد بَڑُن وَنان۔ شَہری دَہہ وار بَڑُن، 2001-2011، چھُ 31.8 فیصَد۔ نٔو قَصبہٕ: 2,774، یِمَن منٛز 242 قانوٗنی تہٕ 2,532 مَردَم شُمٲری قَصبہٕ (census towns) ٲسۍ۔ ملین-پُلس شَہر تہٕ شَہری اِجماع (urban agglomerations): 2001 منٛز 35، 2011 منٛز 53۔

گَریلوٗ مَعامِلاتَن ہِنٛزِ وَزارَتَھ (Ministry of Home Affairs) ہُنٛد 7 مے 2013 ہُنٛد لوک سَبھا (Lok Sabha) جَواب چھُ بغیر گول شَہری شُمار 1,210,569,573 آبادی منٛز 377,106,125 دِوان۔ سُہ جَوابَس پؠٹھ رِیاسَتی حِصہٕ چھِ رِیاسَتی نمونَن ہِنٛدِس بَریٖفَس منٛز، یَتہِ دُہراونہٕ آمٕتۍ نَہ۔

## گۄڈنیُک PMAY-شَہری، یِتھ کَن دَسَمبَر 2024 منٛز چھَپنہٕ آو

PIB پسِ مَنظَر نوٹ، دَسَمبَر 2024 فایِلَس منٛز تٲریٖخ وول دَستاویٖز: 4,618 شَہری مَقٲمی اِدارَن (urban local bodies) منٛز 1.18 کَروڑ گَر منظوٗر؛ 1.14 کَروڑ زَمیٖنَس پؠٹھ شُروٗع؛ 88.32 لاکھ مُکَمَل یا حَوالہٕ۔ منظوٗریَن منٛز عمودی حِصہٕ: فٲیدٕ گَژھَن وٲلٮ۪ن ہِنٛدٮ۪ن نِگرٲنی منٛز تعمیٖر (Beneficiary Led Construction) لگ بگ 75 لاکھ، شِراکَت تہٕ جایہِ پؠٹھ کَچھی بَستی ہُنٛد دُبارٕ اِنتِظام لگ بگ 18 لاکھ، قَرضَس سۭتۍ جوڑِتھ اِمدادی سِکیٖم (Credit Linked Subsidy Scheme) لگ بگ 25 لاکھ۔ سَرمایہٕ کاری ₹8,07,393 کَروڑ۔ مرکزی مَدَد منظوٗر لگ بگ ₹2 لاکھ کَروڑ۔ مرکزی مَدَد جٲری ₹1.66 لاکھ کَروڑ۔ لگ بگ 89 لاکھ گَر زنانَن ہِنٛدِس ناؤَس پؠٹھ یا مُشتَرِکہٕ مِلکیَتَس منٛز۔ اندازٕ سامان: 570 لاکھ میٹرک ٹَن سیٖمینٛٹ تہٕ 130 لاکھ میٹرک ٹَن سٹیل۔ اندازٕ کٲم: 917 کَروڑ نفر-دۄہ۔

PIB، 24 جُلاے 2025، سُہے بند اِمدادَس پؠٹھ: فی گَر ₹2.67 لاکھ تام؛ درمیٲنی آمدَنی وِنڈو گوٚو 31 مارٕچ 2021 خَتم؛ اِقتِصادی طور کمزوٗر تہٕ کَم آمدَنی وِنڈو گوٚو 31 مارٕچ 2022 خَتم؛ ₹58,885 کَروڑ جٲری 25,04,220 فٲیدٕ گَژھَن وٲلٮ۪ن، قومی رہٲیِشی بینٛک (National Housing Bank)، سٹیٹ بینٛک آف اِنڈیا (State Bank of India)، تہٕ رہٲیِشی تہٕ شَہری ترقی کارپوریشَن (Housing and Urban Development Corporation) ذَریعہٕ۔

## PMAY-شَہری 2.0

کَبینَٹ (Cabinet) منظوٗری 9 اَگَست 2024، دَسَمبَر 2024 نوٹَس منٛز۔ نفاذ 1 سَتَمبَر 2024 پؠٹھ، جُلاے 2025 جَوابَس منٛز، پانٛژھ ؤریَن منٛز 1 کَروڑ مَزیٖد فٲیدٕ گَژھَن وٲلٮ۪ن خٲطرٕ۔ حُکوٗمَتی مَدَد ₹2.30 لاکھ کَروڑ تہٕ ₹10 لاکھ کَروڑ سَرمایہٕ کاری، دَسَمبَر 2024 نوٹَس منٛز۔ عمودی حِصہٕ: فٲیدٕ گَژھَن وٲلٮ۪ن ہِنٛدٮ۪ن نِگرٲنی منٛز تعمیٖر، شِراکَتَس منٛز قٲبلِ بَرداش رہٲیِش (Affordable Housing in Partnership)، قٲبلِ بَرداش کرایَہ وٲل رہٲیِش (Affordable Rental Housing)، سودِ ہِنٛز اِمدادی سِکیٖم (Interest Subsidy Scheme)۔ گۄڈَنیِکٮ۪ن زٕن تحت فی یونِٹ مَدَد: ₹2.50 لاکھ۔ مرکزی حِصہٕ: شُمال-مشرِق، ہِماچَل پردیش، اُتَراکھَنڈ، جۆم تہٕ کٔشیٖر، پُڈوچیری، تہٕ دِلّی خٲطرٕ ₹2.25 لاکھ (90 فیصَد)؛ باقی مرکزی زیرِ اِنتِظام عَلاقَن خٲطرٕ ₹2.50 لاکھ (100 فیصَد)؛ باقی رِیاسَتَن خٲطرٕ ₹1.50 لاکھ (60 فیصَد)۔

سودِ ہِنٛز اِمدادی سِکیٖم، جُلاے 2025 جَواب: سَبھ کھۄتہٕ زِیادٕ ₹1.80 لاکھ؛ آمدَنی ₹9 لاکھ تام؛ قَرضہٕ ₹25 لاکھ تام؛ جائیداد ₹35 لاکھ تام؛ کارپِٹ رَقبہ (carpet area) 120 مُرَبَع میٖٹَر تام؛ گۄڈَنیِکِس ₹8 لاکھَس پؠٹھ 4 فیصَد؛ مُدت 12 ؤریَن تام؛ پانٛژھ ؤری وار قِستہٕ؛ قَرضہٕ چالو تہٕ اَصل رَقَمُک اَدھا کھۄتہٕ زِیادٕ باقی۔

صِرف دَسَمبَر 2024 نوٹ تام: 29 رِیاسَتَن تہٕ مرکزی زیرِ اِنتِظام عَلاقَن پؠٹھ یِمَن یادداشتَہ دستخط کٔرمٕتۍ ٲسۍ، 6 لاکھ کھۄتہٕ زِیادٕ گَرَن خٲطرٕ اُصولی منظوٗری۔ یَتھ صَفَس پؠٹھ چھُنہٕ کانٛہہ پَتِم قومی منظوٗری کُل اِستِمال۔

## AMRUT 2.0، 27 اکتوٗبَر 2021 ہُنٛد ہِدایَت جٲری

اِشارٲتی خرچ ₹2,99,000 کَروڑ، شٲمِل جٲری گۄڈنیُک مِشَنَس خٲطرٕ ₹22,000 کَروڑ، 2021-22 پؠٹھ 2025-26 تام۔ مرکزی حِصہٕ ₹86,760 کَروڑ، شٲمِل سُہ جٲری مِشَنَس خٲطرٕ ₹10,000 کَروڑ۔ 4,700 قانوٗنی قَصبَن (statutory towns) منٛز گَریلوٗ نَلکہٕ۔ 500 AMRUT شَہرَن منٛز سیوریج یا سیپٹیج۔ تجویٖز جوڑ: 2.68 کَروڑ نَلکہٕ تہٕ 2.64 کَروڑ سیوَر یا سیپٹیج جوڑ۔ جٲری نمونہٕ 20:40:40۔ ملین-پُلس شَہرَن خٲطرٕ فنڈَن ہُنٛد دَہ فیصَد عَوامی-نِجی شِراکَتَس (public-private partnership) منٛز۔ گۄڈنیُک AMRUT مِشَن (Atal Mission for Rejuvenation and Urban Transformation): جوٗن 2015، 500 شَہر۔

## ہوشیار شَہر

شُروٗع 25 جوٗن 2015، 100 شَہر، مرکزی خرچ ₹48,000 کَروڑ۔ اصلَن 31 مارٕچ 2020 تام منظوٗر، ژور لَٹہِ تَوسیع 31 مارٕچ 2025 تام۔ MoHUA مِشَن صَفہٕ، 1 دَسَمبَر 2025 تام جسمٲنی پیش رَفت: 8,064 مَنصوٗبَہ، قۭمَت ₹1,64,811 کَروڑ، وَرک آرڈَر سۭتۍ، 7,741 (96 فیصَد)، قۭمَت ₹1,55,386 کَروڑ، مُکَمَل۔ صَفہٕ چھُ مالی اِختِتامَس تام ₹47,458 کَروڑ دعوٕے تہٕ ₹46,093 کَروڑ اِستِمال تہِ ریکارڈ کَران، تَتہِ مرکزی حِصَس جٲری ہُنٛد 97 فیصَد وَننہٕ آمُت، تہٕ چھُ وَنان زِ کانٛہہ نٔو شَہر گَژھِہ نَہ رَٹنہٕ یِنِہ۔

لوک سَبھا غٲر ستارہ سَوال 3148، 6 اَگَست 2026 جَواب، 31 مارٕچ 2026 تام رپورٹ: 8,064، قۭمَت ₹1,64,811 کَروڑ، منٛز 7,790 مَنصوٗبَہ، قۭمَت ₹1,56,257 کَروڑ، مُکَمَل؛ مرکزی مَدَد اِستِمال ₹46,377 کَروڑ؛ 197 شِراکَت مَنصوٗبَہ، قۭمَت ₹9,277 کَروڑ، رَٹنہٕ، 194، قۭمَت ₹9,190 کَروڑ، مُکَمَل؛ تَمام 100 شَہرَن منٛز کمانڈ سینٹَر۔ مَشوَرٕ نَمبَر 27 چھُ 10 جوٗن 2025 تٲریٖخ وول۔

دَسَمبَر 2025 صَفہٕ تہٕ مارٕچ 2026 پارلیمانی جَھلَک چھِ زٕنہٕ سرکٲری۔ پَتِم مَعلوٗمات تٲریٖخ چھِ 31 مارٕچ 2026۔ کانٛہہ تہِ چھُنہٕ سَتَمبَر 2026 ہُنٛد شُمار۔

## اِرادَتَہ سۭتۍ ترنہٕ آمُت

2026 ہِنٛز کانٛہہ شَہری آبادی نَہ۔ قومی گَر قۭمَت اِنڈیکس نَہ۔ رِیاسَتی RERA (Real Estate (Regulation and Development) Act) رجِسٹَرَن پؠٹھ مَنصوٗبَن ہُنٛد شُمار نَہ۔ PLFS (Periodic Labour Force Survey) بے روزگٲری دَرَج نَہ۔ یہِ داواہ نَہ زِ پرٛٮ۪تھ رِیاسَتَن مارٕچ 2025 قٲبلِ بَرداش رہٲیِشی پالیسی (Affordable Housing Policy) آخِری تٲریٖخ پوٗرٕ کٔر۔ مَردَم شُمٲری 2027 ہُنٛد کَبینَٹ خرچ ₹11,718.24 کَروڑ، 25 اپریٖل 2026 ہِنٛدِس PIB نوٹَس پؠٹھ، چھُ مَردَم شُمٲری کارروائی ہِنٛز قۭمَت، فِہرِست کَرنہٕ آمُت تاکِہ سُہ رہٲیِشی سِکیٖمِ ہُنٛد خرچ نَہ ماننہٕ یِیہِ۔
