# کَتہِ چھُ کَوَریج وَنان تہِ سَمٲجی نٔتیٖجَس برٛونٛہہ رُکان

ہِندوستانَن ہِنٛد سَمٲجی پروگرام چھِ نِہ وۄنۍ زیٖادٕ تر خٲلی فِہرِستَن ہِنٛز کَتھ۔ رُکاوَٹ چھِ اَکھ رَجسٹر مُستَفیٖد، یا اَکھ سکوٗل یَتہِ اُستاد چھُ، تہٕ سُہ نٔتیٖجہ درمیان یُس بدلاونَس خٲطرٕ پروگرام چھُ۔ بۄنہٕ کَنہِ سرکٲری نمبر چھِ رُکاوَٹ جایہِ۔ تِم چھِ نَہ رِیاسَتی صلاحیتُک درجہ۔

## پۄشتہِ: نِگرٲنی شُدہ شُر، بدلنہٕ ورٲے وَزن

اَکھ پریس اِنفارمیشن بیورو (Press Information Bureau) نوٹ، یُس پوشَن ٹریکَر (Poshan Tracker) حیثیت مَے 2026 تام ریکارڈ چھُ کران، چھُ وَنان زِ پلیٹ فارمَس پؠٹھ اوس 7.7 کروڑ کھوتہٕ زِیادٕ شُرٮ۪ن خٲطرٕ اَکھ جِندہ پۄشتہِ ریکارڈ، تہٕ 0-5 وَرِیَن ہِنٛد 6.3 کروڑ کھوتہٕ زِیادٕ شُرٮ۪ن خٲطرٕ بَڑُن نِگرٲنی، سَتھ عمر بینڈَس مَنٛز رَجسٹر مُستَفیٖدَن ہُنٛد تَقریٖبَن 94%۔ 31 جُلایی 2026 تام سُہ وِزارَت ہُنٛد پوشَن ماہ (Poshan Maah) نوٹ اوس کھلۍ کائنات تَقریٖبَن 8.94 کروڑ مُستَفیٖدَن، تَقریٖبَن 13.99 لَکھ مَرکَزَن، تہٕ 13.30 لَکھ کٲم کرن والٮ۪ن پؠٹھ۔

NFHS-6، 2023-24 خٲطرٕ خانَدٲنی نٔتیٖجہ سَروے، چھُ وَنان تہِ قَد-عُمر کَمی (stunting) 29.3% (35.5% پؠٹھ)، وَزن-قَد کَمی (wasting) 19.0% (19.3% پؠٹھ)، کَم وَزن (underweight) 31.8% (32.1% پؠٹھ)، تہٕ 6-23 رٮ۪تَن پؠٹھ کٲفی غِذا 15.3% (11.0% پؠٹھ) لَبان۔ بَڑُن نِگرٲنی ہٮ۪کہِ رَجسٹر شُرٮ۪ن مَنٛز تَقریٖبَن عالمی آسِتھ، تہٕ وَزن-عُمر شرح ہٮ۪کہِ مُشکِل سان ہِلان۔ رُکاوَٹ چھِ نَہ یہِ زِ «اَسِہ چھُ نَہ زانان زِ شُر کُس چھُ۔» یہِ چھُ یِہ زِ ناپَس پَتہٕ کیاہ چھُ گَژھان: کھٮ۪ن، گَر نِنِہ وول راشَن، بؠمار، خانَدٲنی غِذا۔

زٕ ڈیزائن حقیقت چھِ سُہ رُکاوَٹ جایہِ تھاوان۔ زَنانن تہٕ شُرٮ۪ن ہِنٛز تَرقی ہِنٛز وِزارَت (Ministry of Women and Child Development) چھِ وَنان زِ شےٚ اَنٛگنواڑی (Anganwadi) خٔدمَتَن مَنٛز ترٛےٚ — ویکسین، صِحَت جانچ، تہٕ حَوالہ — قومی صِحَت مِشَن (National Health Mission) ذریعہٕ چھِ دِنہٕ یِوان، نَہ صِرف اَنٛگنواڑی مَحکمَہ سٕتی۔ اَکھ پۄشتہِ جائزہ یُس صِحَت سیشنَس سٕتی کُنہِ نَہ بِہِہ، سُہ چھُ پیکج ہُنٛد اَدھ رَوان۔ تہٕ سَکشَم بَڑاو چھُ 2 لَکھ مَرکَز، تَقریٖبَن 13.99 لَکھ سِٹاکَس مَنٛز۔ زیٖادٕ تر مَرکَز چھِ نَہ سَرمایَتی بَڑاو حِصَس مَنٛز۔

گَر نِنِہ وول راشَن خٲطرٕ چِہرَس پَہچان (facial recognition) تہٕ اَکھ ماسٹر ٹرینَر سلسلہ (پوشَن بِھی پَڑھائی بِھی (Poshan Bhi Padhai Bhi) تَل 18 مارچ 2026 تام 3,901 رِیاسَتی سطح ماسٹر ٹرینَر) چھِ نَفاذ اَوزار۔ اَکھ ٹرینَر شُمار چھُ نَہ ثُبوت زِ پرَٛتھ کٲم کرن وول ہٮ۪کہِ اَکھ دِنہٕ آمُت دۄہ چِہرَس پَہچان جٲری پوٗرٕ کٔرِتھ۔

## زَچِہ کورس یِم شروع چھِ گَژھان تہٕ پوٗرٕ چھِ نَہ گَژھان

اِدارَتی زَیِش چھِ 90.6%۔ کانٛہہ تہِ زَیِش برٛونٛہہ دیکھ بھال (antenatal care) چھِ 95.9%۔ ژور یا زِیادٕ دورٕ چھِ 65.2%۔ 100 دۄہ یا زِیادٕ خٲطرٕ لۄہ تہٕ فولِک تیزاب (iron and folic acid) چھُ 54.9%، تہٕ 180 دۄہ یا زِیادٕ خٲطرٕ 37.8%، 26.0% پؠٹھ۔ رُکاوَٹ چھُ زیٖٹھ کورس، یَمہِ پَتہٕ زَنان گوڈَے نظامَس سٕتی رَبطَس مَنٛز چھِ۔ عوٲمی سَہوٗلَتَس مَنٛز زَیِش 61.9% پؠٹھ 58.6% تام وَتھ، یِلہِ اِدارَتی زَیِش بَڑۍ، تَتھ کِنۍ بَڑنِہ واجِن فراہمی چھِ نَہ خودبخود اَکھ عوٲمی سَہوٗلَت کَتھ۔ 12-23 رٮ۪تَن پؠٹھ کارڈَن پؠٹھ پوٗرٕ ویکسین چھِ 87.1%، نَہ «کانٛہہ تہِ ویکسین» شرح یُس وِزارَت چھِ وَنان زِ 96% کھوتہٕ ہیوٚر چھُ روزان۔ خسرَک ویکسین ہُنٛد دویِم خوراک چھُ 71.8%، 58.6% پؠٹھ۔ رُکاوَٹ چھُ گوڈنِکِس رَبطَس پؠٹھ پوٗرٕ گَژھنَس کُن رۆو۔

## سکوٗل: تَناسُب چھُ نَہ بَچنِہ واجِن روک

UDISE+ 2025-26 چھُ شاگِرد-اُستاد تَناسُب 10، 12، 17، تہٕ 21 رپورٹ کران، قومی تَعلیٖمی پالِسی (National Education Policy) ہِنٛدِس 30:1 حَوالَس اندر، تہٕ 1,02,73,020 اُستاد۔ ثانَوی برقرار رکاوَٹ چھِ 51.9%۔ ثانَوی سکوٗل ترٛاوُن چھُ 7.0%۔ درمیٲنی سکوٗل ترٛاوُن سلسلہ چھُ 3.5% پؠٹھ 3.6% کُن گَژھان۔ اَکہِ اُستاد والؠ سکوٗل چھِ وَنان تہِ 1,00,843۔ کَمپیوٗٹَر چھِ سکوٗلَن ہِنٛدِس 69.9% مَنٛز۔ اَوسَط سکوٗل چھُ نَہ شایع تَناسُبَس پؠٹھ اُستادن ہُنٛد کَم۔ سُہ شُر یُس ثانَوی تام وَاتان چھُ نَہ یا تَتہِ روزان چھُ نَہ، تہٕ سُہ سکوٗل یَتہِ وَنان تہِ اَکھ اُستاد چھُ، چھِ نَفاذ بَچاو۔ یہِ مردم شُماری چھِ نَہ وَنان زِ تِم شُر ہٮ۪کن پرِتھ۔

## مَزدوٗری: قٲعدٕ چھُ ثُبوتَس برٛونٛہہ

نٔو دیہی مَزدوٗری ایکٹ چھُ پَندَہ دۄہَن ہِنٛز ادٲیگی ڈیوٹی، دیر معاوضہ، تہٕ اَکھ اِستِثنا لیکھان یَلہِ چِہرٕ تصدیٖق ناکام گَژھِہ۔ لوک سبھاہس (Lok Sabha) وَننہٕ آمُت گوڈنِک رٮ۪تُک نفر-دۄہ نمبر چھُ اَکھ پیداوار۔ یہِ چھُ نَہ ہاوان زِ اِستِثنا کوتاہَس لَٹہِ اِستعمال گۆو، یا معاوضہ کوتاہَس لَٹہِ دِنہٕ آو۔ سُہ خَلأ چھُ وُچھنِہ واجِن رُکاوَٹ، تِکیازِ قانوٗنَن چھُ گوڈَے ناکٲمی مُقَرر کٔرمٕژ۔

گریٹ اِنڈین کَمپَنی (Great Indian Company) چھِ کَوَریج سُہ شرحَس کھوتہٕ الگ تھاوان یُس ہِلہِ نَہ، تاکہ اَکھ نَفاذ جائزَس اَکھ مخرج آسِہ۔ یِمَن رُکاوَٹَن آس پاس کٲم ہِنٛد طَریٖقہ چھِ [موقَع نوٹَس](/blog/social-sector-outcomes-opportunity-landscape-ministryofcoal-gov-20260330-660-ks) مَنٛز۔ فِہرِست چھِ [greatindiancompany.com/blog](https://greatindiancompany.com/blog) پؠٹھ۔
