# مائیکرو، لَکُٹ تہٕ درمیٲنۍ پیداوٲری آس پاس ادائیگی، شناخت تہٕ کرؠڈِٹ قٲعدٕ

اَکھ زِیادٕ پیداوٲر شفٹ چھُ وُنہِ تِہ ناکام گَژھان اگر خریدار دیر سان ادا کَرِ یا بینک ادارہٕ وُچھِتھ ہٮ۪کَہِ نہٕ۔ ہِندوستانَس مَنٛز مائیکرو، لَکُٹ تہٕ درمیٲنۍ (Micro, Small and Medium Enterprises, MSME) پیداوٲری خٲطرٕ ریگولیٹری نظر چھُ زیادٕ تر اَکھ شناخت قٲعدہ، اَکھ ادائیگی گھڑی تہٕ اَکھ کرؠڈِٹ درجہٕ بندی قٲعدہ۔

## شناخت چھِ گوڈٕ یِوان

اُدیَم رجسٹریشن (Udyam Registration) چھُ مائیکرو، لَکُٹ تہٕ درمیٲنۍ ادارن خٲطرٕ وزارت ہُنٛد آن لایِن رجسٹر۔ تَس سرٹیفکیٹَس پؠٹھ درجہٕ چھُ 1 اپریٖل 2025 پؠٹھ لاگوٗ سرمایہٕ کاری تہٕ سالانہ کاروبار حدَن پتہٕ چَلان۔ اُدیَم اسِسٹ پلیٹ فارم (Udyam Assist Platform)، جَنؤری 2023 مَنٛز شروع، چھُ غیر رسمی مائیکرو ادارٕ کور کَران۔ ریزرو بینک (Reserve Bank) ہُنٛد ترجیحی شعبہٕ سوال جواب چھُ ونان زِ بینک، ترجیحی شعبہٕ درجہٕ بندی (priority-sector classification) خٲطرٕ، اُدیَم رجسٹریشن سرٹیفکیٹَس (Udyam Registration Certificate) یا اُدیَم اسِسٹ سرٹیفکیٹَس (Udyam Assist Certificate) پؠٹھ ریکارڈ درجہٕ پتہٕ چَلان چھِ۔

14 مے 2026 ہُنٛد پریس اِنفارمیشن بیورو (Press Information Bureau) نوٹ چھُ ونان زِ پرچون تہٕ تھوک تاجر چھِ جُلاے 2021 پؠٹھ اُدیَم رجسٹریشن تہٕ ترجیحی شعبہٕ قرضہٕ خٲطرٕ مائیکرو، لَکُٹ تہٕ درمیٲنۍ دائرَس اندر۔ اَتھے نوٹ چھُ ونان زِ یَلٕہ اَکھ ادارہٕ درجہٕ اَپر حرکت کَرِ، پرانَس درجَس ہِنٛد غیر ٹیکس فائدٕ چھِ ترٛے ؤری جٲری روزان۔

سرٹیفکیٹ بغیر چھُ پتہٕ اوزارَن کِہہ جوڑنَس خٲطرٕ نہٕ۔ سمادھان (Samadhaan) سوال جواب چھُ ونان زِ رجسٹریشن گَژھِ متنازعہ رسیدَس برٛونٛہہ آسُن۔ فائدٕ چھِ نہٕ پٔتمِس واقعَس پؠٹھ لاگوٗ۔

## ادائیگی گھڑی

ایم ایس ایم ای ڈی ایکٹ، 2006 (MSMED Act, 2006) ہُنٛد دفعہ 15 (Section 15) چھُ خریدارَس منگان زِ مائیکرو یا لَکُٹ سپلائرَس لِیکھِتھ متفقہ تاریخَس تام ادا کَرِ، تہٕ یہِ متفقہ مدت ہٮ۪کَہِ نہٕ قبول یا فرضی قبول ہِنٛدِس دۄہَس پؠٹھ 45 دۄہَن کھوتہٕ زِیادٕ آسِتھ۔ یَتہِ لِیکھِتھ معاہدٕ چھُ نہٕ، ادائیگی چھِ مقرر دۄہَس برٛونٛہہ واجب، یُس قبول یا فرضی قبولِ پؠٹھ پَنداہ دۄہَن پتہٕ بیٚیہِ دۄہ چھُ۔ دفعہ 16 (Section 16) چھُ مرکب سود ماہوار آرامَس سۭتۍ ریزرو بینکَن نوٹیفائی کَرنہٕ آمُتِس بینک ریٹَس (bank rate) نِش ترٛے گُنا تھاوان یَلٕہ خریدار یہِ مدت چھوڑِ۔

اِنکم ٹیکس ایکٹ (Income-tax Act) ہُنٛد دفعہ 43B(h) (Section 43B(h))، فنانس ایکٹ، 2023 (Finance Act, 2023) نِش شٲمِل تہٕ تشخیصی وَری 2024-25 (assessment year 2024-25) پؠٹھ لاگوٗ، چھُ خریدار ہِنٛز ٹیکس کٹوتی اَکِس رقمَس خٲطرٕ پَتہٕ کَران یُس مائیکرو یا لَکُٹ ادارَس واجب چھُ یَلٕہ یہِ رقم دفعہ 15 حدَس نیبر ادا گَژھِ۔ یہِ بریف چھُ نہٕ ٹیکس مشورٕ۔ ڈیزائن جوڑ چھُ کافی: ادائیگی قانوٗن تہٕ ٹیکس قانوٗن چھِ دۄشوَے اَتھے گھڑی کُن اِشارٕ کَران، تہٕ دۄشوَے چھِ مائیکرو تہٕ لَکُٹ سپلائرَن پؠٹھ لاگوٗ، پرٮ۪تھ درمیٲنۍ ادارَس پؠٹھ نہٕ۔

وزارت ہُنٛد سمادھان پورٹل، اَکتوٗبَر 2017 مَنٛز شروع، چھُ رجسٹر مائیکرو یا لَکُٹ ادارَس آن لایِن تاخیر سان ادائیگی درخواست دِنُک اِجازت دِوان۔ درخواست چھِ ریاست یا مرکزی علاقہٕ (Union Territory) ہِنٛدِس مائیکرو تہٕ لَکُٹ ادارن ہِنٛدِس سہولت کونسلَس (Micro and Small Enterprises Facilitation Council) کُن گَژھان۔ 14 مے 2026 ہُنٛد پریس اِنفارمیشن بیورو نوٹ چھُ ونان زِ 161 اَتھے کونسل چھِ قٲیِم۔ 26 جوٗن 2026 ہُنٛد نوٹ چھُ ونان زِ جوٗن 2026 تام ٲس پورٹلَس ₹55,244.29 کروڑ دعوٕہ وٲلۍ 2,56,892 درخواست مِلِمٕژ، تہٕ زِ کونسلَن ٲس 58,148 کیس تصفیہ کۆرمُت۔ جوٗن 2025 مَنٛز آو اَکھ آن لایِن تنازعہ حل پورٹل (Online Dispute Resolution portal) شروع کَرنہٕ۔

درخواست چھِ نہٕ فیصلہ۔ کونسل چھِ فیصلہٕ کَران۔ وزارتَس پنُن سمادھان سوال جواب چھُ ونان زِ وزارت چھِ نہٕ تَس عدالتی کامَس مَنٛز مداخلت کَران۔

## کرؠڈِٹ درجہٕ بندی

مائیکرو، لَکُٹ تہٕ درمیٲنۍ ادارن خٲطرٕ تمام بینک قرضہٕ چھِ ریزرو بینک ہِنٛدِس 24 مارٕچ 2025 ہِنٛدِس ہدایتن تَل ترجیحی شعبہٕ (priority sector) قٲبِل، یِتھ 9 فَرؤری 2026 اپڈیٹ مائیکرو، لَکُٹ تہٕ درمیٲنۍ قرضہٕ ماسٹر ڈائریکشنَس (Master Direction) مَنٛز دُبارٕ وننہٕ آو۔ ٹریڈ ریسیویبلز ڈسکاؤنٹنگ سسٹم (Trade Receivables Discounting System) ذریعہ فیکٹرنگ چھُ ترجیحی شعبہٕ خٲطرٕ اہل۔ 14 مے 2026 ہُنٛد نوٹ چھُ ونان زِ بجٹ 2026-27 (Budget 2026-27) چھُ مرکزی پبلک سیکٹر ادارن خٲطرٕ ٹی آر ای ڈی ایس (TReDS) لازِم کَران۔

سی جی ٹی ایم ایس ای (Credit Guarantee Fund Trust for Micro and Small Enterprises, CGTMSE)، سِڈبی (Small Industries Development Bank of India, SIDBI) سۭتۍ، چھُ 1 اپریٖل 2025 یا تَمہِ پتہٕ منظوٗر ضمانتن خٲطرٕ مائیکرو تہٕ لَکُٹ ادارن خٲطرٕ بغیرِ گرویِ کرؠڈِٹ ₹10 کروڑ تام ضمانت دِوان۔ ضمانت چھِ کرؠڈِٹ قٲعدہ۔ 45 دۄہَن ہِنٛز گھڑی چھِ نقد قٲعدہ۔ پیداوٲری چھِ کھاتَن مَنٛز ہاوان یَلٕہ دۄشوَے لاگوٗ آسَن۔
