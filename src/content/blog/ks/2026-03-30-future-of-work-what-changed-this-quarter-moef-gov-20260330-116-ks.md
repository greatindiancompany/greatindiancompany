# جُلٲے پؠٹھ ستمبر 2026 تام ہِندوستانَس منز کٲمِ ہِنٛدِس ڈیٹاہَس منز کیاہ حرکت کٔر

سو سِہِ ماہی یُس ستمبر 2026 تام ختم گژھان چھُ، تَتھ منز چھُ نَہ کانہہ نوٚو مزدور کوڈ۔ ژور کوڈ — اُجرتَن ہُند کوڈ، 2019 (Code on Wages, 2019)، صَنعتی تعلقاتَن ہُند کوڈ، 2020 (Industrial Relations Code, 2020)، سماجی تحفُظَس پؠٹھ کوڈ، 2020 (Code on Social Security, 2020)، تہٕ پیشَہٕ وَرٲے حفاظَت، صحت تہٕ کٲمِ ہِنٛدِ حالاتَن ہُند کوڈ، 2020 (Occupational Safety, Health and Working Conditions Code, 2020) — آیہِ 21 نومبر 2025 نافِذ کَرنہٕ۔ تِمَن 29 مرکزی مزدور قانوٗن مُنَظَم کٔرِ۔ سو اوس پٔتمِس وَریُک قانوٗن۔ یِم سِہِ ماہی ہُند سرکٲری حرکت چھُ غٲر مُنَظَم مزدورَن ہِنٛدِس رجسٹرَس منز۔

## دَرجُک پرنٹ

مزدوری تہٕ روزگارَچ وَزارَت (Ministry of Labour and Employment) کۆر 26 اگست 2021 ای-شرم پورٹل (e-Shram) آدھارَس سۭتۍ جوڈِتھ غٲر مُنَظَم مزدورَن ہُند قومی ڈیٹابیس (Aadhaar-seeded National Database of Unorganised Workers) کِس طورَس پؠٹھ شروٗع۔ دَرج چھُ خود اِقراری بُنیادَس پؠٹھ اَکھ یونیورسل اکاوٗنٹ نمبر (Universal Account Number) دِوان۔ پورٹلَس منز چھِ غٲر مُنَظَم مزدوٗر، تِمَن منز گِگ مزدوٗر (gig workers)، پلیٹ فارم مزدوٗر (platform workers)، تہٕ مہاجِر مزدوٗر (migrant workers)۔

لوک سبھا (Lok Sabha) ہُند اَکھ لِکھِتھ جواب، یُس پریس اِنفارمیشن بیورو (Press Information Bureau) ین 20 جُلٲے 2026 جٲری کوٚر، کۆر 14 جُلٲے 2026 تام دَرج 31.78 کروڑ کھۄتہٕ زِیادٕ۔ ضمیمُک صحیٖح کُل چھُ 31,78,31,749۔

پورٹلَس پانٛژِمِس ؤری پؠٹھ کۆر وَزارَتَن 20 اگست 2026 تام 31.89 کروڑ دَرج رپورٹ۔ زَنانَن ہُند حِصہٕ اوس 54.28 فیصد تہٕ مَردَن ہُند 45.72 فیصد۔ وَہَس حسابَس سۭتۍ ٲسۍ 0.08 فیصد 16–18، 55.21 فیصد 18–40، 24.18 فیصد 40–50، تہٕ 20.53 فیصد 50 کھۄتہٕ زِیادٕ۔

وَزارَتَن یِم گول کروڑ کُل شایع کٔرِ، تِمَن منز چھُ 31.89 کروڑ تہٕ 31.78 کروڑ درمیان فرق تَقریٖبَن 11 لاکھ دَرج۔ یِہ چھُ جٲری کٔرمٕتۍ شُمارَن پؠٹھ اَکھ اصلی اضافہٕ۔ یِہ چھُ تِہ اَتھ بُنیادَس پؠٹھ اَکھ لَکُٹ قدم یُس گوڈَے 31 کروڑ کھۄتہٕ زِیادٕ اوس۔ یِہ چھُ نَہ ثبوٗت زِ پانٛژَن ہفتَن منز تَقریٖبَن 11 لاکھ نوِ نوکری آیہِ بناونہٕ۔ جُلٲے ہُند شمار چھُ ضمیمُک صحیٖح گِنتی۔ اگستُک شمار چھُ وَزارَتُک گول کروڑ کُل، اَمہِ کِنۍ گَژھِ نَہ فرق صحیٖح نفری کِس طورَس پؠٹھ کوٹ کَرنہٕ۔ e-Shram چھُ یِہ خود اِقرار ریکارڈ کَران زِ اَکھ نفر چھُ غٲر مُنَظَم کٲم کَران۔ یِہ چھُ نَہ کانہہ نوکری، اُجرت، یا کٲمِ ہُند گھنٹہٕ ریکارڈ کَران۔

## وہِ جواب چھُ کیاہ ہاوان زِ رجسٹر کِس خٲطرٕ چھُ

20 جُلٲے 2026 ہُند جواب چھُ تِہ بیان کَران زِ ڈیٹابیس کَس سۭتۍ چھُ جوڑنہٕ یِوان۔ وَزارَتَن کۆر e-Shram 21 اکتوبر 2024 «اَکھ-جایہِ-حَل» (One-Stop-Solution) کِس طورَس پؠٹھ شروٗع، بجٹ 2024–25 ہِنٛدِس اعلانَس پتہٕ، تاکہِ دَرج گٔمِت مزدورَس اَکی جایہِ سماجی تحفُظَس سکیٖم وُچھِتھ ہٮ۪کَن۔ جوابَس تام ٲسۍ مرکزی وَزارَتَن ہِنٛدِ پَنداہ سکیٖم ضم یا نقشہٕ کَرنہٕ آمِتۍ۔ نامزد مثالَن منز چھِ پی ایم سوانِدھی (PM SVANidhi)، پی ایم ایس بی وائی (PMSBY)، پی ایم جے جے بی وائی (PMJJBY)، وِکسِت بھارت دیہی روزگار ضمانت (Viksit Bharat rural employment guarantee)، پی ایم اے وائی-گرامین (PMAY-Gramin)، آیوشمان بھارت پی ایم-جے اے وائی (Ayushman Bharat PM-JAY)، پی ایم اے وائی-اَربن (PMAY-Urban)، پی ایم ایم ایس وائی (PMMSY)، پی ایم-کسان (PM-KISAN)، ون نیشن ون ریشن کارڈ (One Nation One Ration Card)، تہٕ پی ایم ایم وی وائی (PMMVY)۔

14 جُلٲے 2026 کۆر وَزارَتَن e-Shramَس پؠٹھ دَرج گٔمٕتِن منز سکیٖم دَرجَن ہِنٛدِ گِنتی رپورٹ، تِمَن منز ون نیشن ون ریشن کارڈَس تحت 24.46 کروڑ، آیوشمان بھارت پی ایم-جے اے وائی تحت 15.10 کروڑ، تہٕ PMSBY تحت 8.54 کروڑ۔ یہِ قطارٕ چھِ اَکھ دۄیِمَس سۭتۍ رَلان۔ اَکھ نفر ہٮ۪کہِ واریاہَن سکیٖمَن منز بیٚہِتھ۔ تِم جمع کَرُن چھُ نَہ یِم سِہِ ماہی منز نوِ حفاظت مِلن والٮ۪ن لوٗکَن ہُند شمار بناوان۔

وہِ جواب چھُ نوکری ہِنٛدِ خٲطرٕ قومی کیریئر سروس (National Career Service)، ہُنرَس خٲطرٕ سِکِل اِنڈیا ڈیجیٹل ہَب (Skill India Digital Hub)، تہٕ پینشنَس خٲطرٕ پردھان منتری شرم یوگی ماندھن (Pradhan Mantri Shram Yogi Maandhan) سۭتۍ جوڑ ریکارڈ کَران۔ رِیاسَتَن تہٕ مرکزی زیرِ اِنتظام علاقَن (Union Territory) ہِنٛدِ مایکرو سائٹ، یِم 29 جنوری 2025 شروٗع کَرنہٕ آیہِ، چھِ مقٲمی دَرج، اَپ ڈیٹ، تصدیٖق، تہٕ تجزیہٕ سنبھالنہٕ خٲطرٕ۔

## یِہ سِہِ ماہی کیاہ چھُ نَہ

زٕ غلط پَرُن چھِ آسان، تہٕ دۄشوَے چھِ غلط۔

گۄڈنیُک، اَکھ بۆڈ e-Shram شمار چھُ نَہ بے روزگٲری منز کمی۔ وَقتی مزدور طاقت سروے (Periodic Labour Force Survey, PLFS) چھُ مزدور طاقتَس منز شِرکت، مزدورَن ہُند آبادی تناسب، تہٕ بے روزگٲری خٲطرٕ گھریلو سروے۔ یَتن رِلیزَن منز استعمال کٔرمُت تازٕ پوٗرٕ معموٗلی حیثیتُک (usual status) وَری چھُ جُلٲے 2023–جوٗن 2024، ییلہٕ 15 تہٕ کھۄتہٕ زِیادٕ وَہَس والٮ۪ن لوٗکَن ہُند معموٗلی حیثیتُک بے روزگٲری شرح 3.2 فیصد اوس۔ سو سروے تہٕ یہِ رجسٹر چھِ الگ سوالَن جواب دِوان۔ یَتن کوٹ کٔرمٕتِس رِلیزَن منز آو نَہ دۄشوَے جُلٲے–ستمبر 2026 منز نوِ سرٕ ڈیزائن کَرنہٕ۔

دوٚیُم، یہِ سِہِ ماہی چھُ نَہ سو لمحہٕ ییلہٕ گِگ مزدورَن ایگریگیٹرَن (aggregator) ہِنٛدِ پونٛسہٕ سۭتۍ فائدٕ مِلُن شروٗع کۆر۔ سماجی تحفُظَس پؠٹھ کوڈ چھُ گِگ تہٕ پلیٹ فارم کٲم متعارف کَران تہٕ مخصوص ایگریگیٹرَن خٲطرٕ اَکھ حصہٕ بینڈ تھاوان، مگر کوڈ چھُ تِہ وَنان زِ سو تٲریخ یَتھ پؠٹھ ایگریگیٹرَن ہُند حصہٕ شروٗع گژھان چھُ، سو چھُ مرکزی حُکوٗمَت (Central Government) اِطلاع کَران۔ 21 نومبر 2025 ہُند پریس نوٹ اوس وَنان زِ منتقلی دوران موجودٕ مزدور ایکٹَن تہٕ تِمَن ہِنٛدِ قٲعدَن ہِنٛدِ متعلقہٕ دفعات چھِ جٲری روزان۔ اقتصادی سروے 2025–26 (Economic Survey 2025–26)، یُس 2026 ہِنٛدِس شروٗعاتَس منز پیش کَرنہٕ آو، کۆر ریکارڈ زِ کوڈ آیہِ اِطلاع کَرنہٕ تہٕ قٲعدٕ ٲسۍ اَمِہ پتہٕ کینٛہَن رؠتَن منز متوقع۔ یہِ صفحہٕ چھُ نَہ کانہہ پتُن قٲعدٕ جٲری گٔمِت مانان یوتہِ تام اوپرِ دِنہٕ آمٕتِس ذرایعن منز اَکھ گزٹ حوالہٕ آسہِ۔

## سِہِ ماہی کِتھۍ کَڑِو استعمال

اگر تُہۍ ہِندوستانَس چِ کٲمِ ہِنٛز کَتھ ہفتہٕ پَتہٕ ہفتہٕ پَرِو، تیلہِ کٔرِو جُلٲے–ستمبر 2026 اَکھ رجسٹری سِہِ ماہی فائل۔ کارآمد جملہٕ چھِ تنگ۔ غٲر مُنَظَم مزدورَن ہُند رجسٹر گو 26 اگست 2026 پانٛژِ ؤری پار۔ سو اوس 20 اگست 31.89 کروڑ پؠٹھ، 14 جُلٲے ہِنٛدِ 31.78 کروڑ کھۄتہٕ زِیادٕ۔ تَتھ پؠٹھ لوٗکَن ہُند اکثریت چھِ زَنانَن، تہٕ اکثریت چھِ 40 کھۄتہٕ کم۔ رجسٹر چھُ موجودٕ سکیٖمَن تہٕ کیریئر سروس تہٕ ہُنر پورٹلَن سۭتۍ نقشہٕ کَرنہٕ یِوان۔ یِہ چھُ نَہ نوکری ہُند رِلیز، تہٕ یِہ چھُ نَہ تِمَن مزدور کوڈ قٲعدَن ہُند بدل یِم 2026 ہِنٛدِس شروٗعاتَس تام تِہ کھُلہٕ نفاذُک سوال ٲسۍ۔
