# دَسَمبَر 2026 ٲخر برٛونٛہہ ہِندوستانٕکۍ ڈیجیٹل ریلَن پؠٹھ کوتاہ چیک کَرُن

24 سَتَمبَر 2026 پؠٹھ نَوَد دۄہ چھِ 23 دَسَمبَر 2026 تام چَلان۔ یہِ چھُ تِمَن دَستاوِزَن ہُند فِہرَست یِم گوڈَے شیڈولَس پؠٹھ موجود چھِ، جمع اَکھ قانوٗنی تٲریٖخ یُس کِھرِکۍ اندر چھُ یِوان۔ یہِ چھُ نَہ پیشین کَرنہٕ آمٕتۍ اعلانَن ہُند فِہرَست۔ اگر صَفہٕ اَپ ڈیٹ گٔیہِ نَہ، صحیح نوٹ چھُ "وۄنۍ تام شایع نَہ،" نَہ اندازٕ۔

## اَکھ تٲریٖخ وول قانوٗنی واقعہ

اِلؠکٹرونِکس تہٕ معلوماتِ ٹؠکنالوجی وَزارَت (Ministry of Electronics and Information Technology) نوٹیفِکیشن G.S.R. 843(E) چھُ 13 نَوَمبَر 2025 تٲریٖخ۔ یہِ چھُ "یَمہِ گزٹ ہُند اشاعت تٲریٖخ پؠٹھ اَکھ ؤری" ڈیجیٹل پَرسَنَل ڈیٹا پروٹؠکشن ایکٹ، 2023 (Digital Personal Data Protection Act, 2023) ہُند دفعہ 6(9) تہٕ دفعہ 27(1)(d) ہُند شُروع پٲٹھۍ مُقَرَر کَران۔ 13 نَوَمبَر 2025 پؠٹھ اَکھ ؤری چھُ 13 نَوَمبَر 2026 پؠٹھ یِوان، یَتھ کِھرِکۍ اندر۔

سَتھ تٲریٖخ برٛونٛہہ، تِم زٕ فقرے ایکٹَس مَنٛز پرِو۔ یہِ چیک لِسٹ چھُ نَہ تِمَن خُلاصہٕ کَران۔ تٲریٖخ پَتہٕ، چیک کٔرِو زِ MeitY کٔہہِ تہِ مزید نوٹس جٲری کٔر یُس شُروع بدلاوان چھُ۔ سُہے نوٹیفِکیشنَن گوڈَے پَتہٕ کِن تٲریٖخ مُقَرَر کٔر، اشاعت پؠٹھ اَٹھارَہ رؠتہٕ، دفعہ 3 پؠٹھ 5، دفعہ 6 ہُند ذیلی دفعہ (1) پؠٹھ (8) تہٕ (10)، دفعہ 7 پؠٹھ 17، تہٕ دفعہ 28 پؠٹھ 34 خٲطرٕ، بیٚیہِ تِم پروویژن یِم یہِ فِہرَست چھُ کَران یِمَن درمیان۔ سُہ پَتہٕ کِن تٲریٖخ چھُ مئی 2027 مَنٛز یِوان۔ یہِ چھُ نَہ 90 دۄہَن ہُند آیٹم۔ نَوَمبَر ہیڈ لایِن ایکٹَس متعلق تُلِو نَہ مُکَمَل شُروع غلط۔

تصدیٖق کٔرِو تہِ زِ ڈیٹا پروٹؠکشن بورڈ (Data Protection Board) نوٹیفِکیشن، 13 نَوَمبَر 2025 ہُند G.S.R. 844(E)، سُہ مَت چھُ یُس تُہۍ اِستِعمال چھِو کَران۔ اَتھ قایم کٔر بورڈ اشاعت تٲریٖخ پؠٹھ، اَمہِ ہُند ہیڈ آفِس قومی رازدٲنی عَلاقَس (National Capital Region) مَنٛز۔ بورڈ چھُ نَہ نَوَمبَر 2026 تخلیٖق۔ صِرِف اُوپر زٕ فقرے چھِ اَکھ ؤری آیٹم۔

ڈیجیٹل پَرسَنَل ڈیٹا پروٹؠکشن قاعِدے، 2025 (Digital Personal Data Protection Rules, 2025)، 14 نَوَمبَر 2025 شایع، تہٕ 16 دَسَمبَر 2025 ہُند تصحیح، چھِ سُہ قاعِدہ مَت یُس ایکٹَس نَذدیٖک تھاوُن چھُ۔ اگر بورڈ یَتھ 90 دۄہ کِھرِکۍ مَنٛز طریقہٕ کار یا فارم شایع کَرِ، یہِ تِمَن قاعِدَن مُقابِل فایِل کٔرِو نَہ خبر پیرافریز مُقابِل۔

## ادایگی: زٕ نٔو سَطر، اَگَستَس سۭتۍ مُقابَلہٕ

نیشنل پیمنٹس کارپوریشن آف اِنڈیا (National Payments Corporation of India, NPCI) ہُند یوٗنِفایڈ پیمنٹس اِنٹرفیس (Unified Payments Interface, UPI) پروڈَکٹ شُمار چھُ ماہانہ سیریز۔ یَتھ مُختصَرَس خٲطرٕ ہَتھَس مَنٛز ٲخری سَطر چھُ اَگَست 2026: 752 بینک چالوٗ، 24,508.96 مِلین لین دین، ₹29,82,355.95 کروڑ قۭمَت۔ سَتَمبَرُک سَطر چھُ 24 سَتَمبَر تام گۄڈنیُک غٲب رؠتہٕ۔ اَکتوٗبَرُک سَطر گَژھِہ تہِ دَسَمبَر ٲخر برٛونٛہہ یِتھ اگر صَفہٕ پَنہٕ معمول ماہانہ عادت تھاوِ۔ یہِ چیک لِسٹ چھُ نَہ اشاعت دۄہ وعدٕ کَران۔

ییلہِ سَطر یِیہِ، ترٛے مُقابَلہٕ ریکارڈ کٔرِو تہٕ سُہ کٲمؠنٹری نظر اَنداز کٔرِو یُس جَدوَل سَہار کٔرِتھ ہؠکِہ نَہ:

- حَجم اَگَست ہُند 24,508.96 مِلین مُقابِل، تہٕ پٔتمِس رؠتہٕ مُقابِل۔
- قۭمَت اَگَست ہُند ₹29,82,355.95 کروڑ مُقابِل۔ اَگَست اوس گوڈَے مئی (₹29,90,424.21 کروڑ) تہٕ جُلایی (₹29,87,880.49 کروڑ) کھۄتہٕ بۄن۔ اُوپر حَجم بۄن قۭمَت سۭتۍ چھُ ممکن نتیجہٕ۔ یہِ چھُ نَہ تضاد۔
- چالوٗ بینک، 752 مُقابِل۔ شُمار بَڑِیہِ پرؠتھ رؠتہٕ اپریل (713) پؠٹھ اَگَست تام۔ رُکُن آسہِ نٔو معلومات۔ بَڑُن آسہِ جاری رُجحان۔

ایپ مارکِٹ حصَہٕ جوڑِو نَہ یوت تام NPCI ہُند پنُن سَتھ رؠتہٕ ایپ سطح جَدوَل ذٔریع آسہِ نَہ۔ اتہِ اِستِعمال کَرنہٕ آمٕتۍ پروڈَکٹ شُمار سَطرَن مَنٛز چھُ نَہ یہِ۔

## رِضامَندی: سَہامَتی ڈیش بورڈَس پؠٹھ اَگَست، سَتَمبَر، تہٕ اَکتوٗبَر

سَہامَتی (Sahamati) ہُند عوٲمی ڈیش بورڈ چھُ اتہِ اِستِعمال کَرنہٕ آمُت پرنٹَس مَنٛز اَکھ رؠتہٕ پَتہٕ چَلان۔ ٲخری سَطر چھُ جُلایی 2026۔

| چیک | جُلایی 2026 رَقَم یُس ہارُن، مِلاوُن، یا وَضاحت کَرُن چھُ |
| --- | --- |
| رؠتہٕ مَنٛز جوڑِتھ کھاتہٕ | 11.68 مِلین |
| نٔو پُورٕ گَژھِتھ رِضامَندی | 44.84 مِلین |
| جوڑِتھ کھاتَن پؠٹھ دِنہٕ آمُت ڈیٹا | 343.78 مِلین |
| کُل جوڑِتھ کھاتہٕ | 326.30 مِلین |
| کُل پُورٕ گَژھِتھ رِضامَندی | 538.32 مِلین |

جُلایی ہُند رِضامَندی کالم اوس جوٗن ہُند 19.30 مِلین کھۄتہٕ کٲفی اُوپر۔ اَگَست سَطرَس خٲطرٕ گۄڈنیُک سُوال چھُ یہِ زِ 44.84 مِلین اَکھ رؠتہٕ چھَلان اوس۔ سَہامَتی ہُند تعریٖف گَژھِ نوٹَس پؠٹھ روزُن: پُورٕ گَژھِتھ رِضامَندی چھِ تِم دَرخاست یِمَن مُقابِل ڈیٹا اَکِس لَٹہِ دِنہٕ آمُت، نَہ پرؠتھ تُلنُک کوشِش۔ اگر کٔہہِ بٔڑ "اُٹھاونہٕ آمٕتۍ رِضامَندی" رَقَم الگ سَہامَتی صَفَس پؠٹھ کوٹ کَرِ، لؠکھِو زِ یہِ کُس سیریز چھُ۔ ہوم پیج تہٕ ڈیش بورڈَن چھِ الگ پیشکش اِستِعمال کَرمٕتۍ۔

دَسَمبَر ٲخر تام ہؠکِہ ڈیش بورڈ معقوٗل پٲٹھۍ کَم از کَم اَگَست تہٕ سَتَمبَر، تہٕ ممکنہ طور اَکتوٗبَر، ہاونُک توقع کَرنہٕ یِتھ۔ اگر یہِ نَہ ہاوِ، ؤنِو۔ فَرَق جُلایی رَن ریٹَس سۭتۍ بَرِو نَہ۔

## شِناخت تہٕ دَستاوِزات: صِرِف اگر نٔو کَٹ شایع گَژھِ

UIDAI ہُند سیچوریشن رِپورٹ یُس اتہِ اِستِعمال چھُ گَژھان چھُ 30 جوٗن 2026 تام: اندازٕ چالوٗ آدھار 1,34,76,52,046، چالوٗ رَسایِی 94.59 فیصد، ناگالینڈ (Nagaland) 62.41 فیصد پؠٹھ۔ یَتھ 90 دۄہ کِھرِکۍ مَنٛز، چیک کٔرِو زِ یونیک آئیڈؠنٹِفِکیشن اتھارٹی آف اِنڈیا (Unique Identification Authority of India, UIDAI) سُہ پی ڈی ایف بدلِیہِ۔ اگر بدلِیہِ، کانٛہہ رِیاسَت مُقابَلہٕ برٛونٛہہ نٔو "تام" تٲریٖخ ریکارڈ کٔرِو۔ اگر بدلِیہِ نَہ، جوٗن رِپورٹ چھُ رِیاسَتی جَدوَل روزان۔ چالوٗ ڈیش بورڈ کُل کوٹ کٔرِو نَہ سُہ دۄہ لؠکھنہٕ بغیر یَتھ تُہۍ یہِ وُچھِو۔

ڈیجی لاکَر (DigiLocker) ہُند ٲخری پارلیمانی کَٹ چھِ بھَرِتھ۔ لوک سَبھاہَن (Lok Sabha) 29 جُلایی 2026 وَن 71.66 کروڑ کھۄتہٕ زیٛادٕ دَرج شُدہ اِستِعمال کَرن وٲل تہٕ 936.03 کروڑ کھۄتہٕ زیٛادٕ دَستاوِزات دَستِیاب۔ راجیہ سَبھا (Rajya Sabha) بیانَن 31 جُلایی 2026 تام وَن 72.43 کروڑ کھۄتہٕ زیٛادٕ اِستِعمال کَرن وٲل، پٔتمہِ ترٛؠن ؤرین مَنٛز تَقریٖبَن 72.86 کروڑ دَستاوِز رَسایِی لین دین، تہٕ 5,437 دَستاوِز قِسم۔ یَتھ کِھرِکۍ مَنٛز نٔو پریس اِنفارمیشن بیورو (Press Information Bureau, PIB) یا پارلیمانی جَواب گَژھِ میٹرکَس سۭتۍ مِلاوُن، پرٛونِس سۭتۍ اوسط کَرنہٕ نَہ۔

اُمنگ (UMANG)، سَتھ راجیہ سَبھا بیانَس مَنٛز 31 جُلایی 2026 تام، ٲس 11.66 کروڑ کھۄتہٕ زیٛادٕ اِستِعمال کَرن وٲل، پٔتمہِ ترٛؠن ؤرین مَنٛز تَقریٖبَن 798 کروڑ لین دین، تہٕ 2,575 خِدمَت۔ اتہِ اِستِعمال کَرنہٕ آمُت موادَس مَنٛز چھُ نَہ ماہانہ اُمنگ شماریاتی کیلنڈر۔ یہِ چیک کٔرِو صِرِف اگر MeitY یا PIB تازٕ شُمار شایع کَرِ۔

## صحت تہٕ تجارت: ذخیرہٕ، نَہ ماہانہ ٹیپ

منجمد صحت اینکر چھُ 20 جُلایی 2026: 94.87 کروڑ اے بی ایچ اے (ABHA) شِناخت، 5.36 لَچھ سہوٗلیت، 10.09 لَچھ پیشہٕ ور، لوک سَبھا جَوابَس پؠٹھ یُس پریس اِنفارمیشن بیورو 24 جُلایی پوسٹ کٔر۔ وَزارَت وَنان زِ رِیاسَت تہٕ ضِلع تفصیٖل dashboard.abdm.gov.in پؠٹھ چھےٚ۔ اگر تُہۍ یَتھ کِھرِکۍ مَنٛز ڈیش بورڈ کھُلِو، پَننِس ریکارڈَس مَنٛز تٲریٖخ نوٹ کٔرِو۔ یہِ 20 جُلایی رَقَم وَنِو نَہ۔

اوپن نیٹ ورک فار ڈیجیٹل کامرس (Open Network for Digital Commerce, ONDC) ہُند پارلیمانی اینکر چھُ جُلایی 2026: 500 مِلین کھۄتہٕ زیٛادٕ کُل آرڈَر، منٹَس پؠٹھ 500 کھۄتہٕ زیٛادٕ آرڈَر، تَقریٖبَن 2 لَچھ یا زیٛادٕ چالوٗ پرچون سوداگَر، یِمَن مَنٛز تَقریٖبَن 60 فیصد لۄکُٹ فروش۔ اَکتوٗبَر دُبارٕ چھاپہٕ ہُند سرکٲری وعدٕ چھُ نَہ۔ ڈیجی دُکان (DigiDukaan) آو حیدَرآباد (Hyderabad) تہٕ جَے پوٗر (Jaipur) مَنٛز چالوٗ وَننہٕ۔ 90 دۄہ چیک چھُ یہِ زِ پَتہٕ کِن حُکوٗمَتی جَواب شَہَر جوڑان چھا یا آرڈَر کُل دُبارٕ وَنان چھا۔ نٔو جَواب ہُند غَیر موجودگی مَطلَب چھُ زِ جُلایی رَقَم وۄنۍ تہِ کھڑٕ چھِ، نَہ زِ نؠٹ ورک رُکِیہِ۔

## نشان لگاونُک لۄکُٹ فِہرَست

- ڈی پی ڈی پی ایکٹ دفعہ 6(9) تہٕ 27(1)(d) 13 نَوَمبَر 2026 برٛونٛہہ پرِو، تہٕ گزٹ تٲریٖخ چیک کٔرِو نَہ ثانوی "نَوَمبَرَس مَنٛز نافذ چھُ گَژھان" سَطر۔
- سَتَمبَر تہٕ اَکتوٗبَر UPI سَطر اَگَست ہُند حَجم، قۭمَت، تہٕ بینک شُمار مُقابِل لاگ کٔرِو۔
- بیٚیہِ سَہامَتی رؠتہٕ جُلایی ہُند 44.84 مِلین پُورٕ گَژھِتھ رِضامَندی مُقابِل لاگ کٔرِو، تہٕ کالم تعریٖف تھٲوِو۔
- 30 جوٗن آدھار سیچوریشن جَدوَل بدلِو صِرِف ییلہِ UIDAI نٔو تٲریٖخ لگاوِ۔
- ڈیجی لاکَر اِستِعمال کَرن وٲل، محفوظ دَستاوِزات، تہٕ ترٛؠن ؤرین رَسایِی ترٛے فیلڈ پٲٹھۍ تھٲوِو۔
- اے بی ڈی ایم (ABDM) ڈیش بورڈ کھُلِو صِرِف تٲریٖخ مُہر سۭتۍ، 20 جُلایی پارلیمانی کُلَن نَذدیٖک۔
