# اَکھ سرمایہ کار ہِندوستانَن ہٕندِ آب پروگرامن منز کیا انڈر رائٹ کٔرِتھ ہؠکِہ

انڈر رائٹَس لائق حقیقت تاریخ والہٕ عوامی بیلنس چھِ۔ تِم یِہ نظر چھِ سَمبھالان زِ سرمایہ کٲم کَتِہ منظور، ٹھیکہ دِنہٕ آمٕژ، جٲری، یا مُکمل چھے۔ تِم قومی آب ٹیرف، رعایت مدت، یا واپسی چھِ نَہ سَمبھالان۔ یَتِہ اَکھ دستاویز خاموش چھے، یِہ بَریٖف خاموش چھُ روزان۔

## یَس کُن تُہۍ اِشارٕ کٔرِتھ ہؠکِو

تُہۍ اَکِس دُبارٕ بیان کرنہٕ آمٕتِس دیہی پروگرامَس کُن اِشارٕ کٔرِتھ ہؠکِو۔ 10 مارچ 2026 تھٲو کابینہٕ ہَن جے جے ایم 2.0 (JJM 2.0) ہُند کُل خرچ ₹8.69 لاکھ کروڑ تہٕ مرکزی امداد ₹3.59 لاکھ کروڑ، دسمبر 2028 تام، تہٕ مُکمل گَژھنہٕ ہُند ٹیسٹ خٔدمَت کُن موڑ۔ اَکھ گرام پنچایت (Gram Panchayat) ہَر گھر جَل (Har Ghar Jal) صرف تَتھ پَتہٕ چھے اعلان کَران یَمہِ پَتہٕ گامَس اندر دیکھ بھال موجود آسِہ۔ ادائیگی الگ سِکیمَن پَتہٕ ایس این اے سپارش (SNA SPARSH) پؠٹھ پَکنِچ چھے۔

تُہۍ اَس نقدَس کُن اِشارٕ کٔرِتھ ہؠکِو یُس ہِلِیہِ، تہٕ وُچھِتھ ہؠکِو زِ سُہ حدَس سٕتۍ کِتھ کَن لَکُٹ چھُ۔ 6 اَگَست 2026 تام اوس مرکزَن مالی سال 2026-27 منز ₹6,154.96 کروڑ جٲری کۆرمُت۔ تَمہِ منز زیادٕ تر، ₹5,289.09 کروڑ، دَہ رِیاستَن معاوضہ اوس، منظور کٲمَس خٲطرٕ یۄس تِمَو گۄڈَے پَنان فنڈ کٔرمٕژ اوس۔ ₹865.87 کروڑ نٔو سِکیم سطح ہٕندِس رستَس ذٔریعہٕ 13 رِیاستَن منز گٔمُت اوس۔ یُس انڈر رائٹنگ میمو خرچ 2026 آرڈر کتاب چھُ سمجھان، سُہ یِہ چھَپ چھُ نَہ پَران۔

تُہۍ اَکِس شہری کتابَس کُن اِشارٕ کٔرِتھ ہؠکِو یُس ٹھیکہ رستَس پؠٹھ برٛونٛہہ چھُ۔ امروت 2.0 (AMRUT 2.0) ہٕندِ منظور منصوبَن، یِم لوک سبھا (Lok Sabha) وَننہٕ ٲمٕتۍ، لاگت ₹1,97,191.85 کروڑ چھے، چلاوُن تہٕ دیکھ بھال سَمیت، مرکزی حِصہ ₹66,638.59 کروڑ سٕتۍ۔ تَمہِ اندر سیوریج تہٕ سیپٹیج ₹71,133.01 کروڑ چھُ، 605 منصوبَن پؠٹھ، تہٕ آب رَسان ₹1,18,968.38 کروڑ چھُ، 3,572 پؠٹھ۔ 9 جُلَے 2026 تام ٹھیکہ دِنہٕ آمُت کیپیکس ₹1,37,130.13 کروڑ اوس تہٕ جسمانی طور مُکمل کٲم ₹66,472 کروڑ ٲسۍ۔ ٹھیکہ دِنہٕ آمُت مگر نہ مُکمل تعمیر بیلنس چھُ۔ سُہ ایکویٹی کہانی چھُ نَہ الا یہِ زَن اَکھ مخصوص ٹھیکہ یِہ وَنِہ۔

تُہۍ ٹریٹمنٹ گنجائش کُن اِشارٕ کٔرِتھ ہؠکِو یۄس ملین لِٹر فی دۄہ منز گَنِتھ چھے۔ جوٗن 2026 تام اوس نَمامی گنگے (Namami Gange) ہَن 6,610 ملین لِٹر فی دۄہ کتابَس منز 4,263 ملین لِٹر فی دۄہ مُکمل کۆرمُت، 219 منز 146 مُکمل منصوبَن پؠٹھ، ہٲوِتھ لاگت ₹35,970 کروڑ پؠٹھ۔ ڈیپارٹمنٹ آبِکِہ ذرایع، دریا ترقی تہٕ گنگا نَو حیات ([jalshakti-dowr.gov.in](https://jalshakti-dowr.gov.in/)) چھُ۔ 21 اپریل 2026 ہٕندِس نوٹَن ووٚن زِ مالی سال 2025-26 اضافَن منز ہائبرڈ اینوئٹی ماڈل (hybrid annuity model) تحت منصوبہ ٲسۍ۔ سُہ جملہ تُہۍ وَنن دِوان چھُ زِ اینوئٹی شکل کینٛہہ گنگا منصوبَن پؠٹھ استعمالَس منز چھے۔ سُہ تُہۍ وَنن چھُ نَہ دِوان زِ ماڈل 605 امروت سیوریج منصوبَن، یا دیہی سِکیمَہ، کور چھُ کَران۔

## یَس کُن تُہۍ اِشارٕ کٔرِتھ ہؠکِو نَہ

تُہۍ دیہی آبَس خٲطرٕ قومی گھریلو ٹیرف کُن اِشارٕ کٔرِتھ ہؠکِو نَہ۔ کابینہ نوٹ نظام چَلاوِتھ تھاونُک لاگت تِمَن بندوبستَن کُن چھُ دِوان یِم رِیاستَس بَناوُن چھِ، تہٕ پَتہٕ گرام پنچایت تہٕ گامُک آب تہٕ صفٲیی کمیٹی (Village Water and Sanitation Committee) کُن۔ سُہ اَس بندوبستَس قیمت چھُ نَہ دِوان۔

تُہۍ دسمبر 2028 تام 24×7 رَسان کُن اِشارٕ کٔرِتھ ہؠکِو نَہ۔ نوٹ عالمگیر جوڑ تہٕ تصدیق اَس تاریخَس پؠٹھ چھُ تھاوان، تہٕ یقینی دۄہس راتَس رَسان 2047 نظرِیَس اندر چھُ تھاوان۔

تُہۍ آئی آئی ایم بنگلور (IIM Bangalore) تہٕ آئی ایل او (ILO) ہُند 59.9 لاکھ براہ راست شخص-وَریَہ عدد لِسٹِڈ کمپنی خٲطرٕ ڈیمانڈ کِس طورَس پؠٹھ اِشارٕ کٔرِتھ ہؠکِو نَہ۔ سُہ پریس نوٹَس منز کوٹ کرنہٕ آمُت اندازہٕ چھُ، شخص-وَریَہ منز، تہٕ سُہ ممکنہ لیبل چھُ۔

تُہۍ جے جے ایم حد، امروت منظور لاگت، تہٕ نَمامی گنگے منصوبہ لاگت اَکِس «آب کیپیکس» منز جمع کٔرِتھ تہٕ سُہ مارکیٹ وَنِتھ ہؠکِو نَہ۔ تِم الگ حد، الگ وَریَہ، تہٕ مرکز تہٕ رِیاست ہٕندِ الگ حِصہ چھِ۔ 2021 امروت کابینہ عدد ₹2,77,000 کروڑ اشارتی خرچ دُبارٕ ژورم حد چھُ، منظور منصوبہ کُل ₹1,97,191.85 کروڑ کھوتہٕ پرٛون۔

## اَکھ میمو یُس چیکَس منز زِندٕ روزِ

بیلنس، تاریخ، تہٕ دستاویز وَنِو۔ وَنِو زِ روپیہ حد چھُ، جٲری چھُ، ٹھیکہ چھُ، یا مُکمل کٲم چھُ۔ ٹھیکہ شکل صرف تَتِہ ناو دِیو یَتِہ نوٹ سُہ ناو چھُ دِوان، یِتھ کَن تازہ گنگا کتاب ہٕندِس حصَس پؠٹھ ہائبرڈ اینوئٹی۔ ٹھہریو یَتِہ صفحہ ٹھہران چھُ۔

دستاویز [کابینہ نوٹ](https://www.pib.gov.in/PressReleasePage.aspx?PRID=2237548)، [6 اَگَست 2026 جٲری](https://www.pib.gov.in/PressReleasePage.aspx?PRID=2295509)، [امروت جواب](https://sansad.in/getFile/lsapps/loksabhaquestions/annex/188/AU3022_tT0QXq.pdf)، [نَمامی گنگے جواب](https://sansad.in/getFile/lsapps/loksabhaquestions/annex/188/AU3190_Ip1l3X.pdf)، تہٕ [مالی سال 2025-26 این ایم سی جی نوٹ](https://www.pib.gov.in/PressReleasePage.aspx?PRID=2254266) چھِ۔
