# مرکز، رِیاسَتہِ، تہٕ شَہر شَہری رہٲیِش کِتھ کَن وَنڈان چھِ

## اَکھ لفظ، ترٛےٚ حُکوٗمَتہٕ

ہِندوستٲنی فایِلَس منٛز «شَہری رہٲیِش» چھُ اکثر اَکھ مرکزی سِکیٖم، اَکھ رِیاسَتی قٲعدٕ، تہٕ اَکھ شَہری مَنصوٗبہٕ یِتھ کَن بیان زِ تِم اَکھ فیصلہٕ ٲسۍ۔ تِم چھِ نَہ۔ اَکھ پالیسی ٹیم یۄس ترٛےٚنہٕ خٲطرٕ اَکھ مالِک شایع کَران چھِ، سۄ آسِہ غلط اَتھ طٔریٖقَس سۭتۍ یُس قانوٗن ساز جانٛچِتھ ہیٚکِہ۔

## MoHUA کِیاہ تھاوان چھُ

وَزارَتَن کوٚر ہوشیار شَہرَن ہُنٛد مِشَن (Smart Cities Mission)، AMRUT (Atal Mission for Rejuvenation and Urban Transformation)، تہٕ شَہری عَلاقَن منٛز سَبَن خٲطرٕ رہٲیِش (Housing for All) 25 جوٗن 2015 شُروٗع۔ سۄ چھِ سِکیٖم ہِدایَت، مرکزی حِصہٕ، تہٕ قومی پیش رَفت بیان تھاوان۔ ہوشیار شَہرَن خٲطرٕ کوٚر تَمہِ اَعلٰی کَمیٹی چیئر، تہٕ اَمِہ ہُنٛد مِشَن صَفہٕ چھُ وۄنۍ 31 مارٕچ 2025 مالی اِختِتام، ₹48,000 کَروڑ خرچَس خلاف ₹47,458 کَروڑ دعوٕے، تہٕ یہِ فیصلہٕ ریکارڈ کَران زِ کانٛہہ نٔو شَہر گَژھِہ نَہ رَٹنہٕ یِنِہ۔ AMRUT 2.0 خٲطرٕ چھُ 27 اکتوٗبَر 2021 ہُنٛد پریٚس اِطلاع بیورو (Press Information Bureau, PIB) جٲری 12 اکتوٗبَر 2021 کَبینَٹ (Cabinet) منظوٗری تہٕ ₹2,99,000 کَروڑ اِشارٲتی خرچ ریکارڈ کَران، 20:40:40 قِستَن منٛز جٲری سۭتۍ۔ PMAY-شَہری 2.0 (Pradhan Mantri Awas Yojana – Urban 2.0) خٲطرٕ چھِ 9 اَگَست 2024 ہُنٛد کَبینَٹ فیصلہٕ تہٕ ₹2.30 لاکھ کَروڑ مَدَد مرکزی۔ سودِ ہِنٛزِ اِمدادی سِکیٖمِ (Interest Subsidy Scheme) ہِنٛزۍ حد آیہِ PIB ذَریعہٕ 24 جُلاے 2025 پارلیمانَس وَننہٕ۔

رہٲیِشی تہٕ شَہری مَعامِلاتَن ہِنٛز وَزارَت (Ministry of Housing and Urban Affairs, MoHUA) چھُ ریٖل اِسٹیٹ (ریگولیشَن اینڈ ڈیوَلَپمَنٹ) ایکٹ (Real Estate (Regulation and Development) Act) تہِ تھاوان، اِنڈیا کوڈَس (India Code) پؠٹھ چَلاون وٲل وَزارَت کِس پٲٹھۍ، 2016 ہُنٛز شُروٗع نوٹیفِکیشَن، 2017 ہُنٛد شُروٗع دَستاویٖز، تہٕ رِیاسَتی اتھارٹی ویب سٲیٹَن ہِنٛز عَوامی فِہرِست۔ سۄ چھِ نَہ مَنصوٗبہٕ رجِسٹَر تھاوان۔ 31 جُلاے 2026 ہُنٛد فورس میجَر (force majeure) مَشوَرٕ چھُ مرکزی دَستاویٖز زِ رِیاسَتی اتھارٹی رجِسٹریشَن کِتھ کَن تَوسیع دِتھ ہیٚکَن۔ تَوسیع حُکُم، اگر کانٛہہ آسِہ، چھُ وِنہِ تہِ اتھارٹی ہُنٛد۔

رجِسٹراڑ جَنرَل (Registrar General)، نَہ زَن MoHUA، چھُ آبادی ہُنٛد شُمار تھاوان۔ MoHUA ہُنٛد شَہری بَڑَن ہُنٛد صَفہٕ چھُ مَردَم شُمٲری 2011 (Census 2011) دُبارٕ شایع کَران۔ مَردَم شُمٲری 2027 ہُنٛد کیلنڈَر تہٕ ₹11,718.24 کَروڑ عملی خرچ چھِ مَردَم شُمٲری مشقِ ہِنٛدۍ PIB جٲری۔ رہٲیِشی وَزارَتی ڈیش بورڈَس گَژھِہ نَہ پرژھنہٕ یِنِہ زِ درمیان شَہری حِصہٕ گَژھِ۔

## رِیاسَت کِیاہ تھاوان چھِ

PMAY-شَہری 2.0 چھُ یادداشت، مانٛگ سَروے، تہٕ قٲبلِ بَرداش رہٲیِشی پالیسی (Affordable Housing Policy) لازم کَران۔ دَسَمبَر 2024 نوٹ ہیٚکہِ اُصولی گَر صِرف یِمَن رِیاسَتَن تہٕ مرکزی زیرِ اِنتِظام عَلاقَن خٲطرٕ گَنِتھ یِمَن دستخط کۆرمُت اوس۔ فی یونِٹ مرکزی مَدَد چھِ پَتہٕ بدلِتھ: سُہ نوٹَس منٛز ناو دِنہٕ آمٕتِس شُمال-مشرِقَس تہٕ پہاڑی تہٕ مَخصوٗص عَلاقَن منٛز ₹2.25 لاکھ، باقی مرکزی زیرِ اِنتِظام عَلاقَن منٛز ₹2.50 لاکھ، باقی رِیاسَتَن منٛز ₹1.50 لاکھ۔ اَکھ رِیاسَتی بریفِنگ یۄس عام زمرٕ رِیاسَتَس منٛز ₹2.50 لاکھ مرکزی چیک کِس پٲٹھۍ حوالہٕ چھِ کَران، سۄ چھِ مرکزُک حِصہٕ فی یونِٹ اَکھ لاکھ بڑاوِتھ وَنان۔

AMRUT چھُ رِیاسَتَس پرژھان زِ شَہری منصوٗبَہ اَکِہ سٹیٹ واٹَر ایکشَن پلانَس (State Water Action Plan) منٛز جمع کَرِ تہٕ اِصلاح شَرط پوٗرٕ کَرِ، شٲمِل جائیداد ٹیکس تہٕ صٲرِف چارج، ترٛیِم ؤریہِ پؠٹھ بِنا رُکاوَٹ فنڈَن خٲطرٕ۔ دُبارٕ اِستِمال ہَدف چھِ رِیاسَتی سطحَس پؠٹھ لیکھنہٕ آمٕتۍ: صاف کَرنہٕ آمٕتِہ اِستِمال شُدٕ آبَس پؠٹھ کَم از کَم شَہرِ ہِنٛزِ آب مانٛگِ ہُنٛد 20 فیصَد تہٕ صنعتی آب مانٛگِ ہُنٛد 40 فیصَد۔

ہوشیار شَہرَن چھُ چیف سیکرٹری (chief secretary) تحت اَکھ ہائے پاوَرڈ سٹیئرِنگ کَمیٹی (High Powered Steering Committee) تھا۔ 10 جوٗن 2025 ہُنٛد مَشوَرٕ نَمبَر 27 چھُ سُہ کَمیٹی پرژھان زِ خاس مَقصَدٕچ کمپَنیَن (special purpose vehicles) ہُنٛد اَمہِ پَتہٕ کِردار فیصلہٕ کَرِ۔ مرکزی اِختِتامَس پَتہٕ چھُ باقی فیصلہٕ رِیاسَتی فیصلہٕ۔

RERA قٲعدٕ، اتھارٹی، تہٕ اپیل ٹریبیونَل (appellate tribunal) چھِ موزوٗن حُکوٗمَتَس سۭتۍ قٲیِم۔ زٕ رِیاسَتہِ ہیٚکَن اَکھ ایکٹ لاگوٗ کٔرِتھ تہٕ ہیٚچھَن وولَس الگ معاہدٕ فارم تہٕ الگ رجِسٹَر ڈیزایِن ہاوِتھ۔ پالیسی موازنہٕ گَژھِہ قٲعدٕ حوالہٕ کَرِتھ، نَہ زَن صِرف ایکٹُک لَکوٗٹ عنوان۔

## شَہر کِیاہ تھاوان چھُ

شَہری مَقٲمی اِدارٕ (urban local body) چھُ PMAY فٲیدٕ گَژھَن وٲلۍ شناخْت کَران تہٕ، رِیاسَتَس سۭتۍ، گَر زَمیٖنَس پؠٹھ شُروٗع کَران۔ گۄڈنیُک سِکیٖمُک دَسَمبَر 2024 نوٹَن جوڑ منظوٗری 4,618 اَتھ کٮ۪ن اِدارَن سۭتۍ۔ AMRUT ہُنٛد نَلکہٕ تہٕ سیوَر جوڑ چھُ شَہری خِدمَت، سِٹی واٹَر ایکشَن پلانَس (City Water Action Plan) منٛز بیان، بونَس پؠٹھ سِٹی واٹَر بیلَنس پلان (City Water Balance Plan) سۭتۍ۔ ہوشیار شَہرَن ہِنٛز خاس مَقصَدٕچ کمپنی کوٚر شَہری تجویٖز نافِذ؛ مَشوَرتی فورم اوس اشتراک اِدارٕ۔ اِنٹیگریٹِڈ کمانڈ سینٹَر چھِ وۄنۍ تَمام 100 چُنِمٕتٮ۪ن شَہرَن منٛز۔ تِم مفید تھاوُن چھُ رِیاسَتی ہِدایَت تحت شَہری چَلاونُک مسلہٕ۔

مَردَم شُمٲری گَر فِہرِست وِنڈو آیہِ پرٛٮ۪تھ رِیاسَتَن تہٕ مرکزی زیرِ اِنتِظام عَلاقَن اپریٖل پؠٹھ سَتَمبَر 2026 اندر چُننہٕ۔ شَہرِ ہِنٛدۍ رہٲیِشی حالَت عَدَد یِن سُہ مَردَم شُمٲری چینلَس ذَریعہٕ، نَہ زَن میوٗنِسِپَل بَجَٹ تَقریٖرَس ذَریعہٕ، یوتام رجِسٹراڑ جَنرَل تِم شایع کَرِ نَہ۔

## رپورٹِنگ قٲعدٕ یُس آڈِٹَس منٛز زِنٛدٕ روزِ

ژور سَطَر شایع کٔرِو تہٕ پانٛژِم رَد کٔرِو۔ مرکزی حِصہٕ منظوٗر تہٕ جٲری، سِکیٖم تہٕ تٲریٖخَس سۭتۍ۔ رِیاسَتی یادداشت تہٕ اِصلاح حالَت، دَستاویٖزَس سۭتۍ۔ شَہری مَنصوٗبہٕ یا فٲیدٕ گَژھَن وولَس ہِنٛز پیش رَفت، ضمیمہ تٲریٖخَس سۭتۍ، مِثالَن ہوشیار شَہرَن خٲطرٕ 31 مارٕچ 2026۔ رِیاسَتی RERA رجِسٹَرَس پؠٹھ قانوٗنی حالَت، رجِسٹریشَن نَمبَرَس سۭتۍ۔ پانٛژِم سَطَر، اَکھ رَلِتھ «شَہری تبدیٖلی سکور»، تَس چھُنہٕ کانٛہہ سرکٲری تعریٖف۔ گریٹ اِنڈین کَمپَنی (Great Indian Company) چھِ نَہ کانٛہہ شایع کَران۔
