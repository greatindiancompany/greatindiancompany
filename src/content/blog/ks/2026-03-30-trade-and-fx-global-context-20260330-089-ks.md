# ہِندوستانٕکۍ FTA بازار تہٕ تِمَن پَتہٕ خِدمَتو ہُند فاضِل

یِم دستاویز حُکوٗمَتھ تہٕ ریزَرو بَینک آف اِنڈیا (Reserve Bank of India, RBI) اصلَس مَنٛز شایع چھِ کَران، تِمَن مَنٛز ہِندوستانُک عالمی مقام چھُ دُنیا سۭتۍ مالُک خَسارَہ، مُتَّحِدہ رِیاسَتھ (United States) تہٕ یوٗرَپَس (Europe) پٮ۪ٹھ مرکوز خِدمَتو ہُند فاضِل، تہٕ ترجیحی تِجارَتی مُعاہِدَن (free trade agreement, FTA) ہُند بَڑان فِہرِست، یِمَن ہُند مالِ سطح کُلَس نِش وُنہِ مُتواضِع چھِ۔ یہِ چھُ بیرونی سیاق۔ اَتھ عالمی تِجارَتُک اندازَہ ضروٗری چھُ نہٕ۔

## پوٗرٕ گٔمُت ؤری یُس ہِندوستان بیرونِ مُلکِ تُلان چھُ

وَزارَتِ تِجارَت تہٕ صَنعَت (Ministry of Commerce and Industry) ہُند 15 اپریل 2026 ہُند شایع چھُ 2025-26 ہُند مال تہٕ خِدمَتو ہُند بَرآمَد US$860.09 billion تھاوان، US$825.26 billion پٮ۪ٹھ 4.22 per cent بَڑِتھ۔ مالِ بَرآمَد ٲس US$441.78 billion، 0.93 per cent بَڑِتھ۔ خِدمَتی بَرآمَد ٲس US$418.31 billion۔ خِدمَتو ہُند فاضِل ٲس US$213.89 billion۔ مالُک خَسارَہ ٲس US$333.19 billion۔ مُجمَل تِجارَتی تُلن ٲس −US$119.30 billion۔

18 اَگَست 2026 ہُند پریس اِنفارمیشن بیورو (Press Information Bureau, PIB) نوٹ چھُ مُجمَل بَرآمَد رِکارڈ US$863.1 billion وَنان، مالِ بَرآمَد US$441.8 billion سۭتۍ۔ مالُک اعداد چھُ اپریلُک کُل، گول کٔرِتھ۔ مُجمَل اعداد چھُ US$860.09 billion نِش الگ۔ یُس «رِکارڈ» کوٹ کَران چھُ، تٔمۍ گَژھِ اَگَستُک نوٹ کوٹ کَرُن تہٕ خَسارَہ تہٕ خِدمَتو ہُند فاضِل خٲطرٕ اپریلُک میز تھاوُن۔ یہِ بَریٖف چھُ دۄشوَے کَران، تہٕ یہِ چھُ تِم دۄن ترٛیِمہِ نَمبَرَس مَنٛز رَلاوان نہٕ۔

اَگَست کِس نوٹُک نُکتہٕ چھُ رِکارڈ سَطرَس کھۄتہٕ تنگ۔ یہِ چھُ دلیل دِوان زِ FTA شَریک چھِ تَتھ بَرآمَدی بُنیادَس اندر تِجارَتی طور اہم گٔمٕتۍ، تہٕ یہِ چھُ مالِ سطح شایع کَران۔

## مُعاہِدَن بورڈَس پٮ۪ٹھ کیاہ چھُ

2025-26 مَنٛز FTA شَریکَن کُن مالِ بَرآمَد، مِلیَن US dollar، وَزارَتِ تِجارَت تَمہِ PIB نوٹَس ذَریعہٕ:

| بازار | بَرآمَد |
| --- | --- |
| آسیان (ASEAN) | 38,416.33 |
| سافٹا (SAFTA) | 25,774.68 |
| مُتَّحِدہ عَرب اِمارَت (UAE) | 37,359.11 |
| مُتَّحِدہ بَرطانیہ (United Kingdom) | 13,444.19 |
| سِنگاپور (Singapore) | 11,863.66 |
| آسٹریلیا (Australia) | 7,284.17 |
| جاپان (Japan) | 6,039.50 |
| جۆمٲہی کوریا (Republic of Korea) | 6,012.03 |
| اومان (Oman) | 4,021.48 |

UAE ہُند اعداد چھُ میزَس مَنٛز سارِوٕے کھۄتہٕ بوڈ اَکہِ FTA مُلکُک بازار۔ نوٹ چھُ ہِندوستان-UAE جامع اِقتِصادی شِراکَت مُعاہِدٕ (Comprehensive Economic Partnership Agreement, CEPA) 1 مٔی 2022 تٲریخ دِوان تہٕ چھُ وَنان زِ دۄطرفہٕ تِجارَت گٔیہِ 2024-25 مَنٛز US$100.06 billion پار، 19.6 per cent بَڑِتھ۔ یہِ چھُ یہِ بیان کٔرِتھ مقصد تِہ درج کَران زِ تِجارَت 2032 تام US$200 billion دوگُن کَرُن۔ سو مقصد چھُ نوٹَس مَنٛز حُکوٗمَتھُک ہَدَف۔ یہِ چھُ نہٕ بَرآمَدُک نتیجہٕ۔

آسٹریلیا ہُند اِقتِصادی تَعاوُن تہٕ تِجارَت مُعاہِدٕ (Economic Cooperation and Trade Agreement, ECTA) چھُ 29 دَسَمبَر 2022 پٮ۪ٹھ نافِذ۔ نوٹ چھُ وَنان زِ دۄطرفہٕ تِجارَت وَوت 2024-25 مَنٛز US$24.1 billion، تہٕ زِ ہِندوستانُک آسٹریلیا کُن بَرآمَد گٔیہِ 2020-21 ہَس مَنٛز US$4 billion پٮ۪ٹھ 2025-26 ہَس مَنٛز US$7.28 billion۔ یہِ چھُ وَنان زِ آسٹریلیاہَن دِژ فَوری صفر ڈیوٹی رَسایِی 98.3 per cent ٹیرف لائنَن پٮ۪ٹھ، تہٕ زِ 2026 پٮ۪ٹھ چھِ ہِندوستٲنۍ بَرآمَد صفر ڈیوٹی خٲطرٕ اَہل۔ ہِندوستان-برطانیہ جامع اِقتِصادی تہٕ تِجارَت مُعاہِدٕ (Comprehensive Economic and Trade Agreement, CETA) چھُ تَمہِ نوٹَس مَنٛز یِتھۍ بیان کرنہٕ یِوان زِ یہِ چھُ تَقریٖبَن 99 per cent ہِندوستٲنۍ بَرآمَد برطانیہَس مَنٛز ڈیوٹی بغٲر داخل گَژھنَس اَہل بناوان۔ اومان CEPA چھُ یِتھۍ بیان کرنہٕ یِوان زِ یہِ چھُ اومان ہٕنٛدِس 98 per cent ٹیرف لائنَن پٮ۪ٹھ رعایت محیط کَران۔

ترجیحی رَسایِی چھِ تَمہِ استعمال گَژھان یَمہِ وَقتَس اصٕل ہُند سَرٹِفِکیٹ جاری سپدِ۔ نوٹ چھُ وَنان زِ یوٗرپی فری ٹریڈ ایسوسی ایشن (European Free Trade Association, EFTA) ہُند تِجارَت تہٕ اِقتِصادی شِراکَت مُعاہِدٕ (Trade and Economic Partnership Agreement, TEPA)، اَکتوٗبَر 2025 پٮ۪ٹھ کاربَند، اوس 7,885 سَرٹِفِکیٹ پٲدٕ کٔرمُت، تہٕ اومان CEPA، جوٗن 2026 مَنٛز نافِذ، اوس 783 پٲدٕ کٔرمُت۔ ہِندوستان-برطانیہ مُعاہِدٕ چھُ اصٕل ہُند خود اعلان، دَرآمَد کَرن والٮ۪س ہُند علم، تہٕ £1,000 کھۄتہٕ کم کھیپَن خٲطرٕ اصٕل ہَنٛدِس دستاویزَن ہُند معافی اجازت دِوان۔ یِم قاعِدٕ چھِ لَکٕٹِن کھیپَن تہٕ ای کامرس خٲطرٕ۔ تِم چھِ نِریَت دِشا (Niryat Disha) سۭتۍ بیٹھان، یُس بَرآمَدِ فَروغِ مشن (Export Promotion Mission) ہُند غَیر مالِیاتی کِھڑکی چھُ، یَتہِ مایکرو، لَکُٹ تہٕ درمیٲنی اِدارَن (micro, small and medium enterprises, MSME) بَرآمَد کَرن والٮ۪ن خٲطرٕ تعمیل تہٕ تِجارَتی میلے ہُند سَہارَتھ ناؤ چھُ۔

اپریل–اَگَست 2026 ہُند کَسٹَمز ڈیٹا چھُ یہِ جاےچ زِ FTA ہٕنٛز کَتھ تہٕ رؠتہٕ وار کَتھ چھَا مِلان۔ تَتھ ڈائریکٹوریٹ جَنرَل آف کَمرشَل اِنٹَلیجَنس اینڈ سَٹیٹِسٹِکس (Directorate General of Commercial Intelligence and Statistics, DGCIS) فوری اندازَس مَنٛز ٲس UAE مالِ بَرآمَد US$13,588.28 million، اَکھ ؤری برٛونٛہہ کھۄتہٕ 10.61 per cent کھۄتہٕ بون۔ مُعاہِدٕ ہیٚکہِ نافِذ آسِتھ، 2025-26 ہُند سطح ہیٚکہِ بوڈ آسِتھ، تہٕ اَگہِ پٲنٛژھ رؠتہٕ ہیٚکَن وُنہِ کھَسِتھ۔ سِنگاپور، یُس تِہ FTA شَریک چھُ، ٲس تِمَن پٲنٛژھ رؠتَن مَنٛز 96.56 per cent بَڑِتھ، US$9,497.67 million تام۔

## خِدمَت چھِ سو عالمی رَسید یُس مالُک بوڈ بِلُک اَکھ بوڈ حِصہٕ ادا چھُ کَران

اپریل–جوٗن 2026 مَنٛز کٔر RBI ہَن مالُک خَسارَہ US$86.1 billion تہٕ خالص خِدمَتی رَسید US$51.6 billion درج۔ ذٲتی مَنتَقِلی کٔر US$42.9 billion شٲمِل۔ جٲری کھاتَہٕ خَسارَہ اوس US$4.2 billion۔ ہِندوستانُک عالمی مالُک مقام تہٕ ہِندوستانُک عالمی خِدمَتو ہُند مقام چھِ اُلٹہٕ سمتَن کُن کَشان، تہٕ سہ ماہی میز چھُ خالص ہاوان۔

سافٹ ویَر سروے چھُ سارِوٕے کھۄتہٕ بٔڑِس خِدمَتی کٲمِچ جاے دِوان۔ 2025-26 مَنٛز گٔیہِ سافٹ ویَر بَرآمَد US$119.7 billion، 54.1 per cent، مُتَّحِدہ رِیاسَتھَس۔ یوٗرَپَن تُل US$70.3 billion، 31.8 per cent۔ مُتَّحِدہ بَرطانیہَن تُل US$34.0 billion۔ اَمریٖکی ڈالَرَن کٔر 72.6 per cent انوٲیِس۔ زَرمُبادَل الحاق شٲمِل کٔرِتھ ٲس سافٹ ویَر بَرآمَد US$239.3 billion۔ یہِ چھُ FTA مالِ فِہرِستَس نِش الگ عالمی نشان، یَتہِ سارِوٕے کھۄتہٕ بٔڑِ اَکہِ مُلکِ سطر چھِ پوٗرٕ ؤری خٲطرٕ UAE پٮ۪ٹھ US$37,359.11 million۔

سرمایہٕ ہُند بہاو چھُ ترٛیِم عالمی جوڑ۔ جوٗن سہ ماہیَس مَنٛز ٲس خالص سِدھی زَرمَبادَل سَرمایَہ کٲری (foreign direct investment, FDI) US$6.1 billion تہٕ پورٹ فولیو سَرمایَہ کٲری ٲس US$9.6 billion ہُند خالص نیرُن۔ پرٛونؠ شکلُک زَخیرٕ نوٹ چھُ، تَتھ شکلَس پٮ۪ٹھ، −US$3.9 billion ہُند سرمایہٕ کھاتہٕ ہاوان، تہٕ US$14.4 billion ہُند مُلٕہ بَندِی نۄقصان، یُس US$4.2 billion ہَس جٲری کھاتَہٕ خَسارَس کھۄتہٕ بوڈ چھُ۔ جُلایی ہَنٛدِس اِبتِدٲیی میزَن ہٲوِتھ پَتہٕ اَکہِ رؠتھَس مَنٛز غَیر مُقیم جمع US$33.5 billion۔ سو اندرُن چھُ اَگَس سہ ماہی شایعَس مَنٛز ترمیم پَتہٕ عالمی تصویرَس مَنٛز بیٹھان۔ تَتھ تام چھُ یہِ اَکھ اِبتِدٲیی خانہٕ، غَیر مَعمولی وزنَس سۭتۍ۔

## سیاق کِتھۍ تھاوُن

یَمہِ وَقتَس تُہۍ 2025-26 پوٗرٕ پٲٹھۍ ضروٗری آسہِ، US$860.09 billion تہٕ خَسارَہ میز استعمال کٔرِو۔ یَمہِ وَقتَس شَریک سطح تہٕ سَرٹِفِکیٹَن ہُند شمار ضروٗری آسہِ، اَگَستُک FTA نوٹ استعمال کٔرِو۔ یَمہِ وَقتَس خِدمَت تہٕ مَنتَقِلی مالُک خَسارَہ پُر کَران وُچھُن ضروٗری آسہِ، RBI سہ ماہی استعمال کٔرِو۔ یَمہِ وَقتَس سوال یہِ آسہِ زِ کُس زَرمُبادَل بازار ہِندوستانٕکۍ کَمپیوٗٹَر خِدمَت چھُ ہیٚوان، سافٹ ویَر سروے استعمال کٔرِو۔ یَمہِ وَقتَس یہِ زانُن ضروٗری آسہِ زِ پوٗرٕ گٔمِتِس ؤریَس مَنٛز مضبوط نظرِ یِوان شَریک چھَا وُنہِ مضبوط، DGCIS رؠتہٕ استعمال کٔرِو۔

اَتھ ڈھانچہِ سۭتۍ جوڑِتھ خطرٕ چھِ [تِجارَتی خَسارَس، زَخیرَن تہٕ سرمایہٕ ہُندِس بہاوَس مَنٛز کیاہ وُچھُن](/blog/trade-and-fx-risk-watch-20260330-079-ks) مَنٛز۔
