# خاندان چھِ وُنۍ تہِ ہِندوستانُک بینک کریڈِٹُک بٔڈِ حِصہٕ ہؠوان

خاندان چھِ وُنۍ تہِ قرض کھاتُک مرکز۔ تِم چھِ نَہ وُنۍ سُہ حِصہٕ یُس تیز گَژھان چھُ۔

مارچ 2026 ہِنٛدِس سالانہ بنیٲدی شماریٲتی ریٹرنَس (Basic Statistical Return) پؠٹھ وُدِیہِ خاندٲنی شعبَس (household sector) ہِنٛد قرضہ وَریَس مقابلہ (year on year) **14.3%**۔ خاندانَن اوس بینک کریڈِٹُک **58.6%** تہٕ تِمَو 2025-26 دوران جمع گژھَن والِس کریڈِٹُک لَگ بَگ ترٛے پانژھَم حِصہٕ۔ افراد پانَے ٲسۍ بینک کریڈِٹُک **47.8%**۔ انفرٲدی کریڈِٹَس اندر اوس زنانَن قرضہ ہؠن والَن ہُنٛد حِصہٕ **24.7%**، اَکھ وَریہِ برٛونٛہہ کِس **23.8%** پیٹھہٕ۔ یہِ ہی ریٹرن چھُ جنسَس لحاظِ افرادَن خٲطرٕ کریڈِٹُک اَکھ ضلع سطح سلسلہ شٲمِل کران۔ پریس نوٹ چھُ قومی حِصہٕ شایع کران، نَہ ضلع جدول۔

ذٲتی قرضہٕ (personal loans)، یِتِہ گَرٕ، تعلیٖم، گاڑِ، کریڈِٹ کارڈ، صارف پائیدار سامان، تہٕ بیٛاکھ ذٲتی قرضہ پٲٹھۍ بیان چھِ، وُدِیہِ **12.9%**۔ یہِ اوس کُل BSR کریڈِٹ وُدھُن **14.1%** کھوتہٕ کم۔ ریزرو بینک آف اِنڈیا (Reserve Bank of India, RBI) چھُ وَنان زِ ذٲتی قرضَو کینٛہہ وَریَن تام اَتھ برٛونٛہہ کُل کھاتہ برٛونٛہہ نِیو۔ تِہنٛد حِصہٕ اوس وُنۍ تہِ بۆڈ، **30.7%** پؠٹھ۔

## تازٕ شعبہٕ چھاپ چھُ اَکھ الگ ذٲتی قرضہ سلسلہ

ماہانہ شعبہٕ رپورٹ چھِ نَہ BSR۔ 31 جُلایی 2026 تام وُدُیو 41 اَہَم بینکَن ہُنٛد ذٲتی قرضہ حِصہ وَریَس مقابلہ **16.2%**، اَکھ وَریہِ برٛونٛہہ کِس 11.9% مقابلہ۔ یہِ اوس 31 مارچ 2026 پؠٹھ تہِ 16.2%، تَتھ وِزِ کِس 11.7% مقابلہ۔ تِمَن ژورَن مہینَن منٛز گۆو صَنعت (industry) 15.0% پیٹھہٕ 20.0% تہٕ خٔدمَت (services) 19.0% پیٹھہٕ 22.9%۔ خاندٲنی حِصَن پنُن رفتار تھاو، ییلہِ زن فرمَو تہٕ خٔدمَتَو سرخی غٲر خوراک شرح (non-food rate) 15.9% پیٹھہٕ 19.1% تام تان۔

جُلایی ہِنٛدِس ذٲتی قرض کھاتَس اندر رُوٗدِ گَرٕ تہٕ گاڑِ ہُنٛد قرضہ (housing and vehicle loans) دۄگنہِ ہندسَن منٛز۔ کریڈِٹ کارڈَس پؠٹھ باقی رقم (credit-card outstanding) تہٕ سۄنَہ زیوَرَن ہُنٛد مقابلَس منٛز قرضَو (loans against gold jewellery) رفتار کم کٔر۔ مارچَس منٛز ٲسۍ سۄنَہ زیوَرَن ہُنٛد قرضہ وُنۍ تہِ مضبوط بیان، تہٕ گَرٕ قرضہ مستقل۔ خوردہ مِلان ٹھنڈ گۆو بغیر ضمانت کارڈ سِرَس پؠٹھ تہٕ سۄنَہ پؠٹھ، تہٕ یہِ ٹھنڈ گۆو نَہ گَرٕ تہٕ گاڑِ پؠٹھ۔

کھُلَس مارچ کھاتُک قیمت آسان چکرَس سۭتۍ بدلاو۔ 9% کھوتہٕ کم سود وول قرضہ اوس مارچ 2026 ہِنٛدِس اَندَس تام باقی کریڈِٹُک **64.2%**، اَکھ وَریہِ برٛونٛہہ کِس **43.9%** مقابلہ۔ یہِ چھُ پوٗرٕ BSR کھاتہ، نَہ صرف خاندٲنی شرح۔

## قرض چھُ زِیادٕ، تہٕ وُنۍ تہِ زیٛادٕ تر غٲر گَرٕ خوردہ

جوٗن 2026 ہِنٛز مالیاتی استحکام رپورٹ (Financial Stability Report) چھِ خاندٲنی شعبہ قرض، بینک تہٕ غٲر بینک قرضَن نِش، سَتَمبَر 2025 ہِنٛدِس اَندَس تام **جی ڈی پی ہُنٛد 45.5%** (45.5% of GDP) تھاوان۔ رپورٹُک چارٹ عنوان چھُ تَتھ سطحَس نسبتاً کم بیان کران، تہٕ چھُ وَنان زِ وُدھُن چھُ زیٛادٕ تر غٲر گَرٕ خوردہ قرضَو (non-housing retail loans) نِش آمُت۔ تِم قرضہ ٲسۍ خاندٲنی باقی قرضَن ہُنٛد **58.4%**۔ استعمالَس خٲطرٕ ہؠونہٕ آمُت قرضہ اوس خاندٲنی قرضَن ہُنٛد لَگ بَگ اَدھ، مارچ 2026 ہِنٛدِس اَندَس تام **49.7%**۔ رپورٹ ہِنٛد زمرٕ چھِ استعمال (ذٲتی قرضہ، کارڈ، صارف پائیدار سامان، تہٕ اَمی پٲٹھۍ کریڈِٹ)، اثاثہ تخلیق (گَرٕ تہٕ گاڑِ ہُنٛد قرضہ)، تہٕ پیداوٲری قرضہ (انفرٲدی کٔشی تہٕ کاروبار قرضہ، تہٕ تعلیٖم) الگ کران۔

یہِ جی ڈی پی شرح چھُ نَہ بینک کریڈِٹُک 58.6% حِصہٕ۔ اَکھ چھُ خاندٲنی قرض قومی پیداوارَس مقابلہ ناپان، غٲر بینک قرضہ دِن وول شٲمِل۔ بیٛاکھ چھُ ناپان زِ بینکنگ نظامُک کوتاہ قرض کھاتہ خاندانَن سۭتۍ چھُ۔ دۄشوَے چھِ وَنان زِ خاندان چھُ اَہَم قرضہ ہؠن وول۔ صرف جی ڈی پی شرح چھِ لیوریج متعلق کَتھ کران۔

## بغیر ضمانت قرضَو پٔتمِن قٲعدَن پتہٕ رفتار کم کٔر، تہٕ خطرٕ گۆو نَہ ختم

یہِ ہی استحکام رپورٹ چھِ وَنان زِ بغیر ضمانت ذٲتی قرضَن (unsecured personal loans) ہِنٛز مانگ پٔتمِن وَریَن منٛز اَتھ حدَس تام مضبوط ٲس زِ زیادتی وُدھُنَس رُکاوَن خٲطرٕ ریگولیٹری قدم آیہِ۔ تَتھ حِصَس منٛز وُدھُن پتہٕ معتدل گۆو۔ قرضُک بغیر ضمانت روپ چھُ وُنۍ تہِ قرضہ ہؠن والِس کمزوری۔ دَہ بٔڈِ بینکَن ہُنٛد سروے، یَتھ منٛز 2025-26 ہِنٛدِس ترٛیِم سہ ماہی منٛز ذٲتی قرضَن ہُنٛد **89%** اوس، وُچھ 2023-24 ہِنٛدِس ترٛیِم سہ ماہی پیٹھہٕ نَو سہ ماہیَن تام نوِ قرضَن ہُنٛد آغاز (originations) تہٕ تازٕ غٲر کارکردہ کھاتہ (fresh non-performing accounts)۔ آغاز تہٕ خرابی دۄشوَے ٲسۍ ₹10 لَکھ کھوتہٕ کم آمدنی والَن قرضہ ہؠن والَن منٛز جمع۔ آغازَن منٛز ₹5 لَکھ کھوتہٕ کم بینڈُک حِصہٕ اوس کم گَژھان۔ پرائم تہٕ پرائم پلس درجہ (prime and prime-plus rating tiers) ٲسۍ آغازَن پؠٹھ غالِب۔

اَکِس خاندانَس خٲطرٕ چھُ عملی پَرُن «خوردہ چھُ پھُلان» کھوتہٕ تنگ۔ بینک کھاتُک بٔڈِ حِصہٕ چھُ وُنۍ تہِ خاندٲنی کریڈِٹ۔ تازٕ تیز گَژھُن چھُ صنعتی تہٕ خٔدمَت کریڈِٹ۔ کارڈ تہٕ سۄنَہ قرضَو اَکِس ورنہِ دۄگنہِ ہندسَن والِس ذٲتی قرض شرحَس اندر ٹھنڈ کٔر۔ تِم قرضہ ہؠن وول یِم نَوِ بغیر ضمانت قرضَن منٛز تہٕ خراب گَژھَن والَن قرضَن منٛز زیٛادٕ تر چھِ نظرِ یِوان، چھِ تَتھ کم آمدنی بینڈَس منٛز یُس سروے ناو چھُ دِوان۔

گریٹ اِنڈین کمپنی (Great Indian Company) چھِ یہِ فرق یِم وجہ شایع کران زِ بانیَن، قرضہ دِن والَن، تہٕ پالیسی پَرن والَن چھُ بار بار اَکھ خوردہ سرخی دِنہٕ یِوان۔ اَمی رپورٹَن ہُنٛد صنعتی تہٕ خٔدمَت پہلو چھُ [کیاہ بدلاو](/blog/credit-growth-what-changed-20260330-005-ks) منٛز۔ اَتھ وُدھُنَس ہیٚٹھ خراب قرضہ کِتھ کَنِ چھِ، سُہ چھُ [خطرٕ نوٹَس](/blog/credit-growth-risk-watch-20260330-075-ks) منٛز۔
