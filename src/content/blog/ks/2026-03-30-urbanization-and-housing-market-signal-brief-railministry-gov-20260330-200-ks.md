# شَہری رہٲیِشَس منٛز بازارُک اِشارٕ کِیاہ چھُ گَژھان

## اِشارٕ چھُ اَکھ قٲعدٕ یُس دویم فَریق قۭمَت دِتھ ہیٚکِہ

MoHUA چھُنہٕ یِمَن جٲری کَرنَن منٛز، یِم یہِ مُختصَر اِستِمال چھُ کَران، کانٛہہ قومی گَر قۭمَت اِنڈیکس شایع کَران۔ «قۭمَت مضبوٗط گَژھان» یا «مانٛگ واپس یِوان» ہُنٛد جملہٕ ہیٚکہِ نَہ تَتہِ جوڑِتھ۔ یِم اِشارٕ صَفَس پؠٹھ چھِ، تِم چھِ اہلیتِ ہِنٛزۍ حد، اَکھ بند اِمداد، تہٕ اَکھ رجِسٹریشَن قانوٗن۔

## سودِ ہِنٛز اِمداد چھِ قَرضہٕ دیٖن وولَس حد وَنان

پریٚس اِطلاع بیورو (Press Information Bureau, PIB) ہُنٛد 24 جُلاے 2025 ہُنٛد مَتن، یُس لوک سَبھا (Lok Sabha) ہُنٛد اَکھ جَواب رپورٹ چھُ کَران، چھُ وزیٖرِ اَعظَمَس ہِنٛدِس رہٲیِشی مَنصوٗبَس – شَہری 2.0 (Pradhan Mantri Awas Yojana – Urban 2.0, PMAY-Urban 2.0) تحت سودِ ہِنٛز اِمدادی سِکیٖم (Interest Subsidy Scheme, ISS) یِتھ کَن تھاوان۔ آمدَنی ₹9 لاکھ تام۔ گَر قَرضہٕ ₹25 لاکھ تام۔ جائیدادِ ہِنٛز قۭمَت ₹35 لاکھ تام۔ کارپِٹ رَقبہ (carpet area) 120 مُرَبَع میٖٹَر تام۔ گۄڈَنیِکِس ₹8 لاکھَس پؠٹھ 4 فیصَد اِمداد، 12 ؤریَن تام، پانٛژھ ؤری وار قِستَن منٛز، سَبھ کھۄتہٕ زِیادٕ ₹1.80 لاکھ۔ قَرضہٕ گَژھِہ چالو آسُن، تہٕ اَصل رَقَمُک اَدھا کھۄتہٕ زِیادٕ باقی آسُن، پرٛٮ۪تھ جٲری کَرنَس وِزِ۔

یہِ چھُ بازارُک حد۔ ₹25 لاکھ کھۄتہٕ زِیادٕ قَرضہٕ، ₹35 لاکھ کھۄتہٕ زِیادٕ جائیداد، یا 120 مُرَبَع میٖٹَر کھۄتہٕ زِیادٕ کارپِٹ رَقبہ چھُ اِمدادَس نٮ۪بَر، اگر گَریلوٗ آمدَنی اِہل تہِ آسِہ۔ اَکھ قَرضہٕ دیٖن وول یُس گۄڈنیُک دۄہ اِمداد کِتابَس منٛز تھاوان چھُ، تہٕ پَتہٕ وُچھان چھُ زِ اَصل رَقَم اَدھا نِشانَس کھۄتہٕ بون یِوان چھُ، سُہ چھُ لیکھنہٕ آمٕتِس جٲری قٲعدَس نٮ۪بَر۔ برٛونٛہِم قَرضَس سۭتۍ جوڑِتھ اِمدادی سِکیٖم (Credit Linked Subsidy Scheme, CLSS)، یۄس ₹2.67 لاکھ تام دِوان ٲس تہٕ درمیٲنی آمدَنی وٲلٮ۪ن قَرضہٕ رَٹن وٲلٮ۪ن خٲطرٕ 31 مارٕچ 2021 تہٕ اِقتِصادی طور کمزوٗر تہٕ کَم آمدَنی وٲلٮ۪ن قَرضہٕ رَٹن وٲلٮ۪ن خٲطرٕ 31 مارٕچ 2022 بند گٔیہِ، چھِ نَہ کانٛہہ پائپ لایِن۔ PIB چھُ ریکارڈ کَران زِ ₹58,885 کَروڑ گوٚو گوڈَے 25,04,220 فٲیدٕ گَژھَن وٲلٮ۪ن جٲری، سُہے بند سِکیٖمَس تحت۔

فٲیدٕ گَژھَن وٲلٮ۪ن ہِنٛدٮ۪ن نِگرٲنی منٛز تعمیٖر (Beneficiary Led Construction) تہٕ شِراکَتَس منٛز قٲبلِ بَرداش رہٲیِش (Affordable Housing in Partnership) تحت ₹2.50 لاکھ فی یونِٹ مَدَد، دَسَمبَر 2024 ہِنٛدِس PIB نوٹَس پؠٹھ، چھِ اَکھ الگ اَوزار۔ یہِ مُختصَر چھُنہٕ سُہ ₹1.80 لاکھَس پؠٹھ جوڑان، تِکیازِ وَزارَتَن چھُنہٕ کانٛہہ کَل کَرنہٕ آمُت مثال شایع کۆرمُت یُس اَکِہ گَریلوٗ خٲطرٕ یہِ زٕ اَکھ اَکھَس پؠٹھ تھاوان آسِہ۔

## RERA چھُ اِظہارُک اِشارٕ، نَہ زَن قۭمَتُک اِشارٕ

ریٖل اِسٹیٹ (ریگولیشَن اینڈ ڈیوَلَپمَنٹ) ایکٹ، 2016 (Real Estate (Regulation and Development) Act, 2016, RERA)، یِتھ کَن اِنڈیا کوڈ (India Code) سُہ ریکارڈ چھُ کَران، چھُ اَکھ ایکٹ سیکٹَر ریگولیٹ تہٕ فروغ دِوانَس، پلاٹ، اپارٹمَنٹ، یا عِمارَتِ ہِنٛز فروخت شفاف بناونَس، اَلاٹی (allottees) بچاونَس، تہٕ اَکھ اتھارٹی، اَکھ فیصلہ کَرن وول اَفسَر (adjudicating officer)، تہٕ اَکھ اپیل ٹریبیونَل (appellate tribunal) بناونَس۔ MoHUA چھُ ریکارڈ کَران زِ پارلیمانَن کوٚر سُہ مارٕچ 2016 منٛز منظوٗر تہٕ صَدرَن (President) دِیُت 25 مارٕچ 2016 منظوٗری۔

دَفہ 4 چھُ لازم کَران زِ اَلاٹیَن نِش وَقتاً فَوقتاً حٲصِل رَقَمَن ہُنٛد سَتَتھ فیصَد اَکِہ شیڈول بینٛکَس (scheduled bank) منٛز الگ کھاتَس منٛز جمع سپدِ تہٕ صِرف تعمیٖرِ ہِنٛزِ قۭمَتَس تہٕ زَمیٖنَس خٲطرٕ اِستِمال سپدِ۔ یہِ قٲعدٕ چھُ بدلاوان زِ ڈیوَلَپَر ہیٚچھَن وٲلٮ۪ن ہِنٛدۍ جمع کِتھ کَن اِستِمال کَرِتھ ہیٚکِہ۔ سُہ چھُنہٕ ہیٚچھَن وولَس وَنان زِ کارپِٹ رَقبَس ہِنٛز قۭمَت کِیاہ آسِہ۔ کارپِٹ رَقبہ چھُ سُہ قانوٗنی پیمانہٕ یُس ایکٹ معیٲری چھُ کَران۔ اَکھ فروخت یۄس وِنہِ تہِ سوٗپَر بِلٹ اَپ رَقبَس (super built-up area) پؠٹھ طے گَژھان چھِ، سۄ چھِ اِظہارُک مسلہٕ، تہٕ یہِ چھُ دَفہ 3 پؠٹھ 19 تحت رجِسٹریشَنُک نُکتہ۔ یہِ رجِسٹریشَن دَفہ ٲسۍ نَہ 1 مے 2016 ہِنٛزِ شُروٗع نوٹیفِکیشَنَس منٛز، تہٕ تِمَن چھُ 2017 ہُنٛد شُروٗع دَستاویٖز محیط، یُس MoHUA تھاوان چھُ۔

رِیاسَتی اتھارٹی، نَہ زَن اَکھ قومی ایکسچینج، چھِ مَنصوٗبَس ہُنٛد رجِسٹَر تھاوان۔ MoHUA چھُ رِیاسَتی تہٕ مرکزی زیرِ اِنتِظام عَلاقَن ہِنٛدِس اتھارٹی سٲیٹَن ہِنٛز فِہرِست شایع کَران۔ «قومی نَہ فروخت شُدٕ ذخیرٕ» ہُنٛد عَدَد چھُنہٕ تَتھ فِہرِستَس منٛز۔

## کِیاہ چھُنہٕ اِشارٕ

25 جوٗن 2015 ہُک شُروٗع دۄہُک زبان، شَہرَن ہِنٛدۍ درجہٕ، تہٕ ہوشیار شَہرَن ہُنٛد مُکَمَلی فیصَد چھِ پروگرام کھاتہٕ۔ تِم چھِ عَوامی مَنصوٗبَہ تہٕ شہری نِظام بیان کَران۔ تِم چھِ نَہ نِجی اپارٹمَنٹِ ہِنٛز لین دٮ۪ن ہِنٛز قۭمَت۔ ہوشیار شَہرَن ہِنٛدِس مِشَنَس (Smart Cities Mission) ہُنٛد مالی اِختِتام 31 مارٕچ 2025، نٔو شَہرَس خٲطرٕ مَزیٖد بَجَٹی پروویجَن ورٲے، چھُ میوٗنِسِپَل ٹیکنالوجی ٹھیکَن خٲطرٕ اَکھ رسدِ ہُنٛد اِشارٕ۔ سُہ چھُ سَرمایہٕ نوٹَس منٛز اِستِمال گَژھان، نَہ زَن یَتہِ۔

حد اَکِہ جِنٛدٕ شَرط نامَس خلاف پرِو۔ اگر شَرط نامہٕ آمدَنی، قَرضہٕ، جائیدادِ ہِنٛز قۭمَت، یا کارپِٹ رَقبَس نٮ۪بَر چھُ، تَیلِہ اِمداد چھِ نَہ پَیداوَرَس ہُنٛد حِصہٕ۔ اگر مَنصوٗبہٕ مُتعلِقہ رِیاسَتی RERA رجِسٹَرَس پؠٹھ چھُنہٕ، تَیلِہ الگ کھاتَس ہُنٛد قٲعدٕ چھُنہٕ وِنہِ تہِ اَتھ طٔریٖقَس سۭتۍ جوڑِتھ یُس ہیٚچھَن وول جانٛچِتھ ہیٚکِہ۔
