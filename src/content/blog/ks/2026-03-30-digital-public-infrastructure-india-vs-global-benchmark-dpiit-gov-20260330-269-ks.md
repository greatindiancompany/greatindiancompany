# ہِندوستانٕکۍ ریل جی ٹوَنٹی ہُندِ پنٛنِس تعریٖفَس مُقابِل

ڈیجیٹل عوٲمی بُنیاد (digital public infrastructure) ہُند مُلکی درجہٕ چھُ مُشتَرِکہِ سکور کارڈ، ہِوی تعریٖف، تہٕ اَکھ شایع کَرن وول یُس سکورَن پَتھ کھڑُن تیار آسہِ ضروٗرت۔ جی ٹوَنٹیَن (G20) کٔر نَہ سُہ سکور کارڈ۔ اَتھ کٔر اَکھ تعریٖف تہٕ اَکھ رضاکارانہ فریم ورک۔ یہِ چھُ واحد بینچ مارک یُس یہِ مُختصَر اِستِعمال کَرِ۔ تِم مُقابَلہٕ یِم ہِندوستان گۄڈنیُک اعلان چھِ کَران، یا یِم برازیل، ایسٹونیا، یا سِنگاپور مُقابِل نُکتہٕ چھِ دِوان، چھِ نَہ بۄن کَنہِ دَستاوِزَن مَنٛز۔

## بینچ مارک مَت

جی ٹوَنٹی نئی دِلّی لیڈَرز ڈِکلیریشن (G20 New Delhi Leaders' Declaration)، 9 سَتَمبَر 2023 منظور، چھُ ڈیجیٹل عوٲمی بُنیاد اَکھ بَدلان تصور تہٕ مُشتَرِکہِ ڈیجیٹل نِظامَن ہُند سیٹ وَنان۔ تِم نِظام چھِ عوٲمی تہٕ نجی دۄشوٕے شعبَن بَناوان تہٕ اِستِعمال کَران۔ تِم چھِ محفوظ تہٕ لَچکدار بُنیادَس پؠٹھ۔ تِم ہؠکَن کھُلِہ معیارات تہٕ مُشَخَصاتَن پؠٹھ، تہٕ کھُلِہ ذٔریع سافٹ ویئرَس پؠٹھ، بَنِتھ۔ تِم ہؠکَن سماجی پیٖمانَس پؠٹھ خِدمَت دِتھ۔

لیڈَرزَن خوش آمدید وَن G20 فریم ورک فار سِسٹَمز آف ڈیجیٹل پبلک اِنفراسٹرکچر (G20 Framework for Systems of Digital Public Infrastructure) ترقی، تعیناتی، تہٕ حکمرانی خٲطرٕ رضاکارانہ تہٕ تجویز کَرنہٕ آمُت فریم ورک پٲٹھۍ۔ تِمَن نوٹ کٔر تہِ ہِندوستانُک منصوٗبہٕ زِ ڈیجیٹل عوٲمی بُنیاد ہُند عالمی ذخیرہٕ بَنِہِ، یُس جی ٹوَنٹی رُکن تہٕ بیٚیہِ رضاکارانہ شیئر کَرَن۔ مثالَن ہُند ذخیرہٕ چھُ نَہ درجہٕ۔

ڈیجیٹل معیشت وَزیرَن ہُند ضمیمہ 19 اَگَست 2023 چھُ چَلانُک تفصیٖل جوڑان۔ یہِ وَنان زِ رُکنَن ہُند طٔریٖقہٕ چھِ الگ، زِ تصور ہؠکِہ نَہ اَکِس نمونَس تام محدوٗد آسِتھ، تہٕ زِ یہِ ہؠکہِ ڈھالِتھ تہٕ الگ ناوَن سۭتۍ حوالہٕ دِتھ۔ لَفٕظ "عوٲمی" ہؠکِہ عوٲمی فایِدہ تہٕ رَسایِی مَطلَب آسِتھ، عوٲمی اتھارٹی ہُند حکمرانی تہٕ نگرانی تَحت۔ پَتہٕ چھُ یہِ ترٛے کٲم فِہرَست کَران یِم مُشتَرِکہِ نِظام پُورٕ کٔرِتھ ہؠکَن:

- شِناخت، تکمیٖلی اعتماد خِدمَتَن سَمیت یِتھ کٔنۍ الیکٹرانک دستخط تہٕ تصدیٖق یِن وٲل اسناد۔
- ادایگی: لوٗکھَن، کاروبارَن، تہٕ حُکوٗمَتَن درمیان پونٛسہٕ ہُند آسان تہٕ فوری منتقلی۔
- رِضامَندی سۭتۍ ڈیٹا شریک کَرُن ییتہِ لاگو آسہِ، عوٲمی تہٕ نجی شعبَن درمیان، لاگو ڈیٹا حکمرانی فریم ورک ہٕنٛزِ حفاظَتَن سۭتۍ۔

منصِفانہ بینچ مارک چھُ پرژھان زِ مُلکَس چھِ کیاہ یِمَن ترٛؠن کٲمَن خٲطرٕ مُشتَرِکہِ، حکمرانی وٲل نِظام، نَہ یہِ زِ اَمہِ ہُند ماہانہ ادایگی حَجم بیٚیِس مُلکَس کھۄتہٕ زیٛادٕ چھا۔

## شِناخت

ہِندوستانُک شِناخت نِظام چھُ آدھار (Aadhaar)، یُس یونیک آئیڈؠنٹِفِکیشن اتھارٹی آف اِنڈیا (Unique Identification Authority of India) چھُ چَلان۔ پریس اِنفارمیشن بیورو (Press Information Bureau) ہُند 29 جُلایی 2026 کھاتہٕ چھُ یہِ بایومیٹرک تہٕ ڈیموگرافِک منفرد ڈیجیٹل شِناخت وَنان تہٕ وَنان زِ 144 کروڑ کھۄتہٕ زیٛادٕ آدھار شِناخت تَیار گٔیہِ۔ سُہے دۄہُک MeitY جَوابُک لوک سَبھا (Lok Sabha) ضمیمہ وَنان زِ اندراج گٔیہِ 2010-11 مَنٛز 0.42 کروڑ پؠٹھ جوٗن 2026 تام 144 کروڑ کھۄتہٕ زیٛادٕ۔

UIDAI ہُند پنُن سیچوریشن رِپورٹ 30 جوٗن 2026 تام چھُ زیٛادٕ محتاط دَستاوِز۔ یہِ چھُ 1,34,76,52,046 چالوٗ آدھار تفویض اندازٕ کَران 2026 ہُند پیشین کَرنہٕ آمُت آبادی 1,42,46,96,577 مُقابِل، یُس یہِ 94.59 فیصد چالوٗ رَسایِی وَنان چھُ۔ رِپورٹ چھُ اندازٕ موتَن خٲطرٕ رجِسٹرار جَنرَل (Registrar General) ہٕنٛزِ موت شرح اِستِعمال کَرِتھ ایڈجسٹ کَران، تہٕ یہِ چھُ ہولڈر آدھارَس پؠٹھ پَتَس حساب گَنان، یِم سَتھ رِیاسَتِ مَنٛز مقیم آسِتھ ہؠکَن نَہ۔ کینٛژن رِیاسَتَن مَنٛز 100 فیصد کھۄتہٕ اُوپر چالوٗ رَسایِی چھُ سَتھ طٔریٖقَس نتیجہٕ، نَہ سُہ دعوٲ زِ پرؠتھ رہائشی تہٕ پَتہٕ تہِ بیٚیُہ نفرَس نمبر چھُ۔

جی ٹوَنٹی کٲم مُقابِل چھُ ہِندوستانَس آبادی پیٖمانَس پؠٹھ مُشتَرِکہِ شِناخت نِظام، سرکٲری بَقیہ سۭتۍ: کینٛہہ رِیاسَت چھِ قومی فیصدَس نِش دوٗر۔ ناگالینڈ (Nagaland) ہُند چالوٗ رَسایِی سَتھ رِپورٹَس مَنٛز چھُ 62.41 فیصد۔ اروناچل پردیش (Arunachal Pradesh) چھُ 77.77 فیصد۔ اَکھ بینچ مارک یُس "ہِندوستانَس چھُ آدھار" پؠٹھ رُکِہ چھُ وجود ٹؠسٹ پاس کَران تہٕ شمولیت ٹؠسٹ ناکام، یُس سُہے شایع کَرن وول دَستاوِز چھُ کَران۔

الیکٹرانک دستخط چھِ جی ٹوَنٹی مَتن مَنٛز شِناختَس نَذدیٖک۔ 29 جُلایی PIB کھاتہٕ وَنان زِ ای سائن (eSign) فراہم کَرن وٲلَن کُل 155.56 کروڑ ای سائن جٲری کٔر، یِمَن مَنٛز 44.23 کروڑ سی ڈیکَن (CDAC) جٲری کٔر، تہٕ زِ پٔتمِس ؤریہِ مَنٛز آیہِ تَقریٖبَن 12.46 کروڑ جٲری کَرنہٕ۔ یہِ چھُ اعتماد خِدمَت سَتھ مَعنَس مَنٛز یُس وَزیرَن ہُند ضمیمہ چھُ وَنان۔ یہِ چھُ نَہ دویُم شِناخت نِظام۔

## ادایگی

یوٗنِفایڈ پیمنٹس اِنٹرفیس (Unified Payments Interface, UPI) چھُ ہِندوستانُک مُشتَرِکہِ ادایگی نِظام۔ نیشنل پیمنٹس کارپوریشن آف اِنڈیا (National Payments Corporation of India, NPCI) ہُند اَگَست 2026 پروڈَکٹ شُمار چھُ 752 بینک چالوٗ، 24,508.96 مِلین لین دین، تہٕ ₹29,82,355.95 کروڑ قۭمَت ہاوان۔ سیریز چھےٚ ماہانہ، عوٲمی، تہٕ تعریٖف شُدہ۔ اَگَست 2018 پؠٹھ چھےٚ سُہ لین دین نؠبر یِم اَکِس کھاتَس ڈؠبِٹ تہٕ کریڈِٹ چھِ کَران۔

حُکوٗمَتَن کٔر تہِ اَکھ بین الاقوٲمی مُقابَلہٕ، تہٕ یہِ گَژھِ یِتھ پٲٹھۍ لیبل لگاونہٕ۔ 29 جُلایی 2026 PIB کھاتہٕ وَنان زِ UPI چھُ ہِندوستانٕکۍ ڈیجیٹل ادایگی ہُند 85 فیصد تہٕ عالمی حقیقی وَقت ڈیجیٹل ادایگی ہُند تَقریٖبَن 50 فیصد چَلان، تہٕ زِ یہِ چھُ دُنیاہُک سَبہٕ کھۄتہٕ بٔڑ ڈیجیٹل ادایگی نِظام۔ تِم جُملہ چھِ حُکوٗمَتِہِ۔ اِطلاع چھےٚ پرن وٲلَن رِیاسَت تہٕ مرکزی زیرِ اِنتِظام عَلاقہ (Union Territory) شُمارَن خٲطرٕ NPCI کُن اِشارٕ کَران۔ یہِ چھےٚ نَہ "تَقریٖبَن 50 فیصد" فقرٕ پَتھ بین الاقوٲمی ورک شیٹ دُبارٕ چھاپان۔ اَکھ بینچ مارک نوٹ ہؠکِہ جُملہ رِپورٹ کٔرِتھ۔ یہِ ہؠکِہ نَہ سَتھ صَفَس پؠٹھ موادَس پؠٹھ اَمہِ آڈِٹ کٔرِتھ۔

یُہ گَریلوٗ آڈِٹ گَژھِتھ ہؠکِہ چھُ ادارَتی وسعت تہٕ حالیہ راستہ۔ UPI پؠٹھ چالوٗ بینک گٔیہِ پروڈَکٹ شُمار صَفَس پؠٹھ اپریل 2026 مَنٛز 713 پؠٹھ اَگَست 2026 مَنٛز 752 تام۔ اَگَستَس مَنٛز حَجم اوس اپریل، مئی، جوٗن، یا جُلایی کھۄتہٕ اُوپر۔ اَگَستَس مَنٛز قۭمَت ٲس نَہ سَتھ کِھرِکۍ ہُند اُچَھ۔ پیٖمانہٕ تہٕ ترکیب چھِ دۄشوٕے وُچھنہٕ یِوان۔ اَکھ "سَبہٕ کھۄتہٕ بٔڑ" لیبل چھُ دویُم حقیقت چھُپاوان۔

جی ٹوَنٹی ادایگی کٲم چھےٚ لوٗکھَن، کاروبارَن، تہٕ حُکوٗمَتَن درمیان "آسان تہٕ فوری منتقلی"۔ UPI ہُند شایع جَدوَل چھُ واریاہ بٔڑ شُمارَس مَنٛز منتقلی ہاوان۔ یہِ چھُ نَہ ہاوان زِ لۄکُٹ سوداگَر کوتاہ قۭمَت چھُ دِوان، یا حُکوٗمَتی ادایگی ہُند کوتاہ حصَہٕ UPI پؠٹھ چھُ پَکان۔ ڈایِرؠکٹ بِنِفِٹ ٹرانسفر (Direct Benefit Transfer, DBT) چھُ وابستہٕ عوٲمی پائپ۔ 29 جُلایی 2026 لوک سَبھا ضمیمہ وَنان زِ ₹52 لَچھ کروڑ کھۄتہٕ زیٛادٕ چھُ DBT ذریعہٕ قومی سطحَس پؠٹھ منتقل کَرنہٕ آمُت۔ DBT چھےٚ نَہ UPI سیریز۔

## رِضامَندی سۭتۍ ڈیٹا شریک کَرُن

یہِ چھےٚ سُہ کٲم ییتہِ سُست مُقابَلہٕ ہِندوستان اَکِس پروڈَکٹَس مَنٛز رَلاوان چھُ۔ دَستاوِز چھِ کَم از کَم ترٛے الگ طریقہٕ وَنان۔

ڈیجی لاکَر (DigiLocker) چھُ اصل جاری کَرن وٲلَن جٲری کَرنہٕ آمٕتۍ مستند دَستاوِزَن ہُند بَٹوہٕ۔ 31 جُلایی 2026 تام آو راجیہ سَبھا (Rajya Sabha) ہَس وَننہٕ زِ اَتھ ٲس 72.43 کروڑ کھۄتہٕ زیٛادٕ دَرج شُدہ اِستِعمال کَرن وٲل تہٕ 5,437 دَستاوِز قِسم یا خِدمَت۔ اے پی آی سیتو (API Setu)، 29 جُلایی PIB کھاتَس مَنٛز، ٲس 8,904 حُکوٗمَتی اے پی آی شایع تاکہِ محکمہٕ مُشتَرِکہِ فِہرَستَس ذریعہٕ شیئر کٔرِتھ ہؠکَن۔ کھاتہٕ جَمع کَرن وول (Account Aggregator) فریم ورک، ریزرو بینک ہُند 28 نَوَمبَر 2025 ہِدایَتن تَحت، چھُ مخصوص مالی معلومات خٲطرٕ رِضامَندی شے ۔ ایگریگیٹرَس گَژھِ نَہ سُہ معلومات تھاوُن، گَژھِ نَہ گاہَکَس ہُند لین دین چَلاوُن، تہٕ گَژھِ صریٖح رِضامَندی پؠٹھ عمل کَرُن۔ سَہامَتی (Sahamati) ہُند جُلایی 2026 عوٲمی ڈیش بورڈ چھُ 326.30 مِلین کُل جوڑِتھ کھاتہٕ تہٕ 538.32 مِلین کُل پُورٕ گَژھِتھ رِضامَندی ہاوان، رِضامَندی کالم یِتھ پٲٹھۍ تعریٖف زِ دَرخاست یِمَن مُقابِل ڈیٹا اَکِس لَٹہِ دِنہٕ آمُت۔

جی ٹوَنٹی سَطر چھےٚ "رِضامَندی سۭتۍ ڈیٹا شریک کَرُن ییتہِ تہِ لاگو آسہِ،" جمع لاگو فریم ورکَس تَحت حفاظَت۔ ہِندوستانُک لاگو فریم ورک چھُ منتقلی مَنٛز۔ ڈیجیٹل پَرسَنَل ڈیٹا پروٹؠکشن ایکٹ، 2023 (Digital Personal Data Protection Act, 2023) چھُ صِرِف تِمَن ٹُکرَن مَنٛز نافذ یِم حُکوٗمَتَن مُقَرَر کٔر۔ 13 نَوَمبَر 2025 ہُند گزٹ نوٹیفِکیشن کٔر ادارَتی پروویژن تہٕ ڈیٹا پروٹؠکشن بورڈ (Data Protection Board) نافذ، تہٕ عملی قانوٗنُک زیٛادٕ حِصَس خٲطرٕ پَتہٕ کِن تٲریٖخ مُقَرَر۔ اَکھ مُلک ہؠکِہ رِضامَندی بنیاد ریل چَلاوِتھ برٛونٛہہ پوٗرٕ قانوٗن آن گَژھِہ۔ یہِ چھُ ترتیب حقیقت، نَہ سُہ دعوٲ زِ قانوٗنی بینچ مارک گوڈَے مُکَمَل چھُ۔

## سنجیدٕ مُقابَلَس وۄنۍ تہِ کوتاہ ضروٗرت آسہِ

وَزیرَن پرٛژھ نِظامَن زِ محفوظ، سیکور، قابلِ اعتماد، جوابدہ، تہٕ شٲمِل آسَن، تہٕ اِنسٲنی حقوٗق، ذٲتی ڈیٹا، رازدٲری، تہٕ دانشورانہ مِلکیَتُک اِحتِرام کَرَن۔ ہِندوستانٕکۍ اشاعت چھِ سَتھ حکمرانی ہُند ٹُکرٕ ہاوان: کھاتہٕ جَمع کَرن وول پابَندی، آیوشمان بھارت ڈیجیٹل مِشن (Ayushman Bharat Digital Mission) سۭتۍ شٲمِل گَژھنہٕ برٛونٛہہ صحت نِظام حِفاظَتی جانچ، تہٕ ہوسٹِنٛگ تہٕ آڈِٹ عمل یِم MeitY ای گورننس پلیٹ فارمَن خٲطرٕ وَن۔ تِم چھِ نَہ اَکھ مُلکَن درمیان آڈِٹ ہاوان۔

تِم چھِ تہِ اَکھ جِسمٲنی انحصار ہاوان یُس جی ٹوَنٹی تعریٖف فرض چھُ کَران تہٕ ہِندوستان الگ چھُ ماپَن۔ جوٗن 2026 تام، 29 جُلایی PIB کھاتہٕ وَنان زِ 2.21 لَچھ گام پنچایت (gram panchayat) بھارت نیٹَس (BharatNet) تَحت خِدمَت خٲطرٕ تَیار کَرنہٕ آمٕتۍ ٲسِہ، ییلہِ بھارت نیٹ حاضِری ہُند نُکتہٕ (points of presence) سَتھ رؠتہٕ دوران 80,472 گام پنچایتَن پؠٹھ چالوٗ ٲسِہ۔ اَکھ ادایگی ریل ہؠکِہ قومی حَجم ریکارڈ پوسٹ کٔرِتھ تہٕ وۄنۍ تہِ پَتَل آسِتھ ییتہِ حاضِری ہُند نُکتہٕ بَند آسہِ۔

جی ٹوَنٹی مَت اِستِعمال کٔرِو کٲم تہٕ حفاظَتَن ہُند چیک لِسٹ پٲٹھۍ۔ سَتھ چیک لِسٹَس پؠٹھ چھُ ہِندوستانَس شِناخت، فوری ادایگی، تہٕ اَکھ کھۄتہٕ زیٛادٕ رِضامَندی یا دَستاوِز ریل خٲطرٕ پروڈکشن نِظام، سرکٲری شُمارَن سۭتۍ۔ سُہے چیک لِسٹَس پؠٹھ چھُ کوریج ناہَموار، اَکھ منایاومُت عالمی حصَہٕ چھُ جَدوَل بغیر حُکوٗمَتی جُملہ، تہٕ ڈیٹا تحفُظ قانوٗن چھُ وۄنۍ تہِ مرحلَن مَنٛز وَتان۔ یہِ چھُ تعریٖفَس سۭتۍ مُقابَلہٕ۔ یہِ چھُ نَہ میڈل جَدوَل۔
