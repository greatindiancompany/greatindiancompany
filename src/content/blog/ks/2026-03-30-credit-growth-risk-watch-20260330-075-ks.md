# خراب قرضہ چھِ کم۔ دباؤ ٹیسٹ چھُ واپسی ہِنٛز وَتھ ہاوان

تیز کریڈِٹ وُدھُن چھُ سَتھ سارِوے کھوتہٕ صاف بینک قرض کھاتَس پؠٹھ لَگان یُس ہِندوستانَن کئی دہٲییَن منٛز رپورٹ کوٚر۔ جوٗن 2026 ہِنٛز مالیاتی استحکام رپورٹ (Financial Stability Report) چھِ تِم وَتھ تہِ لیکھان یِمَن پؠٹھ سُہ کھاتہ خراب گَژھِ، تہٕ سُہ سرمایہ یُس رُوٗزِ اگر یہِ خراب گژھِ۔

## خراب قرضَن ہُنٛد ذخیرہ

شیڈولڈ تجارتی بینکَن (scheduled commercial banks) ہِنٛز کُل غٲر کارکردہ اثاثہ شرح (gross non-performing asset ratio) کم گَیہِ مارچ 2026 منٛز **1.8%**۔ رپورٹ چھِ تَس کئی دہٲیی ہُنٛد کم ترین وَنان تہٕ چھِ وَنان زِ بہتری بینک گروپَن پؠٹھ گَزرِیہِ۔ اَتھ پَتہٕ چھُ سرکٲری چیک پوائنٹَن ہُنٛد اَکھ وَریہِ: رُجحان تہٕ پیش رفت ریلیزَس (Trend and Progress release) منٛز مارچ 2025 ہِنٛدِس اَندَس تام **2.2%** تہٕ سَتَمبَر 2025 ہِنٛدِس اَندَس تام **2.1%**، پتہٕ بیٛاکِس مارچَس پؠٹھ 1.8%۔

سالانہ سلِپیج شرح (annual slippage ratio)، نَوِ خراب اثاثَن ہُنٛد بہاو، کم گَیہِ 2025-26 منٛز **1.2%**۔ یہِ ژور مالی وَریَن تام کم گَژھان اوس، سرکٲری شعبُک تہٕ نجی شعبُک بینکَن برٛونٛہہ کٔرتھ۔ پروویژن کوریج شرح (provisioning coverage ratio) وسیع پٲٹھۍ مستحکم ٲس **75.6%** پؠٹھ۔ کُل غٲر کارکردہ اثاثَن نسبت رائٹ آف (write-offs) رُوٗدِ اُچ، خاص کٔرتھ نجی شعبُک تہٕ غیر ملکی بینکَن پؠٹھ۔ بٔڈِ رائٹ آفَن سۭتۍ کم کُل شرح چھُ صاف ذخیرہ، نَہ ثبوت زِ کِہہ تہِ خراب گۆو نَہ۔

شعبہ معیار بہتر گۆو پریتھ کھُلَس معاشی شعبَس پؠٹھ۔ کٔشی رُوٗد کمزور کھاتہ: مارچ 2026 منٛز کُل غٲر کارکردہ شرح **5.1%**، تہٕ نظام ہِنٛدِ کُل غٲر کارکردہ اثاثَن ہُنٛد **37.2%**۔ صَنعت ہِنٛد ذیلی شعبہ تہٕ ذٲتی قرضہ وسیع پٲٹھۍ بہتر گَیہِ۔ بٔڈِ قرضہ ہؠن وول ٲسۍ کُل کریڈِٹُک **44.5%**، تھوڑا زِیادٕ حِصہٕ، تہٕ کُل خراب قرضَن منٛز تِہنٛد حِصہٕ کم گَژھان رُوٗز۔ تِہنٛز پنِن کُل غٲر کارکردہ شرح ٲس مارچ 2026 ہِنٛدِس اَندَس تام **1.2%**، سَتَمبَر 2024 ہِنٛدِس اَندَس کِس **2.4%** پیٹھہٕ کم۔ گۆڈنِک دباؤ کھاتَو تمام اَکی پٲٹھۍ نَہ ہِرِیہِ۔ رپورٹُک بٔڈِ قرضہ ہؠن والَن ہُنٛد چارٹ چھُ SMA-1 حجم مارچ 2026 منٛز زِیادٕ ہاوان، ییلہِ زن بیٛاکھ خصوصی ذکر حجم (special-mention volumes) سِکُڑِیہِ۔

## سرمایہ، فنڈنگ، تہٕ نَوِ قرضَن ہُنٛد خطرٕ وزن

خطرٕ وزنی اثاثَن پؠٹھ سرمایہ (capital to risk-weighted assets) اوس مارچ 2026 ہِنٛدِس اَندَس تام **17.7%**، تہٕ عام ایکویٹی ٹیر 1 (common equity tier 1) اوس **15.3%**۔ دۄشوَے چھِ کئی دہٲیی ہِنٛد اُچ نکتہ بیان۔ عام ایکویٹی اوس کُل سرمایُک **86%**۔ کریڈِٹ خطرٕ وزن چھِ تمام خطرٕ وزنی اثاثَن ہُنٛد لَگ بَگ **86%**، تہٕ 2025-26 منٛز وُدِیہِ تِم کریڈِٹ خطرٕ وزن کریڈِٹَس پانَس کھوتہٕ وارٕ، ترٛے وَریَن منٛز گۆڈنِکہِ لَٹہِ۔ رپورٹ چھُ یہِ کم خطرٕ نَوِ قرضَن کُن تبدیٖلی پٲٹھۍ پَران۔ ٹیر-1 لیوریج شرح (tier-1 leverage ratio) ٲس **8.2%**۔

لیکویڈیٹی بفر پتلہ گَیہِ تہٕ آرام دہ رُوٗدِ۔ لیکویڈیٹی کوریج شرح (liquidity coverage ratio) ٲس **124.2%** تہٕ خالص مستحکم فنڈنگ شرح (net stable funding ratio) **122.1%**، دۄشوَے ریگولیٹری فرشَن کھوتہٕ اَتھ۔ تنگ نکتہ چھُ فنڈنگ مِلان۔ جمع وُدِیہِ 2025-26 منٛز **11.5%** ییلہِ زن کریڈِٹ وُدُیو **14.5%**، تہٕ مدتی جمع (term deposits) رُوٗدِ جاری تہٕ بچت کھاتَن (current and savings accounts) کھوتہٕ برٛونٛہہ۔ رپورٹ چھِ فنڈنگ اَکھ ابھرَن وول چیلنج وَنان اگرچہ بیلنس شیٹ مضبوط بیان چھِ۔ خالص سود آمدنی (net interest income) وُدِیہِ مارچ 2026 منٛز وَریَس مقابلہ **4.0%**، حجمَن سۭتۍ سہارا، ییلہِ زن مارجِن وسیع پٲٹھۍ اَٹِ رُوٗزِ تِکیازِ فنڈَن ہِنٛز لاگت تہٕ اثاثہ پیداوار اَکی ساتہ کم گَیہِ۔ ٹیکسَس پتہٕ منافع (profit after tax) اوس 2025-26 منٛز **₹4,05,268 کَروڑ**، پٔتمِس وَریُک **₹3,78,163 کَروڑ** مقابلہ۔

## دباؤ ٹیسٹ اصل پٲٹھۍ کیاہ چھُ وَنان

میکرو ٹیسٹ چھُ زٕ وَریَن تام 46 اَہَم بینک محیط کران۔ بنیٲدی منظرَس (baseline) منٛز چھُ تِہنٛد مجموعی کُل غٲر کارکردہ شرح مارچ 2026 کِس **1.8%** پیٹھہٕ مارچ 2028 تام **1.9%** تام وَدان۔ زَن ناسازگار جیو پولیٹیکل کیسَن (adverse geopolitical cases) منٛز چھُ یہِ **3.8%** تہٕ **4.1%** تام وَدان۔

تِمَن 46 بینکَن خٲطرٕ مجموعی خطرٕ وزنی اثاثَن پؠٹھ سرمایہ چھُ ٹیسٹَس منٛز **17.5%** پیٹھہٕ شروع گَژھان (نظام ہِنٛدِس 17.7% عددَس کھوتہٕ رَژھ کم) تہٕ مارچ 2028 تام چھُ بنیٲدی منظرَس منٛز **15.6%**، گۆڈنِکِس ناسازگار کیسَس منٛز **13.3%**، تہٕ دویِمِس منٛز **13.0%** تام کم گَژھان۔ نمُونَس منٛز کانٛہہ بینک چھُ نَہ بنیٲدی منظرَس منٛز 9% کم از کم توڑان۔ اَکھ بینک چھُ گۆڈنِکِس ناسازگار کیسَس منٛز کران، تہٕ زٕ دویِمِس منٛز۔ عام ایکویٹی ٹیر 1 چھُ **15.2%** پیٹھہٕ **13.9%**، **11.6%** تہٕ **11.4%** تام کم گَژھان۔ نمُونَس منٛز پریتھ بینک چھُ ترٛے وَے کیسَن منٛز 5.5% عام ایکویٹی فرشَس کھوتہٕ اَتھ رُزان۔

اَکھ سخت اکی عنصر جھٹکہ، خراب قرض شرحَس پؠٹھ زٕ معیٲری اِنحراف (two standard deviations)، ہؠوِ 46 بینکَن ہِنٛز کُل غٲر کارکردہ شرح **8.1%** تام تہٕ سرمایہ شرح **400 بیسِس پوائنٹ** (basis points) کم کَرِ۔ یہِ نظام وُنۍ تہِ ریگولیٹری کم از کم کھوتہٕ اَتھ ترٛاوِ۔ ژور بینک، شیڈولڈ تجارتی بینک اثاثَن ہِنٛدِ **12%** سۭتۍ، تَتھ جھٹکَس منٛز کم از کم توڑِن۔ اُلٹا ٹیسٹ (reverse test) چھُ وَنان زِ نظام سرمایہ 9% پیٹھہٕ دَبانہٕ خٲطرٕ خراب قرض شرحَس منٛز 4.4 معیٲری اِنحراف وُدھُن لَگِ۔

ارتکاز چھُ تیز کریڈِٹ خطرٕ۔ اگر پریتھ بینکُک ٹاپ ترٛے انفرٲدی قرضہ ہؠن وول ڈیفالٹ کَرِ، نظام ہِنٛز خراب قرض شرح **340 بیسِس پوائنٹ** وُدِ تہٕ سرمایہ شرح **80 بیسِس پوائنٹ** کم گَژھِ۔ ٹاپ ترٛے گروپ ڈیفالٹ چھُ اَمی حصَس منٛز سٲتھی کیس۔ یِم چھِ انتہٲیی مفروضہ ڈیفالٹ، نَہ پیشن گوئی۔

## اَتھ وُدھُنَس سۭتۍ جُڑِتھ غٲر بینک چینل

خٔدمَت کریڈِٹ، جُلایی 2026 ہِنٛز تعیناتی رپورٹَس منٛز سارِوے کھوتہٕ تیز شعبہ، چھُ غٲر بینک فنانس کمپنیَن خٲطرٕ قرضَو سۭتۍ تاننہٕ یِوان۔ استحکام رپورٹَس ہِنٛدِس فنڈ ذرایع جدولَس منٛز ٲسۍ بینکَن نِش قرضہ مارچ 2026 پؠٹھ اَتھ تہٕ درمیٲنی پرتَن وٲل غٲر بینکَن (upper- and middle-layer non-banks) ہِنٛدِ فنڈَن ہُنٛد **30.4%**، اَکھ وَریہِ برٛونٛہہ کِس **28.6%** مقابلہ۔ دباؤ ٹیسٹَس منٛز غٲر بینکَن ہِنٛدِس نمُونَس خٲطرٕ ہیکہِ کُل غٲر کارکردہ شرح مارچ 2026 کِس **2.4%** پیٹھہٕ مارچ 2027 منٛز **2.8%** تام وُدِتھ، بنیٲدی منظرَس منٛز تہِ، تہٕ تِہنٛز سرمایہ شرح ہیکہِ **22.3%** پیٹھہٕ **20.8%** تام کم گَژھِتھ۔ بینک اثاثہ معیار تہٕ غٲر بینک اثاثہ معیار چھِ تَتھ فنڈنگ لکیٖرِ سۭتۍ جُڑِتھ۔

رپورٹَس پؠٹھ پریس نوٹ چھُ وَنان زِ ملکی نظام لچکدار چھُ، سرمایہ چھُ ناسازگار میکرو کیسَن تحت فرشَن کھوتہٕ اَتھ رُزان، تہٕ غٲر بینک چھِ بہتر گَژھان اثاثہ معیارَس سۭتۍ مضبوط رُزان۔ ہیٚٹھُک اعداد چھِ تَتھ جملُک سُہ ورژن یُس اَکھ خطرٕ پَرن وول آڈٹ کٔرِتھ ہیکہِ۔ گریٹ اِنڈین کمپنی (Great Indian Company) چھِ تِم وُدھُن نوٹَن سۭتۍ تھاوان تاکہ اَکھ 19% کریڈِٹ شرح معیار شرح پٲٹھۍ پَرنہٕ یِیہِ نَہ۔ وُدھُن شرحُک پنُن تعریف چھُ [اہم اعدادَن](/blog/credit-growth-key-numbers-20260330-015-ks) منٛز۔
